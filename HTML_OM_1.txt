window.addEventListener('DOMContentLoaded', function() {
    console.log('🔄 Cargando tipo de acceso actual...');
    
    google.script.run
        .withSuccessHandler(function(tipoAcceso) {
            console.log('✅ Tipo de acceso recibido:', tipoAcceso);
            
            const select = document.getElementById('tipoAccesoSelect');
            const statusSpan = document.getElementById('tipoAccesoStatus');
            
            if (select) {
                select.value = tipoAcceso;
            }
            
            if (statusSpan) {
                statusSpan.textContent = '✓ Actual: ' + tipoAcceso;
                statusSpan.style.color = '#27ae60';
            }
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error al cargar tipo de acceso:', error);
            
            const statusSpan = document.getElementById('tipoAccesoStatus');
            if (statusSpan) {
                statusSpan.textContent = '⚠️ Error al cargar tipo de acceso';
                statusSpan.style.color = '#e74c3c';
            }

function cambiarTipoAccesoUsuario() {
    const select = document.getElementById('tipoAccesoSelect');
    const statusSpan = document.getElementById('tipoAccesoStatus');
    
    if (!select) {
        console.error('❌ No se encontró el select de tipo de acceso');
        alert('❌ Error: No se encontró el selector');
        return;
    }
    
    const tipoAcceso = select.value;
    console.log('🔄 Cambiando tipo de acceso a:', tipoAcceso);
    
    // Validar selección
    if (!tipoAcceso || tipoAcceso === '') {
        alert('⚠️ Por favor selecciona un tipo de acceso');
        return;
    }
    
    // Actualizar status (si existe)
    if (statusSpan) {
        statusSpan.textContent = '⏳ Actualizando...';
        statusSpan.style.color = '#f39c12';
    }
    
    // Deshabilitar botón temporalmente
    const boton = event.target;
    boton.disabled = true;
    boton.textContent = '⏳ Procesando...';
    
    const email = document.getElementById('correoUsuario').textContent;
    
    google.script.run
        .withSuccessHandler(function(response) {
            console.log('✅ Respuesta:', response);
            
            // Rehabilitar botón
            boton.disabled = false;
            boton.textContent = '🔄 Cambiar';
            
            if (response.success) {
                if (statusSpan) {
                    statusSpan.textContent = '✓ ' + response.mensaje;
                    statusSpan.style.color = '#27ae60';
                }
                
                // Mostrar alerta de éxito
                alert('✅ Tipo de acceso cambiado exitosamente a: ' + tipoAcceso + '\n\n🔄 La página se recargará para aplicar los cambios.');
                
                // Recargar página después de 1 segundo
                setTimeout(function() {
                    location.reload();
                }, 1000);
                
            } else {
                if (statusSpan) {
                    statusSpan.textContent = '✗ Error: ' + response.error;
                    statusSpan.style.color = '#e74c3c';
                }
                alert('❌ Error al cambiar tipo de acceso: ' + response.error);
            }
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error:', error);
            
            // Rehabilitar botón
            boton.disabled = false;
            boton.textContent = '🔄 Cambiar';
            
            if (statusSpan) {
                statusSpan.textContent = '✗ Error de conexión';
                statusSpan.style.color = '#e74c3c';
            }
            alert('❌ Error de conexión: ' + error.message);
        })
        .cambiarTipoAcceso(email, tipoAcceso);
}

            currentStop.style.opacity = '1';
            currentStop.querySelector('.stop-status').innerHTML = '🔄 En progreso...';
            
            // Actualizar info del viaje
            document.getElementById('icono-proceso-actual').textContent = pasos[currentStep-1].icono;
            document.getElementById('texto-proceso-actual').textContent = pasos[currentStep-1].nombre;
            document.getElementById('detalle-proceso').textContent = pasos[currentStep-1].detalle;
            document.getElementById('contador-paradas').textContent = `Parada ${currentStep}/${totalSteps}`;
            document.getElementById('mensaje-conductor').innerHTML = `<span style="font-size:1.3rem;">🗣️</span><span style="flex:1;">${pasos[currentStep-1].mensajeConductor}</span>`;
            
            // Actualizar estado del conductor
            if (currentStep === 1) document.getElementById('estado-viaje').innerHTML = 'Iniciando procesamiento';
            else if (currentStep === 2) document.getElementById('estado-viaje').innerHTML = 'Cargando configuraciones';
            else if (currentStep === 3) document.getElementById('estado-viaje').innerHTML = 'Procesando horas';
            else if (currentStep === 4) document.getElementById('estado-viaje').innerHTML = 'Analizando KPIs';
            else if (currentStep === 5) document.getElementById('estado-viaje').innerHTML = 'Calculando bonos';
            else if (currentStep === 6) document.getElementById('estado-viaje').innerHTML = 'Integrando datos CDM';
            else if (currentStep === 7) document.getElementById('estado-viaje').innerHTML = 'Generando resultados';
            
            // Calcular porcentaje
            const percentage = Math.round((currentStep / totalSteps) * 100);
            
            // Actualizar barra de progreso
            const progressBar = document.getElementById('progress-bar-viaje');
            if (progressBar) {
                progressBar.style.width = percentage + '%';
            }
            
            // Actualizar texto de porcentaje
            const progressPercentViaje = document.getElementById('progress-percentage-viaje');
            if (progressPercentViaje) {
                progressPercentViaje.textContent = percentage + '% completado';
            }
            
            // Actualizar tiempo restante para esta parada
            const tiempoParada = document.getElementById('tiempo-parada');
            if (tiempoParada) {
                const segundosRestantes = Math.round(pasos[currentStep-1].duracion / 1000);
                tiempoParada.textContent = `~${segundosRestantes} seg`;
            }
        }
        
        // Programar siguiente paso con tiempos más largos
        if (currentStep < totalSteps) {
            window.currentProgressTimeout = setTimeout(advanceProgress, pasos[currentStep-1].duracion);
        } else {
            // Último paso: preparar finalización
            setTimeout(() => {
                const lastStop = document.querySelector(`.uber-stop[data-step="${currentStep}"]`);
                if (lastStop) {
                    lastStop.classList.remove('active-stop');
                    lastStop.classList.add('completed-stop');
                    lastStop.querySelector('.stop-status').innerHTML = '✅ Completado';
                }
                
                document.getElementById('icono-proceso-actual').textContent = '🎉';
                document.getElementById('texto-proceso-actual').textContent = '¡Proceso completado!';
                document.getElementById('detalle-proceso').textContent = 'Todos los datos procesados exitosamente';
                document.getElementById('contador-paradas').textContent = 'Destino alcanzado';
                document.getElementById('progress-percentage-viaje').textContent = '100% - ¡Listo!';
                document.getElementById('progress-percentage-viaje').style.color = '#27ae60';
                document.getElementById('progress-percentage-viaje').style.fontWeight = 'bold';
                document.getElementById('mensaje-conductor').innerHTML = '<span style="font-size:1.3rem;">🎉</span><span style="flex:1;">"Hemos llegado a tu destino. ¡Datos listos!"</span>';
                document.getElementById('estado-conductor').innerHTML = '✅ LLEGAMOS';
                document.getElementById('estado-conductor').style.background = '#27ae60';
                document.getElementById('carro-uber').innerHTML = '🏁';
                
                clearInterval(intervaloMensajes);
            }, pasos[currentStep-1].duracion);
        }
    }
    
    // Guardar intervalo
    window.mensajesInterval = intervaloMensajes;
    
    // Iniciar viaje
    window.currentProgressTimeout = setTimeout(advanceProgress, 1200);
}

  function validarYMostrarAgentes(lob) {
      console.log('🔍 Validando agentes para LOB:', lob);

      if (!datos || !datos.metricasPorLob) {
          mostrarAlerta('❌ No se encontraron datos de métricas', 'danger');
          return;
      }
      
      if (!datos.metricasPorLob[lob]) {
          mostrarAlerta(`❌ No se encontraron métricas para esta LOB: ${lob}`, 'danger');
          return;
      }
      
      const metricas = datos.metricasPorLob[lob];
      
      // ⭐ VALIDACIÓN SEGURA para error
      if (metricas.error) {
          mostrarAlerta(`⚠️ ${metricas.error}`, 'warning');
          return;
      }
      
      // ⭐ DEBUG: Verificar resultados del backend
      if (datos.debug && datos.debug.log) {
          console.log('🔍 LOG DETALLADO DEL BACKEND:', datos.debug.log);
          // Si no hay horas, mostrar alerta con detalles
          if (!datos.horasPorAgente || Object.keys(datos.horasPorAgente).length === 0) {
              console.warn('⚠️ ALERTA: Backend retornó 0 horas.', datos.debug.log);
              // Solo mostrar alerta si se esperaba mes
              if (datos.mesSeleccionado) {
                   mostrarAlerta(`⚠️ No se cargaron horas. Revise la consola (F12) para "LOG DETALLADO".`, 'warning');
              }
          }
      }

      const horasMap = datos.horasPorAgente || {}; // Asegurar objeto
      
      // ⭐ DEBUG KEYS
      if (horasMap && Object.keys(horasMap).length > 0) {
          console.log('🔑 Keys disponibles en horasPorAgente:', Object.keys(horasMap).slice(0, 10)); // Muestra primeras 10
      } else {
          console.warn('⚠️ horasPorAgente está vacío o es null');
      }

      let totalAgentes = 0;
      const agentesUnicos = new Set();
      const categorias = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
      
      categorias.forEach(cat => {
          if (metricas[cat]) {
              metricas[cat].forEach(kpiObj => {
                  const val = kpiObj.validacion || {};
                  
                  (val.empleadosCumplen || []).forEach(emp => {
                      agentesUnicos.add(emp.bmsId);
                      totalAgentes++;
                  });
                  
                  (val.empleadosNoCumplen || []).forEach(emp => {
                      agentesUnicos.add(emp.bmsId);
                      totalAgentes++;
                  });
                  
                  (val.empleadosSinDatos || []).forEach(emp => {
                      agentesUnicos.add(emp.bmsId);
                      totalAgentes++;
                  });
              });
          }
      });
      
      console.log(`📊 Agentes: ${agentesUnicos.size} únicos, ${totalAgentes} registros`);
      
      if (agentesUnicos.size === 0) {
          mostrarAlerta('⚠️ <strong>NO se encontraron agentes para esta LOB</strong>', 'danger');
      } else {
          mostrarAlerta(`✅ Se encontraron <strong>${agentesUnicos.size} agentes únicos</strong> (${totalAgentes} evaluaciones) para: <strong>${lob}</strong>`, 'success');
      }
  }

function toggleProrrateo(checkbox) {
    const bmsId       = checkbox.dataset.bms;
    const totalFull   = parseFloat(checkbox.dataset.totalFull);
    const diasExactos = parseFloat(checkbox.dataset.diasLaborados); // días exactos calculados
    const hSem        = parseFloat(checkbox.dataset.horasSem) || 42; // ⭐ horas HC del agente

    const fila = checkbox.closest('tr');

    if (!bmsId || isNaN(totalFull) || isNaN(diasExactos)) {
        console.error('❌ Datos inválidos:', { bmsId, totalFull, diasExactos });
        return;
    }

    const celdaTotal = fila.querySelector('.celda-total');
    let spanMonto = celdaTotal.querySelector('.monto-display');

    if (!spanMonto) {
        spanMonto = document.createElement('span');
        spanMonto.className = 'monto-display';
        spanMonto.style.cssText = 'font-size:1.3rem;font-weight:bold;color:#27ae60;display:block;margin-bottom:8px;';
        celdaTotal.prepend(spanMonto);
    }

    const checkAprobar = fila.querySelector('.aprobado-check');
    const checkDisputa = fila.querySelector('.dispute-check');

    if (checkbox.checked) {
        // ⭐ NUEVA FÓRMULA
        // horasPorDia = horasSemanales / 5 (depende del HC del agente)
        // diasExactos = horasReales / horasPorDia
        // valorDia    = totalFull / 30
        // bonoProrateado = valorDia * diasExactos

        const horasPorDia    = hSem / 5;
        const valorDia       = totalFull / 30;
        const bonoProrateado = valorDia * diasExactos;
        const diferencia     = totalFull - bonoProrateado;

        console.log(`⚙️ PRORRATEO ACTIVADO: BMS ${bmsId}`);
        console.log(`   HC horas/sem: ${hSem}h → ${horasPorDia.toFixed(2)}h/día`);
        console.log(`   Días exactos: ${diasExactos.toFixed(4)}`);
        console.log(`   Valor por día: $${valorDia.toFixed(2)}`);
        console.log(`   Bono original: $${totalFull.toFixed(2)}`);
        console.log(`   Bono prorrateado: $${bonoProrateado.toFixed(2)}`);
        console.log(`   Diferencia: -$${diferencia.toFixed(2)}`);

        spanMonto.textContent = '$' + bonoProrateado.toLocaleString('es-CO', { minimumFractionDigits: 2 });

        celdaTotal.style.background = 'linear-gradient(135deg, #16a085 0%, #1abc9c 100%)';
        celdaTotal.style.color      = 'white';
        celdaTotal.style.boxShadow  = '0 4px 12px rgba(22,160,133,0.4)';
        celdaTotal.dataset.bonoAjustado = bonoProrateado;

        if (checkAprobar) checkAprobar.dataset.total = bonoProrateado;
        if (checkDisputa) checkDisputa.dataset.total = bonoProrateado;

    } else {
        console.log(`❌ PRORRATEO DESACTIVADO: BMS ${bmsId} → Restaurado a: $${totalFull.toFixed(2)}`);

        spanMonto.textContent = '$' + totalFull.toLocaleString('es-CO', { minimumFractionDigits: 2 });

        celdaTotal.style.background = '#d5f4e6';
        celdaTotal.style.color      = '#000';
        celdaTotal.style.boxShadow  = '';
        celdaTotal.dataset.bonoAjustado = totalFull;

        if (checkAprobar) checkAprobar.dataset.total = totalFull;
        if (checkDisputa) checkDisputa.dataset.total = totalFull;
    }

    actualizarTotalAprobar();
    actualizarTotalGeneralTabla();
}

  // ========================================
  // MÉTRICAS
  // ========================================
function generarMetricas(lobSeleccionada) {
    const empleados = empleadosPorLob[lobSeleccionada];
    
    if (!empleados || Object.keys(empleados).length === 0) {
        console.warn('⚠️ No hay empleados');
        document.getElementById('metrics-grid').style.display = 'none';
        return;
    }
    
    const keys = Object.keys(empleados);
    let totalBonos = 0;
    let totalKPIs = 0;
    let kpisCumplidos = 0;
    let topBono = 0;
    let topNombre = '';
    
    // ⭐⭐⭐ NUEVAS VARIABLES PARA HORAS
    let totalHorasReales = 0;
    let totalHorasEsperadas = 0;
    let agentesConHoras = 0;
    
    console.log('📊 Calculando métricas para LOB:', lobSeleccionada);
    
    keys.forEach(key => {
        const emp = empleados[key];
        const bmsId = emp.bmsId;
        
        let bonoEmpleado = 0;
        let keysDelEmpleado = [];
        let otrasCategoriasDelEmpleado = 0;
        let totalDecreases = 0;
        
        // ========================================
        // ⭐ CALCULAR HORAS PARA ESTE AGENTE
        // ========================================
        if (datos.horasPorAgente && datos.horasPorAgente[bmsId]) {
            const hData = datos.horasPorAgente[bmsId];
            const hReales = hData.horas || 0;
            
            // Obtener horas semanales desde agentesPermitidos
            const agentesPermitidos = datos.agentesPermitidos || {};
            const hSemanales = agentesPermitidos[bmsId]?.horasSemanales || 42;
            
            // Calcular horas esperadas (simplificado: 22 días * horas diarias)
            const hEsperadas = (hSemanales / 5) * 22;
            
            totalHorasReales += hReales;
            totalHorasEsperadas += hEsperadas;
            agentesConHoras++;
        }
        
        // ========================================
        // ⭐ CALCULAR KEYS PRIMERO
        // ========================================
        Object.keys(emp.resultados).forEach(resKey => {
            if (resKey.startsWith('keys_')) {
                const res = emp.resultados[resKey];
                totalKPIs++;
                
                if (!res.sinDatos) {
                    if (res.cumple) {
                        kpisCumplidos++;
                        const bono = res.bono || 0;
                        if (bono > 0) {
                            keysDelEmpleado.push({ cumple: true, bono: bono });
                        }
                    } else {
                        keysDelEmpleado.push({ cumple: false, bono: 0 });
                    }
                }
            }
        });
        
        // ========================================
        // ⭐ CALCULAR OTRAS CATEGORÍAS
        // ========================================
        Object.keys(emp.resultados).forEach(resKey => {
            const res = emp.resultados[resKey];
            
            if (resKey.startsWith('bonusStructure_') || resKey.startsWith('additionalIncentives_')) {
                totalKPIs++;
                
                if (res.cumple && !res.sinDatos) {
                    kpisCumplidos++;
                    const bono = res.bono || 0;
                    if (bono > 0) {
                        otrasCategoriasDelEmpleado += bono;
                    }
                }
            } else if (resKey.startsWith('decreasedKPIs_')) {
                totalKPIs++;
                
                if (res.cumple && !res.sinDatos) {
                    kpisCumplidos++;
                    const bono = res.bono || 0;
                    totalDecreases += bono;
                }
            }
        });
        
        // ========================================
        // ⭐ APLICAR LÓGICA DE KEYS
        // ========================================
        if (keysDelEmpleado.length > 0) {
            const todosKeysCumplen = keysDelEmpleado.every(k => k.cumple);
            
            if (todosKeysCumplen) {
                const totalKeys = keysDelEmpleado.reduce((sum, k) => sum + k.bono, 0);
                bonoEmpleado = Math.max(0, totalKeys) + Math.max(0, otrasCategoriasDelEmpleado) + totalDecreases;
            } else {
                bonoEmpleado = totalDecreases;
            }
        } else {
            bonoEmpleado = Math.max(0, otrasCategoriasDelEmpleado) + totalDecreases;
        }
        
        if (bonoEmpleado < 0) {
            bonoEmpleado = 0;
        }
        
        if (bonoEmpleado > 0) {
            totalBonos += bonoEmpleado;
        }
        
        if (bonoEmpleado > topBono) {
            topBono = bonoEmpleado;
            topNombre = emp.nombre;
        }
    });
    
    const porcentajeCumplimiento = totalKPIs > 0 ? ((kpisCumplidos / totalKPIs) * 100).toFixed(1) : 0;
    
    // ⭐⭐⭐ CALCULAR PORCENTAJE DE HORAS
    const porcentajeHoras = totalHorasEsperadas > 0 
        ? ((totalHorasReales / totalHorasEsperadas) * 100).toFixed(1) 
        : 0;
    
    const colorHoras = porcentajeHoras >= 95 ? '#27ae60' : porcentajeHoras >= 85 ? '#f39c12' : '#e74c3c';
    
    console.log('📊 Resumen de métricas:');
    console.log(`   Total Bonos: $${totalBonos.toFixed(2)}`);
    console.log(`   Total Horas Reales: ${totalHorasReales.toFixed(1)}h`);
    console.log(`   Total Horas Esperadas: ${totalHorasEsperadas.toFixed(1)}h`);
    console.log(`   Porcentaje Horas: ${porcentajeHoras}%`);
    
    const metricsGrid = document.getElementById('metrics-grid');
    metricsGrid.innerHTML = `
        <div class="metric-card success">
            <div class="metric-icon">💰</div>
            <div class="metric-label">Total Bonos</div>
            <div class="metric-value">$${totalBonos.toLocaleString('es-CO', {minimumFractionDigits: 2})}</div>
            <div class="metric-subtitle">Suma total pagada</div>
        </div>
        
        <div class="metric-card">
            <div class="metric-icon">📊</div>
            <div class="metric-label">Cumplimiento</div>
            <div class="metric-value">${porcentajeCumplimiento}%</div>
            <div class="metric-subtitle">${kpisCumplidos} de ${totalKPIs} KPIs</div>
        </div>
        
        <div class="metric-card" style="border-left-color:${colorHoras};">
            <div class="metric-icon">⏱️</div>
            <div class="metric-label">Asistencia</div>
            <div class="metric-value" style="color:${colorHoras};">${porcentajeHoras}%</div>
            <div class="metric-subtitle">${totalHorasReales.toFixed(0)}h de ${totalHorasEsperadas.toFixed(0)}h esperadas</div>
        </div>
        
        <div class="metric-card danger">
            <div class="metric-icon">🏆</div>
            <div class="metric-label">Top Performer</div>
            <div class="metric-value">$${topBono.toLocaleString('es-CO', {minimumFractionDigits: 2})}</div>
            <div class="metric-subtitle">${topNombre.split(',')[0]}</div>
        </div>
    `;
    
    metricsGrid.style.display = 'grid';
    console.log(`✅ Métricas generadas: ${keys.length} empleados, ${agentesConHoras} con horas`);
}






  function generarTop10(lobSeleccionada) {
      // ⭐ PRIMERO: Limpiar Top 10 existente
      const top10Existente = document.querySelector('.top10-container');
      if (top10Existente) {
          top10Existente.remove();
      }
      
      const empleados = empleadosPorLob[lobSeleccionada];
      
      if (!empleados || Object.keys(empleados).length === 0) return;
      
      const ranking = [];
      
      Object.keys(empleados).forEach(key => {
          const emp = empleados[key];
          let totalBono = 0;
          
          Object.values(emp.resultados).forEach(res => {
              if (res.cumple && !res.sinDatos) {
                  totalBono += res.bono || 0;
              }
          });
          
          if (totalBono > 0) {
              ranking.push({ bmsId: emp.bmsId, nombre: emp.nombre, bono: totalBono });
          }
      });
      
      ranking.sort((a, b) => b.bono - a.bono);
      const top10 = ranking.slice(0, 10);
      
      if (top10.length === 0) return;
      
      let html = '<div class="top10-container" style="background:white;border-radius:12px;padding:25px;margin-bottom:25px;box-shadow:0 5px 15px rgba(0,0,0,0.08);"><button class="accordion-header" onclick="toggleAccordion(this)"><h3 style="color:var(--primary);margin:0;">🏆 Top 10 Performers - ' + lobSeleccionada + '</h3><span class="accordion-arrow">▼</span></button><div class="accordion-content" style="display:none;grid-template-columns:1fr;gap:10px;margin-top:20px;">';
      
      top10.forEach((emp, i) => {
          const medalla = i === 0 ? '🥇' : i === 1 ? '🥈' : i === 2 ? '🥉' : (i + 1) + '.';
          const bgColor = i === 0 ? '#ffd700' : i === 1 ? '#c0c0c0' : i === 2 ? '#cd7f32' : '#f8f9fa';
          
          html += '<div style="display:flex;align-items:center;justify-content:space-between;padding:12px 15px;background:' + bgColor + '15;border-radius:8px;border-left:4px solid ' + bgColor + ';"><div style="display:flex;align-items:center;gap:10px;"><span style="font-size:1.5rem;">' + medalla + '</span><div><div style="font-weight:600;color:var(--primary);">' + emp.nombre.split(',')[0] + '</div><div style="font-size:0.85rem;color:#7f8c8d;">BMS ' + emp.bmsId + '</div></div></div><div style="text-align:right;"><div style="font-size:1.3rem;font-weight:700;color:var(--success);">$' + emp.bono.toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</div></div></div>';
      });
      
      html += '</div></div>';
      
      // ⭐ INSERTAR en posición específica (después de indicadores, antes de tabla)
      const indicadores = document.querySelector('.indicadores-container');
      const tableContainer = document.getElementById('table-container');
      
      if (indicadores && tableContainer) {
          indicadores.insertAdjacentHTML('afterend', html);
      } else if (tableContainer) {
          tableContainer.insertAdjacentHTML('beforebegin', html);
      }
      
      console.log(`✅ Top 10 generado`);
  }

          const kpis = metricas[cat] || [];
          
          // ⭐ Contar cuántos agentes ÚNICOS cumplieron AL MENOS 1 KPI de esta categoría
          const agentesCumplen = new Set();
          
          kpis.forEach(kpiObj => {
              const val = kpiObj.validacion || {};
              (val.empleadosCumplen || []).forEach(emp => {
                  agentesCumplen.add(emp.bmsId);
              });
          });
          
          const cumplidos = agentesCumplen.size;
          const total = totalAgentesUnicos;
          const porcentaje = total > 0 ? ((cumplidos / total) * 100).toFixed(1) : 0;
          const color = porcentaje >= 80 ? '#27ae60' : porcentaje >= 50 ? '#f39c12' : '#e74c3c';
          
          indicadores[cat] = { cumplidos, total, porcentaje, color };
      });
      
      const nombres = { 
          keys: 'Keys', 
          bonusStructure: 'Bonus Structure', 
          additionalIncentives: 'Additional Incentives', 
          decreasedKPIs: 'Decreased KPIs' 
      };

      Object.keys(indicadores).forEach(cat => {
          const ind = indicadores[cat];
          html += '<div style="padding:15px;border-radius:8px;background:' + ind.color + '10;border-left:4px solid ' + ind.color + ';"><div style="font-size:0.9rem;color:#7f8c8d;margin-bottom:8px;">' + nombres[cat] + '</div><div style="font-size:2rem;font-weight:700;color:' + ind.color + ';margin-bottom:5px;">' + ind.porcentaje + '%</div><div style="font-size:0.85rem;color:#95a5a6;">' + ind.cumplidos + ' de ' + ind.total + ' agentes</div><div style="margin-top:10px;background:#ecf0f1;border-radius:10px;height:8px;overflow:hidden;"><div style="width:' + ind.porcentaje + '%;height:100%;background:' + ind.color + ';"></div></div></div>';
      });
      
      html += '</div></div>';
      
      // ⭐ INSERTAR en posición específica (antes de la tabla)
      const tableContainer = document.getElementById('table-container');
      if (tableContainer) {
          tableContainer.insertAdjacentHTML('beforebegin', html);

  function crearGraficoKPIs() {
      const kpisCumplimiento = {
          keys: { cumplidos: 0, noCumplidos: 0, sinDatos: 0, valorPagado: 0 },
          bonusStructure: { cumplidos: 0, noCumplidos: 0, sinDatos: 0, valorPagado: 0 },
          additionalIncentives: { cumplidos: 0, noCumplidos: 0, sinDatos: 0, valorPagado: 0 },
          decreasedKPIs: { cumplidos: 0, noCumplidos: 0, sinDatos: 0, valorPagado: 0 }
      };
      
      const lobActual = selectLob.value;
      const empleados = empleadosPorLob[lobActual];
      
      if (!empleados || Object.keys(empleados).length === 0) return;
      
      Object.keys(empleados).forEach(empKey => {
          const emp = empleados[empKey];
          Object.keys(emp.resultados).forEach(resKey => {
              const res = emp.resultados[resKey];
              const categoria = resKey.split('_')[0];
              
              if (!kpisCumplimiento[categoria]) return;
              
              if (res.sinDatos) {
                  kpisCumplimiento[categoria].sinDatos++;
              } else if (res.cumple) {
                  kpisCumplimiento[categoria].cumplidos++;
                  kpisCumplimiento[categoria].valorPagado += res.bono || 0;
              } else {
                  kpisCumplimiento[categoria].noCumplidos++;
              }
          });
      });
      
      const ctx = document.getElementById('chartKPIs');
      if (chartKPIs) chartKPIs.destroy();
      
      const categoriasLabels = ['Keys', 'Bonus Structure', 'Additional Incentives', 'Decreased KPIs'];
      const categoriasKeys = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
      
      chartKPIs = new Chart(ctx, {
          type: 'bar',
          data: {
              labels: categoriasLabels,
              datasets: [
                  { label: 'Cumplidos', data: categoriasKeys.map(k => kpisCumplimiento[k].cumplidos), backgroundColor: 'rgba(39,174,96,0.8)', borderColor: 'rgba(39,174,96,1)', borderWidth: 2 },
                  { label: 'No Cumplidos', data: categoriasKeys.map(k => kpisCumplimiento[k].noCumplidos), backgroundColor: 'rgba(231,76,60,0.8)', borderColor: 'rgba(231,76,60,1)', borderWidth: 2 },
                  { label: 'Sin Datos', data: categoriasKeys.map(k => kpisCumplimiento[k].sinDatos), backgroundColor: 'rgba(149,165,166,0.6)', borderColor: 'rgba(149,165,166,1)', borderWidth: 2 }
              ]
          },
          options: {
              responsive: true,
              maintainAspectRatio: true,
              plugins: {
                  tooltip: { callbacks: { afterLabel: function(context) {
                      const catIndex = context.dataIndex;
                      const catKey = categoriasKeys[catIndex];
                      const valor = kpisCumplimiento[catKey].valorPagado;
                      return 'Valor: $' + valor.toLocaleString('es-CO', {minimumFractionDigits: 2});
                  }}},
                  legend: { display: true, position: 'top' }
              },
              scales: { y: { beginAtZero: true, title: { display: true, text: 'Cantidad' }, ticks: { stepSize: 1 }}}
          }
      });
  }

  function crearGraficoLOBs() {
      const bonosPorLob = {};

      // Iterar a través de todas las LOBs que tienen métricas procesadas en datos.metricasPorLob
      for (const lob in datos.metricasPorLob) {
          bonosPorLob[lob] = 0;
          const metricasDeLob = datos.metricasPorLob[lob];

          // Si hay un error para esta LOB, la saltamos
          if (metricasDeLob.error) {
              continue;
          }

          // Iterar a través de las categorías de KPIs para sumar los bonos
          ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(cat => {
              if (metricasDeLob[cat]) {
                  metricasDeLob[cat].forEach(kpiObj => {
                      const validacion = kpiObj.validacion;
                      if (validacion && validacion.empleadosCumplen) {
                          validacion.empleadosCumplen.forEach(emp => {
                              bonosPorLob[lob] += emp.bono || 0;
                          });
                      }
                  });
              }
          });
      }

      // >>> INICIO DEL CAMBIO: Filtrar LOBs con bonos mayores a 0
      const lobsConBonos = Object.keys(bonosPorLob).filter(lob => bonosPorLob[lob] > 0);
      // <<< FIN DEL CAMBIO

      const ctx = document.getElementById('chartLOBs');
      if (chartLOBs) chartLOBs.destroy();

      // Usar las LOBs filtradas para el gráfico
      const lobsLabels = lobsConBonos;
      const lobsData = lobsConBonos.map(lob => bonosPorLob[lob]);

      if (lobsLabels.length === 0) {
          // Si no hay LOBs con bonos, no se crea el gráfico
          return;
      }

      chartLOBs = new Chart(ctx, {
          type: 'bar',
          data: {
              labels: lobsLabels,
              datasets: [{ label: 'Total Bonos ($)', data: lobsData, backgroundColor: 'rgba(52,152,219,0.8)', borderColor: 'rgba(52,152,219,1)', borderWidth: 2 }]
          },
          options: {
              responsive: true,
              maintainAspectRatio: true,
              plugins: { tooltip: { callbacks: { label: function(context) { return '$' + context.parsed.y.toLocaleString('es-CO', {minimumFractionDigits: 2}); }}}, legend: { display: false }},
              scales: { y: { beginAtZero: true, title: { display: true, text: 'Total ($)' }, ticks: { callback: function(value) { return '$' + value.toLocaleString('es-CO'); }}}, x: { ticks: { autoSkip: false, maxRotation: 45, minRotation: 45 }}}
          }
      });
  }

  function crearGraficoDistribucion() {
      const lobActual = selectLob.value;
      const empleados = empleadosPorLob[lobActual];
      
      if (!empleados || Object.keys(empleados).length === 0) return;
      
      let cumplidos = 0, noCumplidos = 0, sinDatos = 0;
      
      Object.keys(empleados).forEach(empKey => {
          const emp = empleados[empKey];
          Object.values(emp.resultados).forEach(res => {
              if (res.sinDatos) sinDatos++;
              else if (res.cumple) cumplidos++;
              else noCumplidos++;
          });
      });
      
      const ctx = document.getElementById('chartDistribucion');
      if (chartDistribucion) chartDistribucion.destroy();
      
      chartDistribucion = new Chart(ctx, {
          type: 'doughnut',
          data: {
              labels: ['✅ Cumplidos', '❌ No Cumplidos', '⚠️ Sin Datos'],
              datasets: [{ data: [cumplidos, noCumplidos, sinDatos], backgroundColor: ['rgba(39,174,96,0.8)', 'rgba(231,76,60,0.8)', 'rgba(149,165,166,0.6)'], borderColor: ['rgba(39,174,96,1)', 'rgba(231,76,60,1)', 'rgba(149,165,166,1)'], borderWidth: 2 }]

      });
  }

  function crearGraficoKPIPagos() {

      // Recopilar todos los pagos por KPI
      categorias.forEach(cat => {
          const kpis = metricas[cat] || [];
          kpis.forEach(kpiObj => {
              const kpiNombre = kpiObj.kpi;
              const validacion = kpiObj.validacion || {};
              
              if (!kpisPagos[kpiNombre]) {
                  kpisPagos[kpiNombre] = {
                      nombre: kpiNombre,
                      totalPagado: 0,
                      categoria: cat,
                      agentesQueCobraron: 0
                  };
              }
              
              // Sumar bonos de empleados que cumplieron
              (validacion.empleadosCumplen || []).forEach(emp => {
                  kpisPagos[kpiNombre].totalPagado += emp.bono || 0;
                  kpisPagos[kpiNombre].agentesQueCobraron++;
              });
          });
      });
      
      // Convertir a array y ordenar por monto pagado (descendente)
      const kpisArray = Object.values(kpisPagos).sort((a, b) => b.totalPagado - a.totalPagado);

              maintainAspectRatio: true,

                  tooltip: {
                      callbacks: {
                          label: function(context) {
                              const valor = context.parsed.x;

function mostrarTablaPorLob(lobSeleccionada) {
    const metricas = datos.metricasPorLob ? datos.metricasPorLob[lobSeleccionada] : null;
    
    if (!metricas || metricas.error) {
        const container = document.getElementById('table-container');
        container.innerHTML = '<div class="info-message">' + (metricas?.error || 'No hay datos') + '</div>';
        container.style.display = 'block';
        return;
    }
    
    const categorias = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
    const empleados = {};
    
    // ⭐⭐⭐ OBTENER AGENTES PERMITIDOS DEL OBJETO DATOS
    const agentesPermitidos = datos.agentesPermitidos || {};
    
    console.log('📋 Agentes permitidos recibidos:', Object.keys(agentesPermitidos).length);
    if (Object.keys(agentesPermitidos).length > 0) {
        const firstKey = Object.keys(agentesPermitidos)[0];
        console.log('🔍 Muestra Agente:', firstKey, agentesPermitidos[firstKey]);
    } else {
        console.error('❌ agentesPermitidos está VACÍO');
    }
    
    categorias.forEach(cat => {
        const kpis = metricas[cat] || [];
        kpis.forEach((kpiObj, idx) => {
            const valid = kpiObj.validacion || {};
            const tipoValidacion = kpiObj.tipo || 'mayor';
            
            // ⭐ EMPLEADOS QUE CUMPLEN
            (valid.empleadosCumplen || []).forEach(emp => {
                const key = emp.bmsId + '_' + emp.nombre;
                if (!empleados[key]) {
                    // ⭐⭐⭐ OBTENER ROLE Y HORAS DESDE agentesPermitidos
                    const datosAgente = agentesPermitidos[emp.bmsId] || {};
                    
                    empleados[key] = {
                        bmsId: emp.bmsId, 
                        nombre: emp.nombre, 
                        rol: datosAgente.rol || 'Sin rol',           // ⭐⭐⭐ NUEVO
                        horasSemanales: datosAgente.horasSemanales || 42, // ⭐⭐⭐
                        resultados: {}
                    };
                    
                    console.log(`✅ Empleado agregado: ${emp.bmsId} - Role: ${empleados[key].rol}`);
                }
                
                const esKey = cat === 'keys';
                
                empleados[key].resultados[cat + '_' + idx] = {
                    ...emp,
                    cumple: true,
                    sinDatos: false,
                    esKey: esKey,
                    tipoValidacion: tipoValidacion,
                    categoria: cat
                };
            });
            
            // ⭐ EMPLEADOS QUE NO CUMPLEN
            (valid.empleadosNoCumplen || []).forEach(emp => {
                const key = emp.bmsId + '_' + emp.nombre;
                if (!empleados[key]) {
                    const datosAgente = agentesPermitidos[emp.bmsId] || {};
                    
                    empleados[key] = {
                        bmsId: emp.bmsId, 
                        nombre: emp.nombre, 
                        rol: datosAgente.rol || 'Sin rol',           // ⭐⭐⭐ NUEVO
                        horasSemanales: datosAgente.horasSemanales || 42,
                        resultados: {}
                    };
                }
                
                const esKey = cat === 'keys';
                
                empleados[key].resultados[cat + '_' + idx] = {
                    ...emp,
                    cumple: false,
                    sinDatos: false,
                    esKey: esKey,
                    tipoValidacion: tipoValidacion,
                    categoria: cat
                };
            });
            
            // ⭐ EMPLEADOS SIN DATOS
            (valid.empleadosSinDatos || []).forEach(emp => {
                const key = emp.bmsId + '_' + emp.nombre;
                if (!empleados[key]) {
                    const datosAgente = agentesPermitidos[emp.bmsId] || {};
                    
                    empleados[key] = {
                        bmsId: emp.bmsId, 
                        nombre: emp.nombre, 
                        rol: datosAgente.rol || 'Sin rol',           // ⭐⭐⭐ NUEVO
                        horasSemanales: datosAgente.horasSemanales || 42,
                        resultados: {}
                    };
                }
                
                empleados[key].resultados[cat + '_' + idx] = {
                    ...emp,
                    cumple: false,
                    sinDatos: true,
                    esKey: cat === 'keys',
                    tipoValidacion: tipoValidacion,
                    categoria: cat
                };
            });
        });
    });
    
    empleadosPorLob[lobSeleccionada] = empleados;
    
    console.log(`✅ ${Object.keys(empleados).length} empleados procesados`);
    
    // ⭐ CARGAR TODAS LAS LOBS DEL SISTEMA
    mostrarLoading('Cargando LOBs disponibles...');
    
    google.script.run
        .withSuccessHandler(function(todasLasLobs) {
            console.log('✅ Todas las LOBs del sistema:', todasLasLobs);
            generarTablaConLobs(lobSeleccionada, empleados, metricas, categorias, todasLasLobs);
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error al cargar LOBs:', error);
            const todasLasLobs = datos.lobs || [lobSeleccionada];
            generarTablaConLobs(lobSeleccionada, empleados, metricas, categorias, todasLasLobs);
        })
        .obtenerTodasLasLobsDelSistema();
}

// FUNCIÓN AUXILIAR: Filtrar filas por estado de un KPI
// ============================================================
function filtrarColumnaKPI(colIndex, estado) {
  var tabla = document.getElementById('tabla-agentes');
  if (!tabla) return;

  document.querySelectorAll('.kpi-filter-btn[data-col="' + colIndex + '"]').forEach(function(b) {
    var activo = b.dataset.estado === estado;
    b.style.background  = activo ? '#3b82f6' : '#1e293b';
    b.style.color       = activo ? '#fff'    : '#94a3b8';
    b.style.borderColor = activo ? '#3b82f6' : '#334155';
  });

  var filas = tabla.querySelectorAll('tbody tr');
  filas.forEach(function(fila) {
    if (fila.style.borderTop === '2px solid rgb(226, 232, 240)') return;
    if (estado === 'todos') { fila.style.display = ''; return; }

    var celdas = fila.querySelectorAll('td');
    var celda  = celdas[colIndex];
    if (!celda) { fila.style.display = ''; return; }

    var html        = celda.innerHTML || '';
    var esBloqueado = html.includes('Gate ✗') || html.includes('Bloq. por otro Gate');
    var esCumple    = html.includes('Gate ✓') || celda.classList.contains('cumple');
    var esNoCumple  = !esCumple && !esBloqueado;

    var mostrar = false;
    if (estado === 'cumple')    mostrar = esCumple;
    if (estado === 'nocumple')  mostrar = esNoCumple;
    if (estado === 'bloqueado') mostrar = esBloqueado;

    fila.style.display = mostrar ? '' : 'none';
  });
}

  ocultarLoading();

    : { salientes: [], entrantes: [] };
  var agentesTrasladados = traslados.salientes.map(function(t) { return t.bmsId; });
  var agentesPermitidos  = datos.agentesPermitidos || {};
  var totalGeneralTabla  = 0;

  var gatesPorCategoria = {};
  categorias.forEach(function(cat) {
    var kpis = metricas[cat] || [];
    gatesPorCategoria[cat] = kpis.filter(function(k) { return k.esGate === true; });
  });

  var colTDCursor = 7;
  var kpiColMap   = [];
  categorias.forEach(function(cat) {

  function ordenarTabla(columnIndex) {
      const tbody = document.getElementById('tbody-agentes');
      const rows = Array.from(tbody.getElementsByTagName('tr'));
      
      if (ordenActual.columna === columnIndex) {
          ordenActual.ascendente = !ordenActual.ascendente;
      } else {
          ordenActual.columna = columnIndex;
          ordenActual.ascendente = true;
      }
      
      rows.sort((a, b) => {
          let valA, valB;
          
          if (columnIndex === -1) {
              const celdasA = a.getElementsByTagName('td');
              const celdasB = b.getElementsByTagName('td');
              valA = parseFloat(celdasA[celdasA.length - 1].textContent.replace(/[^0-9.-]/g, ''));
              valB = parseFloat(celdasB[celdasB.length - 1].textContent.replace(/[^0-9.-]/g, ''));
          } else {
              valA = a.getElementsByTagName('td')[columnIndex].textContent.trim().toLowerCase();
              valB = b.getElementsByTagName('td')[columnIndex].textContent.trim().toLowerCase();
              if (!isNaN(valA) && !isNaN(valB)) {
                  valA = parseFloat(valA);
                  valB = parseFloat(valB);
              }
          }
          
          if (valA < valB) return ordenActual.ascendente ? -1 : 1;
          if (valA > valB) return ordenActual.ascendente ? 1 : -1;
          return 0;
      });
      
      rows.forEach(row => tbody.appendChild(row));
  }

  function filtrarTabla() {
      const input = document.getElementById('search-input');
      const filter = input.value.toLowerCase().trim();
      const tbody = document.getElementById('tbody-agentes');
      const rows = tbody.getElementsByTagName('tr');
      let visibleCount = 0;
      
      for (let i = 0; i < rows.length; i++) {
          const row = rows[i];
          const bms = row.getAttribute('data-bms') || '';
          const nombre = row.getAttribute('data-nombre') || '';
          
          if (bms.includes(filter) || nombre.includes(filter)) {
              row.style.display = '';
              visibleCount++;
          } else {
              row.style.display = 'none';
          }
      }
      
      const countSpan = document.getElementById('search-count');
      countSpan.textContent = filter ? 'Mostrando ' + visibleCount + ' de ' + rows.length : '';
  }

function generarGraficoHistoricoMeses(datosPorMes) {
    // Ordenar por número de mes
    datosPorMes.sort((a, b) => a.numeroMes - b.numeroMes);
    
    const labels = datosPorMes.map(d => d.mes);
    const data = datosPorMes.map(d => d.total);
    
    const ctx = document.getElementById('chartHistoricalMeses');
    if (chartHistoricalMeses) chartHistoricalMeses.destroy();
    
    chartHistoricalMeses = new Chart(ctx, {
        type: 'bar',
        data: {
            labels: labels,
            datasets: [{
                label: 'Total Bonos ($)',
                data: data,
                backgroundColor: 'rgba(155, 89, 182, 0.8)',
                borderColor: 'rgba(155, 89, 182, 1)',
                borderWidth: 2
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: true,
            plugins: {
                tooltip: {
                    callbacks: {
                        label: function(context) {
                            const index = context.dataIndex;
                            const agentes = datosPorMes[index].agentes;
                            return [
                                'Total: $' + context.parsed.y.toLocaleString('es-CO', {minimumFractionDigits: 2}),
                                'Agentes únicos: ' + agentes
                            ];
                        }
                    }
                },
                legend: {
                    display: false
                },
                title: {
                    display: true,
                    text: 'Histórico de Bonos por Mes',
                    font: {
                        size: 16,
                        weight: 'bold'
                    }
                }
            },
            scales: {
                y: {
                    beginAtZero: true,
                    title: {
                        display: true,
                        text: 'Total Bonos ($)'
                    },
                    ticks: {
                        callback: function(value) {
                            return '$' + value.toLocaleString('es-CO');
                        }
                    }
                },
                x: {
                    title: {
                        display: true,
                        text: 'Mes CDM'
                    }
                }
            }
        }
    });
    
    document.getElementById('historical-chart-container').style.display = 'block';
    console.log('✅ Gráfico histórico generado con', datosPorMes.length, 'meses');
}

function obtenerNombreMesDeFecha(fecha) {
  if (!fecha) return 'General';
  
  console.log('🔍 Obteniendo nombre del mes para fecha:', fecha);
  
  // Buscar nombres de meses en español en la fecha
  const mesesNombres = {
    'enero': 'Enero', 'febrero': 'Febrero', 'marzo': 'Marzo', 'abril': 'Abril',
    'mayo': 'Mayo', 'junio': 'Junio', 'julio': 'Julio', 'agosto': 'Agosto',
    'septiembre': 'Septiembre', 'octubre': 'Octubre', 'noviembre': 'Noviembre', 'diciembre': 'Diciembre'
  };
  
  const fechaLower = fecha.toString().toLowerCase().trim();
  console.log('📅 Fecha en minúsculas:', fechaLower);
  
  // Buscar si la fecha contiene el nombre de algún mes
  for (const mes in mesesNombres) {
    if (fechaLower.includes(mes)) {
      console.log('✅ Mes encontrado:', mes, '->', mesesNombres[mes]);
      return mesesNombres[mes];
    }
  }
  
  // Si no encuentra nombre de mes, intentar con fecha JavaScript
  try {
    const fechaObj = new Date(fecha);
    if (!isNaN(fechaObj.getTime())) {
      const meses = ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 
                    'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'];
      const nombreMes = meses[fechaObj.getMonth()];
      console.log('📅 Mes desde Date object:', fechaObj.getMonth() + 1, '->', nombreMes);
      return nombreMes;
    }
  } catch (e) {
    console.log('⚠️ No se pudo parsear como fecha:', e);
  }
  
  console.log('❌ No se pudo determinar el mes, usando "General"');
  return 'General';
}

function obtenerNumeroMesDeFecha(fecha) {
  if (!fecha) return 0;
  
  console.log('🔍 Obteniendo número del mes para fecha:', fecha);
  
  // Mapeo CORREGIDO: Enero=1, Febrero=2, etc.
  const mesesMap = {
    'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
    'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
    'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
  };
  
  const fechaLower = fecha.toString().toLowerCase().trim();
  
  for (const mes in mesesMap) {
    if (fechaLower.includes(mes)) {
      console.log('✅ Número de mes encontrado:', mes, '->', mesesMap[mes]);
      return mesesMap[mes];
    }
  }
  
  // Si es una fecha tipo "2024-05-15"
  try {
    const fechaObj = new Date(fecha);
    if (!isNaN(fechaObj.getTime())) {
      const numeroMes = fechaObj.getMonth() + 1; // getMonth() retorna 0-11
      console.log('📅 Número de mes desde Date:', numeroMes);
      return numeroMes;
    }
  } catch (e) {
    console.log('⚠️ No se pudo parsear como fecha para número:', e);
  }
  
  console.log('❌ No se pudo determinar número de mes, usando 0');
  return 0;
}

  function aplicarTraslados() {
      const checkboxes = document.querySelectorAll('.recalc-check:checked');
      
      if (checkboxes.length === 0) {
          mostrarAlerta('⚠️ Debes seleccionar al menos un agente para trasladar', 'warning');
          return;
      }
      
      const traslados = [];
      const lobActual = selectLob.value;
      const fechaActual = selectFecha.value;
      const mesActual = mesComparativoActual;
      
      let error = false;
      
      checkboxes.forEach(check => {
          const bmsId = check.getAttribute('data-bms');
          const select = document.querySelector('.recalc-lob-select[data-bms="' + bmsId + '"]');
          const lobDestino = select.value;
          
          if (!lobDestino) {
              mostrarAlerta('⚠️ Debes seleccionar una LOB destino para BMS ' + bmsId, 'warning');
              error = true;
              return;
          }
          
          if (lobDestino === lobActual) {
              mostrarAlerta('⚠️ No puedes trasladar un agente a la misma LOB', 'warning');
              error = true;
              return;
          }
          
          const fila = document.querySelector(`tr[data-bms="${bmsId}"]`);
          const nombre = fila ? fila.querySelector('td:nth-child(4)').textContent.trim().split('\n')[0] : 'Agente ' + bmsId;
          
          traslados.push({
              bmsId: bmsId,
              nombre: nombre,
              lobOriginal: lobActual,
              lobDestino: lobDestino,
              fecha: fechaActual,
              mes: mesActual
          });
      });
      
      if (error || traslados.length === 0) return;
      
      // ⭐ CONFIRMACIÓN DETALLADA
      let mensajeConfirmacion = `⚠️ TRASLADO PERMANENTE EN H_C\n\n`;
      mensajeConfirmacion += `Vas a trasladar ${traslados.length} agente(s):\n\n`;
      
      traslados.forEach((t, i) => {
          mensajeConfirmacion += `${i+1}. ${t.nombre} (BMS ${t.bmsId})\n`;
          mensajeConfirmacion += `   De: ${t.lobOriginal} → A: ${t.lobDestino}\n\n`;
      });
      
      mensajeConfirmacion += `❗ IMPORTANTE:\n`;
      mensajeConfirmacion += `• El agente cambiará de LOB en la hoja H_C\n`;
      mensajeConfirmacion += `• Aparecerá inmediatamente en ${traslados[0].lobDestino}\n`;
      mensajeConfirmacion += `• Se registrará en "Ajustes_Traslados" para auditoría\n\n`;
      mensajeConfirmacion += `¿Continuar?`;
      
      if (!confirm(mensajeConfirmacion)) return;
      
      console.log('🔄 Aplicando traslados:', traslados);
      mostrarLoading('Trasladando ' + traslados.length + ' agente(s)...');
      
      google.script.run
          .withSuccessHandler(function(response) {
              ocultarLoading();
              
              if (response.success) {
                  mostrarAlerta('✅ ' + response.mensaje + '\n\n💡 Los agentes ya están en la nueva LOB. Recarga para ver los cambios.', 'success');
                  
                  // Recargar después de 2 segundos
                  setTimeout(() => {
                      cargarDatosCompletos(lobActual, fechaActual, mesActual);
                  }, 2000);
                  
              } else {
                  mostrarAlerta('❌ Error: ' + (response.error || 'Error desconocido'), 'danger');
              }
          })
          .withFailureHandler(function(error) {
              ocultarLoading();
              mostrarAlerta('❌ Error de conexión: ' + error.message, 'danger');
              console.error('Error completo:', error);
          })
          .trasladarAgentesTemporalmente(traslados);
  }

              horas: horas,

                                  celdaTotal.style.background = '#d5f4e6';

                          }

function calcularProrrateoInstantaneo(checkbox) {
    const bmsId = checkbox.getAttribute('data-bms');
    const horasReales = parseFloat(checkbox.getAttribute('data-horas')) || 0;
    const horasSemanales = parseFloat(checkbox.getAttribute('data-horas-semanales')) || 42;
    const bonoOriginal = parseFloat(checkbox.getAttribute('data-bono-original')) || 0;
    
    console.log(`📊 Prorrateo para BMS ${bmsId}:`, {
        checked: checkbox.checked,
        horasReales,
        horasSemanales,
        bonoOriginal
    });
    
    const fila = document.querySelector(`tr[data-bms="${bmsId}"]`);
    if (!fila) {
        console.error('❌ Fila no encontrada');
        return;
    }
    
    const celdaTotal = fila.querySelector('.celda-total');
    if (!celdaTotal) {
        console.error('❌ Celda total no encontrada');
        return;
    }
    
    if (checkbox.checked) {
        // ========================================
        // ✅ APLICAR PRORRATEO
        // ========================================
        const proporcionTrabajada = horasReales / horasSemanales;
        const diasEquivalentes = proporcionTrabajada * 6;
        const diasTotalesMes = 24;
        const bonoProrateado = (bonoOriginal / diasTotalesMes) * diasEquivalentes;
        
        console.log(`💰 Calculando:`, {
            proporcionTrabajada: proporcionTrabajada.toFixed(2),
            diasEquivalentes: diasEquivalentes.toFixed(1),
            bonoProrateado: bonoProrateado.toFixed(2)
        });
        
        // Actualizar celda con animación
        celdaTotal.style.transition = 'all 0.3s ease';
        celdaTotal.style.background = 'linear-gradient(135deg, #16a085 0%, #1abc9c 100%)';
        celdaTotal.style.color = 'white';
        celdaTotal.style.fontWeight = 'bold';
        celdaTotal.innerHTML = `
            <div style="padding:8px;">
                <div style="font-size:1.15rem;margin-bottom:4px;">
                    $${bonoProrateado.toLocaleString('es-CO', {minimumFractionDigits: 2})}
                </div>
                <div style="font-size:0.7rem;opacity:0.95;margin-bottom:3px;">
                    📊 ${diasEquivalentes.toFixed(1)} días / ${diasTotalesMes} días
                </div>
                <div style="font-size:0.65rem;opacity:0.85;border-top:1px solid rgba(255,255,255,0.3);padding-top:3px;margin-top:3px;">
                    Original: $${bonoOriginal.toLocaleString('es-CO', {minimumFractionDigits: 2})}
                </div>
            </div>
        `;
        
        // Guardar valor prorrateado
        checkbox.setAttribute('data-bono-prorrateado', bonoProrateado);
        
        console.log(`✅ Prorrateo aplicado: $${bonoProrateado.toFixed(2)}`);
        
    } else {
        // ========================================
        // ❌ QUITAR PRORRATEO (Restaurar original)
        // ========================================
        celdaTotal.style.transition = 'all 0.3s ease';
        celdaTotal.style.background = '#d5f4e6';
        celdaTotal.style.color = '#000';
        celdaTotal.style.fontWeight = 'bold';
        celdaTotal.innerHTML = '$' + bonoOriginal.toLocaleString('es-CO', {
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        });
        
        // Limpiar atributo
        checkbox.removeAttribute('data-bono-prorrateado');
        
        console.log(`✅ Valor original restaurado: $${bonoOriginal.toFixed(2)}`);
    }
    
    // Actualizar métricas totales
    actualizarMetricasConProrrateo();
}

// ⭐ FUNCIÓN COMPLETA: CONFIGURAR PRORRATEOS AUTOMÁTICOS
// ========================================
function configurarProrrateosAutomaticos() {
    const checkboxes = document.querySelectorAll('.prorrateo-check');
    
    let habilitados = 0;
    let deshabilitados = 0;
    
    checkboxes.forEach(checkbox => {
        const horasReales = parseFloat(checkbox.getAttribute('data-horas')) || 0;
        const horasSemanales = parseFloat(checkbox.getAttribute('data-horas-semanales')) || 42;
        
        // ⭐ UMBRALES FIJOS SEGÚN JORNADA
        let umbralMinimo;
        if (horasSemanales >= 42) {
            umbralMinimo = 150; // 42h/sem → < 150h = vacaciones (7-15 días)
        } else if (horasSemanales >= 40) {
            umbralMinimo = 143; // 40h/sem → < 143h = vacaciones
        } else if (horasSemanales >= 38) {
            umbralMinimo = 136; // 38h/sem → < 136h = vacaciones
        } else {
            umbralMinimo = 130; // Otras jornadas
        }
        
        const proporcionTrabajada = horasReales / horasSemanales;
        const diasEquivalentes = proporcionTrabajada * 6;
        const diasFaltantes = 24 - diasEquivalentes;
        
        // ⭐ SOLO HABILITAR si tiene MENOS del umbral mínimo
        if (horasReales > 0 && horasReales < umbralMinimo) {
            checkbox.disabled = false;
            checkbox.checked = false; // ⭐ IMPORTANTE: No marcar automáticamente
            checkbox.title = `✅ Habilitado: ${horasReales.toFixed(0)}h (< ${umbralMinimo}h) - Aprox. ${diasFaltantes.toFixed(0)} días menos`;
            checkbox.parentElement.style.opacity = '1';
            checkbox.parentElement.style.cursor = 'pointer';
            habilitados++;
        } else {
            checkbox.disabled = true;
            checkbox.checked = false;
            if (horasReales > 0) {
                checkbox.title = `❌ No requiere: ${horasReales.toFixed(0)}h (≥ ${umbralMinimo}h - Mes completo)`;
            } else {
                checkbox.title = '❌ Sin horas registradas';
            }
            checkbox.parentElement.style.opacity = '0.3';
            checkbox.parentElement.style.cursor = 'not-allowed';
            deshabilitados++;
        }
        
        // ⭐ GARANTIZAR: Restaurar valor original si está deshabilitado
        if (checkbox.disabled && checkbox.checked) {
            checkbox.checked = false;
            const bmsId = checkbox.getAttribute('data-bms');
            const bonoOriginal = parseFloat(checkbox.getAttribute('data-bono-original')) || 0;
            const fila = document.querySelector(`tr[data-bms="${bmsId}"]`);
            if (fila) {
                const celdas = fila.getElementsByTagName('td');
                const celdaTotal = celdas[celdas.length - 3];
                if (celdaTotal) {
                    celdaTotal.textContent = '$' + bonoOriginal.toLocaleString('es-CO', {minimumFractionDigits: 2});
                    celdaTotal.style.background = '#d5f4e6';
                    celdaTotal.style.color = '#000';
                    celdaTotal.style.fontWeight = 'bold';
                }
            }
            checkbox.removeAttribute('data-bono-prorrateado');
        }
    });
    
    console.log(`✅ Checkboxes: ${habilitados} habilitados (< 150h), ${deshabilitados} deshabilitados (≥ 150h)`);
    
    // ⭐ LIMPIAR ARRAY AL RECONFIGURAR
    prorrateosMarcados = [];
    console.log('🧹 Array de prorrateos limpiado por reconfiguración automática');
    
    return habilitados; // ⭐ RETORNAR cantidad para uso externo
}

function actualizarMetricasConProrrateo() {
    const lobActual = selectLob.value;
    if (!lobActual) return;
    
    const todasFilas = document.querySelectorAll('tbody tr[data-bms]');
    let totalGeneral = 0;
    let cantidadProrrateos = 0;
    
    console.log('🔄 Recalculando métricas con prorrateos...');
    
    todasFilas.forEach(fila => {
        const celdaTotal = fila.querySelector('.celda-total');
        if (!celdaTotal) return;
        
        const bonoOriginal = parseFloat(celdaTotal.getAttribute('data-bono-original')) || 0;
        const checkbox = fila.querySelector('.prorrateo-check');
        
        let bonoFinal = bonoOriginal;
        
        if (checkbox && checkbox.checked && !checkbox.disabled) {
            const bonoProrateado = parseFloat(checkbox.getAttribute('data-bono-prorrateado')) || bonoOriginal;
            bonoFinal = bonoProrateado;
            cantidadProrrateos++;
        }
        
        // ⭐ SOLO SUMAR SI ES POSITIVO
        if (bonoFinal > 0) {
            totalGeneral += bonoFinal;
        }
    });
    
    console.log(`📊 Total recalculado: $${totalGeneral.toFixed(2)} (${cantidadProrrateos} prorrateos)`);
    
    // Actualizar métrica visual
    const metricsGrid = document.getElementById('metrics-grid');
    if (metricsGrid && metricsGrid.style.display !== 'none') {
        const metricCards = metricsGrid.querySelectorAll('.metric-card');
        if (metricCards.length > 0) {
            const totalBonosCard = metricCards[0];
            const metricValue = totalBonosCard.querySelector('.metric-value');
            const metricSubtitle = totalBonosCard.querySelector('.metric-subtitle');
            
            metricValue.textContent = '$' + totalGeneral.toLocaleString('es-CO', {minimumFractionDigits: 2});
            
            if (cantidadProrrateos > 0) {
                metricSubtitle.innerHTML = `${cantidadProrrateos} prorrateo(s) aplicado(s)`;
                totalBonosCard.style.borderLeft = '4px solid #16a085';
            } else {
                metricSubtitle.textContent = 'Suma total pagada';
                totalBonosCard.style.borderLeft = '4px solid #27ae60';
            }
        }
    }
}

function obtenerNombreMesDesdeNumero(numeroMes) {
    if (!numeroMes || isNaN(numeroMes)) return 'General';
    
    const meses = [
        'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 
        'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'
    ];
    
    // Asegurar que el número esté entre 1-12
    const numero = parseInt(numeroMes);
    if (numero >= 1 && numero <= 12) {
        return meses[numero - 1];
    }
    
    return 'General';
}

      tieneDisputa      = true;
      comentarioDisputa = disputasPendientes[bmsId].comentario       || '';
      kpisDisputados    = disputasPendientes[bmsId].kpisSeleccionados || [];
      montoDisputa      = parseFloat(disputasPendientes[bmsId].montoDisputa)  || 0;
      bonoKPIsSelec     = parseFloat(disputasPendientes[bmsId].bonoKPIsSelec) || 0;
      countDisp++;
    }

    let horasTrabajadas = 0;
    let ausencias       = 0;

    if (datos && datos.horasPorAgente && datos.horasPorAgente[bmsId]) {
      const hData     = datos.horasPorAgente[bmsId];
      horasTrabajadas = parseFloat(hData.horas)     || parseFloat(hData.horasTotales) || 0;
      ausencias       = parseFloat(hData.ausencias) || 0;
    }

    const desgloseKPIs  = [];
    let totalAdditional = 0;
    let totalDecreased  = 0;

    try {
      const metricas2 = datos.metricasPorLob ? datos.metricasPorLob[lobActual] : null;

      if (metricas2) {
        const categorias2 = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
        const catNombres  = {
          keys:                 'Key',
          bonusStructure:       'Bonus Structure',
          additionalIncentives: 'Additional Incentives',
          decreasedKPIs:        'Decreased KPI'
        };

        categorias2.forEach(function(cat) {
          const kpis2 = metricas2[cat] || [];
          kpis2.forEach(function(kpiObj) {
            const valid = kpiObj.validacion || {};

            const empCumple   = (valid.empleadosCumplen   || []).find(function(e) {
              return String(e.bmsId).trim() === String(bmsId).trim();
            });
            const empNoCumple = (valid.empleadosNoCumplen || []).find(function(e) {
              return String(e.bmsId).trim() === String(bmsId).trim();
            });
            const empSinDatos = (valid.empleadosSinDatos  || []).find(function(e) {
              return String(e.bmsId).trim() === String(bmsId).trim();
            });

            if (!empCumple && !empNoCumple && !empSinDatos) return;
            if (empSinDatos && !empCumple && !empNoCumple)  return;

            const emp    = empCumple || empNoCumple;
            const cumple = !!empCumple;
            const bono   = emp.bono   || 0;
            const valorF = (emp.valorF !== undefined && emp.valorF !== null) ? emp.valorF : '-';

            let scoreObjetivo = kpiObj.score || '-';
            if (kpiObj.esMultiTier && emp.scoreCumplido) {
              scoreObjetivo = emp.scoreCumplido;
              if (emp.participacionUsada) {
                scoreObjetivo += ' (' + (emp.participacionUsada * 100).toFixed(0) + '%)';
              }
            }

            const esGate = kpiObj.esGate === true;

            let nota = '';
            if (cat === 'decreasedKPIs' && cumple)  nota  = 'Penalización';
            if (esGate && cumple)   nota += (nota ? ' | ' : '') + '🔒 Gate cumplido';
            if (esGate && !cumple)  nota += (nota ? ' | ' : '') + '🔒 Gate NO cumplido - bloqueó grupo';

            const tipoEstructura = esGate ? 'Gate' : kpiObj.esMultiTier ? 'Multi-Escala' : 'Fija';
            const categoriaLabel = catNombres[cat];
            const bonoNum        = parseFloat(bono) || 0;

            if (cumple) {
              if (cat === 'additionalIncentives') totalAdditional += bonoNum;
              else if (cat === 'decreasedKPIs')   totalDecreased  += bonoNum;
            }

            desgloseKPIs.push({
              categoria:      categoriaLabel,
              kpi:            kpiObj.kpi,
              scoreObjetivo:  scoreObjetivo,
              valorObtenido:  valorF,
              estado:         cumple ? 'Cumple' : 'No Cumple',
              bono:           bonoNum,
              nota:           nota,
              tipoEstructura: tipoEstructura
            });
          });
        });
      }
    } catch(e) {
      console.error('⚠️ Error desglose BMS ' + bmsId + ':', e);
    }

    totalGeneral += montoFinal;

    aprobaciones.push({
      bmsId:               bmsId,
      nombre:              nombre,
      total:               montoFinal,
      lob:                 lobActual,
      fecha:               fechaActual,
      mes:                 nombreMesActual,
      fueProrateado:       fueProrateado,
      montoOriginal:       montoOriginal,
      diasLaborados:       diasLaborados,
      diferenciaProrrateo: fueProrateado ? (montoOriginal - montoFinal) : 0,
      horasTrabajadas:     horasTrabajadas,
      ausencias:           ausencias,
      tieneDisputa:        tieneDisputa,
      comentarioDisputa:   comentarioDisputa,
      kpisDisputados:      kpisDisputados,
      montoDisputa:        montoDisputa,
      bonoKPIsSelec:       bonoKPIsSelec,
      desgloseKPIs:        desgloseKPIs,
      totalAdditional:     totalAdditional,
      totalDecreased:      totalDecreased
    });
  });

  // ── Mensaje de confirmación ──────────────────────────────
  let msgConf  = '⚠️ CONFIRMACIÓN DE APROBACIÓN\n\n';
  msgConf     += 'LOB: ' + lobActual + ' | Mes: ' + nombreMesActual + '\n';
  if (esModoSemana) {
    msgConf   += '📅 Modo: SEMANA (se detectará automáticamente Semana_1 a 4)\n';
  } else {
    msgConf   += '📋 Modo: MES COMPLETO\n';
  }
  msgConf     += '══════════════════════════════\n\n';
  msgConf     += '👥 Agentes: '  + aprobaciones.length + '\n';
  msgConf     += '💰 Total: $'   + totalGeneral.toLocaleString('es-CO', { minimumFractionDigits: 2 }) + '\n';
  if (countProrr > 0) msgConf += '⚙️ Con prorrateo: ' + countProrr + '\n';
  if (countDisp  > 0) msgConf += '🚨 Con disputa: '   + countDisp  + '\n';
  msgConf += '\n¿Confirmar?';

  if (!confirm(msgConf)) {
    mostrarAlerta('ℹ️ Aprobación cancelada', 'info');
    return;
  }

  mostrarLoading('Guardando aprobaciones...');

  google.script.run
    .withSuccessHandler(function(resp) {
      ocultarLoading();
      if (resp.success) {
        let msg  = '✅ APROBACIÓN REGISTRADA\n\n';
        msg     += '• ' + resp.count + ' agente(s)\n';
        msg     += '• Total: $' + totalGeneral.toLocaleString('es-CO', { minimumFractionDigits: 2 }) + '\n';
        if (resp.semana)               msg += '• 📅 Registrado como: ' + resp.semana + '\n';
        if (resp.esTotalMes)           msg += '• 🎯 ¡Semana 4 completada! TOTAL_MES calculado.\n';
        if (resp.kpisDetalle       > 0) msg += '• ' + resp.kpisDetalle       + ' KPIs en Audit_Detalle\n';
        if (resp.prorrateos        > 0) msg += '• ' + resp.prorrateos        + ' prorrateo(s)\n';
        if (resp.disputasGuardadas > 0) msg += '• ' + resp.disputasGuardadas + ' disputa(s)\n';
        alert(msg);

        mostrarAlerta('✅ ' + resp.mensaje, 'success');
        disputasPendientes = {};
        actualizarBannerDisputas();

        // ⭐ Si fue semana 4 → desactivar checkbox y ocultar banner
        if (resp.esTotalMes) {
          if (chkSemana) {
            chkSemana.checked = false;
            toggleModoSemana(chkSemana);
          }
          // Ocultar banner de semana y mostrar el de "mes completo"
          const bSemana    = document.getElementById('banner-modo-semana');
          const bSinSemana = document.getElementById('banner-sin-semana');
          if (bSemana)    bSemana.style.display    = 'none';
          if (bSinSemana) bSinSemana.style.display = 'flex';
        }

        checkboxes.forEach(function(ch) {
          const bms    = ch.getAttribute('data-bms');
          const apInfo = aprobaciones.find(function(a) { return a.bmsId === bms; });
          const filaEl = document.getElementById('row-' + bms);

          let badges = '<span style="color:#155724;font-weight:bold;font-size:0.85rem;">✓ APROBADO</span>';

          if (resp.semana) {
            badges += '<br><span style="background:#dbeafe;color:#1e40af;font-size:0.7rem;'
              + 'padding:1px 6px;border-radius:3px;font-weight:600;">📅 ' + resp.semana + '</span>';
          }
          if (resp.esTotalMes) {
            badges += '<br><span style="background:#fef9c3;color:#854d0e;font-size:0.7rem;'
              + 'padding:1px 6px;border-radius:3px;font-weight:600;">🎯 TOTAL MES</span>';
          }
          if (apInfo && apInfo.fueProrateado)
            badges += '<br><span style="color:#16a085;font-size:0.7rem;">⚙️ Prorrateado</span>';

            badges += '<br><span style="color:#e74c3c;font-size:0.7rem;">🚨 Con Disputa</span>';

          alert('❌ ERROR\n\n' + resp.error);
          mostrarAlerta('❌ Error: ' + resp.error, 'danger');
        }
      }
    })
    .withFailureHandler(function(err) {
      ocultarLoading();
      alert('❌ ERROR DE CONEXIÓN\n\n' + err.message);
      mostrarAlerta('❌ Error: ' + err.message, 'danger');

function validarYCargarDatos(lob, fecha, mes) {
    const lobActual   = lob;
    const fechaActual = fecha;
    const mesActual   = mes;

    console.log(`🔍 Validando aprobación previa para ${lobActual} (${fechaActual})...`);
    mostrarLoading('Verificando aprobaciones previas...');

    google.script.run
        .withSuccessHandler(function(resultado) {
            console.log('📦 Resultado de validación:', resultado);

            if (resultado.yaAprobado) {
                ocultarLoading();
                mostrarMensajeYaAprobado(resultado);
                return;
            }

            // ⭐ Modo semana: hay semanas previas pero NO está cerrado → permitir continuar
            if (resultado.esModoSemana) {
                console.log(`📅 Modo semana detectado: semana actual=${resultado.semanaActual}, próxima=${resultado.proximaSemana}`);

                // Activar checkbox semana automáticamente
                const chkSemana = document.getElementById('chk-semana');
                if (chkSemana && !chkSemana.checked) {
                    chkSemana.checked = true;
                    if (typeof toggleModoSemana === 'function') toggleModoSemana(chkSemana);
                }

                // Mostrar banner informativo (no bloqueante)
                mostrarAlerta(
                    '📅 <strong>' + resultado.mensaje + '</strong>' +
                    ' — Acumulado: <strong>' + (resultado.detalles.totalAcumulado || '') + '</strong>',
                    'info'
                );
            }

            // En ambos casos (sin semana o con semana parcial) → cargar datos
            console.log('✅ Validación pasada - Cargando datos...');
            cargarDatosCompletos(lobActual, fechaActual, mesActual);
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error en validación:', error);
            ocultarLoading();

            const continuar = confirm(
                '⚠️ ERROR AL VERIFICAR APROBACIONES\n\n' +
                'No se pudo verificar si esta LOB ya fue aprobada.\n\n' +
                'Detalles: ' + error.message + '\n\n' +
                '¿Desea continuar de todas formas?'
            );

            if (continuar) {
                cargarDatosCompletos(lobActual, fechaActual, mesActual);
            }
        })
        .verificarAprobacionExistente(lobActual, fechaActual, mesActual); // ⭐ cambiado a verificarAprobacionExistente
}

function actualizarSoloTabla() {
    console.log('🔄 Actualizando solo tabla y gráficos (RECARGA FORZADA)...');
    
    const lobActual = selectLob.value;
    const fechaActual = selectFecha.value;
    const mesActual = mesComparativoActual;
    
    if (!lobActual || !fechaActual) {
        mostrarAlerta('⚠️ Debes seleccionar LOB y Fecha primero', 'warning');
        return;
    }
    
    // ⭐ LIMPIAR CACHE DE DATOS ANTES DE RECARGAR
    datos = null;
    horasPorAgente = {};
    empleadosPorLob = {};
    
    mostrarLoadingConProgreso();
    
    google.script.run
        .withSuccessHandler(function(datosRecibidos) {
            console.log('✅ Datos FRESCOS recibidos:', datosRecibidos);
            
            if (datosRecibidos.error) {
                mostrarError('Error al actualizar: ' + datosRecibidos.error);
                return;
            }
            
            // ⭐ ASIGNAR DATOS FRESCOS
            datos = datosRecibidos;
            
            if (datosRecibidos.horasPorAgente) {
                horasPorAgente = datosRecibidos.horasPorAgente;
                console.log('✅ Horas actualizadas:', Object.keys(horasPorAgente).length, 'agentes');
            }
            
            // ⭐ VALIDAR Y MOSTRAR AGENTES
            validarYMostrarAgentes(lobActual);
            
            // ⭐ PROCESAR Y MOSTRAR DATOS
            procesarYMostrarDatos(lobActual);
            
            // ⭐ APLICAR DISPUTAS DEL CACHE
            setTimeout(() => {
                aplicarDisputasDesdeCache();
            }, 1000);
            
            ocultarLoading();
            mostrarAlerta('✅ Tabla actualizada con datos FRESCOS de H_C', 'success');
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error al actualizar:', error);
            ocultarLoading();
            mostrarAlerta('❌ Error: ' + error.message, 'danger');
        })
        .obtenerDatosParaFrontend(lobActual, fechaActual, mesActual);
}

    console.log('📍 LOB actual:', lobActual);
    console.log('📍 Fecha actual:', fechaActual);
    
    if (!lobActual || !fechaActual) {
        console.warn('⚠️ No hay LOB o Fecha seleccionada');
        return;
    }
    
    const cacheKey = `disputas_${lobActual}_${fechaActual}`;
    console.log('🔑 Cache key:', cacheKey);
    
    try {
        // Método 1: Cache específico
        let disputasCache = localStorage.getItem(cacheKey);
        let disputas = [];
        
        console.log('📦 Cache específico raw:', disputasCache);
        
        if (disputasCache) {
            disputas = JSON.parse(disputasCache);
            console.log('✅ Disputas del cache específico:', disputas);
        }
        
        // Método 2: Cache global (backup)
        if (disputas.length === 0) {
            const todasDisputasRaw = localStorage.getItem('disputasEnviadas');
            console.log('📦 Cache global raw:', todasDisputasRaw);
            
            if (todasDisputasRaw) {
                const todas = JSON.parse(todasDisputasRaw);
                console.log('📋 Total disputas en cache global:', todas.length);
                
                disputas = todas.filter(d => {
                    const match = d.lob === lobActual && d.fecha === fechaActual;
                    console.log(`   Comparando: ${d.lob} === ${lobActual} && ${d.fecha} === ${fechaActual} = ${match}`);
                    return match;
                });
                
                console.log('✅ Disputas filtradas del global:', disputas);
            }
        }
        
        if (disputas.length === 0) {
            console.log('📭 No hay disputas en cache para esta LOB y fecha');
            return;
        }
        
        console.log(`🎯 Aplicando ${disputas.length} disputa(s)...`);
        
        // Esperar a que la tabla esté completamente cargada
        setTimeout(() => {
            let marcadas = 0;
            
            disputas.forEach(disputa => {
                console.log(`   Buscando fila para BMS ${disputa.bmsId}...`);
                const fila = document.querySelector(`tr[data-bms="${disputa.bmsId}"]`);
                
                if (fila) {
                    console.log(`   ✅ Fila encontrada para BMS ${disputa.bmsId}`);
                    marcarComoDisputadoCompleto(disputa);
                    marcadas++;
                } else {
                    console.warn(`   ⚠️ Fila NO encontrada para BMS ${disputa.bmsId}`);
                }
            });
            
            console.log(`✅ Total marcadas: ${marcadas} de ${disputas.length}`);
            
            if (marcadas > 0) {
                mostrarAlerta(`🚨 ${marcadas} agente(s) con disputas registradas`, 'warning');
            }
        }, 500);
        
    } catch (e) {
        console.error('❌ Error al aplicar disputas desde cache:', e);
        console.error('Stack:', e.stack);
    }
}

function marcarComoDisputadoCompleto(disputa) {

    if (!fila) {

        // Intentar con ID alternativo
        const filaAlt = document.getElementById(`row-${disputa.bmsId}`);
        if (!filaAlt) {
            console.error(`❌ Tampoco se encontró con ID row-${disputa.bmsId}`);
            return;
        }
        console.log(`✅ Fila encontrada con ID alternativo`);
        fila = filaAlt;
    }
    
    // 1. Estilo de fila
    fila.style.background = 'linear-gradient(135deg, #f8d7da 0%, #f5c6cb 100%)';
    fila.style.borderLeft = '5px solid #e74c3c';
    fila.style.boxShadow = '0 4px 12px rgba(231, 76, 60, 0.3)';
    console.log(`   ✓ Estilo de fila aplicado`);
    
    // 2. Badge en nombre (columna 4)
    const celdaNombre = fila.querySelector('td:nth-child(4)');
    if (celdaNombre) {
        // Verificar si ya tiene el badge
        const badgeExistente = celdaNombre.querySelector('.badge-disputado');

            const badge = document.createElement('div');
            badge.className = 'badge-disputado';
            badge.style.cssText = 'margin-top:8px;display:inline-block;';

                    🚨 DISPUTADO
                </span>
            `;
            celdaNombre.appendChild(badge);
            console.log(`   ✓ Badge agregado`);
        } else {
            console.log(`   ℹ️ Badge ya existe`);
        }
    } else {
        console.warn(`   ⚠️ Celda de nombre no encontrada`);
    }
    
    // 3. Deshabilitar checkbox de disputa
    const checkbox = fila.querySelector('.dispute-check');
    if (checkbox) {
        checkbox.disabled = true;
        checkbox.checked = false;
        checkbox.style.cursor = 'not-allowed';
        checkbox.style.opacity = '0.5';
        checkbox.title = `Ya disputado el ${new Date(disputa.fechaDisputa).toLocaleString('es-CO')}`;
        console.log(`   ✓ Checkbox de disputa deshabilitado`);
    } else {
        console.warn(`   ⚠️ Checkbox de disputa no encontrado`);
    }
    
    // 4. Actualizar celda de comentario
    const celdaComentario = fila.querySelector('.celda-comentario-disputa');
    if (celdaComentario) {
        const textoSpan = celdaComentario.querySelector('.texto-comentario');
        if (textoSpan) {
            const comentarioCorto = disputa.comentario.length > 40 
                ? disputa.comentario.substring(0, 40) + '...' 
                : disputa.comentario;
            
            textoSpan.textContent = comentarioCorto;
            textoSpan.style.color = '#721c24';
            textoSpan.style.fontStyle = 'normal';
            textoSpan.style.fontWeight = '600';
            
            celdaComentario.style.background = '#f8d7da';
            celdaComentario.style.borderLeft = '3px solid #e74c3c';
            celdaComentario.style.padding = '8px';
            celdaComentario.title = `DISPUTADO:\n${disputa.comentario}\n\nFecha: ${new Date(disputa.fechaDisputa).toLocaleString('es-CO')}\nUsuario: ${disputa.usuarioDisputa || 'N/A'}`;
            console.log(`   ✓ Celda de comentario actualizada`);
        }
    } else {
        console.warn(`   ⚠️ Celda de comentario no encontrada`);
    }
    
    // 5. Deshabilitar checkbox de aprobar
    const checkboxAprobar = fila.querySelector('.aprobado-check');
    if (checkboxAprobar) {
        checkboxAprobar.disabled = true;
        checkboxAprobar.checked = false;
        checkboxAprobar.style.cursor = 'not-allowed';
        checkboxAprobar.style.opacity = '0.3';
        console.log(`   ✓ Checkbox de aprobar deshabilitado`);
    }
    
    console.log(`✅ BMS ${disputa.bmsId} marcado completamente como disputado`);
}

// ── cargarKPIsEnModal ─────────────────────────────────────────
function cargarKPIsEnModal(bmsId, bonoTotal, kpisYaSeleccionados) {
  var contenedor = document.getElementById('modal-disputa-kpis');
  contenedor.innerHTML = '<div style="text-align:center;padding:20px;color:#94a3b8;font-size:0.85rem;">Cargando KPIs...</div>';

  // IDs previamente seleccionados
  var prevSelecSet = new Set((kpisYaSeleccionados || []).map(function(k) { return k.kpi; }));

  // Extraer KPIs desde datos globales
  var lobActual = selectLob ? selectLob.value : '';
  var kpisDelAgente = [];

  try {
    var metricas = datos && datos.metricasPorLob ? datos.metricasPorLob[lobActual] : null;

    if (metricas) {
      var catNombres = {
        keys:                 'Key',
        bonusStructure:       'Bonus',
        additionalIncentives: 'Additional',
        decreasedKPIs:        'Decreased'
      };
      var catColores = {
        keys:                 '#3b82f6',
        bonusStructure:       '#16a34a',
        additionalIncentives: '#f59e0b',
        decreasedKPIs:        '#ef4444'
      };

      ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(function(cat) {
        var kpis = metricas[cat] || [];
        kpis.forEach(function(kpiObj) {
          var valid    = kpiObj.validacion || {};
          var empC     = (valid.empleadosCumplen   || []).find(function(e) { return String(e.bmsId).trim() === String(bmsId).trim(); });
          var empNC    = (valid.empleadosNoCumplen || []).find(function(e) { return String(e.bmsId).trim() === String(bmsId).trim(); });
          var emp      = empC || empNC;

          if (!emp) return;  // agente no aparece en este KPI

          var cumple   = !!empC;
          var bono     = parseFloat(emp.bono) || 0;
          var valorObt = (emp.valorF !== undefined && emp.valorF !== null) ? emp.valorF : '-';
          var score    = kpiObj.score || '-';
          var esGate   = kpiObj.esGate === true;

          kpisDelAgente.push({
            kpi:            kpiObj.kpi,
            categoria:      catNombres[cat],
            catKey:         cat,
            color:          catColores[cat],
            score:          score,
            valorObtenido:  valorObt,
            bono:           bono,
            cumple:         cumple,
            esGate:         esGate
          });
        });
      });
    }
  } catch(e) {
    console.error('Error cargando KPIs en modal:', e);
  }

  // Render
  if (kpisDelAgente.length === 0) {
    contenedor.innerHTML = '<div style="text-align:center;padding:20px;color:#94a3b8;font-size:0.85rem;">No se encontraron KPIs para este agente</div>';
    return;
  }

  contenedor.innerHTML = '';

  kpisDelAgente.forEach(function(k) {
    var checked   = prevSelecSet.has(k.kpi) ? 'checked' : '';
    var bgEstado  = k.cumple ? '#f0fdf4' : '#fef2f2';
    var textEstado= k.cumple ? '#166534' : '#991b1b';
    var labelEst  = k.cumple ? 'Cumple' : 'No cumple';

    var gateHtml  = k.esGate
      ? '<span style="background:#7c3aed;color:#fff;font-size:0.65rem;padding:2px 6px;border-radius:4px;margin-left:4px;">🔒 Gate</span>'
      : '';

    var bonoStr   = k.bono !== 0
      ? '$' + Math.abs(k.bono).toLocaleString('es-CO', { minimumFractionDigits: 2 })
      : '$0';
    if (k.bono < 0) bonoStr = '-' + bonoStr;

    var card = document.createElement('label');
    card.style.cssText = 'display:flex;align-items:flex-start;gap:10px;padding:10px 12px;background:#fff;border:1.5px solid #e2e8f0;border-radius:8px;cursor:pointer;transition:all .15s;';
    card.innerHTML = '\
      <input type="checkbox" class="kpi-disputa-check" data-kpi="' + k.kpi + '"\
        data-bono="' + k.bono + '" data-cumple="' + k.cumple + '"\
        data-categoria="' + k.categoria + '" data-score="' + k.score + '"\
        data-valor="' + k.valorObtenido + '"\
        ' + checked + '\
        style="margin-top:3px;width:16px;height:16px;cursor:pointer;accent-color:#dc2626;flex-shrink:0;">\
      <div style="flex:1;min-width:0;">\
        <div style="display:flex;align-items:center;flex-wrap:wrap;gap:4px;margin-bottom:4px;">\
          <span style="font-weight:600;font-size:0.85rem;color:#0f172a;">' + k.kpi + '</span>\
          ' + gateHtml + '\
          <span style="background:' + k.color + '20;color:' + k.color + ';font-size:0.65rem;padding:2px 7px;border-radius:4px;font-weight:600;">' + k.categoria + '</span>\
        </div>\
        <div style="display:flex;gap:12px;font-size:0.75rem;color:#64748b;">\
          <span>Score: <b>' + k.score + '</b></span>\
          <span>Valor: <b>' + k.valorObtenido + '</b></span>\
          <span>Bono: <b style="color:#0f172a;">' + bonoStr + '</b></span>\
          <span style="background:' + bgEstado + ';color:' + textEstado + ';padding:1px 7px;border-radius:4px;font-weight:600;">' + labelEst + '</span>\
        </div>\
      </div>';

    // Hover
    card.addEventListener('mouseenter', function() { card.style.borderColor = '#dc2626'; card.style.background = '#fff5f5'; });
    card.addEventListener('mouseleave', function() {
      var isChecked = card.querySelector('input').checked;
      card.style.borderColor = isChecked ? '#dc2626' : '#e2e8f0';
      card.style.background  = isChecked ? '#fff5f5' : '#fff';
    });

    contenedor.appendChild(card);
  });

  // Resaltar los ya seleccionados
  actualizarResumenKPIs(bonoTotal);

  // Eventos de cambio
  contenedor.querySelectorAll('.kpi-disputa-check').forEach(function(chk) {
    chk.addEventListener('change', function() {
      var card = chk.closest('label');
      card.style.borderColor = chk.checked ? '#dc2626' : '#e2e8f0';
      card.style.background  = chk.checked ? '#fff5f5' : '#fff';
      actualizarResumenKPIs(bonoTotal);
    });
    // Estado inicial
    if (chk.checked) {
      var card = chk.closest('label');
      card.style.borderColor = '#dc2626';
      card.style.background  = '#fff5f5';
    }
  });
}



function actualizarResumenKPIs(bonoTotal) {
  var checks     = document.querySelectorAll('.kpi-disputa-check:checked');
  var totalBono  = 0;
  var countSelec = checks.length;

  checks.forEach(function(chk) {
    totalBono += Math.abs(parseFloat(chk.getAttribute('data-bono')) || 0);
  });

  // ⭐ ACTUALIZAR TOPE GLOBAL según KPIs seleccionados
  if (countSelec > 0) {
    _topeMaximoDisputa = totalBono;  // tope = suma de KPIs seleccionados
  } else {
    // Sin KPIs seleccionados, volver al bono total del agente
    var elDisplay = document.getElementById('modal-bono-total-display');
    if (elDisplay) {
      var txt = (elDisplay.textContent || '0')
        .replace(/[^0-9.,]/g, '').replace(',', '.');
      _topeMaximoDisputa = parseFloat(txt) || 0;
    }
  }

  // Actualizar display KPIs seleccionados
  var elKpisDisplay = document.getElementById('modal-bono-kpis-display');
  if (elKpisDisplay) {
    elKpisDisplay.textContent =
      '$' + totalBono.toLocaleString('es-CO', { minimumFractionDigits: 2 });
  }

  // Resumen texto
  var resumen    = document.getElementById('modal-kpis-resumen');
  var resumenTxt = document.getElementById('modal-kpis-resumen-texto');
  if (resumen && resumenTxt) {
    if (countSelec > 0) {
      resumen.style.display = 'block';
      resumenTxt.innerHTML  =
        '<b>' + countSelec + '</b> KPI(s) seleccionado(s) — Bono: <b>$' +
        totalBono.toLocaleString('es-CO', { minimumFractionDigits: 2 }) + '</b>';
    } else {
      resumen.style.display = 'none';
    }
  }

  // Auto-sugerir monto si campo vacío
  var inputMonto = document.getElementById('modal-disputa-monto');
  if (inputMonto && countSelec > 0 &&
      (!inputMonto.value || parseFloat(inputMonto.value) === 0)) {
    inputMonto.value = totalBono.toFixed(2);
  }

  generarBotonesRapidos(_topeMaximoDisputa);
  validarMontoDisputa();
}

function generarBotonesRapidos(tope) {
  var cont = document.getElementById('modal-monto-rapido');
  if (!cont) return;
  cont.innerHTML = '';

  var topeReal = tope || _topeMaximoDisputa || 0;
  if (topeReal <= 0) return;

  [100, 75, 50, 25].forEach(function(pct) {
    var monto = (topeReal * pct / 100).toFixed(2);
    var btn   = document.createElement('button');
    btn.type  = 'button';
    btn.textContent = pct + '%';
    btn.title = '$' + parseFloat(monto).toLocaleString('es-CO', { minimumFractionDigits: 2 });
    btn.style.cssText =
      'padding:6px 12px;background:#f8fafc;color:#475569;border:1px solid #e2e8f0;' +
      'border-radius:6px;cursor:pointer;font-size:0.75rem;font-weight:600;transition:all .15s;';
    btn.addEventListener('mouseenter', function() {
      btn.style.background  = '#fef2f2';
      btn.style.borderColor = '#fca5a5';
      btn.style.color       = '#dc2626';
    });
    btn.addEventListener('mouseleave', function() {
      btn.style.background  = '#f8fafc';
      btn.style.borderColor = '#e2e8f0';
      btn.style.color       = '#475569';
    });
    btn.onclick = function() {
      document.getElementById('modal-disputa-monto').value = monto;
      validarMontoDisputa();
    };
    cont.appendChild(btn);
  });
}


function aplicarMontoRapido(valor) {
  document.getElementById('modal-disputa-monto').value = valor.toFixed(2);
  validarMontoDisputa();
}

// ── validarMontoDisputa ───────────────────────────────────────





function validarMontoDisputa() {
  var inputMonto = document.getElementById('modal-disputa-monto');
  var hint       = document.getElementById('modal-monto-hint');

  if (!inputMonto) return false;

  var monto = parseFloat(inputMonto.value) || 0;
  var tope  = _topeMaximoDisputa;  // ⭐ Usa la variable global

  if (monto <= 0) {
    inputMonto.style.borderColor = '#e2e8f0';
    if (hint) hint.textContent = '';
    return false;
  }

  if (tope > 0 && monto > tope) {
    inputMonto.style.borderColor = '#ef4444';
    if (hint) hint.innerHTML =
      '<span style="color:#ef4444;">⚠️ El monto supera el tope ($' +
      tope.toLocaleString('es-CO', { minimumFractionDigits: 2 }) + ')</span>';
    return false;
  }

  inputMonto.style.borderColor = '#16a34a';
  if (hint) hint.innerHTML =
    '<span style="color:#16a34a;">✓ Monto válido</span>';
  return true;
}


function actualizarResumenKPIs() {
  const checks = document.querySelectorAll('.kpi-disputa-check:checked');
  let totalKPIs = 0;
  let nombresKPIs = [];

  checks.forEach(ch => {
    totalKPIs += parseFloat(ch.getAttribute('data-bono')) || 0;
    nombresKPIs.push(ch.getAttribute('data-kpi'));
  });

  // Actualizar display KPIs
  document.getElementById('modal-bono-kpis-display').textContent =
    '$' + totalKPIs.toLocaleString('es-CO', { minimumFractionDigits: 2 });

  // Mostrar resumen
  const resumen    = document.getElementById('modal-kpis-resumen');
  const resumenTxt = document.getElementById('modal-kpis-resumen-texto');

  if (checks.length > 0) {
    resumen.style.display    = 'block';
    resumenTxt.innerHTML     =
      `<strong>${checks.length} KPI(s) seleccionado(s)</strong> · Bono: <strong style="color:#dc2626;">$${totalKPIs.toLocaleString('es-CO', { minimumFractionDigits: 2 })}</strong>`;
  } else {
    resumen.style.display = 'none';
  }

  // Sugerir monto automáticamente si el campo está vacío
  const inputMonto = document.getElementById('modal-disputa-monto');
  if (!inputMonto.value || inputMonto.value === '0') {
    inputMonto.value = totalKPIs > 0 ? totalKPIs.toFixed(2) : '';
    validarMontoDisputa();
  }
}

function toggleDisputeComment(elementOrBmsId, nombre, total) {
  // Soportar ambos modos de llamada:
  // Modo A (desde tabla): toggleDisputeComment(checkboxElement)
  // Modo B (directo):     toggleDisputeComment(bmsId, nombre, total)
  let bmsId, _nombre, _total;

  if (elementOrBmsId && typeof elementOrBmsId === 'object' && elementOrBmsId.tagName) {
    // Modo A: recibió el elemento checkbox
    const el = elementOrBmsId;
    bmsId   = el.getAttribute('data-bms');
    _nombre = el.getAttribute('data-nombre');
    _total  = parseFloat(el.getAttribute('data-total')) || 0;

    // Si el usuario DESMARCA el checkbox, limpiar disputa pendiente
    if (!el.checked) {
      if (disputasPendientes && disputasPendientes[bmsId]) {
        delete disputasPendientes[bmsId];
        actualizarBannerDisputas();
        // Quitar badge visual si existe
        var badge = document.getElementById('disputa-badge-' + bmsId);
        if (badge) badge.style.display = 'none';
      }
      return; // No abrir modal al desmarcar
    }
  } else {
    // Modo B: recibió parámetros directos
    bmsId   = elementOrBmsId;
    _nombre = nombre;
    _total  = parseFloat(total) || 0;
  }

  if (!bmsId) {
    console.error('toggleDisputeComment: bmsId no encontrado');
    return;
  }

  _disputaBmsActual = bmsId;

  // Llenar cabecera del modal
  var elBms    = document.getElementById('modal-disputa-bms');
  var elNombre = document.getElementById('modal-disputa-nombre');
  var elBono   = document.getElementById('modal-bono-total-display');
  var elComent = document.getElementById('modal-disputa-comentario');
  var elCont   = document.getElementById('modal-disputa-contador');
  var elMonto  = document.getElementById('modal-disputa-monto');
  var elResumen= document.getElementById('modal-kpis-resumen');

  if (!elBms || !elNombre || !elBono) {
    console.error('toggleDisputeComment: elementos del modal no encontrados. ¿Está el modal en el DOM?');
    alert('Error: El modal de disputa no está disponible. Recarga la página.');
    return;
  }

  elBms.textContent    = 'BMS: ' + bmsId;
  elNombre.textContent = _nombre || '—';
  elBono.textContent   = '$' + _total.toLocaleString('es-CO', { minimumFractionDigits: 2 });

  // Restaurar valores previos si ya existe una disputa para este agente
  var previo = (disputasPendientes && disputasPendientes[bmsId]) || {};
  if (elComent) {
    elComent.value = previo.comentario || '';
    if (elCont) elCont.textContent = (previo.comentario || '').length;
  }
  if (elMonto) {
    elMonto.value = previo.montoDisputa || '';
  }
  if (elResumen) {
    elResumen.style.display = 'none';
  }

  // Cargar KPIs del agente en el modal
  cargarKPIsEnModal(bmsId, _total, previo.kpisSeleccionados || []);

  // Mostrar modal
  var modal = document.getElementById('modal-disputa');
  if (!modal) {
    console.error('toggleDisputeComment: #modal-disputa no encontrado en el DOM');
    alert('Error: modal no encontrado. Recarga la página.');
    return;
  }
  modal.style.display = 'flex';

  // Cerrar al hacer clic fuera
  modal.onclick = function(e) {
    if (e.target === modal) cerrarModalDisputa();
  };
}
function generarCeldaDisputar(bmsId, nombre, totalFinal) {
  var nombreSafe = (nombre || '').replace(/"/g, '&quot;');
  var html = '<td style="text-align:center;padding:6px;">';

  // Si ya tiene disputa pendiente, mostrar badge directo
  if (disputasPendientes && disputasPendientes[bmsId]) {
    var d = disputasPendientes[bmsId];
    html += '<div style="display:flex;flex-direction:column;align-items:center;gap:4px;">';
    html += '<span id="disputa-badge-' + bmsId + '" style="background:#dc2626;color:#fff;padding:3px 10px;border-radius:8px;font-size:0.7rem;font-weight:700;cursor:pointer;" onclick="toggleDisputeComment(this.closest(\'tr\').querySelector(\'.dispute-check\'))">';
    html += '🚨 $' + (d.montoDisputa || 0).toLocaleString('es-CO', { minimumFractionDigits: 2 });
    html += '</span>';
    html += '</div>';
  } else {
    html += '<input type="checkbox" class="dispute-check"';
    html += ' data-bms="' + bmsId + '"';
    html += ' data-nombre="' + nombreSafe + '"';
    html += ' data-total="' + totalFinal + '"';
    html += ' onchange="toggleDisputeComment(this)"';
    html += ' style="width:20px;height:20px;cursor:pointer;accent-color:#dc2626;">';
    // Badge oculto por defecto, se mostrará cuando se guarde la disputa
    html += '<span id="disputa-badge-' + bmsId + '" style="display:none;background:#dc2626;color:#fff;padding:3px 8px;border-radius:8px;font-size:0.68rem;font-weight:700;margin-top:4px;display:none;cursor:pointer;"></span>';
  }

  html += '</td>';
  return html;
}

function abrirModalDisputa() {
    if (!disputaActual) return;
    
    const modal = document.getElementById('modal-disputa');
    const nombreDiv = document.getElementById('modal-disputa-nombre');
    const bmsDiv = document.getElementById('modal-disputa-bms');
    const comentario = document.getElementById('modal-disputa-comentario');
    const contador = document.getElementById('modal-disputa-contador');
    
    // Rellenar datos
    nombreDiv.textContent = disputaActual.nombre;
    bmsDiv.textContent = 'BMS: ' + disputaActual.bmsId + ' | Bono: $' + disputaActual.total.toLocaleString('es-CO', {minimumFractionDigits: 2});
    
    // Si no tiene comentario previo, limpiar
    if (!disputasPendientes[disputaActual.bmsId]) {
        comentario.value = '';
        contador.textContent = '0';
    } else {
        contador.textContent = comentario.value.length;
    }
    
    // Contador de caracteres
    comentario.oninput = function() {
        contador.textContent = this.value.length;
        
        if (this.value.length >= 10) {
            this.style.borderColor = '#27ae60';
        } else if (this.value.length > 0) {
            this.style.borderColor = '#f39c12';
        } else {
            this.style.borderColor = '#dfe6e9';
        }
    };
    
    // Mostrar modal
    modal.style.display = 'flex';
    
    // Focus en textarea
    setTimeout(() => comentario.focus(), 100);
}

      kpi:           chk.getAttribute('data-kpi'),
      categoria:     chk.getAttribute('data-categoria'),

      valorObtenido: chk.getAttribute('data-valor')

  // Validación 3

    console.warn('❌ FALLO: monto <= 0');
    mostrarAlerta('⚠️ Ingresa un monto válido mayor a $0', 'warning');
    return;
  }
  console.log('✅ Validación monto > 0 OK');

  // Validación 4 — validarMontoDisputa()

    console.warn('   tope usado:', typeof _topeMaximoDisputa !== 'undefined' ? _topeMaximoDisputa : '???');
    mostrarAlerta('⚠️ El monto supera el tope permitido', 'warning');
    return;

  // Calcular bono total KPIs
  var bonoKPIsSelec = kpisSelec.reduce(function(acc, k) {

function enviarDisputas() {
    const disputasArray = Object.values(disputasPendientes);
    
    if (disputasArray.length === 0) {
        mostrarAlerta('⚠️ No hay disputas pendientes para enviar', 'warning');
        return;
    }
    
    // Confirmación
    let mensaje = `📨 ENVIAR DISPUTAS\n\n`;
    mensaje += `Vas a enviar ${disputasArray.length} disputa(s):\n\n`;
    
    disputasArray.forEach((d, i) => {
        mensaje += `${i+1}. ${d.nombre} (BMS ${d.bmsId})\n`;
        mensaje += `   "${d.comentario.substring(0, 50)}${d.comentario.length > 50 ? '...' : ''}"\n\n`;
    });
    
    mensaje += `¿Confirmas el envío?`;
    
    if (!confirm(mensaje)) return;
    
    console.log('📤 Enviando disputas:', disputasArray);
    mostrarLoading('Enviando ' + disputasArray.length + ' disputa(s)...');
    
    google.script.run
        .withSuccessHandler(function(response) {
            ocultarLoading();
            
            if (response.success) {
                console.log('✅ Disputas enviadas exitosamente');
                
                // ⭐ GUARDAR EN CACHE CON IDENTIFICADOR ÚNICO
                const cacheKey = `disputas_${selectLob.value}_${selectFecha.value}`;
                const disputasParaCache = disputasArray.map(d => ({
                    bmsId: d.bmsId,
                    nombre: d.nombre,
                    total: d.total,
                    comentario: d.comentario,
                    lob: d.lob,
                    fecha: d.fecha,
                    fechaDisputa: new Date().toISOString(),
                    usuarioDisputa: document.getElementById('nombreUsuario').textContent
                }));
                
                // Obtener disputas existentes en cache
                let todasLasDisputas = [];
                try {
                    const cacheExistente = localStorage.getItem('disputasEnviadas');
                    if (cacheExistente) {
                        todasLasDisputas = JSON.parse(cacheExistente);
                    }
                } catch (e) {
                    console.warn('⚠️ Error al leer cache existente:', e);
                    todasLasDisputas = [];
                }
                
                // Agregar nuevas disputas
                todasLasDisputas.push(...disputasParaCache);
                
                // Guardar en localStorage
                localStorage.setItem('disputasEnviadas', JSON.stringify(todasLasDisputas));
                localStorage.setItem(cacheKey, JSON.stringify(disputasParaCache));
                
                console.log('💾 Disputas guardadas en cache:', disputasParaCache.length);
                console.log('📦 Cache key:', cacheKey);
                
                mostrarAlerta('✅ ' + disputasArray.length + ' disputa(s) enviada(s) exitosamente. Recargando tabla...', 'success');
                
                // ⭐ LIMPIAR PENDIENTES
                disputasPendientes = {};
                
                // ⭐ RECARGAR SOLO LA TABLA (no toda la página)
                setTimeout(() => {
                    actualizarSoloTabla();
                }, 1500);
                
            } else {
                mostrarAlerta('❌ Error: ' + response.error, 'danger');
            }
        })
        .withFailureHandler(function(error) {
            ocultarLoading();
            mostrarAlerta('❌ Error: ' + error.message, 'danger');
            console.error('Error completo:', error);
        })
        .guardarDisputas(disputasArray);
}

// ⭐ 8. CARGAR DISPUTAS DESDE CACHE AL INICIAR
function cargarDisputasDesdeCache() {
    const cacheData = localStorage.getItem('disputasEnviadas');
    
    if (!cacheData) {
        console.log('📭 No hay disputas en cache');
        return;
    }
    
    try {
        disputasEnviadas = JSON.parse(cacheData);
        console.log('📦 Disputas cargadas desde cache:', disputasEnviadas.length);
        
        const lobActual = selectLob.value;
        const fechaActual = selectFecha.value;
        
        // Marcar las que corresponden a esta LOB y fecha
        disputasEnviadas.forEach(disputa => {
            if (disputa.lob === lobActual && disputa.fecha === fechaActual) {
                setTimeout(() => marcarComoDisputado(disputa.bmsId), 500);
            }
        });
        
    } catch (e) {
        console.error('❌ Error al parsear cache:', e);
        localStorage.removeItem('disputasEnviadas');
    }
}
