    <script>
        // Variables globales
        let resultadosSimulacion = null;

        // Inicializar al cargar la página
        $(document).ready(function() {
            cargarOMs();
            cargarMeses();
        });

        // Función para cargar lista de OMs
        function cargarOMs() {
            google.script.run
                .withSuccessHandler(function(resultado) {
                    if (resultado.error) {
                        mostrarError('Error al cargar OMs: ' + resultado.error);
                        return;
                    }
                    
                    const selectOM = $('#select-om');
                    selectOM.empty();
                    
                    if (resultado.oms && resultado.oms.length > 0) {
                        resultado.oms.forEach(function(om) {
                            selectOM.append(`<option value="${om}">${om}</option>`);
                        });
                    } else {
                        selectOM.append('<option value="">No hay OMs disponibles</option>');
                    }
                    
                    // Cargar LOBs cuando se seleccione OM
                    selectOM.on('change', function() {
                        cargarLOBsOM($(this).val());
                    });
                })
                .withFailureHandler(function(error) {
                    mostrarError('Error al cargar OMs: ' + error);
                })
                .OBTENER_LISTA_OMS();
        }

        // Función para cargar LOBs de un OM específico
        function cargarLOBsOM(nombreOM) {
            if (!nombreOM) return;
            
            google.script.run
                .withSuccessHandler(function(resultado) {
                    if (resultado.error) {
                        mostrarError('Error al cargar LOBs: ' + resultado.error);
                        return;
                    }
                    
                    const selectLOBs = $('#select-lobs');
                    selectLOBs.empty();
                    selectLOBs.append('<option value="all">Todas las LOBs</option>');
                    
                    if (resultado.lobs && resultado.lobs.length > 0) {
                        resultado.lobs.forEach(function(lob) {
                            selectLOBs.append(`<option value="${lob}">${lob}</option>`);
                        });
                    } else {
                        selectLOBs.append('<option value="">No hay LOBs disponibles</option>');
                    }
                })
                .withFailureHandler(function(error) {
                    mostrarError('Error al cargar LOBs: ' + error);
                })
                .OBTENER_LOBS_OM(nombreOM);
        }

        // Función para cargar meses disponibles
        function cargarMeses() {
            google.script.run
                .withSuccessHandler(function(resultado) {
                    if (resultado.error) {
                        mostrarError('Error al cargar meses: ' + resultado.error);
                        return;
                    }
                    
                    const selectMes = $('#select-mes');
                    selectMes.empty();
                    
                    if (resultado.meses && resultado.meses.length > 0) {
                        resultado.meses.forEach(function(mes) {
                            selectMes.append(`<option value="${mes.valor}">${mes.nombre}</option>`);
                        });
                        
                        // Seleccionar el mes actual por defecto
                        const mesActual = new Date().getMonth() + 1;
                        selectMes.val(mesActual);
                    } else {
                        selectMes.append('<option value="">No hay meses disponibles</option>');
                    }
                })
                .withFailureHandler(function(error) {
                    mostrarError('Error al cargar meses: ' + error);
                })
                .OBTENER_MESES_DISPONIBLES();
        }

        // Función para ejecutar simulación
        function ejecutarSimulacion() {
            const om = $('#select-om').val();
            const mes = $('#select-mes').val();
            const lobs = $('#select-lobs').val() || ['all'];
            
            if (!om || !mes) {
                mostrarError('Por favor, selecciona OM y Mes');
                return;
            }
            
            // Mostrar loading
            $('#loading').show();
            $('#results').hide();
            
            // Obtener configuración PVE
            const presupuestoPVE = parseFloat($('#pve-presupuesto').val()) || 0;
            const porcentaje3m = parseFloat($('#pve-porcentaje3m').val()) || 0;
            const porcentajeMM = parseFloat($('#pve-porcentajeMM').val()) || 0;
            
            const params = {
                om: om,
                mes: parseInt(mes),
                lobs: lobs,
                presupuestoPVE: presupuestoPVE,
                porcentaje3m: porcentaje3m,
                porcentajeMM: porcentajeMM
            };
            
            google.script.run
                .withSuccessHandler(function(resultado) {
                    $('#loading').hide();
                    
                    if (resultado.error) {
                        mostrarError('Error en simulación: ' + resultado.error);
                        return;
                    }
                    
                    if (!resultado.success) {
                        mostrarError('Error en simulación: ' + (resultado.error || 'Error desconocido'));
                        return;
                    }
                    
                    resultadosSimulacion = resultado;
                    mostrarResultados(resultado);
                })
                .withFailureHandler(function(error) {
                    $('#loading').hide();
                    mostrarError('Error en simulación: ' + error);
                })
                .SIMULAR_BONOS_CON_PVE(params);
        }

        // Función para mostrar resultados de la simulación
        function mostrarResultados(resultado) {
            const resultsDiv = $('#results');
            resultsDiv.empty().show();
            
            // Header con información básica
            let html = `
                <div class="summary-cards">
                    <div class="card">
                        <h3>OM</h3>
                        <div class="value">${resultado.om}</div>
                        <div class="subtext">Mes: ${resultado.mes}</div>
                    </div>
                    <div class="card">
                        <h3>Total General</h3>
                        <div class="value">${formatCurrency(resultado.totalGeneral)}</div>
                        <div class="subtext">Bonos totales</div>
                    </div>
                    <div class="card">
                        <h3>Total Bonos</h3>
                        <div class="value positive">${formatCurrency(resultado.totalBono)}</div>
                        <div class="subtext">Bonos positivos</div>
                    </div>
                    <div class="card">
                        <h3>Total Penalizaciones</h3>
                        <div class="value negative">${formatCurrency(resultado.totalAdd)}</div>
                        <div class="subtext">Bonos negativos</div>
                    </div>
                    <div class="card">
                        <h3>Total Agentes</h3>
                        <div class="value">${resultado.totalAgentes}</div>
                        <div class="subtext">Agentes procesados</div>
                    </div>
                    <div class="card">
                        <h3>LOBs Procesadas</h3>
                        <div class="value">${resultado.lobs.length}</div>
                        <div class="subtext">LOBs analizadas</div>
                    </div>
                </div>
            `;
            
            // Si hay configuración PVE, mostrar resumen PVE
            if (resultado.esPVE) {
                const porcentajeUsado = resultado.porcentajeUsado || 
                    (resultado.totalGeneral / resultado.presupuestoPVE * 100).toFixed(1);
                
                html += `
                    <div class="pve-summary">
                        <h3>📈 Análisis PVE</h3>
                        <div class="pve-grid">
                            <div class="pve-item">
                                <div class="label">Presupuesto PVE</div>
                                <div class="value">${formatCurrency(resultado.presupuestoPVE)}</div>
                            </div>
                            <div class="pve-item">
                                <div class="label">Total Requerido</div>
                                <div class="value">${formatCurrency(resultado.totalGeneral)}</div>
                                <div style="font-size: 14px; margin-top: 5px;">${porcentajeUsado}% del presupuesto</div>
                            </div>
                            <div class="pve-item">
                                <div class="label">Límite 3 meses</div>
                                <div class="value">${formatCurrency(resultado.limite3m || 0)}</div>
                                <div style="font-size: 14px; margin-top: 5px;">${resultado.porcentaje3m || 0}% del presupuesto</div>
                            </div>
                            <div class="pve-item">
                                <div class="label">Límite Mes Actual</div>
                                <div class="value">${formatCurrency(resultado.limiteMM || 0)}</div>
                                <div style="font-size: 14px; margin-top: 5px;">${resultado.porcentajeMM || 0}% del presupuesto</div>
                            </div>
                        </div>
                        
                        ${resultado.totalGeneral > resultado.presupuestoPVE ? 
                            `<div class="alert alert-danger" style="margin-top: 20px;">
                                ⚠️ <strong>SOBREPASA PRESUPUESTO:</strong> El total requerido (${formatCurrency(resultado.totalGeneral)}) 
                                supera el presupuesto PVE (${formatCurrency(resultado.presupuestoPVE)})
                                <br>Factor de ajuste aplicado: ${(resultado.factorAjuste * 100).toFixed(1)}%
                            </div>` : 
                            `<div class="alert alert-success" style="margin-top: 20px;">
                                ✅ <strong>DENTRO DEL PRESUPUESTO:</strong> El total requerido (${formatCurrency(resultado.totalGeneral)}) 
                                está dentro del presupuesto PVE (${formatCurrency(resultado.presupuestoPVE)})
                                <br>Se utilizó el ${porcentajeUsado}% del presupuesto disponible
                            </div>`}
                    </div>
                `;
            }
            
            // Tabla por LOB
            if (resultado.lobs && resultado.lobs.length > 0) {
                html += `
                    <div class="chart-container">
                        <h3 class="chart-title">📊 Distribución por LOB</h3>
                        <canvas id="chart-lobs" height="100"></canvas>
                    </div>
                    
                    <div class="table-container">
                        <h3 class="chart-title">📋 Detalle por LOB</h3>
                        <table class="table ${resultado.esPVE ? 'pve-table' : ''}">
                            <thead>
                                <tr>
                                    <th>LOB</th>
                                    <th>Agentes</th>
                                    <th>Salarios Totales</th>
                                    <th>Total Bonos</th>
                                    <th>Total Penalizaciones</th>
                                    <th>Total LOB</th>
                                    ${resultado.esPVE ? '<th>Ajuste PVE</th><th>Total Ajustado</th>' : ''}
                                    <th>Acciones</th>
                                </tr>
                            </thead>
                            <tbody>
                `;
                
                let totalAgentes = 0;
                let totalSalarios = 0;
                let totalBonos = 0;
                let totalPenalizaciones = 0;
                let totalGeneral = 0;
                let totalAjustado = 0;
                
                resultado.lobs.forEach((lob, index) => {
                    totalAgentes += lob.hc;
                    totalSalarios += lob.totalSalarios;
                    totalBonos += lob.totalBono;
                    totalPenalizaciones += Math.abs(lob.totalAdd);
                    totalGeneral += lob.total;
                    
                    // Calcular ajuste PVE si aplica
                    const ajustePVE = resultado.esPVE ? (lob.total * (resultado.factorAjuste || 1)) : lob.total;
                    totalAjustado += ajustePVE;
                    
                    html += `
                        <tr>
                            <td><strong>${lob.lob}</strong></td>
                            <td>${lob.hc}</td>
                            <td>${formatCurrency(lob.totalSalarios)}</td>
                            <td class="positive">${formatCurrency(lob.totalBono)}</td>
                            <td class="negative">${formatCurrency(Math.abs(lob.totalAdd))}</td>
                            <td><strong>${formatCurrency(lob.total)}</strong></td>
                            ${resultado.esPVE ? `
                                <td>${resultado.factorAjuste ? (resultado.factorAjuste * 100).toFixed(1) + '%' : '100%'}</td>
                                <td><strong class="${ajustePVE < lob.total ? 'negative' : 'positive'}">
                                    ${formatCurrency(ajustePVE)}
                                </strong></td>
                            ` : ''}
                            <td>
                                <button class="btn btn-primary" onclick="verDetalleLOB(${index})" style="padding: 8px 12px; font-size: 12px;">
                                    👁️ Ver Detalle
                                </button>
                            </td>
                        </tr>
                    `;
                });
                
                // Fila de totales
                html += `
                        <tr style="background: #f8f9fa; font-weight: 600;">
                            <td><strong>TOTALES</strong></td>
                            <td><strong>${totalAgentes}</strong></td>
                            <td><strong>${formatCurrency(totalSalarios)}</strong></td>
                            <td class="positive"><strong>${formatCurrency(totalBonos)}</strong></td>
                            <td class="negative"><strong>${formatCurrency(totalPenalizaciones)}</strong></td>
                            <td><strong>${formatCurrency(totalGeneral)}</strong></td>
                            ${resultado.esPVE ? `
                                <td></td>
                                <td><strong>${formatCurrency(totalAjustado)}</strong></td>
                            ` : ''}
                            <td></td>
                        </tr>
                    </tbody>
                </table>
            </div>
                `;
                
                // Crear gráfico
                setTimeout(() => {
                    crearGraficoLOB(resultado.lobs);
                }, 100);
            }
            
            // Si hay errores, mostrarlos
            if (resultado.errores && resultado.errores.length > 0) {
                html += `
                    <div class="alert alert-danger">
                        ⚠️ <strong>Se encontraron algunos errores:</strong><br>
                        ${resultado.errores.join('<br>')}
                    </div>
                `;
            }
            
            resultsDiv.html(html);
        }

        // Función para crear gráfico de distribución por LOB
        function crearGraficoLOB(lobs) {
            const ctx = document.getElementById('chart-lobs').getContext('2d');
            const labels = lobs.map(lob => lob.lob);
            const data = lobs.map(lob => lob.total);
            const colores = generarColores(lobs.length);
            
            new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Total por LOB ($)',
                        data: data,
                        backgroundColor: colores,
                        borderColor: colores.map(color => color.replace('0.7', '1')),
                        borderWidth: 1,
                        borderRadius: 5
                    }]
                },
                options: {
                    responsive: true,
                    plugins: {
                        legend: {
                            display: false
                        },
                        tooltip: {
                            callbacks: {
                                label: function(context) {
                                    return `Total: ${formatCurrency(context.raw)}`;
                                }
                            }
                        }
                    },
                    scales: {
                        y: {
                            beginAtZero: true,
                            ticks: {
                                callback: function(value) {
                                    return '$' + value.toLocaleString();
                                }
                            }
                        }
                    }
                }
            });
        }

        // Función para ver detalle de una LOB
        function verDetalleLOB(index) {
            if (!resultadosSimulacion || !resultadosSimulacion.lobs || !resultadosSimulacion.lobs[index]) {
                mostrarError('No hay datos disponibles para esta LOB');
                return;
            }
            
            const lob = resultadosSimulacion.lobs[index];
            let html = `
                <h3>Detalle LOB: ${lob.lob}</h3>
                <div class="summary-cards" style="margin-bottom: 20px;">
                    <div class="card">
                        <h3>Agentes</h3>
                        <div class="value">${lob.hc}</div>
                    </div>
                    <div class="card">
                        <h3>Salarios Totales</h3>
                        <div class="value">${formatCurrency(lob.totalSalarios)}</div>
                    </div>
                    <div class="card">
                        <h3>Total Bonos</h3>
                        <div class="value positive">${formatCurrency(lob.totalBono)}</div>
                    </div>
                    <div class="card">
                        <h3>Total Penalizaciones</h3>
                        <div class="value negative">${formatCurrency(Math.abs(lob.totalAdd))}</div>
                    </div>
                    <div class="card">
                        <h3>Total LOB</h3>
                        <div class="value">${formatCurrency(lob.total)}</div>
                    </div>
                </div>
            `;
            
            // Configuración de la LOB
            if (lob.config) {
                html += `
                    <div class="alert alert-success">
                        <strong>Configuración LOB:</strong><br>
                        Salario Base: ${formatCurrency(lob.config.salarioBase)}<br>
                        % Bono Global: ${(lob.config.porcentajeBono * 100).toFixed(1)}%
                    </div>
                `;
            }
            
            // Agentes
            if (lob.agentes && lob.agentes.length > 0) {
                html += `
                    <h4>Agentes (${lob.agentes.length})</h4>
                    <div class="table-container">
                        <table class="table">
                            <thead>
                                <tr>
                                    <th>BMS ID</th>
                                    <th>Nombre</th>
                                    <th>Salario</th>
                                    <th>KPIs Cumplidos</th>
                                    <th>Total Bono</th>
                                </tr>
                            </thead>
                            <tbody>
                `;
                
                lob.agentes.forEach(agente => {
                    html += `
                        <tr>
                            <td>${agente.bmsId}</td>
                            <td>${agente.nombre}</td>
                            <td>${formatCurrency(agente.salario)}</td>
                            <td>${agente.kpisCumplidos}/${agente.kpisEvaluados}</td>
                            <td class="${agente.totalBono >= 0 ? 'positive' : 'negative'}">${formatCurrency(agente.totalBono)}</td>
                        </tr>
                    `;
                });
                
                html += `
                            </tbody>
                        </table>
                    </div>
                `;
            }
            
            // Mostrar en modal o en nueva sección
            alert(html.replace(/<[^>]*>/g, '').substring(0, 2000) + '...');
        }

        // Función para mostrar error
        function mostrarError(mensaje) {
            const errorHTML = `
                <div class="alert alert-danger">
                    ❌ <strong>Error:</strong> ${mensaje}
                </div>
            `;
            
            $('#results').html(errorHTML).show();
        }

        // Función para formatear moneda
        function formatCurrency(amount) {
            if (typeof amount !== 'number') amount = parseFloat(amount) || 0;
            return '$' + amount.toLocaleString('es-CO', {
                minimumFractionDigits: 0,
                maximumFractionDigits: 0
            });
        }

        // Función para generar colores
        function generarColores(cantidad) {
            const colores = [];
            const hueStep = 360 / cantidad;
            
            for (let i = 0; i < cantidad; i++) {
                const hue = i * hueStep;
                colores.push(`hsla(${hue}, 70%, 60%, 0.7)`);
            }
            
            return colores;
        }
    </script>
