/**
 * Punto de entrada de la Web App
 */
function doGet() {
    var payload = null;

    try {
        var folder;
        try {
            folder = DriveApp.getFolderById(CACHE_FOLDER_ID);
        } catch (e) {
            folder = DriveApp.getRootFolder();
        }

        var files = folder.getFilesByName(CACHE_FILENAME);
        if (files.hasNext()) {
            var file = files.next();
            var jsonContent = file.getBlob().getDataAsString();
            payload = JSON.parse(jsonContent);
            payload.source = "DRIVE CACHE";
            
            // ⭐ LOG DE VERIFICACIÓN
            Logger.log("✅ Caché cargado desde Drive");
            Logger.log("📊 Métricas del caché: FT=" + payload.metrics.ft + " | Cases=" + payload.metrics.cases + " | CR=" + payload.metrics.cr);
            Logger.log("📦 Raw Data: " + (payload.rawData ? payload.rawData.length : 0) + " filas");
        }
    } catch (e) {
        Logger.log("⚠️ Error leyendo caché: " + e.toString());
    }

    // ⭐ SI NO HAY CACHÉ O ESTÁ VACÍO, CARGAR EN VIVO
    if (!payload || !payload.rawData || payload.rawData.length === 0) {
        Logger.log("📥 Caché vacío o inexistente. Cargando datos en vivo...");
        payload = obtenerDashboardEnVivo();
        
        if (!payload || !payload.rawData || payload.rawData.length === 0) {
            Logger.log("❌ ERROR CRÍTICO: No hay datos ni en caché ni en vivo");
            return HtmlService.createHtmlOutput(
                '<html><body><h2 style="color:red;text-align:center;">Error: No se pudieron cargar los datos</h2>' +
                '<p style="text-align:center;">Por favor, ejecuta <b>actualizarCacheDrive()</b> en el editor de Apps Script.</p></body></html>'
            );
        }
    }

    try {
        var template = HtmlService.createTemplateFromFile('Index');
        template.payloadJson = JSON.stringify(payload);
        template.userEmail = Session.getActiveUser().getEmail();
        
        return template.evaluate()
            .setTitle('Dashboard de Métricas')
            .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    } catch (e) {
        Logger.log("❌ Error renderizando template: " + e.toString());
        return HtmlService.createHtmlOutput('Error renderizado: ' + e.toString());
    }
}

function obtenerDashboardEnVivo() {
    try {
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        var sheet = ss.getSheetByName('Data');
        if (!sheet) {
            Logger.log("❌ ERROR: No se encontró la hoja 'Data'");
            return null;
        }

        var fullData = sheet.getDataRange().getValues();
        Logger.log("✅ fullData obtenido: " + fullData.length + " filas totales");
        
        var dataRows = fullData.slice(1);
        Logger.log("✅ dataRows (sin headers): " + dataRows.length + " filas");
        
        if (dataRows.length === 0) {
            Logger.log("⚠️ ADVERTENCIA: La hoja 'Data' no tiene filas de datos");
            return null;
        }
        
        var headers = fullData[0];

        // ⭐ FT ESTÁ EN COLUMNA AL (índice 37) - FIJO
        var idxFT = 37;

        // Búsqueda robusta para AJ/Stage
        var idxAJ = -1;
        for (var i = headers.length - 1; i >= 0; i--) {
            var h = headers[i].toString().trim().toUpperCase();
            if (h === "STAGE" || h === "AJ") {
                idxAJ = i;
                break;
            }
        }
        if (idxAJ === -1) idxAJ = 35;

        Logger.log("LIVE | FT (AL): " + idxFT + " | Stage (AJ): " + idxAJ + " (" + headers[idxAJ] + ")");
        
        var filters = {};
        ['U', 'I', 'AJ', 'N', 'V', 'W', 'J', 'S', 'AM'].forEach(function (colName) {
            var colIdx = (colName === 'AJ') ? idxAJ : (colName === 'AM' ? 38 : obtenerIndiceFijo(colName));
            filters[colName] = obtenerValoresUnicosArray(dataRows, colIdx, colName);
        });

        Logger.log("🔄 Llamando a procesarFilas con " + dataRows.length + " filas...");
        var processed = procesarFilas(dataRows, idxFT, idxAJ);

        return {
            filters: filters,
            rawData: processed.rawData,
            metrics: processed.metrics,
            timestamp: new Date().toString(),
            source: "LIVE RAW (OPTIMIZED)"
        };
    } catch (e) {
        Logger.log("❌ Error en obtenerDashboardEnVivo: " + e.toString());
        Logger.log("Stack: " + e.stack);
        return null;
    }
}

function procesarFilas(dataRows, idxFT, idxAJ) {
    // ⭐ VALIDACIÓN CRÍTICA
    if (!dataRows || !Array.isArray(dataRows) || dataRows.length === 0) {
        Logger.log("⚠️ ERROR: dataRows está vacío o no es un array");
        Logger.log("dataRows recibido: " + JSON.stringify(dataRows));
        return {
            rawData: [],
            metrics: {
                ft: 0,
                cases: 0,
                ct: 0,
                cr: 0
            }
        };
    }

    Logger.log("✅ Procesando " + dataRows.length + " filas...");

    var totalFT = 0, totalCases = 0, sumCT = 0, countCT = 0;

    var rawData = dataRows.map(function (row) {
        // ⭐ FT ESTÁ EN COLUMNA AL (índice 37)
        var ftVal = row[idxFT];
        var ftNumeric = 0;
        if (typeof ftVal === 'number') {
            ftNumeric = ftVal;
        } else if (typeof ftVal === 'string') {
            var clean = ftVal.trim().toLowerCase();
            if (clean === 'si' || clean === 'yes' || clean === 'true') {
                ftNumeric = 1;
            } else {
                var parsed = parseFloat(clean);
                if (!isNaN(parsed)) ftNumeric = parsed;
            }
        } else if (ftVal === true) {
            ftNumeric = 1;
        }

        var isFT = ftNumeric > 0;
        if (isFT) totalFT += ftNumeric;
        totalCases++;

        // ⭐ CT está en columna AJ (índice 34)
        var ctVal = parseFloat(row[34]); 
        if (!isNaN(ctVal)) { sumCT += ctVal; countCT++; }

        var monthVal = row[20];
        var monthStr = '';
        if (monthVal !== null && monthVal !== undefined) {
            if (typeof monthVal === 'number') {
                monthStr = Math.floor(monthVal).toString();
            } else {
                monthStr = String(monthVal).trim();
            }
        }

        // ⭐ YEAR está en columna AM (índice 38)
        var yearVal = row[38];
        var yearStr = '';
        if (yearVal !== null && yearVal !== undefined) {
            if (typeof yearVal === 'number') {
                yearStr = Math.floor(yearVal).toString();
            } else {
                yearStr = String(yearVal).trim();
            }
        }

        return [
            row[0],  // 0: CASE
            formatearFecha(row[6]),  // 1: DATE
            String(row[8] || '').trim(),  // 2: COUNTRY
            String(row[9] || '').trim(),  // 3: OWNER (J)
            String(row[idxAJ] || '').trim(), // 4: STAGE (AJ)
            String(row[13] || '').trim(), // 5: STATUS
            String(row[18] || '').trim(), // 6: WEEK
            monthStr, // 7: MONTH (U)
            String(row[21] || '').trim(), // 8: SUP
            String(row[22] || '').trim(), // 9: AGENT
            formatearFecha(row[26]), // 10: TRIP_DATE
            ftNumeric, // 11: FT (AL - 37)
            row[32], // 12: TRIP_HOUR
            row[33], // 13: CR_PLACEHOLDER (Ya no se usa)
            row[34], // 14: CT (AJ - 34)
            yearStr  // 15: YEAR (AM - 38)
        ];
    });

    // ⭐ CALCULAR CR = (FT / TOTAL_CASES) * 100
    var crCalculado = totalCases > 0 ? (totalFT / totalCases) * 100 : 0;

    // ⭐ LOGS DETALLADOS DEL CÁLCULO
    Logger.log("========================================");
    Logger.log("📊 CÁLCULO DE CR (Conversion Rate)");
    Logger.log("========================================");
    Logger.log("✅ Total FT Cases: " + totalFT);
    Logger.log("✅ Total Cases: " + totalCases);
    Logger.log("📐 Fórmula CR: (FT / CASES) * 100");
    Logger.log("📐 Cálculo: (" + totalFT + " / " + totalCases + ") * 100");
    Logger.log("📐 Resultado: " + (totalFT / totalCases) + " * 100");
    Logger.log("🎯 CR FINAL: " + crCalculado.toFixed(2) + "%");
    Logger.log("========================================");

    return {
        rawData: rawData,
        metrics: {
            ft: totalFT,
            cases: totalCases,
            ct: countCT > 0 ? sumCT / countCT : 0,
            cr: crCalculado  // ✅ YA ESTÁ EN PORCENTAJE (0-100)
        }
    };
}

function obtenerIndiceFijo(colName) {
    var map = { 'I': 8, 'J': 9, 'N': 13, 'S': 18, 'U': 20, 'V': 21, 'W': 22 };

}

/**
 * ⭐ FUNCIÓN DE ACTUALIZACIÓN CON VALIDACIÓN DE USUARIO
 */
function actualizarCacheDrive() {
    try {
        // ⭐ VALIDAR USUARIO AUTORIZADO
        var userEmail = Session.getActiveUser().getEmail();
        Logger.log("Usuario intentando actualizar: " + userEmail);
        
        if (USUARIOS_AUTORIZADOS.indexOf(userEmail) === -1) {
            throw new Error("🚫 NO ERES DIGNO. Solo usuarios autorizados pueden actualizar.");
        }

        Logger.log("🚀 Iniciando actualización optimizada por usuario autorizado: " + userEmail);
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        var sheet = ss.getSheetByName('Data');
        if (!sheet) throw new Error("❌ Hoja 'Data' no encontrada");

        var fullData = sheet.getDataRange().getValues();
        var dataRows = fullData.slice(1);
        if (dataRows.length === 0) return "⚠️ La hoja 'Data' está vacía.";
        var headers = fullData[0];

        // ⭐ FT ESTÁ EN COLUMNA AL (índice 37) - FIJO
        var idxFT = 37;

        var idxAJ = -1;
        for (var j = headers.length - 1; j >= 0; j--) {
            var hj = headers[j].toString().trim().toUpperCase();
            if (hj === "STAGE" || hj === "AJ") {
                idxAJ = j;
                break;
            }
        }
        if (idxAJ === -1) idxAJ = 35;

        Logger.log("CACHE | FT (AL): " + idxFT + " | Stage (AJ): " + idxAJ + " (" + headers[idxAJ] + ")");
        
        // ⭐ INCLUIR 'AM' (Year) EN FILTROS
        var filtrosInit = { 'U': new Set(), 'I': new Set(), 'AJ': new Set(), 'N': new Set(), 'V': new Set(), 'W': new Set(), 'J': new Set(), 'S': new Set(), 'AM': new Set() };

        dataRows.forEach(function (row) {
            ['U', 'I', 'AJ', 'N', 'V', 'W', 'J', 'S', 'AM'].forEach(function (key) {
                var idx = (key === 'AJ') ? idxAJ : (key === 'AM' ? 38 : obtenerIndiceFijo(key));
                var val = row[idx];
                if (val instanceof Date) val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
                var valStr = '';
                if (val !== null && val !== undefined) {
                    if ((key === 'U' || key === 'AM') && typeof val === 'number') valStr = Math.floor(val).toString();
                    else valStr = val.toString().trim();
                }
                if (valStr !== '') filtrosInit[key].add(valStr);
            });
        });

        var processed = procesarFilas(dataRows, idxFT, idxAJ);

        for (var k in filtrosInit) {
            var arr = Array.from(filtrosInit[k]);
            // ⭐ Ordenar descendente para U (Meses) y AM (Years)
            if (k === 'U' || k === 'AM') {
                arr.sort(function (a, b) { return b.localeCompare(a, undefined, { numeric: true }); });
            } else {
                arr.sort();
            }
            filtrosInit[k] = arr;
        }

        var payload = {
            filters: filtrosInit,
            rawData: processed.rawData,
            metrics: processed.metrics,
            timestamp: new Date().toString(),
            source: "DRIVE CACHE (v3 - FT en AL)"
        };

        var jsonString = JSON.stringify(payload);
        var folder;
        try { folder = DriveApp.getFolderById(CACHE_FOLDER_ID); } catch (e) { folder = DriveApp.getRootFolder(); }

        var files = folder.getFilesByName(CACHE_FILENAME);
        if (files.hasNext()) {
            files.next().setContent(jsonString);
        } else {
            folder.createFile(CACHE_FILENAME, jsonString, MimeType.PLAIN_TEXT);
        }

        return "✅ Actualización exitosa: " + processed.rawData.length + " filas procesadas. FT desde columna AL.";
    } catch (e) {
        throw new Error("❌ Error en actualización: " + e.toString());
    }
}

}

function obtenerValoresUnicosArray(data, idx, colName) {
    if (idx === -1) return [];
    var set = new Set();
    data.forEach(function (r) {
        var v = r[idx];
        if (v instanceof Date) v = Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd');
        if (v && v.toString().trim() !== '') set.add(v.toString().trim());
    });
    var arr = Array.from(set);
    // ⭐ Ordenar descendente para U (Meses) y AM (Years)
    if (colName === 'U' || colName === 'AM') {
        arr.sort(function (a, b) { return b.localeCompare(a, undefined, { numeric: true }); });
    } else {
        arr.sort();
    }
    return arr;
}

function formatearFecha(date) {
    if (date instanceof Date) return Utilities.formatDate(date, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    return date;
}

function guardarFechaEnSheet(fecha) {
    try {
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        var sheet = ss.getSheetByName('FechaActualizacion') || ss.insertSheet('FechaActualizacion');
        sheet.getRange('A1:A2').setValues([['Última Actualización'], [fecha]]);
        return true;
    } catch (e) { return false; }
}

function obtenerFechaDeSheet() {
    try {
        var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('FechaActualizacion');
        return sheet ? sheet.getRange('A2').getValue().toString() : '';
    } catch (e) { return ''; }
}

/**
 * Obtiene un mensaje motivacional aleatorio según el quartil
 * @param {string} quartil - 'Q1', 'Q2', 'Q3', o 'Q4'
 * @return {string} Mensaje motivacional
 */
function obtenerMensajeMotivacional(quartil) {
    try {
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        var sheetMSG = ss.getSheetByName('MSG');
        
        if (!sheetMSG) {
            Logger.log("❌ ERROR: No se encontró la hoja 'MSG'");
            return "¡Sigue adelante! 💪";
        }
        
        var columna = '';
        
        // Determinar qué columna usar según el quartil
        if (quartil === 'Q1' || quartil === 'Q2') {
            // Mejores performers: Columna D
            columna = 'D';
        } else if (quartil === 'Q3' || quartil === 'Q4') {
            // Necesitan motivación: Columna A
            columna = 'A';
        } else {
            columna = 'A'; // Default
        }
        
        // Obtener todos los mensajes de la columna
        var mensajes = sheetMSG.getRange(columna + '1:' + columna + '200').getValues();
        
        // Filtrar mensajes vacíos
        var mensajesFiltrados = [];
        for (var i = 0; i < mensajes.length; i++) {
            var msg = mensajes[i][0];
            if (msg && msg.toString().trim() !== '') {
                mensajesFiltrados.push(msg.toString().trim());
            }
        }
        
        if (mensajesFiltrados.length === 0) {
            return "¡Excelente trabajo! 🌟";
        }
        
        // Seleccionar un mensaje aleatorio
        var indiceAleatorio = Math.floor(Math.random() * mensajesFiltrados.length);
        var mensajeSeleccionado = mensajesFiltrados[indiceAleatorio];
        
        Logger.log("✅ Mensaje seleccionado (" + quartil + ", Columna " + columna + "): " + mensajeSeleccionado);
        
        return mensajeSeleccionado;
        
    } catch (e) {
        Logger.log("❌ Error obteniendo mensaje motivacional: " + e.toString());
        return "¡Sigue así! 💪";
    }
}
