var CACHE_FILENAME = "dashboard_cache_metrics_v2.json";
var CACHE_FOLDER_ID = "1SrPxDlsXxd-ohwiYh1QSaQx8R60kdji3";

/**
 * Punto de entrada de la Web App
 */
function doGet() {
    var payload = null;

    try {
        var folder;
        try {
            folder = DriveApp.getFolderById(CACHE_FOLDER_ID);
        } catch (e) {
            folder = DriveApp.getRootFolder();
        }

        var files = folder.getFilesByName(CACHE_FILENAME);
        if (files.hasNext()) {
            var file = files.next();
            var jsonContent = file.getBlob().getDataAsString();
            payload = JSON.parse(jsonContent);
            payload.source = "DRIVE CACHE";
        }
    } catch (e) {
        Logger.log("Error leyendo caché: " + e.toString());
    }

    if (!payload || !payload.rawData || payload.rawData.length === 0) {
        payload = obtenerDashboardEnVivo();
        if (!payload || !payload.rawData || payload.rawData.length === 0) {
            return HtmlService.createHtmlOutput(
                '<html><body><h2 style="color:red;text-align:center;">Error: No se pudieron cargar los datos</h2>' +
                '<p style="text-align:center;">Por favor, ejecuta <b>actualizarCacheDrive()</b> en el editor de Apps Script.</p></body></html>'
            );
        }
    }

    try {
        var template = HtmlService.createTemplateFromFile('Index');
        template.payloadJson = JSON.stringify(payload);
        return template.evaluate()
            .setTitle('Dashboard de Métricas')
            .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    } catch (e) {
        return HtmlService.createHtmlOutput('Error renderizado: ' + e.toString());
    }
}

function obtenerDashboardEnVivo() {
    try {
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        var sheet = ss.getSheetByName('Data');
        if (!sheet) return null;

        var fullData = sheet.getDataRange().getValues();
        var dataRows = fullData.slice(1);
        var headers = fullData[0];

        // Función auxiliar para búsqueda dinámica
        var findIdx = function (name) {
            var idx = headers.indexOf(name);
            if (idx === -1) {
                for (var i = 0; i < headers.length; i++) {
                    if (headers[i].toString().trim().toUpperCase() === name.toUpperCase()) return i;
                }
            }
            return idx;
        };

        var idxFT = findIdx("FTS"); if (idxFT === -1) idxFT = findIdx("FT"); if (idxFT === -1) idxFT = 37;

        // Búsqueda robusta para AJ: Priorizar búsqueda desde el final para evitar columna K (10)
        var idxAJ = -1;
        for (var i = headers.length - 1; i >= 0; i--) {
            var h = headers[i].toString().trim().toUpperCase();
            if (h === "STAGE" || h === "AJ") {
                idxAJ = i;
                break;
            }
        }
        if (idxAJ === -1) idxAJ = 35; // Fallback a AJ real

        Logger.log("LIVE | FT: " + idxFT + " (" + headers[idxFT] + ") | AJ/Stage: " + idxAJ + " (" + headers[idxAJ] + ")");
        var filters = {};
        ['U', 'I', 'AJ', 'N', 'V', 'W', 'J', 'S'].forEach(function (colName) {
            var colIdx = (colName === 'AJ') ? idxAJ : obtenerIndiceFijo(colName);
            filters[colName] = obtenerValoresUnicosArray(dataRows, colIdx, colName); // Pass colName for sorting
        });

        var processed = procesarFilas(dataRows, idxFT, idxAJ);

        return {
            filters: filters,
            rawData: processed.rawData,
            metrics: processed.metrics,
            timestamp: new Date().toString(),
            source: "LIVE RAW (OPTIMIZED)"
        };
    } catch (e) {
        Logger.log("Error en obtenerDashboardEnVivo: " + e.toString());
        return null;
    }
}

function procesarFilas(dataRows, idxFT, idxAJ) {
    var totalFT = 0, totalCases = 0, sumCT = 0, countCT = 0, sumCR = 0, countCR = 0;

    var rawData = dataRows.map(function (row) {
        var ftVal = row[idxFT];
        var ftNumeric = 0;
        if (typeof ftVal === 'number') {
            ftNumeric = ftVal;
        } else if (typeof ftVal === 'string') {
            var clean = ftVal.trim().toLowerCase();
            if (clean === 'si' || clean === 'yes' || clean === 'true') {
                ftNumeric = 1;
            } else {
                var parsed = parseFloat(clean);
                if (!isNaN(parsed)) ftNumeric = parsed;
            }
        } else if (ftVal === true) {
            ftNumeric = 1;
        }

        var isFT = ftNumeric > 0;
        if (isFT) totalFT += ftNumeric;
        totalCases++;

        var ctVal = parseFloat(row[34]); if (!isNaN(ctVal)) { sumCT += ctVal; countCT++; }
        var crVal = parseFloat(row[33]); if (!isNaN(crVal)) { sumCR += crVal; countCR++; }

        var monthVal = row[20];
        var monthStr = '';
        if (monthVal !== null && monthVal !== undefined) {
            if (typeof monthVal === 'number') {
                monthStr = Math.floor(monthVal).toString(); // ⭐ Forzar ENTERO para meses
            } else {
                monthStr = String(monthVal).trim();
            }
        }

        return [
            row[0],  // CASE
            formatearFecha(row[6]),  // DATE
            String(row[8] || '').trim(),  // COUNTRY
            String(row[9] || '').trim(),  // OWNER (J)
            String(row[idxAJ] || '').trim(), // STAGE (AJ)
            String(row[13] || '').trim(), // STATUS
            String(row[18] || '').trim(), // WEEK
            monthStr, // ⭐ MONTH (U) - ENTERO
            String(row[21] || '').trim(), // SUP
            String(row[22] || '').trim(), // AGENT
            formatearFecha(row[26]), // TRIP_DATE
            ftNumeric, // FT
            row[32], // TRIP_HOUR
            row[33], // CR
            row[34]  // CT
        ];
    });

    return {
        rawData: rawData,
        metrics: {
            ft: totalFT,
            cases: totalCases,
            ct: countCT > 0 ? sumCT / countCT : 0,
            cr: countCR > 0 ? sumCR / countCR : 0
        }
    };
}

function obtenerIndiceFijo(colName) {
    var map = { 'I': 8, 'J': 9, 'N': 13, 'S': 18, 'U': 20, 'V': 21, 'W': 22 };
    return map[colName] || -1;
}

function actualizarCacheDrive() {
    try {
        Logger.log("🚀 Iniciando actualización optimizada...");
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        var sheet = ss.getSheetByName('Data');
        if (!sheet) throw new Error("❌ Hoja 'Data' no encontrada");

        var fullData = sheet.getDataRange().getValues();
        var dataRows = fullData.slice(1);
        if (dataRows.length === 0) return "⚠️ La hoja 'Data' está vacía.";
        var headers = fullData[0];

        var findIdx = function (name) {
            var idx = headers.indexOf(name);
            if (idx === -1) {
                for (var i = 0; i < headers.length; i++) {
                    if (headers[i].toString().trim().toUpperCase() === name.toUpperCase()) return i;
                }
            }
            return idx;
        };

        var idxFT = findIdx("FTS"); if (idxFT === -1) idxFT = findIdx("FT"); if (idxFT === -1) idxFT = 37;

        // Búsqueda robusta para AJ: Priorizar búsqueda desde el final para evitar columna K (10)
        var idxAJ = -1;
        for (var j = headers.length - 1; j >= 0; j--) {
            var hj = headers[j].toString().trim().toUpperCase();
            if (hj === "STAGE" || hj === "AJ") {
                idxAJ = j;
                break;
            }
        }
        if (idxAJ === -1) idxAJ = 35; // Fallback a AJ real

        Logger.log("CACHE | FT: " + idxFT + " (" + headers[idxFT] + ") | AJ/Stage: " + idxAJ + " (" + headers[idxAJ] + ")");
        var filtrosInit = { 'U': new Set(), 'I': new Set(), 'AJ': new Set(), 'N': new Set(), 'V': new Set(), 'W': new Set(), 'J': new Set(), 'S': new Set() };

        // Cargar filtros
        dataRows.forEach(function (row) {
            ['U', 'I', 'AJ', 'N', 'V', 'W', 'J', 'S'].forEach(function (key) {
                var idx = (key === 'AJ') ? idxAJ : obtenerIndiceFijo(key);
                var val = row[idx];
                if (val instanceof Date) val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
                var valStr = '';
                if (val !== null && val !== undefined) {
                    if (key === 'U' && typeof val === 'number') valStr = Math.floor(val).toString(); // ⭐ MESES ENTEROS
                    else valStr = val.toString().trim();
                }
                if (valStr !== '') filtrosInit[key].add(valStr);
            });
        });

        var processed = procesarFilas(dataRows, idxFT, idxAJ);

        for (var k in filtrosInit) {
            var arr = Array.from(filtrosInit[k]);
            if (k === 'U') {
                arr.sort(function (a, b) { return b.localeCompare(a, undefined, { numeric: true }); });
            } else {
                arr.sort();
            }
            filtrosInit[k] = arr;
        }

        var payload = {
            filters: filtrosInit,
            rawData: processed.rawData,
            metrics: processed.metrics,
            timestamp: new Date().toString(),
            source: "DRIVE CACHE (SUM FT)"
        };

        var jsonString = JSON.stringify(payload);
        var folder;
        try { folder = DriveApp.getFolderById(CACHE_FOLDER_ID); } catch (e) { folder = DriveApp.getRootFolder(); }

        var files = folder.getFilesByName(CACHE_FILENAME);
        if (files.hasNext()) {
            files.next().setContent(jsonString);
        } else {
            folder.createFile(CACHE_FILENAME, jsonString, MimeType.PLAIN_TEXT);
        }

        return "✅ Actualización exitosa: " + processed.rawData.length + " filas. FT Sumado.";
    } catch (e) {
        throw new Error("❌ Error en actualización: " + e.toString());
    }
}

function obtenerValoresUnicosArray(data, idx, colName) {
    if (idx === -1) return [];
    var set = new Set();
    data.forEach(function (r) {
        var v = r[idx];
        if (v instanceof Date) v = Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd');
        if (v && v.toString().trim() !== '') set.add(v.toString().trim());
    });
    var arr = Array.from(set);
    if (colName === 'U') {
        arr.sort(function (a, b) { return b.localeCompare(a, undefined, { numeric: true }); });
    } else {
        arr.sort();
    }
    return arr;
}

function formatearFecha(date) {
    if (date instanceof Date) return Utilities.formatDate(date, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    return date;
}

function guardarFechaEnSheet(fecha) {
    try {
        var ss = SpreadsheetApp.getActiveSpreadsheet();
        var sheet = ss.getSheetByName('FechaActualizacion') || ss.insertSheet('FechaActualizacion');
        sheet.getRange('A1:A2').setValues([['Última Actualización'], [fecha]]);
        return true;
    } catch (e) { return false; }
}

function obtenerFechaDeSheet() {
    try {
        var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('FechaActualizacion');
        return sheet ? sheet.getRange('A2').getValue().toString() : '';
    } catch (e) { return ''; }
}
