        function calcularMetricasPorAgente(rows) {
            var agentMap = {};

            rows.forEach(function (row) {
                var agent = row[COLS.OWNER]; // CAMBIADO A OWNER
                if (!agent) return;

                if (!agentMap[agent]) {
                    agentMap[agent] = {
                        agent: agent,
                        supervisor: row[COLS.SUP] || 'N/A',
                        ft: 0,
                        cases: 0,
                        ctSum: 0,
                        ctCount: 0,
                        crSum: 0,
                        crCount: 0
                    };
                }

                var a = agentMap[agent];
                a.cases++;

                // ⭐ SUMA ROBUSTA DE FT (Columna AL - 37)
                var ftVal = parseFloat(row[COLS.FT]);
                if (!isNaN(ftVal)) {
                    a.ft += ftVal;
                }

                var ct = parseNumberRobust(row[COLS.CT]);
                if (ct !== null) {
                    a.ctSum += ct;
                    a.ctCount++;
                }

                var cr = parseNumberRobust(row[COLS.CR]);
                if (cr !== null) {
                    a.crSum += cr;
                    a.crCount++;
                }
            });

            // Calcular promedios y clasificar
            var agentes = [];
            for (var key in agentMap) {
                var a = agentMap[key];
                agentes.push({
                    agent: a.agent,
                    supervisor: a.supervisor,
                    ft: a.ft,
                    cases: a.cases,
                    ct: a.ctCount > 0 ? a.ctSum / a.ctCount : 0,
                    cr: a.crCount > 0 ? a.crSum / a.crCount : 0
                });
            }

            // Clasificar por quartiles para cada métrica
            return clasificarPorQuartiles(agentes);
        }

        function clasificarPorQuartiles(agentes) {
            if (agentes.length < 4) return { agentes: agentes, stats: null };

            // CORRECCIÓN CRÍTICA: Para FT, mayor es mejor
            var ftValues = agentes.map(a => a.ft).sort((a, b) => b - a); // DESCENDENTE
            var ctValues = agentes.map(a => a.ct).sort((a, b) => a - b);
            var crValues = agentes.map(a => a.cr).sort((a, b) => a - b);

            var stats = {
                ft: calculateStats(ftValues, true), // true = higher is better
                ct: calculateStats(ctValues, false),
                cr: calculateStats(crValues, false)
            };

            agentes.forEach(function (agent) {
                // FT: Mayor = Q1, Menor = Q4
                agent.ftQuartile = getQuartileLabel(agent.ft, stats.ft, true);
                agent.ftOutlier = getOutlierStatus(agent.ft, stats.ft, true);

                // CT & CR: Menor = Q1 (mejor), Mayor = Q4 (peor)
                agent.ctQuartile = getQuartileLabel(agent.ct, stats.ct, false);
                agent.ctOutlier = getOutlierStatus(agent.ct, stats.ct, false);
                agent.crQuartile = getQuartileLabel(agent.cr, stats.cr, false);
                agent.crOutlier = getOutlierStatus(agent.cr, stats.cr, false);
            });

            return { agentes: agentes, stats: stats };
        }

        function getQuartileLabel(value, stats, higherIsBetter) {
            if (higherIsBetter) {
                // Para FT: Mayor = Q1 (mejor)
                if (value >= stats.q3) return 'Q1';
                if (value >= stats.median) return 'Q2';
                if (value >= stats.q1) return 'Q3';
                return 'Q4';
            } else {
                // Para CT/CR: Menor = Q1 (mejor)
                if (value <= stats.q1) return 'Q1';
                if (value <= stats.median) return 'Q2';
                if (value <= stats.q3) return 'Q3';
                return 'Q4';
            }
        }

        function getOutlierStatus(value, stats, higherIsBetter) {
            if (value > stats.upperFence) {
                return higherIsBetter ? 'HIGH_GOOD' : 'HIGH';
            }
            if (value < stats.lowerFence) {
                return higherIsBetter ? 'LOW' : 'LOW_GOOD';
            }
            return 'NONE';
        }

        function procesarDatos(rows) {
            console.time('  📊 Procesamiento');

            var resultado = {

        function calcularKPIs(rows) {
            var ft = 0, cases = 0, sumCT = 0, countCT = 0, sumCR = 0, countCR = 0;

            for (var i = 0; i < rows.length; i++) {
                var row = rows[i];
                cases++;

                var ftVal = parseFloat(row[COLS.FT]);
                if (!isNaN(ftVal)) {
                    ft += ftVal;
                }

                var ct = parseNumberRobust(row[COLS.CT]);
                if (ct !== null) {
                    sumCT += ct;
                    countCT++;
                }

                var cr = parseNumberRobust(row[COLS.CR]);
                if (cr !== null) {
                    sumCR += cr;
                    countCR++;
                }
            }

            return {
                ft: ft,
                cases: cases,
                ct: countCT ? sumCT / countCT : 0,
                cr: countCR ? sumCR / countCR : 0
            };
        }

        function agruparDatos(rows, colIdx) {
            var grupos = {};

            for (var i = 0; i < rows.length; i++) {
                var row = rows[i];
                var key = row[colIdx] || 'N/A';

                if (!grupos[key]) {
                    grupos[key] = {
                        label: key,
                        ft: 0,
                        cases: 0,
                        ctSum: 0,
                        ctCount: 0,
                        crSum: 0,
                        crCount: 0
                    };
                }

                var g = grupos[key];
                g.cases++;

                var ftVal = parseFloat(row[COLS.FT]);
                if (!isNaN(ftVal)) {
                    g.ft += ftVal;
                }

                var ct = parseNumberRobust(row[COLS.CT]);
                if (ct !== null) {
                    g.ctSum += ct;
                    g.ctCount++;
                }

                var cr = parseNumberRobust(row[COLS.CR]);
                if (cr !== null) {
                    g.crSum += cr;
                    g.crCount++;
                }
            }

            var resultado = [];
            for (var key in grupos) {
                var g = grupos[key];
                resultado.push({
                    label: g.label,
                    ft: g.ft,
                    cases: g.cases,
                    ct: g.ctCount ? g.ctSum / g.ctCount : 0,
                    cr: g.crCount ? g.crSum / g.crCount : 0
                });
            }

            resultado.sort(function (a, b) {
                return a.label > b.label ? 1 : -1;
            });

            return resultado;
        }

        function generarDesgloses(rows) {

        function analizarMejoresTiempos(rows) {
            var dias = {}, horas = {}, diaHora = {}, paises = {};
            var diasSemana = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];

            for (var i = 0; i < rows.length; i++) {
                var row = rows[i];
                var ftVal = parseFloat(row[COLS.FT]);
                var isFT = !isNaN(ftVal) && ftVal > 0;

                // Distribución por País (Columna I/COUNTRY)
                var pais = row[COLS.COUNTRY] || 'Desconocido';
                if (!paises[pais]) paises[pais] = { label: pais, count: 0, ft: 0, ctSum: 0, ctCount: 0, crSum: 0, crCount: 0 };
                var p = paises[pais];
                p.count++;
                if (!isNaN(ftVal)) p.ft += ftVal;

                var ct = parseNumberRobust(row[COLS.CT]);
                if (ct !== null) { p.ctSum += ct; p.ctCount++; }
                var cr = parseNumberRobust(row[COLS.CR]);
                if (cr !== null) { p.crSum += cr; p.crCount++; }

                var fechaStr = row[COLS.TRIP_DATE];
                if (!fechaStr) continue;

                var parts = fechaStr.split('-');
                if (parts.length < 3) continue;

                var dateObj = new Date(parts[0], parts[1] - 1, parts[2]);
                var diaNombre = diasSemana[dateObj.getDay()];

                if (!dias[diaNombre]) dias[diaNombre] = { label: diaNombre, count: 0, ft: 0, ctSum: 0, ctCount: 0, crSum: 0, crCount: 0 };
                var d = dias[diaNombre];
                d.count++;
                if (!isNaN(ftVal)) d.ft += ftVal;
                if (ct !== null) { d.ctSum += ct; d.ctCount++; }
                if (cr !== null) { d.crSum += cr; d.crCount++; }

                var hora = row[COLS.TRIP_HOUR];
                var horaNum = -1;

                if (typeof hora === 'number') {
                    if (hora < 1) horaNum = Math.floor(hora * 24);
                    else horaNum = Math.floor(hora);
                } else if (typeof hora === 'string' && hora.includes(':')) {
                    horaNum = parseInt(hora.split(':')[0]);
                }

                if (horaNum >= 0 && horaNum <= 23) {
                    var horaStr = (horaNum < 10 ? '0' + horaNum : horaNum) + ':00';
                    if (!horas[horaStr]) horas[horaStr] = { label: horaStr, count: 0, ft: 0, ctSum: 0, ctCount: 0, crSum: 0, crCount: 0 };
                    var h = horas[horaStr];
                    h.count++;
                    if (!isNaN(ftVal)) h.ft += ftVal;
                    if (ct !== null) { h.ctSum += ct; h.ctCount++; }
                    if (cr !== null) { h.crSum += cr; h.crCount++; }

                    var dh = diaNombre + ' ' + horaStr;
                    if (!isNaN(ftVal)) diaHora[dh] = (diaHora[dh] || 0) + ftVal;
                }
            }

            function getMaxFT(obj) {
                var maxKey = '-', maxVal = 0;
                for (var k in obj) {
                    if (obj[k].ft > maxVal) {
                        maxVal = obj[k].ft;
                        maxKey = k;
                    }
                }
                return { label: maxKey, count: maxVal };
            }

            function toArrayExtended(obj) {
                var arr = [];
                for (var k in obj) {
                    var item = obj[k];
                    arr.push({
                        label: item.label,
                        count: item.count,
                        ft: item.ft,
                        ct: item.ctCount > 0 ? item.ctSum / item.ctCount : 0,
                        cr: item.crCount > 0 ? item.crSum / item.crCount : 0
                    });
                }
                arr.sort(function (a, b) { return b.ft - a.ft; });
                return arr;
            }

            return {
                bestDay: getMaxFT(dias),
                bestHour: getMaxFT(horas),
                bestTime: { label: getMaxKey(diaHora), count: diaHora[getMaxKey(diaHora)] || 0 },
                allDays: toArrayExtended(dias),
                allHours: toArrayExtended(horas),
                allCountries: toArrayExtended(paises)
            };
        }

        function getMaxKey(obj) {
            var maxKey = '-', maxVal = 0;
            for (var k in obj) {
                if (obj[k] > maxVal) { maxVal = obj[k]; maxKey = k; }
            }
            return maxKey;
        }

                return aplicarFiltroFila(r, filtros);
            });

                return;

                container.innerHTML = '<p style="text-align:center;padding:30px;">Insuficientes agentes para análisis de quartiles.</p>';
                return;

                // Peso por FT Quartile (Q4 = 4, Q1 = 1)

                    (b.ctOutlier !== 'NONE' ? 1 : 0) +

        function actualizarComparacion() {
            var weekBaseEl = document.getElementById('weekBase');
            var weekCompareEl = document.getElementById('weekCompare');

            if (!weekBaseEl || !weekCompareEl) return;

            var weekBase = weekBaseEl.value;
            var weekCompare = weekCompareEl.value;

            var container = document.getElementById('comparison-results');

            if (!weekBase || !weekCompare || weekBase === weekCompare) {
                container.innerHTML = '<p style="text-align:center;color:#999;padding:20px;">ℹ️ Por favor selecciona dos semanas diferentes para comparar.</p>';
                return;
            }

            var dataBase = historicalData[weekBase];
            var dataCompare = historicalData[weekCompare];

            if (!dataBase || !dataCompare) return;

            // Crear mapas para búsqueda rápida
            var agentMapBase = {};
            dataBase.agentes.forEach(function (a) { agentMapBase[a.agent] = a; });

            var agentMapCompare = {};
            dataCompare.agentes.forEach(function (a) { agentMapCompare[a.agent] = a; });

            // Encontrar agentes en ambas semanas
            var evoluciones = [];

            for (var agent in agentMapBase) {
                if (agentMapCompare[agent]) {
                    var base = agentMapBase[agent];
                    var compare = agentMapCompare[agent];
                    var qScoreBase = getQuartileScore(base.ftQuartile);
                    var qScoreCompare = getQuartileScore(compare.ftQuartile);

                    var evolution = {
                        agent: agent,
                        supervisor: base.supervisor,
                        ftBase: parseFloat(base.ft) || 0,
                        ftCompare: parseFloat(compare.ft) || 0,
                        ftDelta: (parseFloat(compare.ft) || 0) - (parseFloat(base.ft) || 0),
                        qBase: base.ftQuartile,
                        qCompare: compare.ftQuartile,
                        evolution: qScoreCompare - qScoreBase // + mejoró, - empeoró
                    };

                    evoluciones.push(evolution);
                }
            }

            // Ordenar: 1. Mayor mejora de quartil, 2. Mayor mejora de FT
            evoluciones.sort(function (a, b) {
                if (b.evolution !== a.evolution) return b.evolution - a.evolution;
                return b.ftDelta - a.ftDelta;
            });

            // Resumen Estadístico
            var mejoraron = evoluciones.filter(e => e.evolution > 0).length;
            var empeoraron = evoluciones.filter(e => e.evolution < 0).length;
            var iguales = evoluciones.filter(e => e.evolution === 0).length;

            var html = '<div class="evolution-summary-grid">';
            html += '<div class="evo-card evo-up"><div class="evo-val">+' + mejoraron + '</div><div class="evo-lbl">Mejoraron Quartil</div></div>';
            html += '<div class="evo-card evo-same"><div class="evo-val">' + iguales + '</div><div class="evo-lbl">Sin Cambio Q</div></div>';
            html += '<div class="evo-card evo-down"><div class="evo-val">' + empeoraron + '</div><div class="evo-lbl">Empeoraron Quartil</div></div>';
            html += '</div>';

            // Tabla
            html += '<div style="overflow-x:auto;max-height:500px;overflow-y:auto;margin-top:20px;">';
            html += '<table class="unified-table">';
            html += '<thead>';
            html += '<tr>';
            html += '<th rowspan="2">Owner</th>';
            html += '<th rowspan="2">Supervisor</th>';
            html += '<th colspan="2" style="background:#bdc3c7;color:#2c3e50;">' + weekBase + ' (Base)</th>';
            html += '<th colspan="2" style="background:#34495e;color:white;">' + weekCompare + ' (Objetivo)</th>';
            html += '<th rowspan="2">Δ Evolución</th>';
            html += '<th rowspan="2">Δ FT Cases</th>';
            html += '</tr>';
            html += '<tr>';
            html += '<th style="background:#bdc3c7;color:#2c3e50;">Q</th><th style="background:#bdc3c7;color:#2c3e50;">FT</th>';
            html += '<th style="background:#34495e;color:white;">Q</th><th style="background:#34495e;color:white;">FT</th>';
            html += '</tr>';
            html += '</thead><tbody>';

            evoluciones.forEach(function (e) {
                var rowStyle = '';
                var evoIcon = '➡️';
                var evoClass = '';
                var evoText = 'Igual';

                if (e.evolution > 0) {
                    evoIcon = '🚀';
                    rowStyle = 'background-color:rgba(46, 204, 113, 0.1);';
                    evoClass = 'evolution-up';
                    evoText = 'Mejoró +' + e.evolution;
                } else if (e.evolution < 0) {
                    evoIcon = '🔻';
                    rowStyle = 'background-color:rgba(231, 76, 60, 0.1);';
                    evoClass = 'evolution-down';
                    evoText = 'Bajó ' + e.evolution;
                }

                html += '<tr style="' + rowStyle + '">';
                html += '<td style="text-align:left;font-weight:bold;">' + e.agent + '</td>';
                html += '<td style="text-align:left;color:#7f8c8d;">' + e.supervisor + '</td>';

                // Base
                html += '<td>' + renderQuartileBadge(e.qBase) + '</td>';
                html += '<td style="color:#7f8c8d;">' + e.ftBase + '</td>';

                // Compare
                html += '<td>' + renderQuartileBadge(e.qCompare) + '</td>';
                html += '<td style="font-weight:bold;">' + e.ftCompare + '</td>';

                // Evolución
                html += '<td style="font-weight:bold;" class="' + evoClass + '">' + evoIcon + ' ' + evoText + '</td>';
                html += '<td style="font-weight:bold;color:' + (e.ftDelta >= 0 ? '#27ae60' : '#c0392b') + '">' + (e.ftDelta > 0 ? '+' : '') + e.ftDelta + '</td>';

        function getQuartileScore(qLabel) {
            // Q1 = 4 puntos (Mejor), Q4 = 1 punto (Peor)
            if (qLabel === 'Q1') return 4;
            if (qLabel === 'Q2') return 3;
            if (qLabel === 'Q3') return 2;
            if (qLabel === 'Q4') return 1;
            return 0;
        }

        function calcularEvolucion(qBase, qCompare) {
            var map = { 'Q1': 4, 'Q2': 3, 'Q3': 2, 'Q4': 1 };
            return map[qCompare] - map[qBase];
        }

        function parseNumberRobust(v) {
            if (v === null || v === undefined || v === '') return null;
            if (typeof v === 'number') return v;
            if (typeof v === 'string') {
                var clean = v.replace('%', '').replace(',', '').trim();
                var num = parseFloat(clean);
                if (!isNaN(num)) {
                    if (v.includes('%')) return num / 100;
                    return num;
                }
            }
            return null;
        }

        function aplicarFiltroFila(row, filtros) {
            if (filtros.week && row[COLS.WEEK] !== filtros.week) return false;
            // outreachType removido
            if (filtros.caseOwner && row[COLS.OWNER] !== filtros.caseOwner) return false;
            if (filtros.status && String(row[COLS.STATUS] || '').trim() !== filtros.status) return false;
            if (filtros.begin && String(row[COLS.MONTH] || '').trim() !== filtros.begin) return false;
            if (filtros.supervisor && String(row[COLS.SUP] || '').trim() !== filtros.supervisor) return false;

            // AJ es el nuevo Stage
            if (filtros.stageAJ && String(row[COLS.STAGE] || '').trim() !== filtros.stageAJ) return false;
            return true;
        }

        function calculateStats(values) {
            values.sort(function (a, b) { return a - b; });
            var n = values.length;

            var mean = values.reduce(function (a, b) { return a + b }, 0) / n;
            var q1 = getPercentile(values, 25);
            var median = getPercentile(values, 50);
            var q3 = getPercentile(values, 75);
            var iqr = q3 - q1;

            return {
                mean: mean, q1: q1, median: median, q3: q3, iqr: iqr,
                lowerFence: q1 - (1.5 * iqr),
                upperFence: q3 + (1.5 * iqr),
                min: values[0],
                max: values[n - 1]
            };
        }

        function getPercentile(sorted, p) {
            var idx = (p / 100) * (sorted.length - 1);
            var lower = Math.floor(idx);
            var upper = Math.ceil(idx);
            var weight = idx - lower;
            return sorted[lower] * (1 - weight) + sorted[upper] * weight;
        }

        function renderAnalysisHeatmap(ta) {
            var container = document.getElementById('analysis-heatmap');
            if (!container) return;

            var diasOrder = ['Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado', 'Domingo'];
            var matrix = {};

            // Re-procesar Matrix Day/Hour desde rawDataGlobal filtrada (Solo FT)
            var filtros = getFiltros();
            rawDataGlobal.forEach(function (row) {
                if (!aplicarFiltroFila(row, filtros)) return;

                // ⭐ SUMA ROBUSTA DE FT (Columna AL - 37)
                var ftVal = parseFloat(row[COLS.FT]);
                if (isNaN(ftVal) || ftVal <= 0) return;
                var isFT = true;

                var fechaStr = row[COLS.TRIP_DATE];
                if (!fechaStr) return;
                var parts = fechaStr.split('-');
                var dateObj = new Date(parts[0], parts[1] - 1, parts[2]);
                var dia = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'][dateObj.getDay()];

                var horaNum = -1;
                var h = row[COLS.TRIP_HOUR];
                if (typeof h === 'number') {
                    horaNum = h < 1 ? Math.floor(h * 24) : Math.floor(h);
                } else if (typeof h === 'string' && h.includes(':')) {
                    horaNum = parseInt(h.split(':')[0]);
                }

                if (horaNum >= 0 && horaNum <= 23) {
                    if (!matrix[dia]) matrix[dia] = {};
                    matrix[dia][horaNum] = (matrix[dia][horaNum] || 0) + ftVal; // SUMAR EN LUGAR DE CONTAR
                }
            });

            // Encontrar max para escala de colores
            var maxVal = 0;
            for (var d in matrix) {
                for (var h in matrix[d]) {
                    if (matrix[d][h] > maxVal) maxVal = matrix[d][h];
                }
            }

            var html = '<table class="heatmap-table"><thead><tr><th>Día / Hora</th>';
            for (var h = 0; h < 24; h++) html += '<th>' + (h < 10 ? '0' + h : h) + '</th>';
            html += '</tr></thead><tbody>';

            diasOrder.forEach(function (dia) {
                html += '<tr><td class="day-label">' + dia + '</td>';
                for (var h = 0; h < 24; h++) {
                    var val = (matrix[dia] && matrix[dia][h]) ? matrix[dia][h] : 0;
                    var pct = maxVal > 0 ? (val / maxVal) : 0;
                    var bg = pct > 0 ? 'rgba(118, 75, 162, ' + (0.1 + pct * 0.9) + ')' : 'transparent';
                    var color = pct > 0.5 ? 'white' : '#333';
                    html += '<td class="heatmap-cell" style="background:' + bg + '; color:' + color + '" title="' + dia + ' ' + h + ':00 - ' + val + ' FT Cases">' + (val > 0 ? val : '') + '</td>';
                }
                html += '</tr>';
            });

            html += '</tbody></table>';
            html += '<p style="text-align:right; font-size:0.8em; margin-top:5px; color:#666;">* Los colores representan la densidad de <strong>FT Cases</strong> (más oscuro = más volumen)</p>';
            container.innerHTML = html;
        }

        function renderMainView(type, data, container) {
            if (!data || data.length === 0) {
                container.innerHTML += '<p style="text-align:center;">No hay datos.</p>';
                return;
            }

            var details = document.createElement('details');
            details.open = true;

            var summary = document.createElement('summary');
            summary.textContent = '📈 Gráfico & Tabla';
            details.appendChild(summary);

            var content = document.createElement('div');
            content.className = 'details-content';

            var chartId = 'chart-' + type;
            content.innerHTML = '<div class="chart-wrapper"><canvas id="' + chartId + '"></canvas></div>';

            renderHorizontalTable('Principal', data, content);
            details.appendChild(content);
            container.appendChild(details);

            requestAnimationFrame(function () {
                requestAnimationFrame(function () {
                    renderComboChart(chartId, data);
                });
            });
        }

        function renderBreakdownAccordion(bd, container) {
            var details = document.createElement('details');
            var summary = document.createElement('summary');
            summary.textContent = '📋 Desglose';
            details.appendChild(summary);

            var content = document.createElement('div');
            content.className = 'details-content';

            var breakdowns = [
                { key: 'I', title: 'Country (I)' },
                { key: 'AJ', title: 'Stage (AJ)' },
                { key: 'N', title: 'Status (N)' }
            ];
            for (var i = 0; i < breakdowns.length; i++) {
                var item = breakdowns[i];
                var div = document.createElement('div');
                div.style.marginBottom = '25px';
                renderHorizontalTable('Columna ' + item.key + ' - ' + item.title, bd[item.key], div);
                content.appendChild(div);
            }

            details.appendChild(content);
            container.appendChild(details);
        }

        function renderHorizontalTable(title, data, container) {
            if (!data || data.length === 0) {
                container.innerHTML = '<p>No hay datos para: ' + title + '</p>';
                return;
            }

            var h4 = document.createElement('h4');
            h4.textContent = title;
            h4.style.color = '#667eea';
            h4.style.marginBottom = '8px';
            container.appendChild(h4);

            var div = document.createElement('div');
            div.className = 'table-container';

            var html = '<table><thead><tr><th>Metric</th>';
            for (var i = 0; i < data.length; i++) {
                html += '<th>' + data[i].label + '</th>';
            }
            html += '</tr></thead><tbody>';

            var metrics = [
                { label: 'FT Cases', key: 'ft', format: function (v) { return v; } },
                { label: 'Total Cases', key: 'cases', format: function (v) { return v; } },
                { label: 'CT %', key: 'ct', format: function (v) { return (v * 100).toFixed(2) + '%'; } },
                { label: 'CR %', key: 'cr', format: function (v) { return (v * 100).toFixed(2) + '%'; } }
            ];

            for (var m = 0; m < metrics.length; m++) {
                var metric = metrics[m];
                html += '<tr><td><strong>' + metric.label + '</strong></td>';
                for (var i = 0; i < data.length; i++) {
                    html += '<td>' + metric.format(data[i][metric.key]) + '</td>';
                }
                html += '</tr>';
            }

            html += '</tbody></table>';
            div.innerHTML = html;
            container.appendChild(div);
        }

        function renderComboChart(id, data) {
            var canvas = document.getElementById(id);
            if (!canvas) return;

            var ctx = canvas.getContext('2d');

            if (charts[id]) {
                charts[id].destroy();
            }

            var labels = [];
            var ftData = [];
            var casesData = [];
            var ctData = [];
            var crData = [];

            for (var i = 0; i < data.length; i++) {
                labels.push(data[i].label);
                ftData.push(data[i].ft);
                casesData.push(data[i].cases);
                ctData.push(data[i].ct * 100);
                crData.push(data[i].cr * 100);
            }

            charts[id] = new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: labels,
                    datasets: [
                        {
                            label: 'FT Cases',
                            data: ftData,
                            backgroundColor: 'rgba(102,126,234,0.7)',
                            yAxisID: 'y'
                        },
                        {
                            label: 'Total Cases',
                            data: casesData,
                            backgroundColor: 'rgba(118,75,162,0.7)',
                            yAxisID: 'y'
                        },
                        {
                            type: 'line',
                            label: 'CT %',
                            data: ctData,
                            borderColor: '#ff9f43',
                            borderWidth: 3,
                            yAxisID: 'y1',
                            fill: false
                        },
                        {
                            type: 'line',
                            label: 'CR %',
                            data: crData,
                            borderColor: '#eb3b5a',
                            borderWidth: 3,
                            yAxisID: 'y1',
                            fill: false
                        }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        y: {
                            type: 'linear',
                            position: 'left',
                            title: { display: true, text: 'Cases' }
                        },
                        y1: {
                            type: 'linear',
                            position: 'right',
                            title: { display: true, text: '%' },
                            grid: { drawOnChartArea: false },
                            ticks: {
                                callback: function (value) {
                                    return value + '%';
                                }
                            }
                        }
                    }
                }
            });
        }

        function cambiarTab(type) {
            var tabs = document.querySelectorAll('.tab');
            for (var i = 0; i < tabs.length; i++) {
                tabs[i].classList.remove('active');
            }

            var contents = document.querySelectorAll('.tab-content');
            for (var i = 0; i < contents.length; i++) {
                contents[i].classList.remove('active');
            }

            var btn = document.querySelector("button[onclick=\"cambiarTab('" + type + "')\"]");
            if (btn) btn.classList.add('active');

            var content = document.getElementById(type);
            if (content) content.classList.add('active');

            renderCurrentTab();
        }
