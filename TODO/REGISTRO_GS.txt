function REGISTRO_obtenerLobsPorOM(nombreOM) {
    const cacheKey = `lobs_om_v3_hc_${normalizarTexto(nombreOM)}`;
    return getCached(cacheKey, () => {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaHC = ss.getSheetByName('H_C');
        if (!hojaHC) return [];

        const dataHC = hojaHC.getDataRange().getValues();
        const lobsSet = new Set();
        const omNormalizado = normalizarTexto(nombreOM);

        console.log(`🔍 [Registry] Escaneando H_C para OM: ${nombreOM}`);

        for (let i = 1; i < dataHC.length; i++) {
            const omFila = dataHC[i][6];  // Col G (índice 6) - OM
            const lobFila = dataHC[i][1]; // Col B (índice 1) - LOB
            
            if (!omFila || !lobFila) continue;
            if (normalizarTexto(omFila) === omNormalizado) {
                const lobTrimmed = String(lobFila).trim();
                if (lobTrimmed && lobTrimmed !== '-') lobsSet.add(lobTrimmed);
            }
        }

        return Array.from(lobsSet).sort();
    }, 3600);
}

function REGISTRO_obtenerLobsPorRegion(region) {
    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hojaHC = ss.getSheetByName('H_C');
    if (!hojaHC) return [];

    const data = hojaHC.getDataRange().getValues();
    const lobsSet = new Set();
    const regionNorm = normalizarTexto(region);

    // Buscar encabezados dinamicamente
    const headers = data[0];
    let colLOB = 1;    // Col B por defecto
    let colRegion = 8; // Col I por defecto

    headers.forEach(function(h, i) {
        const hn = normalizarTexto(String(h));
        if (hn === 'lob' || hn === 'sub lob' || hn === 'sublob') colLOB = i;
        if (hn === 'region') colRegion = i;
    });

    Logger.log('Buscando LOBs para region: "' + region + '" | colLOB=' + colLOB + ' | colRegion=' + colRegion);

    for (let i = 1; i < data.length; i++) {
        const regionFila = normalizarTexto(String(data[i][colRegion] || ''));
        const lobFila = String(data[i][colLOB] || '').trim();

        if (regionFila === regionNorm && lobFila) {
            lobsSet.add(lobFila);
        }
    }

    const resultado = Array.from(lobsSet).sort();
    Logger.log('LOBs en H_C para region "' + region + '": ' + resultado.length + ' -> ' + JSON.stringify(resultado));
    return resultado;
}

function REGISTRO_contarAgentesPorLob(lob) {
    const cacheKey = `count_agentes_${normalizar(lob)}`;
    return getCached(cacheKey, function () {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaHC = ss.getSheetByName('H_C');
        const data = hojaHC.getDataRange().getValues();
        const lobNorm = normalizar(lob);
        let count = 0;

        const headers = data[0].map(h => String(h).toLowerCase());
        const colStatus = headers.findIndex(h => h.includes('status') || h.includes('estado'));

        for (let i = 1; i < data.length; i++) {
            if (normalizar(data[i][1]) === lobNorm) {
                if (colStatus > -1) {
                    const estado = String(data[i][colStatus]).toLowerCase();
                    if (estado === 'activo' || estado === 'active' || estado === 'alta') count++;
                } else {
                    count++;
                }
            }
        }
        return count;
    }, 3600);
}

/**
 * Obtiene todos los agentes de una LOB desde el objeto de agentes permitidos
 */
function obtenerTodosLosAgentesDeLOB(lob, agentesPermitidos) {
    const lobNorm = normalizar(lob);
    const result = [];
    
    for (const id in agentesPermitidos) {
        if (normalizar(agentesPermitidos[id].lob) === lobNorm) {
            result.push({ bmsId: id, nombre: agentesPermitidos[id].nombre });
        }
    }
    
    return result;
}

function filtrarLobsPorColumna(nombreHoja, colIndex, valorBusqueda) {
    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hoja = ss.getSheetByName(nombreHoja);
    if (!hoja) return [];
    const data = hoja.getDataRange().getValues();
    const lobs = new Set();
    const valNorm = normalizarTexto(valorBusqueda);

    for (let i = 1; i < data.length; i++) {
        const valFila = normalizarTexto(data[i][colIndex]);
        if (valFila === valNorm && valNorm !== '') {
            const lob = String(data[i][1]).trim();
            if (lob) lobs.add(lob);
        }
    }
    return Array.from(lobs).sort();
}

function normalizarKey(str) {
    return String(str || '').toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-z0-9]/g, '');
}

function normalizar(s) {
    if (!s) return "";
    return String(s).normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim();
}

/**
 * Función unificada de normalización para textos y LOBs.
 * Mantiene coherencia entre simulación y validación.
 */
function normalizarTexto(s) {
    if (!s) return '';
    return String(s)
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "") // Quitar acentos
        .toLowerCase()
        .trim()
        .replace(/\s+/g, ' '); // Unificar espacios
}

function REGISTRO_obtenerResponsablesPorLob(lob) {
    try {
        console.log(`🔍 [Optimizado] Buscando responsables para LOB: ${lob}`);

        const cacheKey = `resp_lob_${normalizarKey(lob)}`;
        const datosLOB = getCached(cacheKey, function () {
            const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
            const hojaHC = ss.getSheetByName('H_C');
            const dataHC = hojaHC.getDataRange().getValues();
            const lobNorm = normalizarKey(lob);

            for (let i = 1; i < dataHC.length; i++) {
                if (normalizarKey(dataHC[i][1]) === lobNorm) {
                    return {
                        encontrado: true,
                        supervisor: String(dataHC[i][4] || '').trim(),
                        accm: String(dataHC[i][5] || '').trim(),
                        om: String(dataHC[i][6] || '').trim()
                    };
                }
            }
            return { encontrado: false };
        }, 3600);

        if (!datosLOB.encontrado) {
            return { supervisor: '', accm: '', om: '', error: 'LOB no encontrada en H_C' };
        }

        const usuario = REGISTRO_obtenerDatosUsuarioActivo();
        if (usuario.rol === 'OM') {
            const omAsignadoNorm = normalizarTexto(datosLOB.om);
            const usuarioNombreNorm = normalizarTexto(usuario.nombre);

            if (omAsignadoNorm !== usuarioNombreNorm) {
                return {
                    supervisor: '', accm: '', om: '',
                    error: `Esta LOB está asignada a otro OM (${datosLOB.om}).`
                };
            }
        }
        return datosLOB;

    } catch (error) {
        return { supervisor: '', accm: '', om: '', error: error.toString() };
    }
}

function REGISTRO_obtenerSalarioPorLob(lob, usuario) {
    try {
        const userNorm = usuario ? normalizarTexto(usuario.nombre || '') : '';
        const rol      = usuario ? (usuario.rol || '') : '';
        const esAdmin  = (rol === 'Reporting Analyst' || rol === 'Audit');

        // ⭐ v5 para invalidar cache anterior
        const cacheKey = `salario_lob_v5_${normalizarKey(lob)}_${normalizarKey(userNorm)}`;

        return getCached(cacheKey, function () {
            const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
            const hoja = ss.getSheetByName('Configuracion');

            if (!hoja) {
                Logger.log('❌ No existe la hoja Configuracion');
                return { encontrado: false };
            }

            const data    = hoja.getDataRange().getValues();
            const lobNorm = normalizarTexto(lob);

            Logger.log(`\n🔍 obtenerSalarioPorLob v5:`);
            Logger.log(`   LOB buscada: "${lob}" → "${lobNorm}"`);
            Logger.log(`   Usuario: "${userNorm}" | Rol: "${rol}" | esAdmin: ${esAdmin}`);
            Logger.log(`   Total filas (sin header): ${data.length - 1}`);

            const _parse = (val) => {
                if (val === null || val === undefined || val === '') return 0;
                if (typeof val === 'number') return val;
                // ⭐ FIX: limpiar formato colombiano (puntos de miles, coma decimal)
                return parseFloat(
                    String(val)
                        .replace(/\./g, '')   // quitar puntos de miles
                        .replace(',', '.')    // coma decimal → punto
                        .replace(/[^\d.]/g, '') // quitar cualquier otro símbolo
                ) || 0;
            };

            const filasCandidatas = [];

            for (let i = 1; i < data.length; i++) {
                const lobFila = normalizarTexto(String(data[i][2] || ''));
                const omFila  = normalizarTexto(String(data[i][1] || ''));

                Logger.log(`   Fila ${i+1}: LOB="${data[i][2]}"(${lobFila}) | OM="${data[i][1]}"(${omFila})`);

                // ⭐ PASO 1: LOB debe coincidir exactamente
                if (lobFila !== lobNorm) continue;

                // ⭐ PASO 2: verificar acceso
                // Admin → acepta cualquier OM
                // OM    → comparación EXACTA (sin includes — evita falsos matches)
                let omMatch = false;

                if (esAdmin || !userNorm) {
                    omMatch = true;
                } else {
                    // Comparación exacta primero
                    omMatch = (omFila === userNorm);

                    // Si no hay match exacto, intentar match por apellido+nombre
                    // (ej: "Ospina, Michel" vs "Michel Ospina")
                    if (!omMatch && omFila && userNorm) {
                        const partsOm   = omFila.split(/[,\s]+/).filter(Boolean).sort();
                        const partsUser = userNorm.split(/[,\s]+/).filter(Boolean).sort();
                        omMatch = JSON.stringify(partsOm) === JSON.stringify(partsUser);
                    }
                }

                if (!omMatch) {
                    Logger.log(`   ⚠️ Fila ${i+1}: LOB coincide pero OM no: "${omFila}" ≠ "${userNorm}"`);
                    continue;
                }

                // ⭐ PASO 3: parsear salario y porcentaje
                const salario    = _parse(data[i][3]);
                const porcentaje = _parse(data[i][4]);

                Logger.log(`   Fila ${i+1}: raw salario="${data[i][3]}" → ${salario} | raw %="${data[i][4]}" → ${porcentaje}`);

                if (salario === 0) {
                    Logger.log(`   ⚠️ Fila ${i+1}: Salario=0, ignorando`);
                    continue;
                }

                filasCandidatas.push({
                    indice:     i + 1,
                    mes:        String(data[i][0] || '').trim(),
                    om:         String(data[i][1] || '').trim(),
                    lob:        String(data[i][2] || '').trim(),
                    salario:    salario,
                    porcentaje: porcentaje
                });

                Logger.log(`   ✅ Candidata fila ${i+1}: Mes="${data[i][0]}" | Salario=$${salario} | %=${porcentaje}`);
            }

            if (filasCandidatas.length === 0) {
                // ⭐ FIX: si no encontró con match de OM pero es OM,
                // intentar buscar solo por LOB (puede que el OM no esté bien escrito)
                Logger.log(`   ⚠️ Sin candidatas con match OM. Intentando solo por LOB...`);

                for (let i = 1; i < data.length; i++) {
                    const lobFila = normalizarTexto(String(data[i][2] || ''));
                    if (lobFila !== lobNorm) continue;

                    const salario    = _parse(data[i][3]);
                    const porcentaje = _parse(data[i][4]);

                    if (salario === 0) continue;

                    Logger.log(`   🔄 Fallback fila ${i+1}: OM="${data[i][1]}" | Salario=$${salario} | %=${porcentaje}`);

                    filasCandidatas.push({
                        indice:     i + 1,
                        mes:        String(data[i][0] || '').trim(),
                        om:         String(data[i][1] || '').trim(),
                        lob:        String(data[i][2] || '').trim(),
                        salario:    salario,
                        porcentaje: porcentaje
                    });
                }
            }

            if (filasCandidatas.length === 0) {
                Logger.log(`   ❌ Sin candidatas para LOB="${lob}"`);
                return { encontrado: false };
            }

            // ⭐ TOMAR LA PRIMERA candidata válida (más antigua = más confiable)
            // La hoja Configuracion solo debería tener 1 fila por LOB
            // Si hay varias, tomamos la primera que coincide
            const mejor = filasCandidatas[0];

            Logger.log(`   🏆 Usando fila ${mejor.indice}: Mes="${mejor.mes}" | OM="${mejor.om}" | Salario=$${mejor.salario} | %=${mejor.porcentaje}`);

            return {
                encontrado:  true,
                salario:     mejor.salario,
                porcentaje:  mejor.porcentaje,
                mes:         mejor.mes,
                om:          mejor.om,
                lob:         mejor.lob,
                filaIndex:   mejor.indice
            };

        }, 300); // ⭐ Cache reducido a 5 minutos para evitar datos viejos

    } catch (e) {
        Logger.log('❌ Error en REGISTRO_obtenerSalarioPorLob: ' + e.toString());
        return { encontrado: false };
    }
}

                    Logger.log(`        Razón: ${razonPermiso}`);
                    
                    if (tienePermiso) {
                        filasCoincidentesPermiso++;
                        
                        let mesFinal = mesFila;
                        
                        if (mesFila instanceof Date) {
                            mesFinal = Utilities.formatDate(mesFila, Session.getScriptTimeZone(), 'yyyy-MM-dd');
                            Logger.log(`        📅 Fecha convertida: "${mesFila}" → "${mesFinal}"`);
                        } else if (mesFila) {
                            mesFinal = String(mesFila).trim();
                            Logger.log(`        📅 Ya es String: "${mesFinal}"`);
                        } else {
                            Logger.log(`        ⚠️ Mes/Fecha VACÍO en columna A`);
                            continue;
                        }
                        
                        meses.add(mesFinal);
                        Logger.log(`        ✅ MES AGREGADO: "${mesFinal}"`);
                    }
                } else if (i <= 5) {
                    Logger.log(`\n  ❌ Fila ${i+1}: LOB NO coincide`);
                    Logger.log(`     Esperado: "${lob}" (normalizado: "${lobNorm}")`);
                    Logger.log(`     Encontrado: "${lobFila}" (normalizado: "${lobFilaNorm}")`);
                }
            }
            
            Logger.log(`\n📊 Resumen de hoja ${nombreHoja}:`);
            Logger.log(`   Filas procesadas: ${filasProcesadas}`);
            Logger.log(`   Filas con LOB coincidente: ${filasCoincidentesLOB}`);
            Logger.log(`   Filas con permiso: ${filasCoincidentesPermiso}`);
        });
        
        const resultado = Array.from(meses).sort();
        
        Logger.log(`\n🎯 ========== RESULTADO FINAL ==========`);
        Logger.log(`✅ Total meses encontrados: ${resultado.length}`);
        
        if (resultado.length > 0) {
            Logger.log(`📅 Meses encontrados:`);
            resultado.forEach((mes, idx) => {
                Logger.log(`   ${idx + 1}. "${mes}"`);
            });
        } else {
            Logger.log(`\n⚠️ ========== NO SE ENCONTRARON MESES ==========`);
        }
        
        Logger.log(`=========================================\n`);
        
        return resultado;
        
    } catch (error) {
        Logger.log(`\n❌ ========== ERROR ==========`);
        Logger.log(`Mensaje: ${error.message}`);
        Logger.log(`Stack: ${error.stack}`);
        Logger.log(`================================\n`);
        return [];
    }
}

function REGISTRO_guardarRegistroMetricas(datos) {
    const lock = LockService.getScriptLock();
    try {
        if (!lock.tryLock(30000)) {
            return { success: false, error: '⚠️ Servidor ocupado. Intente nuevamente.' };
        }
        console.log('🔒 Guardado seguro iniciado...');

        if (!datos || !datos.mes || !datos.lob || !datos.tipoHoja) {
            return { success: false, error: 'Faltan campos obligatorios' };
        }
        
        if (datos.bonus) {
            const kpisUnicos = {};
            datos.bonus.forEach(kpi => {
                const nombre = (kpi.nombre || '').trim().toUpperCase();
                if (nombre && !kpisUnicos[nombre]) {
                    kpisUnicos[nombre] = parseFloat(kpi.porcentaje) || 0;
                }
            });
            let totalPct = Object.values(kpisUnicos).reduce((sum, pct) => sum + pct, 0);
            if (totalPct > 100) return { success: false, error: `Bonus Structure (${totalPct.toFixed(1)}%) excede 100%` };
        }

        // 🔒 Guardar configuración SOLO si salario NO es null/undefined
        if (datos.salario !== null && datos.salario !== undefined && datos.porcentaje !== undefined) {
            console.log('💰 Salario presente — guardando en Configuracion');
            REGISTRO_guardarConfiguracion(datos.mes, datos.om, datos.lob, datos.salario, datos.porcentaje);
        } else {
            console.log('🔒 Salario bloqueado — no se actualiza Configuracion');
        }

        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hoja = ss.getSheetByName(datos.tipoHoja);
        if (!hoja) return { success: false, error: 'Hoja no encontrada' };

        const data = hoja.getDataRange().getValues();
        const mesNorm = normalizarKey(datos.mes);
        const lobNorm = normalizarKey(datos.lob);
        let filaIndex = -1;

        for (let i = 1; i < data.length; i++) {
            if (normalizarKey(data[i][0]) === mesNorm && normalizarKey(data[i][1]) === lobNorm) {
                filaIndex = i + 1;
                break;
            }
        }

        const fila = _construirFila(datos, hoja);

        // ⭐ NUEVO: ASEGURAR TÍTULOS EN REGISTRO / REGISTRO_PISOS
        const headers = [['Comentario', 'Hora Registro']];
        if (datos.tipoHoja === 'Registro') {
            const headerRange = hoja.getRange(1, 45, 1, 2); // Columnas AS y AT
            if (!headerRange.getValues()[0][0]) {
                headerRange.setValues(headers);
                headerRange.setFontWeight('bold');
            }
        } else if (datos.tipoHoja === 'Registro_pisos') {
            const headerRange = hoja.getRange(1, 127, 1, 2); // Columnas DW y DX
            if (!headerRange.getValues()[0][0]) {
                headerRange.setValues(headers);
                headerRange.setFontWeight('bold');
            }
        }

        if (filaIndex > 0) {
            hoja.getRange(filaIndex, 1, 1, fila.length).setValues([fila]);
            return { success: true, mensaje: 'Registro actualizado correctamente' };
        } else {
            hoja.appendRow(fila);
            return { success: true, mensaje: 'Nuevo registro creado exitosamente' };
        }

    } catch (error) {
        console.error('❌ Error guardando:', error);
        return { success: false, error: error.toString() };
    } finally {
        lock.releaseLock();
        console.log('🔓 Bloqueo liberado');
    }
}


function _construirFila(datos, hoja) {
    let fila = [];
    
    const _formatNum = (val) => {
        if (val === undefined || val === null || val === '') return '';
        // Convertir a string y reemplazar punto por coma
        return String(val).replace('.', ',');
    };

    const norm = (arr, i) => {
        const k = (arr && arr[i]) ? arr[i] : {};
        let sc = k.score || '';
        
        // ⭐ NUEVO FIX: Anexar ",Gate" si es Gate (solo aplicable a Bonus Structure)
        if (k.gate) {
            sc = sc + ',Gate';
        }
        
        // Comprobar que no empieza ya con el operador para evitar dobles '>= >='
        if (k.comparacion && sc) {
            const hasOp = /^[<>=]+/.test(sc.trim());
            if (!hasOp) {
                sc = k.comparacion + sc.replace('%', '');
            }
            // Asegurar coma en el score si es numérico después del operador
            sc = sc.replace(/(\d+)\.(\d+)/g, '$1,$2');
        }
        
        return { 
            n: k.nombre || '',
            s: sc || '',
            p: _formatNum((k.porcentaje || '').toString().replace('%', '')),
            part: _formatNum((k.participacion || '').toString().replace('%', ''))
        };
    };

    if (datos.tipoHoja === 'Registro_pisos') {
        fila.push(datos.mes, datos.lob, datos.accm || '', datos.om || '');
        
        const numAgentes = REGISTRO_contarAgentesPorLob(datos.lob);
        fila.push(numAgentes || '');
        
        // 1. Keys (3 slots: Index 5-10)
        for (let i = 0; i < 3; i++) { 
            const k = norm(datos.keys, i); 
            fila.push(k.n, k.s); 
        }
        
        fila.push(''); // Separador (Index 11 - Columna L)
        
        // 2. Bonus Structure (17 slots: Index 12-79, inicia en Columna M)
        for (let i = 0; i < 17; i++) { 
            const k = norm(datos.bonus, i); 
            fila.push(k.n, k.s, k.p, k.part);
        }
        
        fila.push(''); // Separador (Index 80)
        
        // 3. Additional Incentive (5 slots: Index 81-100)
        for (let i = 0; i < 5; i++) { 
            const k = norm(datos.additional, i); 
            fila.push(k.n, k.s, k.p, k.part);
        }
        
        fila.push(''); // Separador (Index 101)
        
        // 4. Decreased (5 slots: Index 102-121)
        for (let i = 0; i < 5; i++) { 
            const k = norm(datos.decreased, i); 
            fila.push(k.n, k.s, k.p, k.part);
        }

        // 5. Comentarios y Hora (Columna DW/DX - Index 126/127)
        while (fila.length < 126) fila.push('');
        fila[126] = datos.comentarios || '';
        fila[127] = new Date();
    }
    else if (datos.tipoHoja === 'Registro') {
        fila.push(datos.mes, datos.lob, datos.accm || '', datos.om || '');
        fila.push('');
        
        for (let i = 0; i < 2; i++) { 
            const k = norm(datos.keys, i); 
            fila.push(k.n, k.s); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 4; i++) { 
            const k = norm(datos.bonus, i); 
            fila.push(k.n, k.s, k.p); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 3; i++) { 
            const k = norm(datos.additional, i); 
            fila.push(k.n, k.s, k.p); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 2; i++) { 
            const k = norm(datos.decreased, i); 
            fila.push(k.n, k.s, k.p); 
        }

        // 5. Comentarios y Hora (Columna AS/AT - Index 44/45)
        while (fila.length < 44) fila.push('');
        fila[44] = datos.comentarios || '';
        fila[45] = new Date();

        // 6. Participación (Para no romper las columnas de "Registro", se agregan al final a partir de AU)
        for (let i = 0; i < 4; i++) { const k = norm(datos.bonus, i); fila.push(k.part); }
        for (let i = 0; i < 3; i++) { const k = norm(datos.additional, i); fila.push(k.part); }
        for (let i = 0; i < 2; i++) { const k = norm(datos.decreased, i); fila.push(k.part); }
    }
    
    return fila;
}

function REGISTRO_obtenerEstructuraRegistrada(lob, mes, usuario) {
    try {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojas = [ss.getSheetByName('Registro'), ss.getSheetByName('Registro_pisos')];
        const lobNorm = normalizarKey(lob);
        const mesNorm = normalizarKey(mes);

        for (const hoja of hojas) {
            if (!hoja) continue;
            const nombre = hoja.getName();
            const data = hoja.getDataRange().getValues();
            const userNorm = normalizarTexto(usuario.nombre);
            const rol = usuario.rol;
            
            for (let i = 1; i < data.length; i++) {
                let mesFila = data[i][0];
                if (mesFila instanceof Date) {
                    mesFila = Utilities.formatDate(mesFila, Session.getScriptTimeZone(), "yyyy-MM-dd");
                }
                
                if (normalizarKey(mesFila) === mesNorm && normalizarKey(data[i][1]) === lobNorm) {
                    // Validar OM si el rol no es administrador
                    const omFila = normalizarTexto(data[i][3]);
                    const tieneAcceso = (rol === 'Reporting Analyst' || rol === 'Audit' || omFila === userNorm);
                    
                    const _parseNum = (val) => {
                        if (!val) return 0;
                        // Convertir coma a punto para que JS lo entienda como número
                        const s = String(val).replace(',', '.').replace(/[^0-9.-]/g, '');
                        return parseFloat(s) || 0;
                    };

                    const isPisos = (nombre === 'Registro_pisos');
                    const kpis = { keys: [], bonusStructure: [], additionalIncentives: [], decreasedKPIs: [] };
                    
                    if (!isPisos) {
                        let idxPart = 46; // Leer participación desde AU
                        const _getPart = () => {
                            if (data[i][idxPart] !== undefined && data[i][idxPart] !== '') {
                                const p = _parseNum(data[i][idxPart]);
                                idxPart++;
                                return p;
                            }
                            idxPart++;
                            return 100;
                        };

                        // 1. Keys (Index 5-8)
                        for (let c = 5; c <= 8; c += 2) {
                            if (data[i][c]) {
                                kpis.keys.push({ 
                                    nombre: String(data[i][c]).trim(),
                                    score: String(data[i][c + 1] || '').trim()
                                });
                            }
                        }
                        // 2. Bonus (Index 10-21)
                        for (let c = 10; c <= 21; c += 3) {
                            if (data[i][c]) {
                                const _scoreRaw = String(data[i][c + 1] || '').trim();
                                const _scoreLow = _scoreRaw.toLowerCase().replace(/\s/g, '');
                                const _esGate = _scoreLow.endsWith(',key') || _scoreLow.endsWith(',gate');
                                const _scoreLimpio = _esGate ? _scoreRaw.replace(/\s*,\s*(key|gate)\s*$/i, '').trim() : _scoreRaw;
                                kpis.bonusStructure.push({
                                    nombre: String(data[i][c]).trim(),
                                    score: _scoreLimpio,
                                    porcentaje: _parseNum(data[i][c + 2]),
                                    participacion: _getPart(),
                                    esGate: _esGate
                                });
                            } else { idxPart++; }
                        }
                        // 3. Additional (Index 23-31)
                        for (let c = 23; c <= 31; c += 3) {
                            if (data[i][c]) {
                                kpis.additionalIncentives.push({
                                    nombre: String(data[i][c]).trim(),
                                    score: String(data[i][c + 1] || '').trim(),
                                    porcentaje: _parseNum(data[i][c + 2]),
                                    participacion: _getPart()
                                });
                            } else { idxPart++; }
                        }
                        // 4. Decreased (Index 33-38)
                        for (let c = 33; c <= 38; c += 3) {
                            if (data[i][c]) {
                                kpis.decreasedKPIs.push({
                                    nombre: String(data[i][c]).trim(),
                                    score: String(data[i][c + 1] || '').trim(),
                                    porcentaje: _parseNum(data[i][c + 2]),
                                    participacion: _getPart()
                                });
                            } else { idxPart++; }
                        }
                    } else {
                        // COMPLEX (Registro_pisos)
                        // 1. Keys (Index 5-10)
                        for (let c = 5; c <= 9; c += 2) {
                            const val = data[i][c];
                            if (!val || val === 'Keys') continue;
                            kpis.keys.push({
                                nombre: String(val).trim(),
                                score: String(data[i][c + 1] || '').trim()
                            });
                        }
                        // 2. Bonus (Index 12-79)
                        let bonusCount = 0;
                        for (let c = 12; c < 80 && bonusCount < 17; c += 4) {
                            const val = data[i][c];
                            if (!val || val === 'Bonus Structure') continue;
                            const _scoreRawP = String(data[i][c + 1] || '').trim();
                            const _scoreLowP = _scoreRawP.toLowerCase().replace(/\s/g, '');
                            const _esGateP = _scoreLowP.endsWith(',key') || _scoreLowP.endsWith(',gate');
                            const _scoreLimpioP = _esGateP ? _scoreRawP.replace(/\s*,\s*(key|gate)\s*$/i, '').trim() : _scoreRawP;
                            kpis.bonusStructure.push({
                                nombre: String(val).trim(),
                                score: _scoreLimpioP,
                                porcentaje: _parseNum(data[i][c + 2]),
                                participacion: _parseNum(data[i][c + 3]),
                                esGate: _esGateP
                            });
                            bonusCount++;
                        }
                        // 3. Additional (Index 81-100)
                        let additionalCount = 0;
                        for (let c = 81; c < 101 && additionalCount < 5; c += 4) {
                            const val = data[i][c];
                            if (!val || val === 'Additional Incentive') continue;
                            kpis.additionalIncentives.push({
                                nombre: String(val).trim(),
                                score: String(data[i][c + 1] || '').trim(),
                                porcentaje: _parseNum(data[i][c + 2]),
                                participacion: _parseNum(data[i][c + 3])
                            });
                            additionalCount++;
                        }
                        // 4. Decreased (Index 102-121)
                        let decreasedCount = 0;
                        for (let c = 102; c < 122 && decreasedCount < 5; c += 4) {
                            const val = data[i][c];
                            if (!val || val === 'Decreased') continue;
                            kpis.decreasedKPIs.push({
                                nombre: String(val).trim(),
                                score: String(data[i][c + 1] || '').trim(),
                                porcentaje: _parseNum(data[i][c + 2]),
                                participacion: _parseNum(data[i][c + 3])
                            });
                            decreasedCount++;
                        }
                    }

                    const omEncontrado = String(data[i][3]).trim();
                    const lobEncontrado = String(data[i][1]).trim();
                    const configSalario = REGISTRO_obtenerSalarioPorLob(lobEncontrado, usuario);

                    let fechaFormateada = data[i][0];
                    if (fechaFormateada instanceof Date) {
                        fechaFormateada = Utilities.formatDate(fechaFormateada, Session.getScriptTimeZone(), 'yyyy-MM-dd');
                    } else {
                        fechaFormateada = String(fechaFormateada).trim();
                    }

                    const salarioFinal = configSalario.encontrado ? _parseNum(configSalario.salario) : 0;
                    const porcentajeFinal = configSalario.encontrado ? _parseNum(configSalario.porcentaje) : 0;

                    let horaReq = data[i][isPisos ? 127 : 45];
                    if (horaReq instanceof Date) {
                        horaReq = Utilities.formatDate(horaReq, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
                    }

                    console.log(`✅ Estructura encontrada para ${lobEncontrado} (${nombre})`);
                    return {
                        encontrado: true,
                        tipoHoja: nombre,
                        registroOriginal: data[i].map(c => c instanceof Date ? Utilities.formatDate(c, Session.getScriptTimeZone(), 'yyyy-MM-dd') : c),
                        kpis: kpis,
                        salario: salarioFinal,

                        hc: data[i][4],
                        lob: lobEncontrado,
                        mes: String(data[i][0]).trim(),
                        fecha: fechaFormateada,
                        accm: String(data[i][2]).trim(),
                        om: omEncontrado,
                        comentarios: String(data[i][isPisos ? 126 : 44] || '').trim(),
                        horaRegistro: horaReq || ''
                    };
                }
            }
        }
        return { encontrado: false };
    } catch (e) { 
        return { encontrado: false, error: e.toString() }; 
    }
}

        let totalGeneral = 0, totalAgentesGeneral = 0, totalBonoMaximo = 0;

        lobsRegistradas.forEach(lob => {
            const meses = REGISTRO_obtenerMesesRegistrados(lob, usuario);
            if (!meses.length) return;

            const mesReciente = meses[meses.length - 1];

            // ✅ USA la función correcta con CDM real
            const sim = REGISTRO_simularConMetricasReales(lob, mesReciente, mesActualNumero);
            if (!sim || !sim.success) {
                console.log(`⚠️ Error en LOB "${lob}": ${sim?.error}`);
                return;
            }

            lobsSimuladas.push({
                lob:             lob,
                mes:             mesReciente,
                hc:              sim.hc || 0,
                bonoPorAgente:   sim.hc > 0 ? sim.inversionReal / sim.hc : 0,
                totalBonus:      sim.desglosePorCategoria?.bonusStructure
                                    ?.reduce((s, k) => s + (k.montoTotal || 0), 0) || 0,
                totalAdditional: sim.desglosePorCategoria?.additionalIncentives
                                    ?.reduce((s, k) => s + (k.montoTotal || 0), 0) || 0,
                totalLOB:        sim.inversionReal || 0,
                inversionTeorica: sim.inversionTeorica || 0,
                eficiencia:      sim.eficienciaReal || 0,
                tasaCumplimiento: sim.estadisticas?.tasaCumplimiento || 0
            });

            totalGeneral       += sim.inversionReal || 0;
            totalAgentesGeneral += sim.hc || 0;
            totalBonoMaximo    += sim.inversionTeorica || 0;
        });

        return {
            success: true,
            lobsSimuladas,
            totalGeneral,
            resumen: {
                totalAgentes:    totalAgentesGeneral,
                bonoMaximoTotal: totalBonoMaximo,
                totalBonus:      lobsSimuladas.reduce((s, l) => s + l.totalBonus, 0),
                totalAdditional: lobsSimuladas.reduce((s, l) => s + l.totalAdditional, 0)
            },
            usuario: { nombre: usuario.nombre, rol: usuario.rol }
        };

    } catch (error) {
        return { success: false, error: error.toString() };
    }
}


function REGISTRO_simularTodasLOBsPorOM(nombreOM, mesActualNumero) {
    try {
        if (!nombreOM) {
            const usuarioActivo = REGISTRO_obtenerDatosUsuarioActivo();
            nombreOM = usuarioActivo?.nombre || '';
        }

        if (!mesActualNumero || mesActualNumero < 1 || mesActualNumero > 12) {
            mesActualNumero = parseInt(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'M'));
        }

        Logger.log(`\n🚀 ========== SIMULACIÓN MASIVA OM: ${nombreOM} ==========`);
        Logger.log(`📅 Mes actual CDM: ${mesActualNumero}`);

        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojas = [ss.getSheetByName('Registro'), ss.getSheetByName('Registro_pisos')];
        const zonaHoraria = Session.getScriptTimeZone();

        const lobMesMap = {};

        hojas.forEach(hoja => {
            if (!hoja) return;
            const datos = hoja.getDataRange().getValues();
            Logger.log(`\n📊 Escaneando hoja: ${hoja.getName()} (${datos.length - 1} filas)`);
            let filasOM = 0;

            for (let i = 1; i < datos.length; i++) {
                const lobFila = String(datos[i][1] || '').trim();
                const omFila  = String(datos[i][3] || '').trim();
                if (!lobFila || normalizarTexto(omFila) !== normalizarTexto(nombreOM)) continue;
                filasOM++;

                let mesFila = datos[i][0];
                if (mesFila instanceof Date) {
                    mesFila = Utilities.formatDate(mesFila, zonaHoraria, 'yyyy-MM-dd');
                } else {
                    mesFila = String(mesFila || '').trim();
                }
                if (!mesFila) continue;

                const numMesEst = _resolverNumeroMesDesdeValor(datos[i][0]);
                if (!numMesEst) continue;

                const lobKey = normalizarTexto(lobFila);
                const existing = lobMesMap[lobKey];

                // ⭐ PRIORIDAD: el mes más cercano al actual sin pasarlo
                // Si numMesEst <= mesActual, es candidato válido
                // Gana el candidato válido con número más alto (más reciente)
                // Si todos son futuros (> mesActual), usar el más bajo (más antiguo disponible)
                const esValido     = numMesEst <= mesActualNumero;
                const existeValido = existing && existing.numMesEst <= mesActualNumero;

                let reemplazar = false;

                if (!existing) {
                    reemplazar = true;
                } else if (esValido && !existeValido) {
                    // El nuevo es válido (pasado/actual) y el existente es futuro → reemplazar
                    reemplazar = true;
                } else if (esValido && existeValido) {
                    // Ambos válidos → preferir el más reciente (mayor número)
                    reemplazar = numMesEst > existing.numMesEst;
                } else if (!esValido && !existeValido) {
                    // Ambos son futuros → preferir el más antiguo (menor número, más cercano al actual)
                    reemplazar = numMesEst < existing.numMesEst;
                }
                // Si nuevo es futuro y existente es válido → NO reemplazar

                if (reemplazar) {
                    lobMesMap[lobKey] = { lob: lobFila, mesEstructura: mesFila, numMesEst };
                    Logger.log(`   ✅ LOB: "${lobFila}" | Mes: "${mesFila}" | NumMes: ${numMesEst} | ${esValido ? '✅ válido' : '⚠️ futuro'}`);
                } else {
                    Logger.log(`   ⏭️ LOB: "${lobFila}" | Mes: "${mesFila}" | NumMes: ${numMesEst} | descartado (ya hay mejor: ${existing?.numMesEst})`);
                }
            }
            Logger.log(`   → Filas del OM en ${hoja.getName()}: ${filasOM}`);
        });

        const lobsEncontradas = Object.values(lobMesMap);
        Logger.log(`\n📋 Total LOBs para OM "${nombreOM}": ${lobsEncontradas.length}`);
        lobsEncontradas.forEach(item => {
            Logger.log(`   → "${item.lob}" | estructura: "${item.mesEstructura}" | numMes: ${item.numMesEst} | ${item.numMesEst <= mesActualNumero ? '✅' : '⚠️ futuro'}`);
        });

        if (lobsEncontradas.length === 0) {
            return { success: false, error: `No se encontraron LOBs para el OM "${nombreOM}"` };
        }

        const resultados = [];
        let totalAgentesGeneral = 0, totalInversionReal = 0, totalInversionTeorica = 0;

        for (const item of lobsEncontradas) {
            Logger.log(`\n🔍 ===== Simulando: "${item.lob}" =====`);
            Logger.log(`   Estructura: "${item.mesEstructura}" (mes ${item.numMesEst}) | CDM mes: ${mesActualNumero}`);
            try {
                const simResult = REGISTRO_simularConMetricasReales(
                    item.lob,
                    item.mesEstructura,
                    mesActualNumero
                );

                if (!simResult || !simResult.success) {
                    Logger.log(`⚠️ Error: ${simResult?.error}`);
                    resultados.push({
                        lob: item.lob, mes: item.mesEstructura,
                        error: simResult?.error || 'Sin datos',
                        totalLOB: 0, hc: 0
                    });
                    continue;
                }

                Logger.log(`   inversionReal: $${simResult.inversionReal?.toLocaleString()} | lobEnCDM: ${simResult.lobEnCDM}`);

                resultados.push({
                    lob:                  item.lob,
                    mes:                  item.mesEstructura,
                    fecha:                item.mesEstructura,
                    hc:                   simResult.hc || 0,
                    salarioBase:          simResult.configuracion?.salario || 0,
                    porcentajeBono:       simResult.configuracion?.porcentaje || 0,
                    bonoMaximoPorAgente:  simResult.configuracion?.bonoMaximo || 0,
                    inversionReal:        simResult.inversionReal,
                    inversionTeorica:     simResult.inversionTeorica,
                    totalLOB:             simResult.inversionReal,
                    eficiencia:           simResult.eficienciaReal || 0,
                    tasaCumplimiento:     simResult.estadisticas?.tasaCumplimiento || 0,
                    ahorro:               simResult.ahorro || 0,
                    estadisticas:         simResult.estadisticas || {},
                    desglosePorCategoria: simResult.desglosePorCategoria || {},
                    top10Ganadores:       simResult.top10Ganadores || [],
                    tipoHoja:             simResult.tipoHoja || ''
                });

                totalAgentesGeneral   += simResult.hc || 0;
                totalInversionReal    += simResult.inversionReal;
                totalInversionTeorica += simResult.inversionTeorica;

            } catch (err) {
                Logger.log(`❌ Excepción: ${err.message}`);
                resultados.push({
                    lob: item.lob, mes: item.mesEstructura,
                    error: err.message, totalLOB: 0, hc: 0
                });
            }
        }

        Logger.log(`\n🎯 ========== RESUMEN FINAL ==========`);
        Logger.log(`   Inv. real:    $${totalInversionReal.toLocaleString()}`);
        Logger.log(`   Inv. teórica: $${totalInversionTeorica.toLocaleString()}`);
        Logger.log(`   Eficiencia:   ${totalInversionTeorica > 0 ? ((totalInversionReal/totalInversionTeorica)*100).toFixed(2) : 0}%`);

        const lobsExitosas     = resultados.filter(r => !r.error);
        const lobsConError     = resultados.filter(r =>  r.error);
        const eficienciaGlobal = totalInversionTeorica > 0
            ? (totalInversionReal / totalInversionTeorica) * 100 : 0;

        return {
            success:        true,
            om:             nombreOM,
            totalLOBs:      lobsEncontradas.length,
            lobsProcesadas: lobsExitosas.length,
            lobsConError:   lobsConError.length,
            resultados:     lobsExitosas.sort((a, b) => b.totalLOB - a.totalLOB),
            lobsErrores:    lobsConError,
            resumen: {
                totalAgentes:      totalAgentesGeneral,
                inversionTotal:    totalInversionReal,
                inversionTeorica:  totalInversionTeorica,
                bonoMaximoTotal:   totalInversionTeorica,
                ahorro:            totalInversionTeorica - totalInversionReal,
                eficienciaGlobal,
                promedioPorAgente: totalAgentesGeneral > 0
                    ? totalInversionReal / totalAgentesGeneral : 0
            },
            fechaSimulacion: new Date().toISOString()
        };

    } catch (error) {
        Logger.log(`❌ Error general: ${error.message}\nStack: ${error.stack}`);
        return { success: false, error: error.toString() };
    }
}

function _resolverNumeroMesDesdeValor(valor) {
    if (!valor) return null;
    if (valor instanceof Date) return valor.getMonth() + 1;
    if (typeof valor === 'number') return valor >= 1 && valor <= 12 ? valor : null;

    const s = String(valor).trim();

    if (/^\d{4}-\d{2}-\d{2}/.test(s)) {
        const n = parseInt(s.split('-')[1]);
        return n >= 1 && n <= 12 ? n : null;
    }

    // ⭐ toLowerCase() maneja mayúsculas: "Abril" → "abril", "MAYO" → "mayo"
    const mapa = {
        enero:1, january:1,
        febrero:2, february:2,
        marzo:3, march:3,
        abril:4, april:4,
        mayo:5, may:5,
        junio:6, june:6,
        julio:7, july:7,
        agosto:8, august:8,
        septiembre:9, september:9,
        octubre:10, october:10,
        noviembre:11, november:11,
        diciembre:12, december:12
    };
    const clave = s.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').trim();
    return mapa[clave] || null;
}

function REGISTRO_obtenerEstructuraTodasLOBsPorOM(nombreOM) {

        const omNorm = normalizar(nombreOM);
        const hojas = [ss.getSheetByName('Registro'), ss.getSheetByName('Registro_pisos')];
        
        const lobMesMap = {};
        hojas.forEach(hoja => {
            if (!hoja) return;
            const data = hoja.getDataRange().getValues();
            for (let i = 1; i < data.length; i++) {
                if (normalizar(data[i][3]) === omNorm) {
                    const lobKey = normalizar(data[i][1]);
                    const mesFila = data[i][0] instanceof Date 
                        ? Utilities.formatDate(data[i][0], Session.getScriptTimeZone(), 'yyyy-MM-dd')
                        : String(data[i][0] || '').trim();
                    if (!lobMesMap[lobKey] || mesFila > lobMesMap[lobKey].mes) {
                        lobMesMap[lobKey] = { lob: String(data[i][1]).trim(), mes: mesFila };
                    }
                }
            }
        });

        const estructuras = [];
        for (const item of Object.values(lobMesMap)) {
            const est = REGISTRO_obtenerEstructuraRegistrada(item.lob, item.mes, usuario);
            if (est.encontrado) estructuras.push(est);
        }

        return { success: true, estructuras: estructuras };
    } catch (e) {
        return { success: false, error: e.toString() };
    }
}

function REGISTRO_simularTodasLOBsPorRegion(region) {

        const mesActualNumero = parseInt(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'M'));

        const dataConfig = hojaConfig.getDataRange().getValues();
        // Col A=Mes, B=OM, C=LOB, D=Salario_Base, E=%_Bono
        const lobsEnConfig = {};

        for (let i = 1; i < dataConfig.length; i++) {
            const lobConfig  = String(dataConfig[i][2] || '').trim();
            const salario    = String(dataConfig[i][3] || '').trim();
            const porcentaje = String(dataConfig[i][4] || '').trim();

            if (!lobConfig || !salario || salario === '0' || salario === '') continue;

            // Guardar la entrada mas reciente por LOB (la ultima en la hoja)
            lobsEnConfig[normalizarTexto(lobConfig)] = {
                lob:        lobConfig,
                salario:    parseFloat(salario.replace(',', '.'))   || 0,
                porcentaje: parseFloat(porcentaje.replace(',', '.')) || 0
            };
        }

        Logger.log('LOBs en Configuracion con salario: ' + Object.keys(lobsEnConfig).length);
        Logger.log(JSON.stringify(Object.values(lobsEnConfig).map(function(l) { return l.lob; })));

        if (Object.keys(lobsEnConfig).length === 0) {
            return { success: false, error: 'No hay LOBs con salario configurado en Configuracion' };
        }

        // ========================================
        // PASO 2: Si hay region, filtrar por H_C
        // Si el usuario es Reporting Analyst sin region especifica, usar todas las de Configuracion
        // ========================================
        let lobsASimular = Object.values(lobsEnConfig).map(function(l) { return l.lob; });

        if (region && region !== 'TODAS') {
            const lobsDeRegion = REGISTRO_obtenerLobsPorRegion(region);
            Logger.log('LOBs en H_C para region "' + region + '": ' + lobsDeRegion.length);
            Logger.log(JSON.stringify(lobsDeRegion));

            // Filtrar: solo LOBs que estan EN la region Y EN Configuracion
            lobsASimular = lobsDeRegion.filter(function(lob) {
                return !!lobsEnConfig[normalizarTexto(lob)];
            });

            Logger.log('LOBs finales (region interseccion Configuracion): ' + lobsASimular.length);
            Logger.log(JSON.stringify(lobsASimular));

            if (lobsASimular.length === 0) {
                return {
                    success: false,
                    error: 'Ninguna LOB de la region "' + region + '" tiene salario configurado en Configuracion'
                };
            }
        }

        // ========================================
        // PASO 3: Para cada LOB, encontrar su estructura mas reciente
        // buscando en Registro y Registro_pisos
        // ========================================
        const hojas = [ss.getSheetByName('Registro'), ss.getSheetByName('Registro_pisos')];
        const zonaHoraria = Session.getScriptTimeZone();
        const lobMesMap = {};

        hojas.forEach(function(hoja) {
            if (!hoja) return;
            const data = hoja.getDataRange().getValues();

            for (let i = 1; i < data.length; i++) {
                const lobFila = String(data[i][1] || '').trim();
                if (!lobFila) continue;

                // Solo procesar si esta en nuestra lista de LOBs a simular
                const lobNorm = normalizarTexto(lobFila);
                const estaEnLista = lobsASimular.some(function(l) {
                    return normalizarTexto(l) === lobNorm;
                });
                if (!estaEnLista) continue;

                let mesFila = data[i][0];
                if (mesFila instanceof Date) {
                    mesFila = Utilities.formatDate(mesFila, zonaHoraria, 'yyyy-MM-dd');
                } else {
                    mesFila = String(mesFila || '').trim();
                }
                if (!mesFila) continue;

                const numMesEst = _resolverNumeroMesDesdeValor(data[i][0]);
                if (!numMesEst) continue;

                const existing = lobMesMap[lobNorm];
                const esValido     = numMesEst <= mesActualNumero;
                const existeValido = existing && existing.numMesEst <= mesActualNumero;

                let reemplazar = false;
                if (!existing) {
                    reemplazar = true;
                } else if (esValido && !existeValido) {
                    reemplazar = true;
                } else if (esValido && existeValido) {
                    reemplazar = numMesEst > existing.numMesEst;
                } else if (!esValido && !existeValido) {
                    reemplazar = numMesEst < existing.numMesEst;
                }

                if (reemplazar) {
                    lobMesMap[lobNorm] = {
                        lob:        lobFila,
                        mes:        mesFila,
                        numMesEst:  numMesEst
                    };
                }
            }
        });

        // ========================================
        // PASO 4: LOBs que estan en Configuracion pero NO en Registro
        // las incluimos con mes calculado desde Configuracion
        // ========================================
        lobsASimular.forEach(function(lob) {
            const lobNorm = normalizarTexto(lob);
            if (!lobMesMap[lobNorm]) {

                for (let i = 1; i < dataConfig.length; i++) {
                    const lobConf = normalizarTexto(String(dataConfig[i][2] || ''));
                    if (lobConf === lobNorm) {
                        const mesConf = String(dataConfig[i][0] || '').trim();

                        if (mesConf && numMes) {

                        }
                        break;
                    }
                }
            }
        });

        const lobsConEstructura = Object.values(lobMesMap);
        Logger.log('LOBs con estructura para simular: ' + lobsConEstructura.length);

        if (lobsConEstructura.length === 0) {
            return {
                success: false,
                error: 'Ninguna LOB tiene estructura registrada'
            };
        }

        // ========================================

        // ========================================

            Logger.log('\n🔍 Simulando: "' + item.lob + '" | Estructura: "' + item.mes + '" | CDM mes: ' + mesActualNumero);

            try {
                const simResult = REGISTRO_simularConMetricasRealesMasivo(
                    item.lob,
                    item.mes,
                    mesActualNumero
                );

                if (!simResult || !simResult.success) {

                    resultados.push({
                        lob:      item.lob,
                        mes:      item.mes,
                        error:    simResult ? simResult.error : 'Error desconocido',
                        totalLOB: 0,
                        hc:       0
                    });
                    continue;
                }

                resultados.push({
                    lob:                  item.lob,
                    mes:                  item.mes,
                    fecha:                item.mes,
                    hc:                   simResult.hc || 0,
                    salarioBase:          simResult.configuracion ? simResult.configuracion.salario     : 0,
                    porcentajeBono:       simResult.configuracion ? simResult.configuracion.porcentaje  : 0,
                    bonoMaximoPorAgente:  simResult.configuracion ? simResult.configuracion.bonoMaximo  : 0,
                    inversionReal:        simResult.inversionReal     || 0,
                    inversionTeorica:     simResult.inversionTeorica  || 0,
                    totalLOB:             simResult.inversionReal     || 0,
                    eficiencia:           simResult.eficienciaReal    || 0,
                    tasaCumplimiento:     simResult.estadisticas ? simResult.estadisticas.tasaCumplimiento : 0,
                    ahorro:               simResult.ahorro            || 0,
                    estadisticas:         simResult.estadisticas      || {},
                    desglosePorCategoria: simResult.desglosePorCategoria || {},
                    top10Ganadores:       simResult.top10Ganadores    || [],
                    tipoHoja:             simResult.tipoHoja          || ''
                });

                totalAgentes          += simResult.hc            || 0;
                totalInversionReal    += simResult.inversionReal  || 0;
                totalInversionTeorica += simResult.inversionTeorica || 0;

            } catch (err) {
                Logger.log('Excepcion en "' + item.lob + '": ' + err.message);
                resultados.push({
                    lob:      item.lob,
                    mes:      item.mes,
                    error:    err.message,
                    totalLOB: 0,
                    hc:       0
                });
            }
        }

        const lobsExitosas     = resultados.filter(function(r) { return !r.error; });
        const lobsConError     = resultados.filter(function(r) { return  r.error; });
        const eficienciaGlobal = totalInversionTeorica > 0
            ? (totalInversionReal / totalInversionTeorica) * 100 : 0;

        Logger.log('\n🎯 RESULTADO REGION "' + region + '":');
        Logger.log('   Teorico:    $' + totalInversionTeorica);
        Logger.log('   Real CDM_' + mesActualNumero + ': $' + totalInversionReal);
        Logger.log('   Eficiencia: ' + eficienciaGlobal.toFixed(1) + '%');

        return {
            success:        true,
            region:         region,
            totalLOBs:      lobsASimular.length,
            lobsProcesadas: lobsExitosas.length,
            lobsConError:   lobsConError.length,
            resultados:     lobsExitosas.sort(function(a, b) { return b.totalLOB - a.totalLOB; }),
            lobsErrores:    lobsConError,
            resumen: {
                totalAgentes:     totalAgentes,
                inversionTotal:   totalInversionReal,
                inversionTeorica: totalInversionTeorica,
                ahorro:           totalInversionTeorica - totalInversionReal,
                eficienciaGlobal: eficienciaGlobal,
                promedioPorAgente: totalAgentes > 0 ? totalInversionReal / totalAgentes : 0
            },
            fechaSimulacion: new Date().toISOString()
        };

    } catch (error) {
        Logger.log('Error general: ' + error.message + '\nStack: ' + error.stack);
        return { success: false, error: error.toString() };
    }
}




 

function REGISTRO_obtenerLobsRegistradasPorRegion(region) {
    // Paso 1: obtener todas las LOBs de la region desde H_C
    const lobsEnHC = REGISTRO_obtenerLobsPorRegion(region);
    
    if (lobsEnHC.length === 0) {
        Logger.log('No se encontraron LOBs en H_C para region: ' + region);
        return [];
    }

    Logger.log('LOBs en H_C: ' + lobsEnHC.length);

    // Paso 2: filtrar solo las que tienen configuracion (salario) en hoja Configuracion
    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hojaConfig = ss.getSheetByName('Configuracion');
    
    if (!hojaConfig) {
        Logger.log('No existe hoja Configuracion — retornando LOBs de H_C sin filtrar');
        return lobsEnHC;
    }

    const dataConfig = hojaConfig.getDataRange().getValues();
    const lobsEnConfig = new Set();

    for (let i = 1; i < dataConfig.length; i++) {
        const lobConfig = String(dataConfig[i][2] || '').trim(); // Col C = LOB
        const salario = String(dataConfig[i][3] || '').trim();   // Col D = Salario_Base

        // Solo contar si tiene salario configurado
        if (lobConfig && salario && salario !== '0' && salario !== '') {
            lobsEnConfig.add(lobConfig);
        }
    }

    Logger.log('LOBs con configuracion: ' + lobsEnConfig.size);

    // Paso 3: interseccion — LOBs que estan en H_C (region) Y tienen configuracion
    const resultado = lobsEnHC.filter(function(lob) {
        // Comparacion tolerante: normalizar ambos lados
        const lobNorm = normalizarTexto(lob);
        for (const lobConf of lobsEnConfig) {
            if (normalizarTexto(lobConf) === lobNorm) return true;
        }
        return false;
    });

    Logger.log('LOBs finales (H_C interseccion Configuracion) para region "' + region + '": ' + resultado.length);
    Logger.log(JSON.stringify(resultado));

    return resultado;
}
 
function REGISTRO_simularTodasLOBsPorRegionConPVE(region, pve, porc) {
    const sim = REGISTRO_simularTodasLOBsPorRegion(region);
    if (!sim.success) return sim;
    const prep = (pve * porc) / 100;
    sim.resultados = sim.resultados.map(r => ({
        ...r,
        pctDelPVE: prep > 0 ? parseFloat((r.totalLOB / prep * 100).toFixed(2)) : 0
    }));
    const inv = sim.resumen.inversionTotal;
    const pct = prep > 0 ? (inv / prep * 100) : 0;
    sim.pveConfig = { pveTotal: pve, porcPVE: porc, presupuestoDisponible: prep };
    sim.estadoPresupuesto = {
        texto: pct > 100 ? '❌ EXCEDE' : '✅ DENTRO',
        color: pct > 100 ? '#e74c3c' : '#27ae60',
        pctUsado: parseFloat(pct.toFixed(2)),
        inversionTotal: inv,
        diferencia: prep - inv
    };
    return sim;
}

/**
 * Función central de validación de KPI con paridad de cálculos:
 * - Ajuste de 42h proporcional al salario.
 * - Prorrateo por horas reales vs esperadas (Horas_N).
 */

function REGISTRO_validarKPI1(llave, condicionScore, porcentajePunto, dataCDM, dataUsuario, agentesPermitidos, lobActual, numeroMes = null, fechaSeleccionada = null, esDecrease = false, esProrrateado = true) {
    const resultado = { 
        llave, 
        condicionScore, 
        totalCoincidencias: 0, 
        totalCumplenCondicion: 0, 
        empleadosCumplen: [], 
        empleadosNoCumplen: [], 
        empleadosSinDatos: [] 
    };

    if (!condicionScore || condicionScore === 'N/A' || condicionScore.trim() === '') {
        return resultado;
    }

    const normalizar = s => String(s || '')
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase()
        .trim()
        .replace(/\s+/g, ' ');

    const llaveNorm     = normalizar(llave);
    const lobActualNorm = normalizar(lobActual);

    const tieneOperador = /(<=|>=|<|>|=)/.test(condicionScore);
    const esSoloNumero  = /^\d+([.,]\d+)?$/.test(condicionScore.trim());
    const esBonoFijo    = !tieneOperador && !esSoloNumero;

    let valorBonoFijo  = 0;
    let operador       = null;
    let valorCondicion = 0;

    if (esSoloNumero) {
        Logger.log(`⚠️ Score "${condicionScore}" sin operador en KPI "${llave}" → bono $0`);
        const todosAgentes = obtenerTodosLosAgentesDeLOB(lobActual, agentesPermitidos);
        todosAgentes.forEach(ag => {
            resultado.empleadosSinDatos.push({
                bmsId: ag.bmsId, nombre: ag.nombre,
                valorF: '-', bono: 0, sinDatos: true,
                razon: `Score "${condicionScore}" sin operador`
            });
        });
        return resultado;

    } else if (esBonoFijo) {
        valorBonoFijo = parseFloat(condicionScore.replace(/[^\d.]/g, '')) || 0;
        Logger.log(`💰 Bono Fijo: ${llave} = $${valorBonoFijo}`);

    } else {
        const match = condicionScore.match(/(<=|>=|<|>|=)\s*(\d+(?:[.,]\d+)?)/);
        if (!match) {
            Logger.log('❌ Condición no válida:', condicionScore);
            return resultado;
        }
        operador       = match[1];
        valorCondicion = parseFloat(match[2].replace(',', '.'));
    }

    const todosLosAgentes = obtenerTodosLosAgentesDeLOB(lobActual, agentesPermitidos);

    Logger.log(`\n🔍 REGISTRO_validarKPI1`);
    Logger.log(`   KPI: "${llave}" → norm: "${llaveNorm}"`);
    Logger.log(`   LOB: "${lobActual}" → norm: "${lobActualNorm}"`);
    Logger.log(`   Condición: ${condicionScore} | Decrease: ${esDecrease}`);
    Logger.log(`   Agentes en LOB: ${todosLosAgentes.length}`);

    // ========================================
    // CONFIGURACIÓN LOB
    // ========================================
    let salarioBaseLOB       = 0;
    let porcentajeBonoGlobal = 0;

    try {
        const ssConfig   = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaConfig = ssConfig.getSheetByName('Configuracion');
        if (!hojaConfig) { Logger.log('❌ No hoja Configuracion'); return resultado; }

        const dataConfig = hojaConfig.getDataRange().getValues();
        let encontrado   = false;

        for (let c = 1; c < dataConfig.length; c++) {
            const lobConfig = normalizar(String(dataConfig[c][2] || ''));
            if (lobConfig !== lobActualNorm) continue;
            const sRaw = String(dataConfig[c][3] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
            const pRaw = String(dataConfig[c][4] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
            salarioBaseLOB       = parseFloat(sRaw);
            porcentajeBonoGlobal = parseFloat(pRaw);
            if (salarioBaseLOB > 0 && porcentajeBonoGlobal > 0) {
                encontrado = true;
                Logger.log(`✅ Config LOB: $${salarioBaseLOB} | ${porcentajeBonoGlobal}%`);
                break;
            }
        }
        if (!encontrado) {
            Logger.log(`❌ Config no encontrada para "${lobActual}"`);
            return resultado;
        }

    } catch (error) {
        Logger.log(`❌ Error Configuracion: ${error.message}`);
        return resultado;
    }

    // ========================================
    // HORAS MÁXIMAS DESDE H_C
    // ========================================
    const horasMaxima = Math.max(
        ...Object.values(agentesPermitidos).map(a => a.horasSemanales || 0)
    ) || 42;

    // ========================================
    // ⭐⭐⭐ DETECCIÓN DE COLUMNAS CDM
    // Basada en el log real:
    //   Col 0: Mes
    //   Col 1: CCMSID  ← BMS ID
    //   Col 2: NombreAgente
    //   Col 3: KPI     ← nombre del KPI solo ✅
    //   Col 4: SubLOB  ← LOB sola ✅
    //   Col 5: KPI_Value ← valor real ✅
    //   Col 6: Key     ← BMS+LOB+KPI concatenado (NO es Llave útil)
    //   Col 7: Score   ← mismo valor que KPI_Value
    //   Col 8: Key2    ← ACCM+LOB+KPI (NO es Score objetivo)
    // ========================================
    const headers = dataCDM[0];

    let idxBms = headers.indexOf('BMS ID');
    if (idxBms === -1) idxBms = headers.indexOf('CCMSID');
    if (idxBms === -1) idxBms = 1;
    
    let idxValH = headers.indexOf('Columna H');
    if (idxValH === -1) idxValH = headers.indexOf('KPI_Value');
    if (idxValH === -1) idxValH = headers.indexOf('Score');
    if (idxValH === -1) idxValH = 7;
    
    let idxValF = headers.indexOf('Columna F');
    if (idxValF === -1) idxValF = 5;
    
    let idxLlave = headers.indexOf('Llave');
    if (idxLlave === -1) idxLlave = headers.indexOf('KPI');
    if (idxLlave === -1) idxLlave = 8;

    let idxSubLOB = headers.indexOf('SubLOB');
    if (idxSubLOB === -1) idxSubLOB = 2;

    // ========================================
    // ⭐ MATCH: solo usa col KPI + col SubLOB
    // El log confirma que col 3 tiene el KPI solo y col 4 tiene la LOB sola
    // NO usar col Llave (col 6) que tiene BMS+LOB+KPI concatenado
    // ========================================
    const _kpiMatch = (row) => {
        const llaveCDMRaw = String(row[idxLlave] || '').trim();
        const llaveCDM = normalizar(llaveCDMRaw);
        let kpiExtraidoCDM = '';
        const posLOB = llaveCDM.indexOf(lobActualNorm);
        if (posLOB !== -1) {
            kpiExtraidoCDM = llaveCDM.substring(posLOB + lobActualNorm.length).trim();
        } else {
            kpiExtraidoCDM = llaveCDM;
        }
        return kpiExtraidoCDM === llaveNorm;
    };

    const _subLobMatch = (row) => {
        const subLob = normalizar(String(row[idxSubLOB] || '').trim());
        // Si está vacío, no filtrar (aceptar)
        return !subLob || subLob === lobActualNorm;
    };

    // ========================================
    // VERIFICAR SI EL KPI EXISTE EN CDM
    // ========================================
    const kpiExisteEnCDM = !esBonoFijo && dataCDM.slice(1).some(row =>
        _kpiMatch(row) && _subLobMatch(row)
    );

    Logger.log(`🔍 KPI "${llave}" existe en CDM para LOB "${lobActual}": ${kpiExisteEnCDM ? '✅ SÍ' : '❌ NO'}`);

    if (!esBonoFijo && !kpiExisteEnCDM) {
        // Diagnóstico: mostrar qué hay en CDM para esta LOB
        Logger.log(`\n   🔎 Valores KPI disponibles en CDM para LOB "${lobActual}":`);
        const kpisDisponibles = new Set();
        const lobsDisponibles = new Set();

        for (let i = 1; i < Math.min(dataCDM.length, 50); i++) {
            const lobFila = normalizar(String(dataCDM[i][idxSubLOB] || ''));
            const kpiFila = String(dataCDM[i][idxLlave] || '').trim();
            lobsDisponibles.add(String(dataCDM[i][idxSubLOB] || '').trim());
            if (lobFila === lobActualNorm && kpiFila) kpisDisponibles.add(kpiFila);
        }

        Logger.log(`   KPIs encontrados para esta LOB: ${JSON.stringify([...kpisDisponibles])}`);
        Logger.log(`   LOBs disponibles en CDM: ${JSON.stringify([...lobsDisponibles])}`);
        Logger.log(`   KPI buscado (norm): "${llaveNorm}"`);
        Logger.log(`   LOB buscada (norm): "${lobActualNorm}"`);
    }

    const agentesConDatos = new Set();

    // ========================================
    // CASO A: BONO FIJO
    // ========================================
    if (esBonoFijo) {
        Logger.log(`💰 Asignando bono fijo a todos (${todosLosAgentes.length} agentes)`);

        todosLosAgentes.forEach(ag => {
            let salario = salarioBaseLOB;
            let salarioPersonalizado = false;

            for (let u = 1; u < dataUsuario.length; u++) {
                if (String(dataUsuario[u][0]).trim() === ag.bmsId) {
                    const sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                    const sNum = parseFloat(sRaw) || 0;
                    if (sNum > 0) {
                        salario = sNum;
                        if (sNum !== salarioBaseLOB) salarioPersonalizado = true;
                    }
                    break;
                }
            }

            const horasAgente = agentesPermitidos[ag.bmsId]?.horasSemanales || horasMaxima;
            const factorHoras = horasMaxima > 0 ? horasAgente / horasMaxima : 1;
            const bonoBase    = salario * (porcentajeBonoGlobal / 100);
            let   bonoFinal   = bonoBase * porcentajePunto * factorHoras;

            if (esDecrease) bonoFinal = -Math.abs(bonoFinal);

            resultado.totalCoincidencias++;
            resultado.totalCumplenCondicion++;
            resultado.empleadosCumplen.push({
                bmsId: ag.bmsId, nombre: ag.nombre,
                valorF: valorBonoFijo, bono: bonoFinal,
                horasContrato: horasAgente, horasMaxima, factorHoras, salarioPersonalizado
            });
        });

        return resultado;
    }

    // ========================================
    // CASO B: KPI CON CONDICIÓN
    // ========================================
    Logger.log(`\n🔄 Escaneando ${dataCDM.length - 1} filas CDM...`);
    let cntKpiMatch = 0;
    let cntLobMatch = 0;
    let cntAgMatch  = 0;

    for (let row = 1; row < dataCDM.length; row++) {
        const filaData = dataCDM[row];

        // Filtro 1: KPI coincide (col 3, nombre exacto)
        if (!_kpiMatch(filaData)) continue;
        cntKpiMatch++;

        // Filtro 2: SubLOB coincide (col 4)
        if (!_subLobMatch(filaData)) continue;
        cntLobMatch++;

        // Filtro 3: BMS en agentesPermitidos
        // ⭐ El CCMSID puede venir como número — normalizar a string
        const bmsRaw    = String(filaData[idxBms] || '').trim();
        // Algunos CDMs guardan "5321650" otros "5321650US Spa..." — extraer solo dígitos si hay texto
        const bmsIdFila = /^\d+$/.test(bmsRaw) ? bmsRaw : (bmsRaw.match(/^(\d+)/)?.[1] || bmsRaw);

        if (!agentesPermitidos[bmsIdFila]) {
            // Intentar también con el valor tal cual
            if (!agentesPermitidos[bmsRaw]) {
                Logger.log(`   ⚠️ Fila ${row + 1}: BMS "${bmsIdFila}" (raw:"${bmsRaw}") no en agentesPermitidos`);
                continue;
            }
        }
        const bmsIdFinal = agentesPermitidos[bmsIdFila] ? bmsIdFila : bmsRaw;
        cntAgMatch++;

        // Filtro 4: LOB del agente en H_C
        const lobAgente = normalizar(agentesPermitidos[bmsIdFinal].lob);
        if (lobAgente !== lobActualNorm) {
            Logger.log(`   ⚠️ BMS ${bmsIdFinal}: LOB en H_C="${lobAgente}" ≠ "${lobActualNorm}"`);
            continue;
        }

        // Filtro 5: sin duplicados
        if (agentesConDatos.has(bmsIdFinal)) continue;
        agentesConDatos.add(bmsIdFinal);
        resultado.totalCoincidencias++;

        // ⭐ Leer KPI_Value (col 5) — este es el valor real medido
        const valorRaw      = String(filaData[idxValH] || '').replace('%', '').trim();
        const valorNumerico = parseFloat(valorRaw.replace(',', '.'));

        Logger.log(`   ✅ Fila ${row + 1}: BMS=${bmsIdFinal} | KPI_Value="${filaData[idxValH]}" → ${valorNumerico}`);

        if (isNaN(valorNumerico) || valorNumerico === 0) {
            resultado.empleadosSinDatos.push({
                bmsId: bmsIdFinal, nombre: agentesPermitidos[bmsIdFinal].nombre,
                valorF: 0, bono: 0, sinDatos: true, salarioPersonalizado: false
            });
            continue;
        }

        // Evaluar condición
        let cumple = false;
        switch (operador) {
            case '>':  cumple = valorNumerico >  valorCondicion; break;
            case '>=': cumple = valorNumerico >= valorCondicion; break;
            case '<':  cumple = valorNumerico <  valorCondicion; break;
            case '<=': cumple = valorNumerico <= valorCondicion; break;
            case '=':  cumple = valorNumerico === valorCondicion; break;
        }

        // Determinar salario
        let salario = salarioBaseLOB;
        let salarioPersonalizado = false;
        for (let u = 1; u < dataUsuario.length; u++) {
            if (String(dataUsuario[u][0]).trim() === bmsIdFinal) {
                const sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                const sNum = parseFloat(sRaw) || 0;
                if (sNum > 0) {
                    salario = sNum;
                    if (sNum !== salarioBaseLOB) salarioPersonalizado = true;
                }
                break;
            }
        }
        if (salario === 0) salario = salarioBaseLOB;

        // Calcular bono con prorrateo por horas H_C
        const horasAgente = agentesPermitidos[bmsIdFinal]?.horasSemanales || horasMaxima;
        const factorHoras = horasMaxima > 0 ? horasAgente / horasMaxima : 1;
        const bonoBase    = salario * (porcentajeBonoGlobal / 100);
        let   bonoFinal   = bonoBase * porcentajePunto * factorHoras;

        if (esDecrease && cumple) bonoFinal = -Math.abs(bonoFinal);

        Logger.log(`   💰 ${agentesPermitidos[bmsIdFinal].nombre}: ${valorNumerico} ${operador} ${valorCondicion} → ${cumple ? '✅ CUMPLE' : '❌ NO'} | $${bonoFinal.toFixed(2)}`);

        const item = {
            bmsId:                bmsIdFinal,
            nombre:               agentesPermitidos[bmsIdFinal].nombre,
            valorF:               filaData[idxValF],
            valorReal:            valorNumerico,
            bono:                 cumple ? bonoFinal : 0,
            horasContrato:        horasAgente,
            horasMaxima,
            factorHoras,
            salarioPersonalizado
        };

        if (cumple) {
            resultado.totalCumplenCondicion++;
            resultado.empleadosCumplen.push(item);
        } else {
            resultado.empleadosNoCumplen.push(item);
        }
    }

    Logger.log(`\n   📊 Escaneo: KPI=${cntKpiMatch} LOB=${cntLobMatch} Ag=${cntAgMatch} Procesados=${agentesConDatos.size}`);

    // ========================================
    // AGENTES SIN REGISTRO EN CDM
    // ========================================
    todosLosAgentes.forEach(ag => {
        if (!agentesConDatos.has(ag.bmsId)) {
            resultado.empleadosSinDatos.push({ 
                bmsId:   ag.bmsId, 
                nombre:  ag.nombre, 
                valorF:  '-', 
                bono:    0, 
                sinDatos: true,
                salarioPersonalizado: false
            });
        }
    });

    Logger.log(`📊 FINAL "${llave}": cumplen=${resultado.totalCumplenCondicion} noCumplen=${resultado.empleadosNoCumplen.length} sinDatos=${resultado.empleadosSinDatos.length}\n`);

    return resultado;
}


function REGISTRO_validarKPIsPorCategoria(registro, lob, dataCDM, dataUsuario, hoja, agentesPermitidos, numeroMes = null, fechaSeleccionada = null) {
    const resultadosBrutos = { 
        keys: [], 
        bonusStructure: [], 
        additionalIncentives: [], 
        decreasedKPIs: [] 
    };

    const procesar = (cat, start, end, inc, conParticipacion) => {
        for (let c = start; c <= end; c += inc) {

            const valorCelda = registro[c];
            if (valorCelda === null || valorCelda === undefined || valorCelda === '') continue;

            const kpi = String(valorCelda).trim();
            if (!kpi || kpi === '-') continue;

            const scoreRaw = String(registro[c + 1] || '').trim();
            let score = scoreRaw;

            // ⭐ PARIDAD OM: Detectar Gate (termina en ",key" o ",gate")
            const scoreRawLower = score.toLowerCase().replace(/\s/g, '');
            const esGate = cat === 'bonusStructure' && (scoreRawLower.endsWith(',key') || scoreRawLower.endsWith(',gate'));
            if (esGate) {
                score = score.replace(/\s*,\s*(key|gate)\s*$/i, '').trim();
                Logger.log(`🔒 Gate detectado: KPI="${kpi}" | scoreRaw="${scoreRaw}" | scoreLimpio="${score}"`);
            }

            // Normalizar coma decimal
            score = score.replace(/(\d),(\d)/g, '$1.$2');

            let peso = parseFloat(String(registro[c + 2] || '0').replace('%', '')) || 0;
            if (peso > 1) peso /= 100;

            let participation = 1;
            if (conParticipacion) {
                const participacionRaw = String(registro[c + 3] || '100');
                const participacionNum = parseFloat(participacionRaw.replace('%', '').trim()) || 100;
                participation = participacionNum > 1 ? participacionNum / 100 : participacionNum;
                Logger.log(`   Participación col ${c + 3}: "${participacionRaw}" → ${(participation * 100).toFixed(0)}%`);
            }

            Logger.log(`\n📋 [${cat}] KPI="${kpi}" | score="${score}" | peso=${(peso*100).toFixed(1)}% | part=${(participation*100).toFixed(0)}% | esGate=${esGate}`);

            const esDecrease = cat === 'decreasedKPIs';

            const validacion = REGISTRO_validarKPI1(
                kpi,
                score,
                peso * participation,
                dataCDM,
                dataUsuario,
                agentesPermitidos,
                lob,
                numeroMes,
                fechaSeleccionada,
                esDecrease
            );

            resultadosBrutos[cat].push({ 
                kpi, 
                score,
                scoreOriginal: scoreRaw,
                porcentaje: peso, 
                participacion: participation,
                validacion,
                esDecrease,
                esGate
            });
        }
    };

    if (hoja === 'Registro' || hoja === 'Registro_Estandar') {
        procesar('keys', 5, 9, 2, false);
        procesar('bonusStructure', 10, 21, 3, false);
        procesar('additionalIncentives', 23, 31, 3, false);
        procesar('decreasedKPIs', 33, 38, 3, false);

        // Standard Registro: no multi-tier consolidation needed, apply gate logic directly
        resultadosBrutos.bonusStructure = REGISTRO_aplicarLogicaGate(resultadosBrutos.bonusStructure);

        return resultadosBrutos;
    } else {
        // Registro_pisos (Multi-Tier con participación)
        Logger.log('\n⭐ Procesando Registro_pisos (Multi-Tier)');

        procesar('keys',                  5,  10,  2, false);
        procesar('bonusStructure',        12,  79,  4, true);
        procesar('additionalIncentives',  81, 100,  4, true);
        procesar('decreasedKPIs',        102, 121,  4, true);

        return REGISTRO_consolidarPisos(resultadosBrutos);
    }
}



// ========================================
// ⭐⭐⭐ LÓGICA DE GATE (PARIDAD OM)
// Si un Gate no cumple, bloquea solo ESE KPI para el agente
// ========================================
function REGISTRO_aplicarLogicaGate(bonusStructureKPIs) {
    if (!bonusStructureKPIs || bonusStructureKPIs.length === 0) return bonusStructureKPIs;

    Logger.log(`\n🔒 ========== LÓGICA GATE ACTIVE ==========`);
    
    // 1. Identificar agentes bloqueados (fallan cualquier Gate en Bonus Structure)
    const agentesBloqueados = new Set();
    const gates = bonusStructureKPIs.filter(kpiObj => kpiObj.esGate === true);

    if (gates.length === 0) {
        Logger.log('✅ No hay Gates activos');
        return bonusStructureKPIs;
    }

    gates.forEach(gate => {
        const val = gate.validacion || {};
        (val.empleadosNoCumplen || []).forEach(emp => agentesBloqueados.add(emp.bmsId));
        (val.empleadosSinDatos  || []).forEach(emp => agentesBloqueados.add(emp.bmsId));
    });

    if (agentesBloqueados.size === 0) {
        Logger.log('✅ Todos los agentes pasaron los Gates');
        return bonusStructureKPIs;
    }

    Logger.log(`🚫 Agentes bloqueados por Gate: ${agentesBloqueados.size}`);

    // 2. Aplicar el bloqueo: Cero bono en todos los KPIs para estos agentes
    // NOTA: Esta función se aplica a bonusStructure. 
    // El llamador debe encargarse de que impacte el total.
    bonusStructureKPIs.forEach(kpiObj => {
        const val = kpiObj.validacion || {};
        
        (val.empleadosCumplen || []).forEach(emp => {
            if (agentesBloqueados.has(emp.bmsId)) {
                emp.bono = 0;
                emp.bloqueadoPorGate = true;
            }
        });
    });

    Logger.log('===================================\n');
    return bonusStructureKPIs;
}

function REGISTRO_consolidarPisos(resultadosBrutos) {
    const resultadosConsolidados = {};

    // ⭐ Función de consolidación reutilizable para cualquier categoría
    const consolidarCategoria = (cat) => {
        const grupos = {};
        resultadosBrutos[cat].forEach(item => {
            if (!grupos[item.kpi]) grupos[item.kpi] = [];
            grupos[item.kpi].push(item);
        });

        return Object.keys(grupos).map(nombreKpi => {
            const tiers = grupos[nombreKpi];

            // Si solo hay 1 tier, retornarlo tal cual
            if (tiers.length === 1) return tiers[0];

            const maxParticipacion = Math.max(...tiers.map(t => t.participacion || 1));

            console.log(`\n📊 Consolidando multi-tier [${cat}]: ${nombreKpi}`);
            console.log(`   Tiers: ${tiers.length}`);
            console.log(`   Participaciones: ${tiers.map(t => ((t.participacion || 1) * 100).toFixed(0) + '%').join(', ')}`);
            console.log(`   Max participación: ${(maxParticipacion * 100).toFixed(0)}%`);

            // ⭐ PARIDAD OM: esGate se preserva desde el primer tier
            const consolidado = {
                kpi: nombreKpi,
                score: "Multi-Escala",
                porcentaje: tiers[0].porcentaje,
                participacion: maxParticipacion,
                esMultiTier: true,
                esGate: tiers[0].esGate === true,   // ⭐ PRESERVAR esGate
                scoresPorTier: tiers.map(t => ({
                    score: t.score,
                    condicion: t.score,
                    participacion: t.participacion
                })),
                validacion: {
                    totalCoincidencias: tiers[0].validacion.totalCoincidencias,
                    totalCumplenCondicion: 0,
                    empleadosCumplen: [],
                    empleadosNoCumplen: [],
                    empleadosSinDatos: tiers[0].validacion.empleadosSinDatos || []
                }
            };

            const mejorResultadoAgente = {};

            // ⭐ PARIDAD OM: SELECCIONAR MEJOR BONO POR AGENTE (bono > comparison)
            tiers.forEach(tier => {
                console.log(`\n   🔍 Procesando tier: ${tier.score} (participación: ${((tier.participacion || 1) * 100).toFixed(0)}%)`);

                (tier.validacion.empleadosCumplen || []).forEach(e => {
                    console.log(`      Agente ${e.bmsId}: bono=$${e.bono.toFixed(2)}`);

                    // ⭐ MEJORA: Para bonos buscamos el Máx. Para penalidades buscamos la más fuerte (Mín).
                    const esPenalizacion = e.bono < 0;
                    const esMejor = !mejorResultadoAgente[e.bmsId] || 
                                    (esPenalizacion ? e.bono < mejorResultadoAgente[e.bmsId].bono : e.bono > mejorResultadoAgente[e.bmsId].bono);
                    
                    if (esMejor) {
                        mejorResultadoAgente[e.bmsId] = {
                            ...e,
                            scoreCumplido: tier.score,
                            participacionUsada: tier.participacion
                        };

                        console.log(`      ✅ Mejor resultado actualizado: tier="${tier.score}", part=${((tier.participacion || 1) * 100).toFixed(0)}%`);
                    }
                });

                (tier.validacion.empleadosNoCumplen || []).forEach(e => {
                    if (!mejorResultadoAgente[e.bmsId]) {
                        mejorResultadoAgente[e.bmsId] = {
                            ...e,
                            bono: 0,
                            scoreCumplido: tier.score,
                            participacionUsada: tier.participacion
                        };
                    }
                });
            });

            // Clasificar resultados consolidados
            for (let id in mejorResultadoAgente) {
                const resultado = mejorResultadoAgente[id];

                console.log(`\n   👤 Agente ${id}:`);
                console.log(`      scoreCumplido: ${resultado.scoreCumplido}`);
                console.log(`      participacionUsada: ${resultado.participacionUsada ? ((resultado.participacionUsada) * 100).toFixed(0) + '%' : 'undefined'}`);
                console.log(`      bono: $${resultado.bono.toFixed(2)}`);

                if (resultado.bono > 0) {
                    consolidado.validacion.empleadosCumplen.push(resultado);
                    consolidado.validacion.totalCumplenCondicion++;
                } else {
                    consolidado.validacion.empleadosNoCumplen.push(resultado);
                }
            }

            console.log(`✅ Consolidado: ${consolidado.validacion.empleadosCumplen.length} cumplen | esGate=${consolidado.esGate}`);

            return consolidado;
        });
    };

    // Keys: no se consolidan (nunca son multi-tier)
    resultadosConsolidados['keys'] = resultadosBrutos['keys'];

    // ⭐ Las 3 categorías restantes usan la misma lógica de consolidación
    resultadosConsolidados['bonusStructure']      = consolidarCategoria('bonusStructure');
    resultadosConsolidados['additionalIncentives'] = consolidarCategoria('additionalIncentives');
    resultadosConsolidados['decreasedKPIs']        = consolidarCategoria('decreasedKPIs');

    // ⭐ PARIDAD OM: aplicar lógica de Gate después de consolidar
    resultadosConsolidados['bonusStructure'] = REGISTRO_aplicarLogicaGate(resultadosConsolidados['bonusStructure']);

    return resultadosConsolidados;
}

function REGISTRO_obtenerAgentesPermitidos(filtroACCM = null) {
  const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
  const hojaHC = ss.getSheetByName('H_C');
  if (!hojaHC) return {};
  
  const dataHC = hojaHC.getDataRange().getValues();
  const headers = dataHC[0];
  const agentes = {};
  
  const normalizarLocal = s => String(s || '').normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim().replace(/\s+/g, ' ');
  
  // ⭐ DETECCIÓN DINÁMICA DE COLUMNAS (PARIDAD OM)
  let idxBMS = -1, idxLOB = -1, idxNombre = -1, idxRol = -1, idxHoras = -1;
  
  headers.forEach((h, i) => {
    const header = String(h).toLowerCase().trim();
    if (header.includes('bms') || header.includes('ccms') || header.includes('id')) idxBMS = i;
    if (header.includes('lob')) idxLOB = i;
    if (header.includes('nombre') || header.includes('name')) idxNombre = i;
    if (header.includes('rol') || header.includes('role')) idxRol = i;
    if (header.includes('horas') || header.includes('hours') || header === 'h') idxHoras = i;
  });

  // Fallbacks (basados en índices típicos de H_C: A=0, B=1, C=2, D=3, J=9)
  if (idxBMS     === -1) idxBMS     = 0;
  if (idxLOB     === -1) idxLOB     = 1;
  if (idxNombre  === -1) idxNombre  = 2;
  if (idxRol     === -1) idxRol     = 3;
  if (idxHoras   === -1) idxHoras   = 9;

  // PARSEO ROBUSTO DE HORAS
  const parsearHoras = (valor) => {
    if (typeof valor === 'number' && !isNaN(valor) && valor > 0) return valor;
    if (!valor) return 42; 
    const n = parseFloat(String(valor).replace(',', '.'));
    return (!isNaN(n) && n > 0) ? n : 42;
  };

  for (let i = 1; i < dataHC.length; i++) {
    const bms = String(dataHC[i][idxBMS]).trim();
    if (!bms) continue;
    
    // Si hay filtro ACCM, aplicarlo
    if (filtroACCM) {
        const accmFila = String(dataHC[i][5] || '').trim(); // col F = ACCM
        if (normalizarLocal(accmFila) !== normalizarLocal(filtroACCM)) continue;
    }

    agentes[bms] = { 
        nombre: String(dataHC[i][idxNombre]).trim(), 
        lob: dataHC[i][idxLOB], 
        rol: String(dataHC[i][idxRol] || '').trim(),
        horasSemanales: parsearHoras(dataHC[i][idxHoras])
    };
  }
  
  Logger.log(`✅ REGISTRO_obtenerAgentesPermitidos: ${Object.keys(agentes).length} agentes cargados`);
  return agentes;
}

function REGISTRO_obtenerDiasLaborablesMes(numeroMes, anio) {
    const primerDia = new Date(anio, numeroMes - 1, 1);
    const ultimoDia = new Date(anio, numeroMes, 0);
    
    let diasLaborables = 0;
    let dia = new Date(primerDia);
    
    while (dia <= ultimoDia) {
        const diaSemana = dia.getDay();
        if (diaSemana !== 0 && diaSemana !== 6) {
            diasLaborables++;
        }
        dia.setDate(dia.getDate() + 1);
    }
    
    Logger.log(`📅 Mes ${numeroMes}/${anio}: ${diasLaborables} días laborables`);
    return diasLaborables;
}

        errorSimulacion: sim.success ? null : sim.error
      });
    }
    return { lobs: resultados };
  } catch (e) {
    return { error: e.toString() };
  }
}

/**
 * Orquestador principal para el frontend de OM
 */
function OM_obtenerDatosParaFrontend(lob, numeroMes, fechaSeleccionada) {
  try {
    const usuario = OM_obtenerDatosUsuarioActivo();
    if (!usuario) return { error: "No se identificó el usuario activo" };
    
    // Si numeroMes es nulo, intentar resolverlo
    if (!numeroMes) {
      const mesesMap = {
        'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
        'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
        'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
      };
      const mesActual = new Date().toLocaleString('es-ES', { month: 'long' }).toLowerCase();
      numeroMes = mesesMap[mesActual];
    }

    const resultado = {
      usuario: usuario,
      metricasPorLob: {},
      horasRealesPorAgente: {},
      regiones: OM_obtenerTodasLasRegiones()
    };

    // Determinar LOBs a procesar
    let lobsAProcesar = [];
    let lobsParaDropdown = [];

    if (usuario.rol === 'Reporting Analyst' || usuario.rol === 'Audit') {
      lobsParaDropdown = OM_obtenerTodasLasLobs();
    } else if (usuario.rol === 'OM') {
      lobsParaDropdown = OM_obtenerLobsPorOM(usuario.nombre);
    } else if (usuario.rol === 'ACCM') {
      lobsParaDropdown = OM_obtenerLobsPorACCM(usuario.nombre);
    } else {
      lobsParaDropdown = [usuario.lob];
    }
    
    resultado.lobs = lobsParaDropdown;

    if (!lob) {
      // Si no hay lob seleccionada, solo retornamos los datos iniciales
      return resultado;
    }

    if (lob.startsWith("REGION:")) {
      const regionNombre = lob.replace("REGION:", "").trim();
      const resumenRegion = OM_obtenerResumenLobsPorRegion(regionNombre, numeroMes);
      if (resumenRegion.error) return { error: resumenRegion.error };
      lobsAProcesar = resumenRegion.lobs.filter(l => l.tieneCDM).map(l => l.lob);
      if (lobsAProcesar.length === 0) return { error: `No se encontraron LOBs con datos CDM en la región ${regionNombre}` };
    } else if (lob === "TODAS LAS LOBS") {
      lobsAProcesar = lobsParaDropdown;
    } else {
      lobsAProcesar = [lob];
    }

    // Procesar cada LOB usando la lógica de simulación
    for (const lobActual of lobsAProcesar) {
      const sim = OM_simularConMetricasReales(lobActual, null, numeroMes); // null mes porque simularConMetricasReales lo resuelve
      if (sim.success) {
        // Flatten categories and include important summary metrics for frontend compatibility
        resultado.metricasPorLob[lobActual] = {
          ...sim.desglosePorCategoria,
          success: true,
          inversionTeorica: sim.inversionTeorica,
          inversionReal: sim.inversionReal,
          hc: sim.hc,
          mes: sim.mes,
          numeroMes: sim.numeroMes,
          ahorro: sim.ahorro,
          eficienciaReal: sim.eficienciaReal,
          estadisticas: sim.estadisticas,
          configuracion: sim.configuracion
        };
        // Combinar horas reales para el frontend
        const horas = OM_obtenerHorasPorAgente(lobActual, numeroMes);
        Object.assign(resultado.horasRealesPorAgente, horas);
      } else {
        resultado.metricasPorLob[lobActual] = { error: sim.error };
      }
    }

    return resultado;
  } catch (error) {
    return { error: error.message };
  }
}

/**
 * ⭐ FUNCIÓN: Orquestador de validaciones por categoría (Alineada con OM)
 */
function OM_validarKPIsPorCategoria(registro, lob, dataCDM, dataUsuario, hoja, agentesPermitidos, numeroMes = null, fechaSeleccionada = null) {
    const resultadosBrutos = { 
        keys: [], 
        bonusStructure: [], 
        additionalIncentives: [], 
        decreasedKPIs: [] 
    };

    const procesar = (cat, start, end, inc, conParticipacion) => {
        for (let c = start; c <= end; c += inc) {
            if (registro[c] && registro[c] !== '-') {
                const kpi = String(registro[c]).trim();
                const scoreRaw = String(registro[c + 1]).trim();

                // ================================================================
                // ✅ DETECTAR GATE: termina en ",key" o ",gate" ignorando espacios/mayúsculas (PARIDAD OM)
                // ================================================================
                const scoreRawLower = scoreRaw.toLowerCase().replace(/\s/g, '');
                const esGate = cat === 'bonusStructure' && (scoreRawLower.endsWith(',key') || scoreRawLower.endsWith(',gate'));

                let score = scoreRaw;

                if (esGate) {
                    // Limpiar ",key" o ",gate" del score
                    score = scoreRaw.replace(/\s*,\s*(key|gate)\s*$/i, '').trim();
                    Logger.log(`🔒 Gate detectado: KPI="${kpi}" | scoreRaw="${scoreRaw}" | scoreLimpio="${score}"`);
                }

                // Normalizar coma decimal DESPUÉS de limpiar el gate (PARIDAD OM)
                score = score.replace(/(\d),(\d)/g, '$1.$2');
                let peso = parseFloat(String(registro[c + 2] || '0').replace('%', '')) || 0;
                if (peso > 1) peso /= 100;

                let participation = 1;
                if (conParticipacion) {
                    const pRaw = String(registro[c + 3] || '100');
                    const pNum = parseFloat(pRaw.replace('%', '').trim()) || 100;
                    participation = pNum > 1 ? pNum / 100 : pNum;
                }
                
                const validacion = OM_validarKPI1(
                    kpi, 
                    score, 
                    peso * participation, 
                    dataCDM, 
                    dataUsuario, 
                    agentesPermitidos, 
                    lob, 
                    numeroMes, 
                    cat === 'decreasedKPIs'
                );
                
                resultadosBrutos[cat].push({ 
                    kpi, 
                    score, 
                    porcentaje: peso, 
                    participacion: participation, 
                    validacion,
                    esDecrease: cat === 'decreasedKPIs',
                    esGate: esGate
                });
            }
        }
    };

    if (hoja === 'Registro') {
        // Estructura Estándar (Triplets)
        procesar('keys', 5, 8, 3, false);
        procesar('bonusStructure', 10, 21, 3, false);
        procesar('additionalIncentives', 23, 31, 3, false);
        procesar('decreasedKPIs', 33, 38, 3, false);

        // ⭐ PARIDAD: Aplicar Gate
        const resGate = OM_aplicarLogicaGate(resultadosBrutos.bonusStructure, resultadosBrutos.additionalIncentives);
        resultadosBrutos.bonusStructure      = resGate.bonusStructure;
        resultadosBrutos.additionalIncentives = resGate.additionalIncentives;

        return resultadosBrutos;
    } else {
        // Estructura Registro_pisos (Multi-Tier)
        procesar('keys', 5, 10, 4, false);
        procesar('bonusStructure', 12, 79, 4, true); 
        procesar('additionalIncentives', 81, 100, 4, true);
        procesar('decreasedKPIs', 102, 121, 4, true);
        
        const consolidado = OM_consolidarPisos(resultadosBrutos);
        
        // ⭐ PARIDAD: Aplicar Gate
        const resGateCons = OM_aplicarLogicaGate(consolidado.bonusStructure, consolidado.additionalIncentives);
        consolidado.bonusStructure      = resGateCons.bonusStructure;
        consolidado.additionalIncentives = resGateCons.additionalIncentives;

        return consolidado;
    }
}

/**
 * ⭐ FUNCIÓN CORE: validarKPI1 (Alineada con OM)
 * Calcula el bono individual para un KPI específico.
 */

function OM_validarKPI1(llave, condicionScore, porcentajePunto, dataCDM, dataUsuario, agentesPermitidos, lobActual, numeroMes = null, esDecrease = false) {
    const resultado = { 
        llave, 
        condicionScore, 
        totalCoincidencias: 0, 
        totalCumplenCondicion: 0, 
        empleadosCumplen: [], 
        empleadosNoCumplen: [], 
        empleadosSinDatos: [] 
    };

    if (!condicionScore || condicionScore === 'N/A' || condicionScore.trim() === '') {
        return resultado;
    }

    const normalizar = s => String(s || '')
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase()
        .trim()
        .replace(/\s+/g, ' ');
    
    const llaveNorm     = normalizar(llave);
    const lobActualNorm = normalizar(lobActual);

    // ========================================
    // ⭐ DETECTAR TIPO DE CONDICIÓN
    // - Tiene operador (>=, <=, >, <, =) → condición normal
    // - Número puro sin operador (ej: "7.5") → INVÁLIDO → bono $0
    // - Texto con $ sin operador → bono fijo real
    // ========================================
    const tieneOperador = /(<=|>=|<|>|=)/.test(condicionScore);
    const esSoloNumero  = /^\d+([.,]\d+)?$/.test(condicionScore.trim());
    const esBonoFijo    = !tieneOperador && !esSoloNumero;

    let valorBonoFijo  = 0;
    let operador       = null;
    let valorCondicion = 0;

    if (esSoloNumero) {
        // ⭐ Número puro sin operador → bono $0 para todos
        Logger.log(`⚠️ Score "${condicionScore}" sin operador en KPI "${llave}" → bono $0`);
        const todosAgentes = OM_obtenerTodosLosAgentesDeLOB(lobActual, agentesPermitidos);
        todosAgentes.forEach(ag => {
            resultado.empleadosSinDatos.push({
                bmsId:    ag.bmsId,
                nombre:   ag.nombre,
                valorF:   '-',
                bono:     0,
                sinDatos: true,
                razon:    `Score "${condicionScore}" sin operador — agrega >= o <= en el Registro`
            });
        });
        return resultado;

    } else if (esBonoFijo) {
        valorBonoFijo = parseFloat(condicionScore.replace(/[^\d.]/g, '')) || 0;
        Logger.log(`💰 Bono Fijo: ${llave} = $${valorBonoFijo}`);

    } else {
        const match = condicionScore.match(/(<=|>=|<|>|=)\s*(\d+(?:[.,]\d+)?)/);
        if (!match) {
            Logger.log('❌ Condición no válida:', condicionScore);
            return resultado;
        }
        operador       = match[1];
        valorCondicion = parseFloat(match[2].replace(',', '.'));
    }

    const todosLosAgentes = OM_obtenerTodosLosAgentesDeLOB(lobActual, agentesPermitidos);

    // ========================================
    // CONFIGURACIÓN LOB
    // ========================================
    const configLOB          = OM_obtenerConfiguracionDesdeConfiguracion(lobActual);
    const salarioBaseLOB     = configLOB.encontrado ? configLOB.salario    : 2150000;
    const porcentajeBonoGlobal = configLOB.encontrado ? configLOB.porcentaje : 30;

    // ========================================
    // HORAS MÁXIMAS DESDE H_C
    // ========================================
    const horasMaxima = Math.max(
        ...Object.values(agentesPermitidos).map(a => a.horasSemanales || 0)
    ) || 42;

    Logger.log(`⏱️ Horas máximas H_C: ${horasMaxima}h`);

    // ========================================
    // COLUMNAS CDM
    // ========================================
    const headers = dataCDM[0];
    const getIdx  = (names, def) => {
        const norm = names.map(n => String(n).toLowerCase().trim());
        const idx  = headers.findIndex(h => norm.includes(String(h || '').toLowerCase().trim()));
        return idx === -1 ? def : idx;
    };

    const idxBms   = getIdx(['BMS ID', 'CCMSID', 'ID'], 1);
    const idxValH  = getIdx(['Columna H', 'Score', 'KPI_Value'], 7);
    const idxValF  = getIdx(['Columna F', 'Value'], 5);
    const idxLlave = getIdx(['Llave', 'KPI'], 8);
    const idxSubLOB = getIdx(['SubLOB'], 2);

    // ========================================
    // ⭐ VERIFICAR SI EL KPI EXISTE EN CDM
    // ========================================
    const kpiExisteEnCDM = !esBonoFijo && dataCDM.slice(1).some(row => {
        const llaveCDM = normalizar(String(row[idxLlave] || '').trim());
        let kpiExtraido = '';
        const posLOB = llaveCDM.indexOf(lobActualNorm);
        if (posLOB !== -1) {
            kpiExtraido = llaveCDM.substring(posLOB + lobActualNorm.length).trim();
        } else {
            kpiExtraido = llaveCDM;
        }
        return kpiExtraido === llaveNorm;
    });

    if (!esBonoFijo && !kpiExisteEnCDM) {
        Logger.log(`⚠️ KPI "${llave}" NO existe en CDM para LOB "${lobActual}" → todos $0`);
    }

    const agentesConDatos = new Set();

    // ========================================
    // CASO A: BONO FIJO
    // ========================================
    if (esBonoFijo) {
        todosLosAgentes.forEach(ag => {
            let salario = salarioBaseLOB;
            let salarioPersonalizado = false;

            for (let u = 1; u < dataUsuario.length; u++) {
                if (String(dataUsuario[u][0]).trim() === ag.bmsId) {
                    const sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                    const sNum = parseFloat(sRaw) || 0;
                    if (sNum > 0) {
                        salario = sNum;
                        if (sNum !== salarioBaseLOB) salarioPersonalizado = true;
                    }
                    break;
                }
            }

            const horasAgente = agentesPermitidos[ag.bmsId]?.horasSemanales || horasMaxima;
            const factorHoras = horasMaxima > 0 ? horasAgente / horasMaxima : 1;
            const bonoBase    = salario * (porcentajeBonoGlobal / 100);
            let   bonoFinal   = bonoBase * porcentajePunto * factorHoras;

            if (esDecrease) bonoFinal = -Math.abs(bonoFinal);

            resultado.totalCoincidencias++;
            resultado.totalCumplenCondicion++;
            resultado.empleadosCumplen.push({
                bmsId: ag.bmsId,
                nombre: ag.nombre,
                valorF: valorBonoFijo,
                bono: bonoFinal,
                horasContrato: horasAgente,
                horasMaxima,
                factorHoras,
                salarioPersonalizado
            });
        });
        return resultado;
    }

    // ========================================
    // CASO B: KPI CON CONDICIÓN
    // ========================================
    for (let row = 1; row < dataCDM.length; row++) {
        const llaveCDM = normalizar(String(dataCDM[row][idxLlave] || '').trim());
        
        let kpiExtraido = '';
        const posLOB = llaveCDM.indexOf(lobActualNorm);
        if (posLOB !== -1) {
            kpiExtraido = llaveCDM.substring(posLOB + lobActualNorm.length).trim();
        } else {
            kpiExtraido = llaveCDM;
        }

        if (kpiExtraido !== llaveNorm) continue;

        const bmsId = String(dataCDM[row][idxBms]).trim();
        if (!agentesPermitidos[bmsId]) continue;

        const lobAgente = normalizar(agentesPermitidos[bmsId].lob);
        if (lobAgente !== lobActualNorm) continue;

        const subLobCDM = normalizar(String(dataCDM[row][idxSubLOB] || ''));
        if (subLobCDM && subLobCDM !== lobActualNorm) continue;

        if (agentesConDatos.has(bmsId)) continue;
        agentesConDatos.add(bmsId);
        resultado.totalCoincidencias++;

        const valorNumerico = parseFloat(String(dataCDM[row][idxValH]).replace('%', '').trim());

        if (isNaN(valorNumerico) || valorNumerico === 0) {
            resultado.empleadosSinDatos.push({
                bmsId,
                nombre:              agentesPermitidos[bmsId].nombre,
                valorF:              0,
                bono:                0,
                sinDatos:            true,
                salarioPersonalizado: false
            });
            continue;
        }

        let cumple = false;
        switch (operador) {
            case '>':  cumple = valorNumerico >  valorCondicion; break;
            case '>=': cumple = valorNumerico >= valorCondicion; break;
            case '<':  cumple = valorNumerico <  valorCondicion; break;
            case '<=': cumple = valorNumerico <= valorCondicion; break;
            case '=':  cumple = valorNumerico === valorCondicion; break;
        }

        let salario = salarioBaseLOB;
        let salarioPersonalizado = false;
        for (let u = 1; u < dataUsuario.length; u++) {
            if (String(dataUsuario[u][0]).trim() === bmsId) {
                const sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                const sNum = parseFloat(sRaw) || 0;
                if (sNum > 0) {
                    salario = sNum;
                    if (sNum !== salarioBaseLOB) salarioPersonalizado = true;
                }
                break;
            }
        }
        if (salario === 0) salario = salarioBaseLOB;

        const horasAgente = agentesPermitidos[bmsId]?.horasSemanales || horasMaxima;
        const factorHoras = horasMaxima > 0 ? horasAgente / horasMaxima : 1;
        const bonoBase    = salario * (porcentajeBonoGlobal / 100);
        let   bonoFinal   = bonoBase * porcentajePunto * factorHoras;

        if (esDecrease && cumple) bonoFinal = -Math.abs(bonoFinal);

        const item = {
            bmsId,
            nombre:               agentesPermitidos[bmsId].nombre,
            valorF:               dataCDM[row][idxValF],
            valorReal:            valorNumerico,
            bono:                 cumple ? bonoFinal : 0,
            horasContrato:        horasAgente,
            horasMaxima,
            factorHoras,
            salarioPersonalizado
        };

        if (cumple) {
            resultado.totalCumplenCondicion++;
            resultado.empleadosCumplen.push(item);
        } else {
            resultado.empleadosNoCumplen.push(item);
        }
    }

    // ========================================
    // ⭐ AGENTES SIN REGISTRO EN CDM
    // Distingue KPI inexistente vs agente sin fila
    // ========================================
    todosLosAgentes.forEach(ag => {
        if (!agentesConDatos.has(ag.bmsId)) {
            if (!kpiExisteEnCDM) {
                resultado.empleadosSinDatos.push({
                    bmsId:    ag.bmsId,
                    nombre:   ag.nombre,
                    valorF:   '-',
                    bono:     0,
                    sinDatos: true,
                    salarioPersonalizado: false,
                    razon:    'KPI no encontrado en CDM — verifica nombre en Registro'
                });
            } else {
                resultado.empleadosNoCumplen.push({
                    bmsId:    ag.bmsId,
                    nombre:   ag.nombre,
                    valorF:   '-',
                    bono:     0,
                    sinDatos: false,
                    salarioPersonalizado: false,
                    razon:    'Agente sin registro en CDM para este KPI'
                });
            }
        }
    });

    Logger.log(`\n📊 RESUMEN KPI "${llave}":`);
    Logger.log(`   KPI existe en CDM: ${kpiExisteEnCDM ? '✅' : '❌'}`);
    Logger.log(`   Coincidencias: ${resultado.totalCoincidencias}`);
    Logger.log(`   Cumplen: ${resultado.totalCumplenCondicion}`);
    Logger.log(`   No cumplen: ${resultado.empleadosNoCumplen.length}`);
    Logger.log(`   Sin datos: ${resultado.empleadosSinDatos.length}\n`);

    return resultado;
}

function fichaKPIPorLob(lob, fechaSeleccionada, numeroMes) {
  try {
    Logger.log('\n🟦 ========== INICIO fichaKPIPorLob ==========');
    Logger.log('LOB: ' + lob);
    Logger.log('Fecha: ' + fechaSeleccionada);
    Logger.log('Mes: ' + numeroMes);

    if (!lob || !fechaSeleccionada) {
      return { error: 'Se requiere LOB y Fecha.' };
    }

    // ── 1. Agentes de H_C ──────────────────────────────────
    const agentes = fk_obtenerAgentes();
    if (!Object.keys(agentes).length) {
      return { error: 'No se encontraron agentes en H_C.' };
    }
    Logger.log('✅ Agentes H_C: ' + Object.keys(agentes).length);

    // ── 2. Mes numérico ────────────────────────────────────
    const mes = numeroMes
      ? parseInt(numeroMes, 10)
      : fk_mesDeFecha(fechaSeleccionada);

    // ── 3. CDM ─────────────────────────────────────────────
    const dataCDM = mes ? fk_obtenerCDM(mes) : [];
    Logger.log('📋 Filas CDM: ' + dataCDM.length);

    // ── 4. Usuarios (salarios personalizados) ──────────────
    const ss = SpreadsheetApp.openById(OM_ID_SS_PRINCIPAL_LOCAL);
    const dataUsuario = ss.getSheetByName('Usuario').getDataRange().getValues();

    // ── 5. Horas reales ────────────────────────────────────
    const horasMap = mes ? fk_obtenerHoras(lob, mes) : {};
    Logger.log('⏱ Agentes con horas: ' + Object.keys(horasMap).length);

    // ── 6. Configuración salario LOB ───────────────────────
    const config = fk_obtenerConfigLOB(lob);
    if (!config) {
      return { error: 'No se encontró configuración de salario para LOB: ' + lob };
    }
    Logger.log('💰 Config LOB: salario=' + config.salarioBase + ' bono=' + config.porcentajeBono + '%');

    // ── 7. Registros (Registro / Registro_pisos) ──────────
    const registros = fk_obtenerRegistros(lob, fechaSeleccionada);
    if (!registros.length) {
      return { error: 'No hay registros para LOB: ' + lob + ' / Fecha: ' + fechaSeleccionada };
    }
    
    // ⭐ PARIDAD: Extraer la fecha exacta del registro encontrado
    const firstReg = registros[0].registro;
    const fechaExacta = firstReg[0] instanceof Date
        ? Utilities.formatDate(firstReg[0], Session.getScriptTimeZone(), 'yyyy-MM-dd')
        : String(firstReg[0]);

    Logger.log('📂 Registro encontrado: ' + fechaExacta + ' (Hoja: ' + registros[0].hoja + ')');

    // ── 8. Horas máximas en H_C (base prorrateo) ──────────
    const horasMaxima = Math.max.apply(null,
      Object.values(agentes).map(function(a) { return a.horasSemanales || 0; }).concat([1])
    );

    // ── 9. Calcular KPIs por categoría para cada registro ──
    const resultadosBrutos = {
      keys: [], bonusStructure: [], additionalIncentives: [], decreasedKPIs: []
    };

    registros.forEach(function(regData) {
      const val = fk_validarCategorias(
        regData.registro,
        lob,
        dataCDM,
        dataUsuario,
        regData.hoja,
        agentes,
        config,
        horasMaxima,
        mes,
        fechaSeleccionada
      );
      ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(function(cat) {
        if (val[cat] && val[cat].length) {
          resultadosBrutos[cat] = resultadosBrutos[cat].concat(val[cat]);
        }
      });
    });

    // ── 10. Construir ficha por agente ────────────────────
    const fichaAgentes = fk_construirFichaAgentes(
      resultadosBrutos, agentes, horasMap, lob
    );

    // ── 11. Totales ────────────────────────────────────────
    const totalGeneral = Object.values(fichaAgentes)
      .reduce(function(s, a) { return s + (a.bonoFinal || 0); }, 0);
    const conBono = Object.values(fichaAgentes)
      .filter(function(a) { return (a.bonoFinal || 0) > 0; }).length;

    Logger.log('💰 Total general: $' + totalGeneral.toFixed(2));
    Logger.log('👥 Agentes totales: ' + Object.keys(fichaAgentes).length);
    Logger.log('✅ Con bono: ' + conBono);
    Logger.log('🟦 ========== FIN fichaKPIPorLob ==========\n');

    return {
      lob:              lob,
      fecha:            fechaSeleccionada,
      mes:              mes,
      fichaAgentes:     fichaAgentes,
      kpisPorCategoria: resultadosBrutos,
      totales: {
        totalGeneral:   totalGeneral,
        totalAgentes:   Object.keys(fichaAgentes).length,
        agentesConBono: conBono,
        promedio: Object.keys(fichaAgentes).length > 0
          ? totalGeneral / Object.keys(fichaAgentes).length
          : 0
      }
    };

  } catch (e) {
    Logger.log('❌ ERROR fichaKPIPorLob: ' + e.message + '\n' + e.stack);
    return { error: e.message };
  }
}


// ============================================================
//  fk_obtenerAgentes()
// ============================================================
function fk_obtenerAgentes() {
  const ss     = SpreadsheetApp.openById(OM_ID_SS_PRINCIPAL_LOCAL);
  const hoja   = ss.getSheetByName('H_C');
  if (!hoja) return {};

  const data    = hoja.getDataRange().getValues();
  const headers = data[0];

  let idxBMS = -1, idxLOB = -1, idxNombre = -1, idxRol = -1, idxHoras = -1;
  headers.forEach(function(h, i) {
    const hh = String(h).toLowerCase().trim();
    if (hh.includes('bms') || hh.includes('ccms') || hh.includes('id')) idxBMS    = i;
    if (hh.includes('lob'))                                               idxLOB    = i;
    if (hh.includes('nombre') || hh.includes('name'))                    idxNombre = i;
    if (hh.includes('rol') || hh.includes('role'))                       idxRol    = i;
    if (hh.includes('horas') || hh.includes('hours') || hh === 'h')     idxHoras  = i;
  });
  if (idxBMS    === -1) idxBMS    = 0;
  if (idxLOB    === -1) idxLOB    = 1;
  if (idxNombre === -1) idxNombre = 2;
  if (idxRol    === -1) idxRol    = 3;
  if (idxHoras  === -1) idxHoras  = 9;

  const parsearH = function(v) {
    if (typeof v === 'number' && !isNaN(v) && v > 0) return v;
    const n = parseFloat(String(v || '').replace(/[^0-9.,]/g, '').replace(',', '.'));
    return (!isNaN(n) && n > 0) ? n : 42;
  };

  const agentes = {};
  for (let i = 1; i < data.length; i++) {
    const bms = String(data[i][idxBMS] || '').trim();
    if (!bms) continue;
    const lob   = data[i][idxLOB];
    const horas = parsearH(data[i][idxHoras]);
    const rol   = String(data[i][idxRol] || '').trim() || 'Sin rol';

    agentes[bms] = {
      nombre:         String(data[i][idxNombre] || '').trim() || ('Agente ' + bms),
      rol:            rol,
      lob:            fk_norm(lob),
      lobOriginal:    String(lob || '').trim(),
      horasSemanales: horas,
      supervisor:     String(data[i][4] || '').trim(),
      accm:           String(data[i][5] || '').trim(),
      om:             String(data[i][6] || '').trim()
    };
  }
  return agentes;
}


// ============================================================
//  fk_obtenerHoras(lob, mes)
// ============================================================
function fk_obtenerHoras(lob, mes) {
  if (!mes) return {};
  const ss = SpreadsheetApp.openById(OM_ID_SS_CDM);

  const variantes = [
    'HORAS_' + mes, 'Horas_' + mes,
    'HORAS ' + mes, 'Horas ' + mes,
    'HORAS_' + String(mes).padStart(2, '0'),
    'Horas_' + String(mes).padStart(2, '0')
  ];

  let hoja = null;
  for (const v of variantes) {
    hoja = ss.getSheetByName(v);
    if (hoja) break;
  }
  if (!hoja) return {};

  const data = hoja.getDataRange().getValues();
  if (!data || data.length < 2) return {};

  const headers = data[0];
  let colCCMS = -1, colHoras = -1, colAus = -1, colLOB = -1;
  const colDias = 6;

  headers.forEach(function(h, i) {
    const hh = String(h).trim();
    if (hh === 'CCMSID' || hh === 'BMS ID' || hh === 'ID') colCCMS  = i;
    if (hh === 'horas'  || hh === 'Horas')                 colHoras = i;
    if (hh === 'Ausencias' || hh === 'ausencias')          colAus   = i;
    if (hh === 'SubLOB' || hh === 'LOB')                   colLOB   = i;
  });

  if (colCCMS === -1 || colHoras === -1) return {};

  const toN  = function(v) {
    return typeof v === 'number' ? v : (parseFloat(String(v).replace(',', '.')) || 0);
  };
  const lobN = lob ? String(lob).trim() : null;
  const mapa = {};

  for (let i = 1; i < data.length; i++) {
    const ccms  = String(data[i][colCCMS] || '').trim();
    const bmsId = (ccms.match(/(\d{7})/) || [])[1];
    if (!bmsId) continue;

    if (lobN && colLOB !== -1) {
      if (String(data[i][colLOB] || '').trim() !== lobN) continue;
    }

    const hTot = toN(data[i][colHoras]);
    const hAus = colAus !== -1 ? toN(data[i][colAus]) : 0;
    const dias  = toN(data[i][colDias]);

    mapa[bmsId] = {
      horas:         hTot,
      horasTotales:  hTot,
      ausencias:     hAus,
      efectivas:     Math.max(0, hTot - hAus),
      diasLaborados: dias
    };
  }
  return mapa;
}

  if (!hoja) return null;

  const data = hoja.getDataRange().getValues();

}


// ============================================================
//  fk_obtenerRegistros(lob, fecha)
// ============================================================
function fk_obtenerRegistros(lob, fecha) {
  const ss     = SpreadsheetApp.openById(OM_ID_SS_PRINCIPAL_LOCAL);
  const hojas  = ['Registro', 'Registro_pisos'];
  const result = [];
  const lobN   = fk_norm(lob);
  const fechaN = fk_norm(fecha);

  // ⭐ FIX: solo tomar el PRIMER match encontrado entre ambas hojas
  // para evitar procesar la misma LOB/fecha desde dos hojas
  let encontrado = false;

  for (const nombreHoja of hojas) {
    if (encontrado) break;  // ⭐ parar en cuanto se encuentre en una hoja
    
    const hoja = ss.getSheetByName(nombreHoja);
    if (!hoja) continue;
    const data = hoja.getDataRange().getValues();

    for (let i = 1; i < data.length; i++) {
      const fila   = data[i];
      const fFecha = fila[0];
      const fLob   = fila[1];

      if (!fFecha || !fLob) continue;
      if (fk_norm(String(fLob)) !== lobN) continue;

      const fechaTxt = fFecha instanceof Date
        ? Utilities.formatDate(fFecha, Session.getScriptTimeZone(), 'yyyy-MM-dd')
        : String(fFecha).trim();

      // ⭐ PARIDAD: Intentar match exacto PRIMERO
      let matches = (fk_norm(fechaTxt) === fechaN);

      // ⭐ FALLBACK: Si no hay match exacto, ver si la búsqueda es por NOMBRE DE MES
      if (!matches) {
          const numMesBusqueda = _sim_resolverNumeroMes(fecha); // "Mayo" -> 5
          const numMesRegistro = _sim_resolverNumeroMes(fechaTxt); // "2024-05-15" -> 5
          if (numMesBusqueda && numMesBusqueda === numMesRegistro) {
              matches = true;
          }
      }

      if (!matches) continue;
      
      result.push({ registro: fila, hoja: nombreHoja });
      encontrado = true;
      break;
    }
  }
  return result;
}


// ============================================================
//  fk_validarCategorias(...)
// ============================================================
function fk_validarCategorias(registro, lob, dataCDM, dataUsuario,
                               hoja, agentes, config, horasMaxima, mes, fecha) {
  const brutos = {
    keys: [], bonusStructure: [], additionalIncentives: [], decreasedKPIs: []
  };

  const procesar = function(cat, start, end, inc, conParticipacion) {
    for (let c = start; c <= end; c += inc) {
      if (!registro[c]) continue;
      const kpi      = String(registro[c]).trim();
      const scoreRaw = String(registro[c + 1]).trim();
      if (!kpi || !scoreRaw) continue;

      const scoreRawLower = scoreRaw.toLowerCase().replace(/\s/g, '');
      const esGate = cat === 'bonusStructure' && (scoreRawLower.endsWith(',key') || scoreRawLower.endsWith(',gate'));
      let score    = esGate ? scoreRaw.replace(/\s*,\s*(key|gate)\s*$/i, '').trim() : scoreRaw;
      score = score.replace(/(\d),(\d)/g, '$1.$2');

      let peso = parseFloat(String(registro[c + 2] || '0').replace('%', '')) || 0;
      if (peso > 1) peso /= 100;

      let participation = 1;
      if (conParticipacion) {
        const pRaw = String(registro[c + 3] || '100');
        const pNum = parseFloat(pRaw.replace('%', '').trim()) || 100;
        participation = pNum > 1 ? pNum / 100 : pNum;
      }

      const esDecrease = cat === 'decreasedKPIs';

      const validacion = fk_validarKPI(
        kpi, score, peso * participation,
        dataCDM, dataUsuario,
        agentes, lob,
        config, horasMaxima,
        mes, fecha, esDecrease
      );

      brutos[cat].push({
        kpi, score, scoreOriginal: scoreRaw,
        porcentaje: peso, participacion: participation,
        esGate, validacion, esDecrease
      });
    }
  };

  if (hoja === 'Registro') {
    procesar('keys',                5,  9,  2, false);
    procesar('bonusStructure',      10, 21, 3, false);
    procesar('additionalIncentives',23, 31, 3, false);
    procesar('decreasedKPIs',       33, 38, 3, false);
    const resGate = fk_aplicarGate(brutos.bonusStructure, brutos.additionalIncentives);
    brutos.bonusStructure      = resGate.bonusStructure;
    brutos.additionalIncentives = resGate.additionalIncentives;
    return brutos;
  } else {
    procesar('keys',                5,   10,  2, false);
    procesar('bonusStructure',      12,  79,  4, true);
    procesar('additionalIncentives',81, 100,  4, true);
    procesar('decreasedKPIs',       102, 121, 4, true);
    const consolidado = fk_consolidarMultiTier(brutos);
    
    const resGateCons = fk_aplicarGate(consolidado.bonusStructure, consolidado.additionalIncentives);
    consolidado.bonusStructure      = resGateCons.bonusStructure;
    consolidado.additionalIncentives = resGateCons.additionalIncentives;
    return consolidado;
  }
}


// ============================================================
//  fk_validarKPI(...)
// ============================================================
function fk_validarKPI(llave, condicionScore, porcentajePunto,
                        dataCDM, dataUsuario, agentes, lobActual,
                        config, horasMaxima, mes, fecha, esDecrease) {

  const resultado = {
    llave, condicionScore,
    totalCoincidencias:    0,
    totalCumplenCondicion: 0,
    empleadosCumplen:   [],
    empleadosNoCumplen: [],
    empleadosSinDatos:  []
  };

  if (!condicionScore || condicionScore === 'N/A' || !condicionScore.trim()) {
    return resultado;
  }

  const llaveN = fk_norm(llave);
  const lobN   = fk_norm(lobActual);

  const esBonoFijo = !condicionScore.match(/(<=|>=|<|>|=)/);
  let valorBonoFijo = 0, operador = null, valorCondicion = 0;

  if (esBonoFijo) {
    valorBonoFijo = parseFloat(condicionScore.replace(/[^\d.]/g, '')) || 0;
  } else {
    const match = condicionScore.match(/(<=|>=|<|>|=)\s*(\d+(?:\.\d+)?)/);
    if (!match) return resultado;
    operador       = match[1];
    valorCondicion = parseFloat(match[2]);
  }

  const agentesLOB = Object.entries(agentes)
    .filter(function(kv) { return fk_norm(kv[1].lob) === lobN; })
    .map(function(kv) { return { bmsId: kv[0], nombre: kv[1].nombre }; });

  const getSalario = function(bmsId) {
    for (let u = 1; u < dataUsuario.length; u++) {
      if (String(dataUsuario[u][0]).trim() === bmsId) {
        const s = parseFloat(String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.')) || 0;
        if (s > 0) return { salario: s, personalizado: s !== config.salarioBase };
      }
    }
    return { salario: config.salarioBase, personalizado: false };
  };

  const calcBono = function(bmsId, cumple) {
    const { salario, personalizado } = getSalario(bmsId);
    const bonoBase = salario * (config.porcentajeBono / 100);
    const hAg      = (agentes[bmsId] && agentes[bmsId].horasSemanales) || horasMaxima;
    const factor   = horasMaxima > 0 ? hAg / horasMaxima : 1;
    let bonoFinal  = bonoBase * porcentajePunto * factor;
    if (esDecrease && cumple) bonoFinal = -Math.abs(bonoFinal);
    return { bonoFinal, factor, hAg, salario, personalizado };
  };

  // ── BONO FIJO ─────────────────────────────────────────────
  if (esBonoFijo) {
    agentesLOB.forEach(function(ag) {
      const { bonoFinal, factor, hAg, personalizado } = calcBono(ag.bmsId, true);
      resultado.totalCoincidencias++;
      resultado.totalCumplenCondicion++;
      resultado.empleadosCumplen.push({
        bmsId:             ag.bmsId,
        nombre:            ag.nombre,
        valorF:            valorBonoFijo,
        bono:              bonoFinal,
        horasContrato:     hAg,
        horasMaxima:       horasMaxima,
        factorHoras:       factor,
        salarioPersonalizado: personalizado
      });
    });
    return resultado;
  }

  // ── KPI CON CONDICIÓN ─────────────────────────────────────
  if (!dataCDM || dataCDM.length < 2) return resultado;

  const headers = dataCDM[0] || [];

  let idxBms    = headers.indexOf('BMS ID');
  if (idxBms    === -1) idxBms    = headers.indexOf('CCMSID'); if (idxBms    === -1) idxBms    = 1;
  let idxValH   = headers.indexOf('Columna H');
  if (idxValH   === -1) idxValH   = headers.indexOf('KPI_Value'); if (idxValH === -1) idxValH  = 7;
  let idxValF   = headers.indexOf('Columna F');
  if (idxValF   === -1) idxValF   = 5;
  let idxLlave  = headers.indexOf('Llave');
  if (idxLlave  === -1) idxLlave  = headers.indexOf('KPI'); if (idxLlave === -1) idxLlave = 8;
  let idxSubLOB = headers.indexOf('SubLOB');
  if (idxSubLOB === -1) idxSubLOB = 2;

  const agentesConDatos = new Set();

  for (let row = 1; row < dataCDM.length; row++) {
    const llaveCDM    = fk_norm(String(dataCDM[row][idxLlave] || '').trim());
    const posLOB      = llaveCDM.indexOf(lobN);
    const kpiExtraido = posLOB !== -1
      ? llaveCDM.substring(posLOB + lobN.length).trim()
      : llaveCDM;

    if (kpiExtraido !== llaveN) continue;

    const bmsId = String(dataCDM[row][idxBms]).trim();
    if (!agentes[bmsId]) continue;
    if (fk_norm(agentes[bmsId].lob) !== lobN) continue;

    const subLobCDM = fk_norm(String(dataCDM[row][idxSubLOB] || ''));
    if (subLobCDM && subLobCDM !== lobN) continue;

    agentesConDatos.add(bmsId);
    resultado.totalCoincidencias++;

    const valorNumerico = parseFloat(String(dataCDM[row][idxValH]).replace('%', '').trim());

    if (isNaN(valorNumerico) || valorNumerico === 0) {
      resultado.empleadosSinDatos.push({
        bmsId, nombre: agentes[bmsId].nombre,
        valorF: 0, bono: 0, sinDatos: true, salarioPersonalizado: false
      });
      continue;
    }

    let cumple = false;
    switch (operador) {
      case '>':  cumple = valorNumerico >  valorCondicion; break;
      case '>=': cumple = valorNumerico >= valorCondicion; break;
      case '<':  cumple = valorNumerico <  valorCondicion; break;
      case '<=': cumple = valorNumerico <= valorCondicion; break;
      case '=':  cumple = valorNumerico === valorCondicion; break;
    }

    const { bonoFinal, factor, hAg, personalizado } = calcBono(bmsId, cumple);

    const item = {
      bmsId,
      nombre:            agentes[bmsId].nombre,
      rol:               agentes[bmsId].rol || 'Sin rol',
      supervisor:        agentes[bmsId].supervisor || '',
      accm:              agentes[bmsId].accm || '',
      om:                agentes[bmsId].om   || '',
      valorF:            idxValF !== -1 ? dataCDM[row][idxValF] : '-',
      valorH:            valorNumerico,
      bono:              cumple ? bonoFinal : 0,
      horasContrato:     hAg,
      horasMaxima:       horasMaxima,
      factorHoras:       factor,
      salarioPersonalizado: personalizado
    };

    if (cumple) {
      resultado.totalCumplenCondicion++;
      resultado.empleadosCumplen.push(item);
    } else {
      resultado.empleadosNoCumplen.push(item);
    }
  }

  agentesLOB.forEach(function(ag) {
    if (!agentesConDatos.has(ag.bmsId)) {
      resultado.empleadosSinDatos.push({
        bmsId:   ag.bmsId,
        nombre:  ag.nombre,
        valorF:  '-',
        bono:    0,
        sinDatos: true,
        salarioPersonalizado: false
      });
    }
  });

  return resultado;
}


// ============================================================
//  fk_consolidarMultiTier(brutos)
// ============================================================
function fk_consolidarMultiTier(brutos) {
  const cons = {};

  for (const cat in brutos) {
    if (cat === 'keys') { cons[cat] = brutos[cat]; continue; }

    const grupos = {};
    brutos[cat].forEach(function(item) {
      if (!grupos[item.kpi]) grupos[item.kpi] = [];
      grupos[item.kpi].push(item);
    });

    cons[cat] = Object.keys(grupos).map(function(nombreKpi) {
      const tiers = grupos[nombreKpi];
      if (tiers.length === 1) return tiers[0];

      const maxPart = Math.max.apply(null, tiers.map(function(t) { return t.participacion || 1; }));

      const consolidado = {
        kpi:           nombreKpi,
        score:         'Multi-Escala',
        porcentaje:    tiers[0].porcentaje,
        participacion: maxPart,
        esMultiTier:   true,
        esGate:        tiers[0].esGate === true,
        scoresPorTier: tiers.map(function(t) {
          return { score: t.score, condicion: t.score, participacion: t.participacion };
        }),
        validacion: {
          totalCoincidencias:    tiers[0].validacion.totalCoincidencias,
          totalCumplenCondicion: 0,
          empleadosCumplen:      [],
          empleadosNoCumplen:    [],
          empleadosSinDatos:     tiers[0].validacion.empleadosSinDatos || []
        }
      };

      const mejorPorAgente = {};
      tiers.forEach(function(tier) {
        (tier.validacion.empleadosCumplen || []).forEach(function(e) {
          if (!mejorPorAgente[e.bmsId] || e.bono > mejorPorAgente[e.bmsId].bono) {
            mejorPorAgente[e.bmsId] = Object.assign({}, e, {
              scoreCumplido:      tier.score,
              participacionUsada: tier.participacion
            });
          }
        });
        (tier.validacion.empleadosNoCumplen || []).forEach(function(e) {
          if (!mejorPorAgente[e.bmsId]) {
            mejorPorAgente[e.bmsId] = Object.assign({}, e, {
              bono:               0,
              scoreCumplido:      tier.score,
              participacionUsada: tier.participacion
            });
          }
        });
      });

      Object.values(mejorPorAgente).forEach(function(r) {
        if (r.bono !== 0) {
          consolidado.validacion.empleadosCumplen.push(r);
          consolidado.validacion.totalCumplenCondicion++;
        } else {
          consolidado.validacion.empleadosNoCumplen.push(r);
        }
      });

      return consolidado;
    });
  }
  return cons;
}


// ============================================================
//  fk_aplicarGate(bonusStructureKPIs)
// ============================================================
function fk_aplicarGate(bonusStructureKPIs, additionalIncentivesKPIs) {
  if (!bonusStructureKPIs || !bonusStructureKPIs.length) return { bonusStructure: bonusStructureKPIs, additionalIncentives: additionalIncentivesKPIs };

  const gates = bonusStructureKPIs.filter(function(k) { return k.esGate === true; });
  if (!gates.length) return { bonusStructure: bonusStructureKPIs, additionalIncentives: additionalIncentivesKPIs };

  const agentesBloqueados = new Set();
  gates.forEach(function(gate) {
    const val = gate.validacion || {};
    (val.empleadosNoCumplen || []).forEach(function(e) { agentesBloqueados.add(e.bmsId); });
    (val.empleadosSinDatos || []).forEach(function(e) { agentesBloqueados.add(e.bmsId); });
  });

  if (agentesBloqueados.size > 0) {
    // Bloquear Bonus Structure
    bonusStructureKPIs.forEach(function(kpiObj) {
      const val = kpiObj.validacion || {};
      (val.empleadosCumplen || []).forEach(function(e) {
        if (agentesBloqueados.has(e.bmsId)) {
          e.bono = 0;
          e.bloqueadoPorGate = true;
        }
      });
    });

    // Bloquear Additional Incentives
    if (additionalIncentivesKPIs && additionalIncentivesKPIs.length) {
      additionalIncentivesKPIs.forEach(function(kpiObj) {
        const val = kpiObj.validacion || {};
        (val.empleadosCumplen || []).forEach(function(e) {
          if (agentesBloqueados.has(e.bmsId)) {
            e.bono = 0;
            e.bloqueadoPorGate = true;
          }
        });
      });
    }
  }

  return { bonusStructure: bonusStructureKPIs, additionalIncentives: additionalIncentivesKPIs };
}


// ============================================================
//  fk_construirFichaAgentes(resultados, agentes, horasMap, lob)
// ============================================================
function fk_construirFichaAgentes(resultados, agentes, horasMap, lob) {
  const lobN  = fk_norm(lob);
  const data  = {};
  const cats  = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];

  cats.forEach(function(cat, catIdx) {
    (resultados[cat] || []).forEach(function(kpiObj, kpiIdx) {
      const val  = kpiObj.validacion || {};
      const rKey = cat + '_' + kpiIdx;

      const add = function(emp, cumple, sinDatos) {
        if (!data[emp.bmsId]) {
          data[emp.bmsId] = { bmsId: emp.bmsId, nombre: emp.nombre, resultadosPorKPI: {} };
        }
        data[emp.bmsId].resultadosPorKPI[rKey] = {
          cat,
          kpiNombre:          kpiObj.kpi,
          score:              kpiObj.score,
          esGate:             kpiObj.esGate === true,
          esMultiTier:        kpiObj.esMultiTier === true,
          scoreCumplido:      emp.scoreCumplido      || kpiObj.score,
          participacionUsada: emp.participacionUsada  || kpiObj.participacion || 1,
          valorF:             emp.valorF,
          valorH:             emp.valorH,
          bono:               parseFloat(emp.bono) || 0,
          cumple,
          sinDatos,
          factorHoras:        emp.factorHoras,
          horasContrato:      emp.horasContrato
        };
      };

      (val.empleadosCumplen   || []).forEach(function(e) { add(e, true,  false); });
      (val.empleadosNoCumplen || []).forEach(function(e) { add(e, false, false); });
      (val.empleadosSinDatos  || []).forEach(function(e) { add(e, false, true);  });
    });
  });

  // Asegurar que todos los agentes de la LOB estén presentes
  Object.entries(agentes)
    .filter(function(kv) { return fk_norm(kv[1].lob) === lobN; })
    .forEach(function(kv) {
      if (!data[kv[0]]) {
        data[kv[0]] = { bmsId: kv[0], nombre: kv[1].nombre, resultadosPorKPI: {} };
      }
    });

  const ficha = {};

  Object.values(data).forEach(function(agObj) {
    const bmsId  = agObj.bmsId;
    const agInfo = agentes[bmsId] || {};
    const hData  = horasMap[bmsId] || null;

    let totalKeys       = 0;
    let todosKeysOK     = true;
    let tieneAlgunKey   = false;
    let totalBonus      = 0;
    let totalAdditional = 0;
    let totalDecreased  = 0;

    const kpisDesglose = [];

    Object.values(agObj.resultadosPorKPI).forEach(function(res) {
      const bono = parseFloat(res.bono) || 0;

      kpisDesglose.push({
        cat:               res.cat,
        kpiNombre:         res.kpiNombre,
        score:             res.score,
        scoreCumplido:     res.scoreCumplido,
        esGate:            res.esGate,
        esMultiTier:       res.esMultiTier,
        valorF:            res.valorF,
        valorH:            res.valorH,
        bono,
        cumple:            res.cumple,
        sinDatos:          res.sinDatos,
        factorHoras:       res.factorHoras,
        horasContrato:     res.horasContrato
      });

      if (res.cat === 'keys') {
        if (!res.sinDatos) {
          tieneAlgunKey = true;
          if (res.cumple) totalKeys += bono;
          else            todosKeysOK = false;
        }
      } else if (res.cat === 'decreasedKPIs') {
        if (res.cumple && !res.sinDatos) totalDecreased += bono;
      } else {
        if (res.cumple && !res.sinDatos) {
          if (res.cat === 'bonusStructure')            totalBonus      += bono;
          else if (res.cat === 'additionalIncentives') totalAdditional += bono;
        }
      }
    });

    let bonoFinal      = 0;
    let bloqueadoPorKey = false;
    let tieneGateFallido = kpisDesglose.some(function(k) {
        return k.esGate && !k.cumple && !k.sinDatos;
    });

    if (tieneAlgunKey) {
      if (todosKeysOK) {
        if (tieneGateFallido) {
            bonoFinal = totalDecreased;
        } else {
            bonoFinal = Math.max(0, totalKeys)
                      + Math.max(0, totalBonus)
                      + Math.max(0, totalAdditional)
                      + totalDecreased;
        }
      } else {
        bonoFinal      = totalDecreased;
        bloqueadoPorKey = true;
      }
    } else {
      if (tieneGateFallido) {
          bonoFinal = totalDecreased;
      } else {
          bonoFinal = Math.max(0, totalBonus)
                    + Math.max(0, totalAdditional)
                    + totalDecreased;
      }
    }

    bonoFinal = Math.max(0, bonoFinal);

    ficha[bmsId] = {
      bmsId,
      nombre:           agObj.nombre || agInfo.nombre || ('Agente ' + bmsId),
      rol:              agInfo.rol   || 'Sin rol',
      supervisor:       agInfo.supervisor || '',
      accm:             agInfo.accm       || '',
      om:               agInfo.om         || '',
      horasSemanales:   agInfo.horasSemanales || 42,
      lobOriginal:      agInfo.lobOriginal    || lob,
      horasReales:      hData ? (parseFloat(hData.horas)        || 0) : 0,
      ausencias:        hData ? (parseFloat(hData.ausencias)     || 0) : 0,
      diasLaborados:    hData ? (parseFloat(hData.diasLaborados) || 0) : 0,
      tieneHoras:       !!hData,
      totalKeys:        Math.max(0, totalKeys),
      totalBonusStructure: Math.max(0, totalBonus),
      totalAdditional:  Math.max(0, totalAdditional),
      totalDecreased:   Math.abs(totalDecreased),
      bonoFinal,
      bloqueadoPorKey,
      bloqueadoPorGate: tieneGateFallido,
      tieneAlgunKey,
      todosKeysCumplen: todosKeysOK,
      kpisDesglose
    };
  });

  return ficha;
}


// ============================================================
//  fk_mesDeFecha(fecha)
// ============================================================
function fk_mesDeFecha(fecha) {
  if (!fecha) return null;
  const re = /[-/]/;
  if (re.test(fecha)) {
    const d = new Date(fecha);
    if (!isNaN(d.getTime())) return d.getMonth() + 1;
    const p = fecha.split(/[-/]/);
    if (p.length === 3) {
      if (p[0].length === 4) return parseInt(p[1], 10);
      if (p[2].length === 4) return parseInt(p[1], 10);
    }
  }
  const mesesMap = {
    'enero':1,'febrero':2,'marzo':3,'abril':4,'mayo':5,'junio':6,
    'julio':7,'agosto':8,'septiembre':9,'octubre':10,'noviembre':11,'diciembre':12
  };
  return mesesMap[String(fecha).toLowerCase().trim()] || null;
}


// ============================================================
//  fk_norm(s)
// ============================================================
function fk_norm(s) {
  return String(s || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .trim()
    .replace(/\s+/g, ' ');
}

// ============================================================
//  SIMULADOR_MOTOR.GS regio
//  Motor central de simulación con métricas reales CDM.
//  Depende de: FICHA_KPI.GS (fichaKPIPorLob, fk_*)
//              Código 1 (OM_ID_SS_PRINCIPAL_LOCAL, OM_ID_SS_CDM)
//
//  Funciones exportadas:
//    simularBonos(lob, mes, numeroMes)          ← punto de entrada frontend
//    REGISTRO_simularConMetricasReales(...)      ← usado por simularTodasLOBsPorOM
//    REGISTRO_simularConMetricasRealesMasivo(...) ← usado por simularTodasLOBsPorRegion
// ============================================================


// ============================================================
//  PUNTO DE ENTRADA FRONTEND
//  Llamado desde el HTML con:
//    google.script.run.simularBonos(lob, mes, numeroMes)
//
//  - lob        : nombre de la LOB (string)
//  - mes        : fecha/mes del registro ("2026-02-15" o "Febrero")
//  - numeroMes  : número de mes CDM (1-12), o null para inferirlo
// ============================================================
function simularBonos(lob, mes, numeroMes) {
    try {
        Logger.log('\n🚀 ========== simularBonos ==========');
        Logger.log('LOB: ' + lob);
        Logger.log('Mes registro: ' + mes);
        Logger.log('Mes CDM: ' + numeroMes);

        if (!lob || !mes) {
            return { success: false, error: 'Se requiere LOB y Mes.' };
        }

        // Resolver numeroMes si no viene
        var mesNum = numeroMes
            ? parseInt(numeroMes, 10)
            : _sim_resolverNumeroMes(mes);

        if (!mesNum) {
            mesNum = new Date().getMonth() + 1;
            Logger.log('⚠️ No se pudo resolver mes — usando mes actual: ' + mesNum);
        }

        // Normalizar fecha para que fichaKPIPorLob la encuentre
        var fechaNorm = _sim_normalizarFecha(mes);

        Logger.log('Fecha normalizada: ' + fechaNorm + ' | MesCDM: ' + mesNum);

        // ── MOTOR PRINCIPAL ────────────────────────────────
        var ficha = fichaKPIPorLob(lob, fechaNorm, mesNum);

        if (ficha.error) {
            Logger.log('❌ fichaKPIPorLob error: ' + ficha.error);
            return { success: false, error: ficha.error };
        }

        // ── CONSTRUIR RESPUESTA EN FORMATO QUE ESPERA EL FRONTEND ──
        return _sim_construirRespuestaFrontend(ficha, lob, fechaNorm, mesNum);

    } catch (e) {
        Logger.log('❌ simularBonos excepción: ' + e.message + '\n' + e.stack);
        return { success: false, error: e.message };
    }
}

}

// ============================================================
//  _sim_construirRespuestaFrontend
//  Formato que espera _mostrarSimulacionOMEnModal en el HTML.
// ============================================================
function _sim_construirRespuestaFrontend(ficha, lob, fecha, mesNum) {
    var fichaAgentes    = ficha.fichaAgentes    || {};
    var kpisPorCategoria = ficha.kpisPorCategoria || {};
    var totales         = ficha.totales          || {};
    var config          = fk_obtenerConfigLOB(lob) || {};

    // ── resumenAgentes: formato que lee _mostrarSimulacionOMEnModal ──
    var resumenAgentes = {};
    Object.values(fichaAgentes).forEach(function(ag) {
        resumenAgentes[ag.bmsId] = {
            bmsId:            ag.bmsId,
            nombre:           ag.nombre,
            rol:              ag.rol             || 'Sin rol',
            supervisor:       ag.supervisor      || '',
            accm:             ag.accm            || '',
            om:               ag.om              || '',
            horasSemanales:   ag.horasSemanales  || 42,
            totalKeys:        ag.totalKeys       || 0,
            totalBonus:       ag.totalBonusStructure || 0,
            totalAdditional:  ag.totalAdditional || 0,
            totalDecreased:   -(ag.totalDecreased || 0),  // negativo para cálculo
            bonoFinal:        ag.bonoFinal       || 0,
            bloqueadoPorKey:  ag.bloqueadoPorKey || false,
            tieneAlgunKey:    ag.tieneAlgunKey   || false,
            horasReales:      ag.horasReales     || 0,
            ausencias:        ag.ausencias       || 0,
            factorProrrateo:  ag.horasSemanales && ag.horasSemanales < 42
                ? ag.horasSemanales / 42 : 1,
            // Detalle plano para verDetalleKPIsAgente
            detalle: _sim_aplanarDetalle(ag.kpisDesglose || [])
        };
    });

    // ── metricasPorLob: formato que lee tablaCategoria en _mostrarSimulacionOMEnModal ──
    var metricasPorLob = {};
    metricasPorLob[lob] = {
        keys:                kpisPorCategoria.keys                || [],
        bonusStructure:      kpisPorCategoria.bonusStructure      || [],
        additionalIncentives: kpisPorCategoria.additionalIncentives || [],
        decreasedKPIs:       kpisPorCategoria.decreasedKPIs       || []
    };

    // ── top10 ────────────────────────────────────────────────
    var top10 = Object.values(resumenAgentes)
        .sort(function(a, b) { return b.bonoFinal - a.bonoFinal; })
        .slice(0, 10);

    // ── configuracion ────────────────────────────────────────
    var salario    = config.salarioBase   || 0;
    var porcentaje = config.porcentajeBono || 0;
    var bonoMaximo = salario > 0 && porcentaje > 0 ? (salario * porcentaje) / 100 : 0;

    var configuracion = {
        salario:    salario,
        porcentaje: porcentaje,
        bonoMaximo: bonoMaximo
    };
    return {
        success:             true,
        lob:                 lob,
        fecha:               fecha,
        fechaExacta:         totales.fechaExacta || fecha, // ⭐ De la DB
        fechaSimulacion:     Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'),
        mes:                 fecha,
        mesNum:              mesNum,
        hc:                  totales.totalAgentes || 0,
        totales:             {
            fechaExacta:     totales.fechaExacta || fecha,
            totalAgentes:    totales.totalAgentes    || 0,
            agentesConBono:  totales.agentesConBono  || 0,
            totalGeneral:    totales.totalGeneral     || 0,
            promedioBono:    totales.promedio         || 0
        },
        resumenAgentes:      resumenAgentes,
        metricasPorLob:      metricasPorLob,
        resultadosDetallados: metricasPorLob[lob],
        top10Ganadores:      top10,
        configuracion:       configuracion,
        inversionReal:       totales.totalGeneral   || 0,
        inversionTeorica:    bonoMaximo * (totales.totalAgentes || 0),
        ahorro:              Math.max(0, bonoMaximo * (totales.totalAgentes || 0) - (totales.totalGeneral || 0)),
        eficienciaReal:      bonoMaximo * (totales.totalAgentes || 0) > 0
            ? ((totales.totalGeneral || 0) / (bonoMaximo * (totales.totalAgentes || 0))) * 100
            : 0,
        estadisticas: {
            agentesCobran:   totales.agentesConBono  || 0,
            agentesNoCobran: (totales.totalAgentes   || 0) - (totales.agentesConBono || 0),
            tasaCumplimiento: totales.totalAgentes > 0
                ? ((totales.agentesConBono || 0) / totales.totalAgentes) * 100 : 0
        }
    };
}

    };
}


// ============================================================
//  _sim_construirDesglosePorCategoria
//  Agrupa montos por categoría (formato usado en reporte unificado).
// ============================================================
function _sim_construirDesglosePorCategoria(kpisPorCategoria) {
    var desglose = {
        bonusStructure:      [],
        additionalIncentives: [],
        decreasedKPIs:       []
    };

    var cats = ['bonusStructure', 'additionalIncentives', 'decreasedKPIs'];

    cats.forEach(function(cat) {
        var kpis = kpisPorCategoria[cat] || [];
        kpis.forEach(function(kpiObj) {
            var val     = kpiObj.validacion || {};
            var cumplen = val.empleadosCumplen || [];
            var monto   = cumplen.reduce(function(s, e) {
                return s + Math.abs(e.bono || 0);
            }, 0);
            desglose[cat].push({
                kpi:        kpiObj.kpi,
                score:      kpiObj.score,
                montoTotal: monto,
                cumplen:    cumplen.length,
                esGate:     kpiObj.esGate === true
            });
        });
    });

    return desglose;
}

            sinDatos:     d.sinDatos || false,
            participacion: d.participacionUsada || 1,
            factorHoras:  d.factorHoras || 1
        };
    });
}


// ============================================================
//  _sim_normalizarFecha
//  Convierte cualquier formato de fecha/mes a "yyyy-MM-dd"
//  o nombre de mes, para que fk_obtenerRegistros lo encuentre.
// ============================================================
function _sim_normalizarFecha(valor) {
    if (!valor) return '';

    // Ya es formato yyyy-MM-dd
    if (/^\d{4}-\d{2}-\d{2}/.test(String(valor))) {
        return String(valor).trim();
    }

    // Es un objeto Date
    if (valor instanceof Date) {
        return Utilities.formatDate(valor, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    }

    // Es un nombre de mes o cualquier string — devolver tal cual
    // fk_obtenerRegistros normaliza por su cuenta
    return String(valor).trim();
}


// ============================================================
//  _sim_resolverNumeroMes
//  Extrae el número de mes (1-12) desde fecha o nombre.
// ============================================================
function _sim_resolverNumeroMes(valor) {
    if (!valor) return null;
    if (valor instanceof Date) return valor.getMonth() + 1;

    var s = String(valor).trim();

    // yyyy-MM-dd
    if (/^\d{4}-(\d{2})-\d{2}/.test(s)) {
        return parseInt(s.split('-')[1], 10);
    }

    // Nombre de mes
    var mapa = {
        enero:1, january:1,
        febrero:2, february:2,
        marzo:3, march:3,
        abril:4, april:4,
        mayo:5, may:5,
        junio:6, june:6,
        julio:7, july:7,
        agosto:8, august:8,
        septiembre:9, september:9,
        octubre:10, october:10,
        noviembre:11, november:11,
        diciembre:12, december:12
    };
    var clave = s.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').trim();
    return mapa[clave] || null;
}


// ============================================================
//  _sim_detectarTipoHoja
//  Detecta si la LOB/fecha viene de Registro o Registro_pisos.
// ============================================================
function _sim_detectarTipoHoja(lob, fecha) {
    try {
        var ss      = SpreadsheetApp.openById(OM_ID_SS_PRINCIPAL_LOCAL);
        var hojas   = ['Registro_pisos', 'Registro'];
        var lobN    = fk_norm(lob);
        var fechaN  = fk_norm(fecha);

        for (var h = 0; h < hojas.length; h++) {
            var hoja = ss.getSheetByName(hojas[h]);
            if (!hoja) continue;
            var data = hoja.getDataRange().getValues();
            for (var i = 1; i < data.length; i++) {
                if (fk_norm(String(data[i][1] || '')) !== lobN) continue;
                var fFecha = data[i][0] instanceof Date
                    ? Utilities.formatDate(data[i][0], Session.getScriptTimeZone(), 'yyyy-MM-dd')
                    : String(data[i][0] || '').trim();
                if (fk_norm(fFecha) === fechaN) return hojas[h];
            }
        }
    } catch (e) {
        Logger.log('⚠️ _sim_detectarTipoHoja: ' + e.message);
    }
    return 'Registro';
}

// ============================================================
//  obtenerMesesCDMDisponibles()
//  Llamado desde simOnChangeFecha()
//  Devuelve los meses CDM disponibles en la hoja CDM del SS CDM.
// ============================================================
function obtenerMesesCDMDisponibles() {
    try {
        var ss     = SpreadsheetApp.openById(OM_ID_SS_CDM);
        var hojas  = ss.getSheets();
        var meses  = [];
        var mesesNombres = [
            '', 'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio',
            'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'
        ];

        hojas.forEach(function(h) {
            var nombre = h.getName();
            // CDM_1, CDM_2, ... CDM_12
            var match = nombre.match(/^CDM_(\d{1,2})$/);
            if (match) {
                var num = parseInt(match[1], 10);
                if (num >= 1 && num <= 12) {
                    meses.push({ numero: num, nombre: mesesNombres[num] });
                }
            }
        });

        // Ordenar por número de mes
        meses.sort(function(a, b) { return a.numero - b.numero; });

        // Si no hay hojas CDM_N, retornar meses del año como fallback
        if (meses.length === 0) {
            for (var i = 1; i <= 12; i++) {
                meses.push({ numero: i, nombre: mesesNombres[i] });
            }
        }

        return meses;
    } catch (e) {
        Logger.log('❌ obtenerMesesCDMDisponibles: ' + e.message);
        return [];
    }
}

/**
 * ⭐ LÓGICA GATE ACTIVA PARA OM (PARIDAD)
 */
function OM_aplicarLogicaGate(bonusStr, addInc) {
    if (!bonusStr || !bonusStr.length) return { bonusStructure: bonusStr, additionalIncentives: addInc };
    
    const gates = bonusStr.filter(k => k.esGate === true);
    if (!gates.length) return { bonusStructure: bonusStr, additionalIncentives: addInc };

    const agBloq = new Set();
    gates.forEach(g => {
        const v = g.validacion || {};
        (v.empleadosNoCumplen || []).forEach(e => agBloq.add(e.bmsId));
        (v.empleadosSinDatos  || []).forEach(e => agBloq.add(e.bmsId));
    });

    if (agBloq.size > 0) {
        // Bloquear ambos
        [bonusStr, addInc].forEach(cat => {
            if (!cat) return;
            cat.forEach(kpiObj => {
                const v = kpiObj.validacion || {};
                (v.empleadosCumplen || []).forEach(e => {
                    if (agBloq.has(e.bmsId)) {
                        e.bono = 0;
                        e.bloqueadoPorGate = true;
                    }
                });
            });
        });
    }
    return { bonusStructure: bonusStr, additionalIncentives: addInc };
}
