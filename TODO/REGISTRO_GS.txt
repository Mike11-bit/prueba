function REGISTRO_obtenerLobsPorOM(nombreOM) {
    const cacheKey = `lobs_om_v3_hc_${normalizarTexto(nombreOM)}`;
    return getCached(cacheKey, () => {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaHC = ss.getSheetByName('H_C');
        if (!hojaHC) return [];

        const dataHC = hojaHC.getDataRange().getValues();
        const lobsSet = new Set();
        const omNormalizado = normalizarTexto(nombreOM);

        console.log(`🔍 [Registry] Escaneando H_C para OM: ${nombreOM}`);

        for (let i = 1; i < dataHC.length; i++) {
            const omFila = dataHC[i][6];  // Col G (índice 6) - OM
            const lobFila = dataHC[i][1]; // Col B (índice 1) - LOB
            
            if (!omFila || !lobFila) continue;
            if (normalizarTexto(omFila) === omNormalizado) {
                const lobTrimmed = String(lobFila).trim();
                if (lobTrimmed && lobTrimmed !== '-') lobsSet.add(lobTrimmed);
            }
        }

        return Array.from(lobsSet).sort();
    }, 3600);
}

/**
 * Nueva función solicitada: Cargar LOBs desde Configuracion para el simulador

function REGISTRO_obtenerLobsConfiguradasPorUsuario(usuario) {
    if (!usuario) return [];
    
    const rol = (usuario.rol || '').trim();
    const nombre = (usuario.nombre || '').trim();
    
    console.log(`🔍 [Simulator] Obteniendo LOBs configuradas desde 'Configuracion' para ${nombre} (Rol: ${rol})`);
    
    // El simulador SIEMPRE usa la hoja 'Configuracion'
    if (rol === 'Reporting Analyst' || rol === 'Audit') {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hoja = ss.getSheetByName('Configuracion');
        if (!hoja) return [];
        const data = hoja.getDataRange().getValues();
        const lobs = new Set();
        for (let i = 1; i < data.length; i++) if (data[i][2]) lobs.add(String(data[i][2]).trim());
        return Array.from(lobs).sort();
    } else {
        // Para OM, ACCM, Supervisor, etc. se busca en su columna de Configuracion
        return REGISTRO_obtenerLobsDesdeConfiguracion(nombre, 'OM');
    }
}

function REGISTRO_obtenerLobsDesdeConfiguracion(nombre, rolCol) {

    if (!hoja) return [];

    // En la hoja Configuracion generada por REGISTRO_, OM está en Col B (1).

function REGISTRO_obtenerLobsPorRegion(region) {
    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hojaHC = ss.getSheetByName('H_C');
    if (!hojaHC) return [];

    const data = hojaHC.getDataRange().getValues();
    const lobsSet = new Set();
    const regionNorm = normalizarTexto(region);

    // Buscar encabezados dinamicamente
    const headers = data[0];
    let colLOB = 1;    // Col B por defecto
    let colRegion = 8; // Col I por defecto

    headers.forEach(function(h, i) {
        const hn = normalizarTexto(String(h));
        if (hn === 'lob' || hn === 'sub lob' || hn === 'sublob') colLOB = i;
        if (hn === 'region') colRegion = i;
    });

    Logger.log('Buscando LOBs para region: "' + region + '" | colLOB=' + colLOB + ' | colRegion=' + colRegion);

    for (let i = 1; i < data.length; i++) {
        const regionFila = normalizarTexto(String(data[i][colRegion] || ''));
        const lobFila = String(data[i][colLOB] || '').trim();

        if (regionFila === regionNorm && lobFila) {
            lobsSet.add(lobFila);
        }
    }

    const resultado = Array.from(lobsSet).sort();
    Logger.log('LOBs en H_C para region "' + region + '": ' + resultado.length + ' -> ' + JSON.stringify(resultado));
    return resultado;
}

function REGISTRO_contarAgentesPorLob(lob) {
    const cacheKey = `count_agentes_${normalizar(lob)}`;
    return getCached(cacheKey, function () {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaHC = ss.getSheetByName('H_C');
        const data = hojaHC.getDataRange().getValues();
        const lobNorm = normalizar(lob);
        let count = 0;

        const headers = data[0].map(h => String(h).toLowerCase());
        const colStatus = headers.findIndex(h => h.includes('status') || h.includes('estado'));

        for (let i = 1; i < data.length; i++) {
            if (normalizar(data[i][1]) === lobNorm) {
                if (colStatus > -1) {
                    const estado = String(data[i][colStatus]).toLowerCase();
                    if (estado === 'activo' || estado === 'active' || estado === 'alta') count++;
                } else {
                    count++;
                }
            }
        }
        return count;
    }, 3600);
}

/**
 * Obtiene todos los agentes de una LOB desde el objeto de agentes permitidos
 */
function obtenerTodosLosAgentesDeLOB(lob, agentesPermitidos) {
    const lobNorm = normalizar(lob);
    const result = [];
    
    for (const id in agentesPermitidos) {
        if (normalizar(agentesPermitidos[id].lob) === lobNorm) {
            result.push({ bmsId: id, nombre: agentesPermitidos[id].nombre });
        }
    }
    
    return result;
}

function filtrarLobsPorColumna(nombreHoja, colIndex, valorBusqueda) {
    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hoja = ss.getSheetByName(nombreHoja);
    if (!hoja) return [];
    const data = hoja.getDataRange().getValues();
    const lobs = new Set();
    const valNorm = normalizarTexto(valorBusqueda);

    for (let i = 1; i < data.length; i++) {
        const valFila = normalizarTexto(data[i][colIndex]);
        if (valFila === valNorm && valNorm !== '') {
            const lob = String(data[i][1]).trim();
            if (lob) lobs.add(lob);
        }
    }
    return Array.from(lobs).sort();
}

function normalizarKey(str) {
    return String(str || '').toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-z0-9]/g, '');
}

function normalizar(s) {
    if (!s) return "";
    return String(s).normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim();
}

/**
 * Función unificada de normalización para textos y LOBs.
 * Mantiene coherencia entre simulación y validación.
 */
function normalizarTexto(s) {
    if (!s) return '';
    return String(s)
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "") // Quitar acentos
        .toLowerCase()
        .trim()
        .replace(/\s+/g, ' '); // Unificar espacios
}

function REGISTRO_obtenerResponsablesPorLob(lob) {
    try {
        console.log(`🔍 [Optimizado] Buscando responsables para LOB: ${lob}`);

        const cacheKey = `resp_lob_${normalizarKey(lob)}`;
        const datosLOB = getCached(cacheKey, function () {
            const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
            const hojaHC = ss.getSheetByName('H_C');
            const dataHC = hojaHC.getDataRange().getValues();
            const lobNorm = normalizarKey(lob);

            for (let i = 1; i < dataHC.length; i++) {
                if (normalizarKey(dataHC[i][1]) === lobNorm) {
                    return {
                        encontrado: true,
                        supervisor: String(dataHC[i][4] || '').trim(),
                        accm: String(dataHC[i][5] || '').trim(),
                        om: String(dataHC[i][6] || '').trim()
                    };
                }
            }
            return { encontrado: false };
        }, 3600);

        if (!datosLOB.encontrado) {
            return { supervisor: '', accm: '', om: '', error: 'LOB no encontrada en H_C' };
        }

        const usuario = REGISTRO_obtenerDatosUsuarioActivo();
        if (usuario.rol === 'OM') {
            const omAsignadoNorm = normalizarTexto(datosLOB.om);
            const usuarioNombreNorm = normalizarTexto(usuario.nombre);

            if (omAsignadoNorm !== usuarioNombreNorm) {
                return {
                    supervisor: '', accm: '', om: '',
                    error: `Esta LOB está asignada a otro OM (${datosLOB.om}).`
                };
            }
        }
        return datosLOB;

    } catch (error) {
        return { supervisor: '', accm: '', om: '', error: error.toString() };
    }
}

function REGISTRO_obtenerSalarioPorLob(lob, usuario) {
    try {
        const userNorm = usuario ? normalizarTexto(usuario.nombre || '') : '';
        const rol      = usuario ? (usuario.rol || '') : '';
        const esAdmin  = (rol === 'Reporting Analyst' || rol === 'Audit');

        // ⭐ v4 para invalidar cache viejo
        const cacheKey = `salario_lob_v4_${normalizarKey(lob)}_${normalizarKey(userNorm)}`;

        return getCached(cacheKey, function () {
            const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
            const hoja = ss.getSheetByName('Configuracion');

            if (!hoja) {
                Logger.log('❌ No existe la hoja Configuracion');
                return { encontrado: false };
            }

            const data    = hoja.getDataRange().getValues();
            const lobNorm = normalizarTexto(lob);

            Logger.log(`\n🔍 obtenerSalarioPorLob:`);
            Logger.log(`   LOB buscada: "${lob}" → "${lobNorm}"`);
            Logger.log(`   Usuario: "${userNorm}" | Rol: "${rol}" | esAdmin: ${esAdmin}`);
            Logger.log(`   Total filas (sin header): ${data.length - 1}`);

            // ⭐ HELPER: parsear número desde string o number
            const _parse = (val) => {
                if (val === null || val === undefined || val === '') return 0;
                if (typeof val === 'number') return val;
                return parseFloat(String(val).replace(/[^\d.,]/g, '').replace(',', '.')) || 0;
            };

            // ⭐ RECOPILAR TODAS las filas candidatas (no hacer return al primer match)
            const filasCandidatas = [];

            for (let i = 1; i < data.length; i++) {
                const lobFila = normalizarTexto(String(data[i][2] || ''));
                const omFila  = normalizarTexto(String(data[i][1] || ''));

                Logger.log(`   Fila ${i+1}: LOB="${data[i][2]}"(${lobFila}) | OM="${data[i][1]}"(${omFila})`);

                // ⭐ PASO 1: LOB debe coincidir siempre
                if (lobFila !== lobNorm) continue;

                // ⭐ PASO 2: Verificar acceso por OM
                // Admin/Reporting → acepta cualquier OM
                // OM → comparación tolerante (includes en ambas direcciones)
                const omMatch = esAdmin
                    || !userNorm
                    || omFila === userNorm
                    || omFila.includes(userNorm)
                    || userNorm.includes(omFila);

                if (!omMatch) {
                    Logger.log(`   ⚠️ Fila ${i+1}: LOB coincide pero OM no: "${omFila}" ≠ "${userNorm}"`);
                    continue;
                }

                // ⭐ PASO 3: Parsear salario y porcentaje
                const salario    = _parse(data[i][3]);
                const porcentaje = _parse(data[i][4]);

                if (salario === 0) {
                    Logger.log(`   ⚠️ Fila ${i+1}: Salario=0, ignorando`);
                    continue;
                }

                filasCandidatas.push({
                    indice:     i + 1,
                    mes:        String(data[i][0] || '').trim(),
                    om:         String(data[i][1] || '').trim(),
                    lob:        String(data[i][2] || '').trim(),
                    salario:    salario,
                    porcentaje: porcentaje
                });

                Logger.log(`   ✅ Candidata fila ${i+1}: Mes="${data[i][0]}" | Salario=$${salario} | %=${porcentaje}`);
            }

            // ⭐ Sin candidatas
            if (filasCandidatas.length === 0) {
                Logger.log(`   ❌ Sin candidatas para LOB="${lob}"`);
                return { encontrado: false };
            }

            // ⭐ TOMAR LA ÚLTIMA (más reciente — la hoja se escribe en orden cronológico)
            const mejor = filasCandidatas[filasCandidatas.length - 1];

            Logger.log(`   🏆 Usando fila ${mejor.indice}: Mes="${mejor.mes}" | Salario=$${mejor.salario} | %=${mejor.porcentaje}`);

            return {
                encontrado:  true,
                salario:     mejor.salario,
                porcentaje:  mejor.porcentaje,
                mes:         mejor.mes,
                om:          mejor.om,
                lob:         mejor.lob,
                filaIndex:   mejor.indice
            };

        }, 600); // 10 minutos de cache

    } catch (e) {
        Logger.log('❌ Error en REGISTRO_obtenerSalarioPorLob: ' + e.toString());
        return { encontrado: false };
    }
}

function REGISTRO_obtenerMesesRegistrados(lob, usuario) {
    Logger.log(`\n🔍 ========== OBTENIENDO MESES PARA LOB ==========`);
    Logger.log(`📋 LOB recibida: "${lob}"`);
    Logger.log(`   Tipo: ${typeof lob}`);
    Logger.log(`   Longitud: ${lob ? lob.length : 0}`);
    Logger.log(`👤 Usuario: ${usuario.nombre} (${usuario.rol})`);
    
    const cacheKey = `meses_reg_${normalizarKey(lob)}_${normalizarKey(usuario.nombre)}`;
    
    try {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojas = [ss.getSheetByName('Registro'), ss.getSheetByName('Registro_pisos')];
        const meses = new Set();
        const lobNorm = normalizarKey(lob);
        const userNorm = normalizarTexto(usuario.nombre);
        
        Logger.log(`\n🔑 Claves de búsqueda normalizadas:`);
        Logger.log(`   LOB normalizada (key): "${lobNorm}"`);
        Logger.log(`   Usuario normalizado (texto): "${userNorm}"`);
        Logger.log(`   Rol: ${usuario.rol}`);

        hojas.forEach(hoja => {
            if (!hoja) {
                Logger.log(`\n⚠️ Hoja no encontrada`);
                return;
            }
            
            const nombreHoja = hoja.getName();
            const data = hoja.getDataRange().getValues();
            
            Logger.log(`\n📊 ========== ESCANEANDO HOJA: ${nombreHoja} ==========`);
            Logger.log(`   Total filas (incluyendo encabezado): ${data.length}`);
            Logger.log(`   Filas de datos: ${data.length - 1}`);
            
            Logger.log(`\n📋 Encabezados de la hoja:`);
            data[0].forEach((header, idx) => {
                Logger.log(`   Col ${String.fromCharCode(65 + idx)} (${idx}): "${header}"`);
            });
            
            let filasProcesadas = 0;
            let filasCoincidentesLOB = 0;
            let filasCoincidentesPermiso = 0;
            
            for (let i = 1; i < data.length; i++) {
                filasProcesadas++;
                
                const mesFila = data[i][0];
                const lobFila = data[i][1];
                const accmFila = data[i][2];
                const omFila = data[i][3];
                
                const lobFilaNorm = normalizarKey(lobFila);
                const omFilaNorm = normalizarTexto(omFila);
                const accmFilaNorm = normalizarTexto(accmFila);
                
                if (lobFilaNorm === lobNorm) {
                    filasCoincidentesLOB++;
                    
                    Logger.log(`\n  ✅ Fila ${i+1}: LOB coincide`);
                    Logger.log(`     Columna A (Mes): "${mesFila}" (Tipo: ${typeof mesFila}, Es Date: ${mesFila instanceof Date})`);
                    Logger.log(`     Columna B (LOB): "${lobFila}" → Normalizada: "${lobFilaNorm}"`);
                    Logger.log(`     Columna C (ACCM): "${accmFila}" → Normalizada: "${accmFilaNorm}"`);
                    Logger.log(`     Columna D (OM): "${omFila}" → Normalizada: "${omFilaNorm}"`);
                    
                    let tienePermiso = false;
                    let razonPermiso = '';
                    
                    if (usuario.rol === 'Reporting Analyst' || usuario.rol === 'Audit') {
                        tienePermiso = true;
                        razonPermiso = 'Rol tiene acceso total';
                    } else if (usuario.rol === 'OM') {
                        if (omFilaNorm === userNorm) {
                            tienePermiso = true;
                            razonPermiso = `OM coincide: "${omFila}" = "${usuario.nombre}"`;
                        } else {
                            razonPermiso = `OM NO coincide: "${omFila}" (${omFilaNorm}) ≠ "${usuario.nombre}" (${userNorm})`;
                        }
                    } else if (usuario.rol === 'ACCM') {
                        if (accmFilaNorm === userNorm) {
                            tienePermiso = true;
                            razonPermiso = `ACCM coincide: "${accmFila}" = "${usuario.nombre}"`;
                        } else {
                            razonPermiso = `ACCM NO coincide: "${accmFila}" (${accmFilaNorm}) ≠ "${usuario.nombre}" (${userNorm})`;
                        }
                    } else if (usuario.rol === 'Supervisor') {
                        tienePermiso = true;
                        razonPermiso = 'Supervisor tiene acceso';
                    }
                    
                    Logger.log(`     🔐 Verificación de permisos: ${tienePermiso ? '✅ PERMITIDO' : '❌ DENEGADO'}`);
                    Logger.log(`        Razón: ${razonPermiso}`);
                    
                    if (tienePermiso) {
                        filasCoincidentesPermiso++;
                        
                        let mesFinal = mesFila;
                        
                        if (mesFila instanceof Date) {
                            mesFinal = Utilities.formatDate(mesFila, Session.getScriptTimeZone(), 'yyyy-MM-dd');
                            Logger.log(`        📅 Fecha convertida: "${mesFila}" → "${mesFinal}"`);
                        } else if (mesFila) {
                            mesFinal = String(mesFila).trim();
                            Logger.log(`        📅 Ya es String: "${mesFinal}"`);
                        } else {
                            Logger.log(`        ⚠️ Mes/Fecha VACÍO en columna A`);
                            continue;
                        }
                        
                        meses.add(mesFinal);
                        Logger.log(`        ✅ MES AGREGADO: "${mesFinal}"`);
                    }
                } else if (i <= 5) {
                    Logger.log(`\n  ❌ Fila ${i+1}: LOB NO coincide`);
                    Logger.log(`     Esperado: "${lob}" (normalizado: "${lobNorm}")`);
                    Logger.log(`     Encontrado: "${lobFila}" (normalizado: "${lobFilaNorm}")`);
                }
            }
            
            Logger.log(`\n📊 Resumen de hoja ${nombreHoja}:`);
            Logger.log(`   Filas procesadas: ${filasProcesadas}`);
            Logger.log(`   Filas con LOB coincidente: ${filasCoincidentesLOB}`);
            Logger.log(`   Filas con permiso: ${filasCoincidentesPermiso}`);
        });
        
        const resultado = Array.from(meses).sort();
        
        Logger.log(`\n🎯 ========== RESULTADO FINAL ==========`);
        Logger.log(`✅ Total meses encontrados: ${resultado.length}`);
        
        if (resultado.length > 0) {
            Logger.log(`📅 Meses encontrados:`);
            resultado.forEach((mes, idx) => {
                Logger.log(`   ${idx + 1}. "${mes}"`);

function REGISTRO_guardarRegistroMetricas(datos) {
    const lock = LockService.getScriptLock();
    try {
        if (!lock.tryLock(30000)) {
            return { success: false, error: '⚠️ Servidor ocupado. Intente nuevamente.' };
        }
        console.log('🔒 Guardado seguro iniciado...');

        if (!datos || !datos.mes || !datos.lob || !datos.tipoHoja) {
            return { success: false, error: 'Faltan campos obligatorios' };
        }
        if (datos.bonus) {
            const kpisUnicos = {};
            datos.bonus.forEach(kpi => {
                const nombre = (kpi.nombre || '').trim().toUpperCase();
                if (nombre && !kpisUnicos[nombre]) {
                    kpisUnicos[nombre] = parseFloat(kpi.porcentaje) || 0;
                }
            });
            let totalPct = Object.values(kpisUnicos).reduce((sum, pct) => sum + pct, 0);
            if (totalPct > 100) return { success: false, error: `Bonus Structure (${totalPct.toFixed(1)}%) excede 100%` };
        }

        // Guardar configuración siempre que estén presentes
        if (datos.salario !== undefined && datos.porcentaje !== undefined) {
            REGISTRO_guardarConfiguracion(datos.mes, datos.om, datos.lob, datos.salario, datos.porcentaje);
        }

        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hoja = ss.getSheetByName(datos.tipoHoja);
        if (!hoja) return { success: false, error: 'Hoja no encontrada' };

        const data = hoja.getDataRange().getValues();
        const mesNorm = normalizarKey(datos.mes);
        const lobNorm = normalizarKey(datos.lob);
        let filaIndex = -1;

        for (let i = 1; i < data.length; i++) {
            if (normalizarKey(data[i][0]) === mesNorm && normalizarKey(data[i][1]) === lobNorm) {
                filaIndex = i + 1;
                break;
            }
        }

        const fila = _construirFila(datos, hoja);

        // ⭐ NUEVO: ASEGURAR TÍTULOS EN REGISTRO / REGISTRO_PISOS
        const headers = [['Comentario', 'Hora Registro']];
        if (datos.tipoHoja === 'Registro') {
            const headerRange = hoja.getRange(1, 45, 1, 2); // Columnas AS y AT
            if (!headerRange.getValues()[0][0]) {
                headerRange.setValues(headers);
                headerRange.setFontWeight('bold');
            }
        } else if (datos.tipoHoja === 'Registro_pisos') {
            const headerRange = hoja.getRange(1, 127, 1, 2); // Columnas DW y DX
            if (!headerRange.getValues()[0][0]) {
                headerRange.setValues(headers);
                headerRange.setFontWeight('bold');
            }
        }

        if (filaIndex > 0) {
            hoja.getRange(filaIndex, 1, 1, fila.length).setValues([fila]);
            return { success: true, mensaje: 'Registro actualizado correctamente' };
        } else {
            hoja.appendRow(fila);
            return { success: true, mensaje: 'Nuevo registro creado exitosamente' };
        }

    } catch (error) {
        console.error('❌ Error guardando:', error);
        return { success: false, error: error.toString() };
    } finally {
        lock.releaseLock();
        console.log('🔓 Bloqueo liberado');
    }
}

function _construirFila(datos, hoja) {
    let fila = [];
    
    const _formatNum = (val) => {
        if (val === undefined || val === null || val === '') return '';
        // Convertir a string y reemplazar punto por coma
        return String(val).replace('.', ',');
    };

    const norm = (arr, i) => {
        const k = (arr && arr[i]) ? arr[i] : {};
        let sc = k.score || '';
        
        if (k.comparacion && sc) {
            sc = k.comparacion + sc.replace('%', '');
            // Asegurar coma en el score si es numérico después del operador
            sc = sc.replace(/(\d+)\.(\d+)/g, '$1,$2');
        }
        
        return { 
            n: k.nombre || '',
            s: sc || '',
            p: _formatNum((k.porcentaje || '').toString().replace('%', '')),
            part: _formatNum((k.participacion || '').toString().replace('%', ''))
        };
    };

    if (datos.tipoHoja === 'Registro_pisos') {
        fila.push(datos.mes, datos.lob, datos.accm || '', datos.om || '');
        
        const numAgentes = REGISTRO_contarAgentesPorLob(datos.lob);
        fila.push(numAgentes || '');
        
        // 1. Keys (3 slots: Index 5-10)
        for (let i = 0; i < 3; i++) { 
            const k = norm(datos.keys, i); 
            fila.push(k.n, k.s); 
        }
        
        fila.push(''); // Separador (Index 11 - Columna L)
        
        // 2. Bonus Structure (17 slots: Index 12-79, inicia en Columna M)
        for (let i = 0; i < 17; i++) { 
            const k = norm(datos.bonus, i); 
            fila.push(k.n, k.s, k.p, k.part);
        }
        
        fila.push(''); // Separador (Index 80)
        
        // 3. Additional Incentive (5 slots: Index 81-100)
        for (let i = 0; i < 5; i++) { 
            const k = norm(datos.additional, i); 
            fila.push(k.n, k.s, k.p, k.part);
        }
        
        fila.push(''); // Separador (Index 101)
        
        // 4. Decreased (5 slots: Index 102-121)
        for (let i = 0; i < 5; i++) { 
            const k = norm(datos.decreased, i); 
            fila.push(k.n, k.s, k.p, k.part);
        }

        // 5. Comentarios y Hora (Columna DW/DX - Index 126/127)
        while (fila.length < 126) fila.push('');
        fila[126] = datos.comentarios || '';
        fila[127] = new Date();
    }
    else if (datos.tipoHoja === 'Registro') {
        fila.push(datos.mes, datos.lob, datos.accm || '', datos.om || '');
        fila.push('');
        
        for (let i = 0; i < 2; i++) { 
            const k = norm(datos.keys, i); 
            fila.push(k.n, k.s); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 4; i++) { 
            const k = norm(datos.bonus, i); 
            fila.push(k.n, k.s, k.p); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 3; i++) { 
            const k = norm(datos.additional, i); 
            fila.push(k.n, k.s, k.p); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 2; i++) { 
            const k = norm(datos.decreased, i); 
            fila.push(k.n, k.s, k.p); 
        }

        // 5. Comentarios y Hora (Columna AS/AT - Index 44/45)
        while (fila.length < 44) fila.push('');
        fila[44] = datos.comentarios || '';
        fila[45] = new Date();

        // 6. Participación (Para no romper las columnas de "Registro", se agregan al final a partir de AU)
        for (let i = 0; i < 4; i++) { const k = norm(datos.bonus, i); fila.push(k.part); }
        for (let i = 0; i < 3; i++) { const k = norm(datos.additional, i); fila.push(k.part); }
        for (let i = 0; i < 2; i++) { const k = norm(datos.decreased, i); fila.push(k.part); }
    }
    
    return fila;
}

function REGISTRO_obtenerEstructuraRegistrada(lob, mes, usuario) {
    try {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojas = [ss.getSheetByName('Registro'), ss.getSheetByName('Registro_pisos')];
        const lobNorm = normalizarKey(lob);
        const mesNorm = normalizarKey(mes);

        for (const hoja of hojas) {
            if (!hoja) continue;
            const nombre = hoja.getName();
            const data = hoja.getDataRange().getValues();
            const userNorm = normalizarTexto(usuario.nombre);
            const rol = usuario.rol;
            
            for (let i = 1; i < data.length; i++) {
                let mesFila = data[i][0];
                if (mesFila instanceof Date) {
                    mesFila = Utilities.formatDate(mesFila, Session.getScriptTimeZone(), "yyyy-MM-dd");
                }
                
                if (normalizarKey(mesFila) === mesNorm && normalizarKey(data[i][1]) === lobNorm) {
                    // Validar OM si el rol no es administrador
                    const omFila = normalizarTexto(data[i][3]);
                    const tieneAcceso = (rol === 'Reporting Analyst' || rol === 'Audit' || omFila === userNorm);
                    
                    const _parseNum = (val) => {
                        if (!val) return 0;
                        // Convertir coma a punto para que JS lo entienda como número
                        const s = String(val).replace(',', '.').replace(/[^0-9.-]/g, '');
                        return parseFloat(s) || 0;
                    };

                    const isPisos = (nombre === 'Registro_pisos');
                    const kpis = { keys: [], bonusStructure: [], additionalIncentives: [], decreasedKPIs: [] };
                    
                    if (!isPisos) {
                        let idxPart = 46; // Leer participación desde AU
                        const _getPart = () => {
                            if (data[i][idxPart] !== undefined && data[i][idxPart] !== '') {
                                const p = _parseNum(data[i][idxPart]);
                                idxPart++;
                                return p;
                            }
                            idxPart++;
                            return 100;
                        };

                        // 1. Keys (Index 5-8)
                        for (let c = 5; c <= 8; c += 2) {
                            if (data[i][c]) {
                                kpis.keys.push({ 
                                    nombre: String(data[i][c]).trim(),
                                    score: String(data[i][c + 1] || '').trim()
                                });
                            }
                        }
                        // 2. Bonus (Index 10-21)
                        for (let c = 10; c <= 21; c += 3) {
                            if (data[i][c]) {
                                kpis.bonusStructure.push({
                                    nombre: String(data[i][c]).trim(),
                                    score: String(data[i][c + 1] || '').trim(),
                                    porcentaje: _parseNum(data[i][c + 2]),
                                    participacion: _getPart()
                                });
                            } else { idxPart++; }
                        }
                        // 3. Additional (Index 23-31)
                        for (let c = 23; c <= 31; c += 3) {
                            if (data[i][c]) {
                                kpis.additionalIncentives.push({
                                    nombre: String(data[i][c]).trim(),
                                    score: String(data[i][c + 1] || '').trim(),
                                    porcentaje: _parseNum(data[i][c + 2]),
                                    participacion: _getPart()
                                });
                            } else { idxPart++; }
                        }
                        // 4. Decreased (Index 33-38)
                        for (let c = 33; c <= 38; c += 3) {
                            if (data[i][c]) {
                                kpis.decreasedKPIs.push({
                                    nombre: String(data[i][c]).trim(),
                                    score: String(data[i][c + 1] || '').trim(),
                                    porcentaje: _parseNum(data[i][c + 2]),
                                    participacion: _getPart()
                                });
                            } else { idxPart++; }
                        }
                    } else {
                        // COMPLEX (Registro_pisos)
                        // 1. Keys (Index 5-10)
                        for (let c = 5; c <= 9; c += 2) {
                            const val = data[i][c];
                            if (!val || val === 'Keys') continue;
                            kpis.keys.push({
                                nombre: String(val).trim(),
                                score: String(data[i][c + 1] || '').trim()
                            });
                        }
                        // 2. Bonus (Index 12-79)
                        let bonusCount = 0;
                        for (let c = 12; c < 80 && bonusCount < 17; c += 4) {
                            const val = data[i][c];
                            if (!val || val === 'Bonus Structure') continue;
                            kpis.bonusStructure.push({
                                nombre: String(val).trim(),
                                score: String(data[i][c + 1] || '').trim(),
                                porcentaje: _parseNum(data[i][c + 2]),
                                participacion: _parseNum(data[i][c + 3])
                            });
                            bonusCount++;
                        }
                        // 3. Additional (Index 81-100)
                        let additionalCount = 0;
                        for (let c = 81; c < 101 && additionalCount < 5; c += 4) {
                            const val = data[i][c];
                            if (!val || val === 'Additional Incentive') continue;
                            kpis.additionalIncentives.push({
                                nombre: String(val).trim(),
                                score: String(data[i][c + 1] || '').trim(),
                                porcentaje: _parseNum(data[i][c + 2]),
                                participacion: _parseNum(data[i][c + 3])
                            });
                            additionalCount++;
                        }
                        // 4. Decreased (Index 102-121)
                        let decreasedCount = 0;
                        for (let c = 102; c < 122 && decreasedCount < 5; c += 4) {
                            const val = data[i][c];
                            if (!val || val === 'Decreased') continue;
                            kpis.decreasedKPIs.push({
                                nombre: String(val).trim(),
                                score: String(data[i][c + 1] || '').trim(),
                                porcentaje: _parseNum(data[i][c + 2]),
                                participacion: _parseNum(data[i][c + 3])
                            });
                            decreasedCount++;
                        }
                    }

                    const omEncontrado = String(data[i][3]).trim();
                    const lobEncontrado = String(data[i][1]).trim();
                    const configSalario = REGISTRO_obtenerSalarioPorLob(lobEncontrado, usuario);

                        fechaFormateada = Utilities.formatDate(fechaFormateada, Session.getScriptTimeZone(), 'yyyy-MM-dd');

                const lobFila = String(datos[i][1] || '').trim();
                const omFila  = String(datos[i][3] || '').trim();
                if (!lobFila || normalizarTexto(omFila) !== normalizarTexto(nombreOM)) continue;
                filasOM++;

                let mesFila = datos[i][0];
                if (mesFila instanceof Date) {
                    mesFila = Utilities.formatDate(mesFila, zonaHoraria, 'yyyy-MM-dd');
                } else {
                    mesFila = String(mesFila || '').trim();
                }
                if (!mesFila) continue;

                const numMesEst = _resolverNumeroMesDesdeValor(datos[i][0]);
                if (!numMesEst) continue;

                const lobKey = normalizarTexto(lobFila);
                const existing = lobMesMap[lobKey];

                // ⭐ PRIORIDAD: el mes más cercano al actual sin pasarlo
                // Si numMesEst <= mesActual, es candidato válido
                // Gana el candidato válido con número más alto (más reciente)
                // Si todos son futuros (> mesActual), usar el más bajo (más antiguo disponible)
                const esValido     = numMesEst <= mesActualNumero;
                const existeValido = existing && existing.numMesEst <= mesActualNumero;

                let reemplazar = false;

                if (!existing) {
                    reemplazar = true;
                } else if (esValido && !existeValido) {
                    // El nuevo es válido (pasado/actual) y el existente es futuro → reemplazar
                    reemplazar = true;
                } else if (esValido && existeValido) {
                    // Ambos válidos → preferir el más reciente (mayor número)
                    reemplazar = numMesEst > existing.numMesEst;
                } else if (!esValido && !existeValido) {
                    // Ambos son futuros → preferir el más antiguo (menor número, más cercano al actual)
                    reemplazar = numMesEst < existing.numMesEst;
                }
                // Si nuevo es futuro y existente es válido → NO reemplazar

                if (reemplazar) {
                    lobMesMap[lobKey] = { lob: lobFila, mesEstructura: mesFila, numMesEst };
                    Logger.log(`   ✅ LOB: "${lobFila}" | Mes: "${mesFila}" | NumMes: ${numMesEst} | ${esValido ? '✅ válido' : '⚠️ futuro'}`);
                } else {
                    Logger.log(`   ⏭️ LOB: "${lobFila}" | Mes: "${mesFila}" | NumMes: ${numMesEst} | descartado (ya hay mejor: ${existing?.numMesEst})`);
                }
            }
            Logger.log(`   → Filas del OM en ${hoja.getName()}: ${filasOM}`);
        });

        const lobsEncontradas = Object.values(lobMesMap);
        Logger.log(`\n📋 Total LOBs para OM "${nombreOM}": ${lobsEncontradas.length}`);
        lobsEncontradas.forEach(item => {
            Logger.log(`   → "${item.lob}" | estructura: "${item.mesEstructura}" | numMes: ${item.numMesEst} | ${item.numMesEst <= mesActualNumero ? '✅' : '⚠️ futuro'}`);
        });

        if (lobsEncontradas.length === 0) {
            return { success: false, error: `No se encontraron LOBs para el OM "${nombreOM}"` };
        }

        const resultados = [];
        let totalAgentesGeneral = 0, totalInversionReal = 0, totalInversionTeorica = 0;

        for (const item of lobsEncontradas) {
            Logger.log(`\n🔍 ===== Simulando: "${item.lob}" =====`);
            Logger.log(`   Estructura: "${item.mesEstructura}" (mes ${item.numMesEst}) | CDM mes: ${mesActualNumero}`);
            try {
                const simResult = REGISTRO_simularConMetricasReales(
                    item.lob,
                    item.mesEstructura,
                    mesActualNumero
                );

                if (!simResult || !simResult.success) {
                    Logger.log(`⚠️ Error: ${simResult?.error}`);
                    resultados.push({
                        lob: item.lob, mes: item.mesEstructura,
                        error: simResult?.error || 'Sin datos',
                        totalLOB: 0, hc: 0
                    });
                    continue;
                }

                Logger.log(`   inversionReal: $${simResult.inversionReal?.toLocaleString()} | lobEnCDM: ${simResult.lobEnCDM}`);

                resultados.push({
                    lob:                  item.lob,
                    mes:                  item.mesEstructura,
                    fecha:                item.mesEstructura,
                    hc:                   simResult.hc || 0,
                    salarioBase:          simResult.configuracion?.salario || 0,
                    porcentajeBono:       simResult.configuracion?.porcentaje || 0,
                    bonoMaximoPorAgente:  simResult.configuracion?.bonoMaximo || 0,
                    inversionReal:        simResult.inversionReal,
                    inversionTeorica:     simResult.inversionTeorica,
                    totalLOB:             simResult.inversionReal,
                    eficiencia:           simResult.eficienciaReal || 0,
                    tasaCumplimiento:     simResult.estadisticas?.tasaCumplimiento || 0,
                    ahorro:               simResult.ahorro || 0,
                    estadisticas:         simResult.estadisticas || {},
                    desglosePorCategoria: simResult.desglosePorCategoria || {},
                    top10Ganadores:       simResult.top10Ganadores || [],
                    tipoHoja:             simResult.tipoHoja || ''
                });

                totalAgentesGeneral   += simResult.hc || 0;
                totalInversionReal    += simResult.inversionReal;
                totalInversionTeorica += simResult.inversionTeorica;

            } catch (err) {
                Logger.log(`❌ Excepción: ${err.message}`);
                resultados.push({
                    lob: item.lob, mes: item.mesEstructura,
                    error: err.message, totalLOB: 0, hc: 0
                });
            }
        }

        Logger.log(`\n🎯 ========== RESUMEN FINAL ==========`);
        Logger.log(`   Inv. real:    $${totalInversionReal.toLocaleString()}`);
        Logger.log(`   Inv. teórica: $${totalInversionTeorica.toLocaleString()}`);
        Logger.log(`   Eficiencia:   ${totalInversionTeorica > 0 ? ((totalInversionReal/totalInversionTeorica)*100).toFixed(2) : 0}%`);

        const lobsExitosas     = resultados.filter(r => !r.error);
        const lobsConError     = resultados.filter(r =>  r.error);
        const eficienciaGlobal = totalInversionTeorica > 0
            ? (totalInversionReal / totalInversionTeorica) * 100 : 0;

        return {
            success:        true,
            om:             nombreOM,
            totalLOBs:      lobsEncontradas.length,
            lobsProcesadas: lobsExitosas.length,
            lobsConError:   lobsConError.length,
            resultados:     lobsExitosas.sort((a, b) => b.totalLOB - a.totalLOB),
            lobsErrores:    lobsConError,
            resumen: {
                totalAgentes:      totalAgentesGeneral,
                inversionTotal:    totalInversionReal,
                inversionTeorica:  totalInversionTeorica,
                bonoMaximoTotal:   totalInversionTeorica,
                ahorro:            totalInversionTeorica - totalInversionReal,
                eficienciaGlobal,
                promedioPorAgente: totalAgentesGeneral > 0
                    ? totalInversionReal / totalAgentesGeneral : 0
            },
            fechaSimulacion: new Date().toISOString()
        };

    } catch (error) {
        Logger.log(`❌ Error general: ${error.message}\nStack: ${error.stack}`);
        return { success: false, error: error.toString() };
    }
}

function _resolverNumeroMesDesdeValor(valor) {
    if (!valor) return null;
    if (valor instanceof Date) return valor.getMonth() + 1;
    if (typeof valor === 'number') return valor >= 1 && valor <= 12 ? valor : null;

    const s = String(valor).trim();

    if (/^\d{4}-\d{2}-\d{2}/.test(s)) {
        const n = parseInt(s.split('-')[1]);
        return n >= 1 && n <= 12 ? n : null;
    }

    // ⭐ toLowerCase() maneja mayúsculas: "Abril" → "abril", "MAYO" → "mayo"
    const mapa = {
        enero:1, january:1,
        febrero:2, february:2,
        marzo:3, march:3,
        abril:4, april:4,
        mayo:5, may:5,
        junio:6, june:6,
        julio:7, july:7,
        agosto:8, august:8,
        septiembre:9, september:9,
        octubre:10, october:10,
        noviembre:11, november:11,
        diciembre:12, december:12
    };
    const clave = s.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').trim();
    return mapa[clave] || null;
}

function REGISTRO_simularTodasLOBsPorRegion(region) {
    try {
        Logger.log('\n🚀 ========== SIMULACIÓN MASIVA REGIÓN: ' + region + ' ==========');

        const mesActualNumero = parseInt(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'M'));
        Logger.log('📅 Mes actual CDM: ' + mesActualNumero);

        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');

        // ========================================
        // PASO 1: Leer LOBs de Configuracion que tengan salario
        // Estas son las unicas que tienen datos validos para simular
        // ========================================
        const hojaConfig = ss.getSheetByName('Configuracion');
        if (!hojaConfig) {
            return { success: false, error: 'No existe la hoja Configuracion' };
        }

        const dataConfig = hojaConfig.getDataRange().getValues();
        // Col A=Mes, B=OM, C=LOB, D=Salario_Base, E=%_Bono
        const lobsEnConfig = {};

        for (let i = 1; i < dataConfig.length; i++) {
            const lobConfig  = String(dataConfig[i][2] || '').trim();
            const salario    = String(dataConfig[i][3] || '').trim();
            const porcentaje = String(dataConfig[i][4] || '').trim();

            if (!lobConfig || !salario || salario === '0' || salario === '') continue;

            // Guardar la entrada mas reciente por LOB (la ultima en la hoja)
            lobsEnConfig[normalizarTexto(lobConfig)] = {
                lob:        lobConfig,
                salario:    parseFloat(salario.replace(',', '.'))   || 0,
                porcentaje: parseFloat(porcentaje.replace(',', '.')) || 0
            };
        }

        Logger.log('LOBs en Configuracion con salario: ' + Object.keys(lobsEnConfig).length);
        Logger.log(JSON.stringify(Object.values(lobsEnConfig).map(function(l) { return l.lob; })));

        if (Object.keys(lobsEnConfig).length === 0) {
            return { success: false, error: 'No hay LOBs con salario configurado en Configuracion' };
        }

        // ========================================
        // PASO 2: Si hay region, filtrar por H_C
        // Si el usuario es Reporting Analyst sin region especifica, usar todas las de Configuracion
        // ========================================
        let lobsASimular = Object.values(lobsEnConfig).map(function(l) { return l.lob; });

        if (region && region !== 'TODAS') {
            const lobsDeRegion = REGISTRO_obtenerLobsPorRegion(region);
            Logger.log('LOBs en H_C para region "' + region + '": ' + lobsDeRegion.length);
            Logger.log(JSON.stringify(lobsDeRegion));

            // Filtrar: solo LOBs que estan EN la region Y EN Configuracion
            lobsASimular = lobsDeRegion.filter(function(lob) {
                return !!lobsEnConfig[normalizarTexto(lob)];
            });

            Logger.log('LOBs finales (region interseccion Configuracion): ' + lobsASimular.length);
            Logger.log(JSON.stringify(lobsASimular));

            if (lobsASimular.length === 0) {
                return {
                    success: false,
                    error: 'Ninguna LOB de la region "' + region + '" tiene salario configurado en Configuracion'
                };
            }
        }

        // ========================================
        // PASO 3: Para cada LOB, encontrar su estructura mas reciente
        // buscando en Registro y Registro_pisos
        // ========================================
        const hojas = [ss.getSheetByName('Registro'), ss.getSheetByName('Registro_pisos')];
        const zonaHoraria = Session.getScriptTimeZone();
        const lobMesMap = {};

        hojas.forEach(function(hoja) {
            if (!hoja) return;
            const data = hoja.getDataRange().getValues();

            for (let i = 1; i < data.length; i++) {
                const lobFila = String(data[i][1] || '').trim();
                if (!lobFila) continue;

                // Solo procesar si esta en nuestra lista de LOBs a simular
                const lobNorm = normalizarTexto(lobFila);
                const estaEnLista = lobsASimular.some(function(l) {
                    return normalizarTexto(l) === lobNorm;
                });
                if (!estaEnLista) continue;

                let mesFila = data[i][0];
                if (mesFila instanceof Date) {
                    mesFila = Utilities.formatDate(mesFila, zonaHoraria, 'yyyy-MM-dd');
                } else {
                    mesFila = String(mesFila || '').trim();
                }
                if (!mesFila) continue;

                const numMesEst = _resolverNumeroMesDesdeValor(data[i][0]);
                if (!numMesEst) continue;

                const existing = lobMesMap[lobNorm];
                const esValido     = numMesEst <= mesActualNumero;
                const existeValido = existing && existing.numMesEst <= mesActualNumero;

                let reemplazar = false;
                if (!existing) {
                    reemplazar = true;
                } else if (esValido && !existeValido) {
                    reemplazar = true;
                } else if (esValido && existeValido) {
                    reemplazar = numMesEst > existing.numMesEst;
                } else if (!esValido && !existeValido) {
                    reemplazar = numMesEst < existing.numMesEst;
                }

                if (reemplazar) {
                    lobMesMap[lobNorm] = {
                        lob:        lobFila,
                        mes:        mesFila,
                        numMesEst:  numMesEst
                    };
                }
            }
        });

        // ========================================
        // PASO 4: LOBs que estan en Configuracion pero NO en Registro
        // las incluimos con mes calculado desde Configuracion
        // ========================================
        lobsASimular.forEach(function(lob) {
            const lobNorm = normalizarTexto(lob);
            if (!lobMesMap[lobNorm]) {
                // No tiene estructura en Registro — buscar su mes en Configuracion
                for (let i = 1; i < dataConfig.length; i++) {
                    const lobConf = normalizarTexto(String(dataConfig[i][2] || ''));
                    if (lobConf === lobNorm) {
                        const mesConf = String(dataConfig[i][0] || '').trim();
                        const numMes  = _resolverNumeroMesDesdeValor(dataConfig[i][0]);
                        if (mesConf && numMes) {
                            lobMesMap[lobNorm] = { lob: lob, mes: mesConf, numMesEst: numMes };
                            Logger.log('LOB sin Registro, usando mes de Configuracion: "' + lob + '" -> ' + mesConf);
                        }
                        break;
                    }
                }
            }
        });

        const lobsConEstructura = Object.values(lobMesMap);
        Logger.log('LOBs con estructura para simular: ' + lobsConEstructura.length);

        if (lobsConEstructura.length === 0) {
            return {
                success: false,
                error: 'Ninguna LOB tiene estructura registrada'
            };
        }

        // ========================================
        // PASO 5: Simular cada LOB
        // ========================================
        const resultados = [];
        let totalAgentes = 0, totalInversionReal = 0, totalInversionTeorica = 0;

        for (var idx = 0; idx < lobsConEstructura.length; idx++) {
            const item = lobsConEstructura[idx];
            Logger.log('\n🔍 Simulando: "' + item.lob + '" | Estructura: "' + item.mes + '" | CDM mes: ' + mesActualNumero);

            try {
                const simResult = REGISTRO_simularConMetricasRealesMasivo(
                    item.lob,
                    item.mes,
                    mesActualNumero
                );

                if (!simResult || !simResult.success) {
                    Logger.log('Error en "' + item.lob + '": ' + (simResult ? simResult.error : 'sin respuesta'));
                    resultados.push({
                        lob:      item.lob,
                        mes:      item.mes,
                        error:    simResult ? simResult.error : 'Error desconocido',
                        totalLOB: 0,
                        hc:       0
                    });
                    continue;
                }

                resultados.push({
                    lob:                  item.lob,
                    mes:                  item.mes,
                    fecha:                item.mes,
                    hc:                   simResult.hc || 0,
                    salarioBase:          simResult.configuracion ? simResult.configuracion.salario     : 0,
                    porcentajeBono:       simResult.configuracion ? simResult.configuracion.porcentaje  : 0,
                    bonoMaximoPorAgente:  simResult.configuracion ? simResult.configuracion.bonoMaximo  : 0,
                    inversionReal:        simResult.inversionReal     || 0,
                    inversionTeorica:     simResult.inversionTeorica  || 0,
                    totalLOB:             simResult.inversionReal     || 0,
                    eficiencia:           simResult.eficienciaReal    || 0,
                    tasaCumplimiento:     simResult.estadisticas ? simResult.estadisticas.tasaCumplimiento : 0,
                    ahorro:               simResult.ahorro            || 0,
                    estadisticas:         simResult.estadisticas      || {},
                    desglosePorCategoria: simResult.desglosePorCategoria || {},
                    top10Ganadores:       simResult.top10Ganadores    || [],
                    tipoHoja:             simResult.tipoHoja          || ''
                });

                totalAgentes          += simResult.hc            || 0;
                totalInversionReal    += simResult.inversionReal  || 0;
                totalInversionTeorica += simResult.inversionTeorica || 0;

            } catch (err) {
                Logger.log('Excepcion en "' + item.lob + '": ' + err.message);
                resultados.push({
                    lob:      item.lob,
                    mes:      item.mes,
                    error:    err.message,
                    totalLOB: 0,
                    hc:       0
                });
            }
        }

        const lobsExitosas     = resultados.filter(function(r) { return !r.error; });
        const lobsConError     = resultados.filter(function(r) { return  r.error; });
        const eficienciaGlobal = totalInversionTeorica > 0
            ? (totalInversionReal / totalInversionTeorica) * 100 : 0;

        Logger.log('\n🎯 RESULTADO REGION "' + region + '":');
        Logger.log('   Teorico:    $' + totalInversionTeorica);
        Logger.log('   Real CDM_' + mesActualNumero + ': $' + totalInversionReal);
        Logger.log('   Eficiencia: ' + eficienciaGlobal.toFixed(1) + '%');

        return {
            success:        true,
            region:         region,
            totalLOBs:      lobsASimular.length,
            lobsProcesadas: lobsExitosas.length,
            lobsConError:   lobsConError.length,
            resultados:     lobsExitosas.sort(function(a, b) { return b.totalLOB - a.totalLOB; }),
            lobsErrores:    lobsConError,
            resumen: {
                totalAgentes:     totalAgentes,
                inversionTotal:   totalInversionReal,
                inversionTeorica: totalInversionTeorica,
                ahorro:           totalInversionTeorica - totalInversionReal,
                eficienciaGlobal: eficienciaGlobal,
                promedioPorAgente: totalAgentes > 0 ? totalInversionReal / totalAgentes : 0
            },
            fechaSimulacion: new Date().toISOString()
        };

    } catch (error) {
        Logger.log('Error general: ' + error.message + '\nStack: ' + error.stack);
        return { success: false, error: error.toString() };
    }
}



function REGISTRO_simularConMetricasRealesMasivo(lob, mesEstructura, numeroMesCDM) {
    try {
        Logger.log(`\n🚀 [MASIVO] LOB: "${lob}" | Estructura: "${mesEstructura}" | CDM: ${numeroMesCDM}`);
 
        if (!lob || !mesEstructura) return { success: false, error: 'LOB y mes son obligatorios' };
 
        const datosUsuarioActivo = REGISTRO_obtenerDatosUsuarioActivo();
        if (!datosUsuarioActivo) return { success: false, error: 'Usuario no encontrado' };
 
        // ⭐ PASO 1: Obtener estructura del Registro (por mes histórico)
        let est = REGISTRO_obtenerEstructuraRegistrada(lob, mesEstructura, datosUsuarioActivo);
        if (!est || !est.encontrado || !est.registroOriginal) {
            return { success: false, error: `No hay estructura para ${lob} en ${mesEstructura}` };
        }
        Logger.log(`✅ Estructura encontrada: ${est.tipoHoja}`);
 
        // ⭐ PASO 2: Cargar CDM con el mes ACTUAL (numeroMesCDM), NO el mes de la estructura
        Logger.log(`📊 Cargando CDM_${numeroMesCDM}...`);
        let dataCDM = null;
        let lobExisteEnCDM = false;
 
        try {
            dataCDM = REGISTRO_obtenerDataCDMCached(numeroMesCDM);
            if (dataCDM && dataCDM.length > 1) {
                Logger.log(`✅ CDM_${numeroMesCDM}: ${dataCDM.length} filas`);
                const lobNorm = normalizarTexto(lob);
                const headers = dataCDM[0];
                let colSubLOB = headers.findIndex(h => {
                    const hn = String(h).toLowerCase().trim();
                    return hn === 'sublob' || hn === 'sub lob' || hn === 'lob';
                });
                if (colSubLOB === -1) colSubLOB = 2;
 
                for (let i = 1; i < dataCDM.length; i++) {
                    if (normalizarTexto(String(dataCDM[i][colSubLOB] || '')) === lobNorm) {
                        lobExisteEnCDM = true;
                        break;
                    }
                }
                Logger.log(`   LOB "${lob}" en CDM_${numeroMesCDM}: ${lobExisteEnCDM ? '✅ SÍ' : '❌ NO'}`);
            } else {
                Logger.log(`⚠️ CDM_${numeroMesCDM} vacío`);
            }
        } catch (e) {
            Logger.log(`⚠️ Error CDM_${numeroMesCDM}: ${e.message}`);
        }
 
        // ⭐ PASO 3: Contar agentes y cargar datos de usuario
        const hc = REGISTRO_contarAgentesPorLob(lob);
        if (hc === 0) return { success: false, error: `Sin agentes en ${lob}` };
 
        const agentesPermitidos = REGISTRO_obtenerAgentesPermitidos();
        const ssU = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaU = ssU.getSheetByName('Usuario');
        const dataUsuario = hojaU ? hojaU.getDataRange().getValues() : [];
 
        // ⭐ PASO 4: Validar KPIs
        // Usamos numeroMesCDM como el "numeroMes" de validación para paridad de horas
        let resultadosV;
        if (lobExisteEnCDM && dataCDM && dataCDM.length > 1) {
            Logger.log(`✅ Validando con CDM_${numeroMesCDM} real`);
            resultadosV = REGISTRO_validarKPIsPorCategoria(
                est.registroOriginal, lob, dataCDM, dataUsuario,
                est.tipoHoja, agentesPermitidos, numeroMesCDM
            );
        } else {
            Logger.log(`⚠️ Sin CDM → bonos cero`);
            resultadosV = validarKPIsConValoresCero(
                est.registroOriginal, lob, hc, est.tipoHoja, agentesPermitidos
            );
        }
 
        if (!resultadosV) return { success: false, error: 'Error en validación de KPIs' };
 
        // ⭐ PASO 5: Consolidar totales
        const todosLosAgentes = obtenerTodosLosAgentesDeLOB(lob, agentesPermitidos);
        const bonosPorAgente = {};
        const agentesConBono = new Set();
 
        todosLosAgentes.forEach(ag => {
            bonosPorAgente[ag.bmsId] = {
                bmsId: ag.bmsId, nombre: ag.nombre,
                totalKeys: 0, totalBonus: 0, totalAdditional: 0, totalDecreased: 0,
                bonoFinal: 0, detalleKPIs: []
            };
        });
 
        const procesarCategoria = (cat) => {
            if (!resultadosV[cat]) return;
            resultadosV[cat].forEach(kpi => {
                if (!kpi.validacion) return;
                (kpi.validacion.empleadosCumplen || []).forEach(e => {
                    const bono = e.bono || 0;
                    if (!bonosPorAgente[e.bmsId]) {
                        bonosPorAgente[e.bmsId] = {
                            bmsId: e.bmsId, nombre: e.nombre,
                            totalKeys: 0, totalBonus: 0, totalAdditional: 0, totalDecreased: 0,
                            bonoFinal: 0, detalleKPIs: []
                        };
                    }
                    if (cat === 'keys')                  bonosPorAgente[e.bmsId].totalKeys      += bono;
                    else if (cat === 'bonusStructure')   bonosPorAgente[e.bmsId].totalBonus     += bono;
                    else if (cat === 'additionalIncentives') bonosPorAgente[e.bmsId].totalAdditional += bono;
                    else if (cat === 'decreasedKPIs')   bonosPorAgente[e.bmsId].totalDecreased += Math.abs(bono);
                    if (bono > 0) agentesConBono.add(e.bmsId);
                    bonosPorAgente[e.bmsId].detalleKPIs.push({
                        categoria: cat, kpi: kpi.kpi,
                        score: e.scoreCumplido || kpi.score || '-',
                        bono, esGate: kpi.esGate || false
                    });
                });
            });
        };
 
        ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(procesarCategoria);
 
        let inversionReal = 0;
        Object.keys(bonosPorAgente).forEach(bmsId => {
            const ag = bonosPorAgente[bmsId];
            ag.bonoFinal = Math.max(0, ag.totalKeys + ag.totalBonus + ag.totalAdditional - ag.totalDecreased);
            inversionReal += ag.bonoFinal;
        });
 
        const afectadosKeys = new Set(
            Object.values(bonosPorAgente)
                .filter(ag => ag.detalleKPIs.some(d => d.esGate && d.bono === 0))
                .map(ag => ag.bmsId)
        );
 
        // ⭐ PASO 6: Teórico (NO cambia — siempre HC × Salario × %Bono)
        const configSalario = REGISTRO_obtenerSalarioPorLob(lob);
        const salarioBase = configSalario.encontrado ? configSalario.salario : 0;
        const porcentajeBono = configSalario.encontrado ? configSalario.porcentaje : 0;
        const inversionTeorica = hc * (salarioBase * (porcentajeBono / 100));
 
        Logger.log(`\n💰 [${lob}] Teórico: $${inversionTeorica.toLocaleString()} | Real CDM_${numeroMesCDM}: $${inversionReal.toLocaleString()}`);
 
        const top10Ganadores = Object.values(bonosPorAgente).sort((a, b) => b.bonoFinal - a.bonoFinal);
 
        const desglosePorCategoria = {
            keys: (resultadosV.keys || []).map(k => ({
                kpi: k.kpi || 'N/A',
                totalCumplen: k.validacion?.totalCumplenCondicion || 0,
                totalNoCumplen: k.validacion?.empleadosNoCumplen?.length || 0,
                totalSinDatos: k.validacion?.empleadosSinDatos?.length || 0
            })),
            bonusStructure: (resultadosV.bonusStructure || []).map(k => ({
                kpi: k.kpi || 'N/A',
                porcentaje: k.porcentaje || 0,
                totalCumplen: k.validacion?.totalCumplenCondicion || 0,
                montoTotal: (k.validacion?.empleadosCumplen || []).reduce((s, e) => s + (e.bono || 0), 0)
            })).sort((a, b) => b.montoTotal - a.montoTotal),
            additionalIncentives: (resultadosV.additionalIncentives || []).map(k => ({
                kpi: k.kpi || 'N/A',
                porcentaje: k.porcentaje || 0,
                totalCumplen: k.validacion?.totalCumplenCondicion || 0,
                montoTotal: (k.validacion?.empleadosCumplen || []).reduce((s, e) => s + (e.bono || 0), 0)
            })).sort((a, b) => b.montoTotal - a.montoTotal),
            decreasedKPIs: (resultadosV.decreasedKPIs || []).map(k => ({
                kpi: k.kpi || 'N/A',
                porcentaje: k.porcentaje || 0,
                totalCumplen: k.validacion?.totalCumplenCondicion || 0,
                montoTotal: (k.validacion?.empleadosCumplen || []).reduce((s, e) => s + Math.abs(e.bono || 0), 0)
            })).sort((a, b) => b.montoTotal - a.montoTotal)
        };
 
        return {
            success: true,
            lob, mes: mesEstructura, numeroMes: numeroMesCDM, hc,
            tipoHoja: est.tipoHoja,
            inversionTeorica,   // ⭐ Teórico: sin cambios
            inversionReal,      // ⭐ Real: CDM del mes actual
            ahorro: inversionTeorica - inversionReal,
            eficienciaReal: inversionTeorica > 0 ? (inversionReal / inversionTeorica) * 100 : 0,
            configuracion: { salario: salarioBase, porcentaje: porcentajeBono, bonoMaximo: salarioBase * (porcentajeBono / 100) },
            estadisticas: {
                totalAgentes: hc,
                agentesCobran: agentesConBono.size,
                agentesNoCobran: hc - agentesConBono.size,
                agentesAfectadosPorKeys: afectadosKeys.size,
                tasaCumplimiento: hc > 0 ? (agentesConBono.size / hc) * 100 : 0
            },
            desglosePorCategoria,
            resultadosDetallados: resultadosV,
            top10Ganadores,
            fechaSimulacion: new Date().toISOString(),
            lobEnCDM: lobExisteEnCDM
        };
 
    } catch (error) {
        Logger.log(`❌ [MASIVO] Error: ${error.message}\nStack: ${error.stack}`);
        return { success: false, error: error.message || error.toString() };
    }
}
 

function REGISTRO_obtenerLobsRegistradasPorRegion(region) {
    // Paso 1: obtener todas las LOBs de la region desde H_C
    const lobsEnHC = REGISTRO_obtenerLobsPorRegion(region);
    
    if (lobsEnHC.length === 0) {
        Logger.log('No se encontraron LOBs en H_C para region: ' + region);
        return [];
    }

    Logger.log('LOBs en H_C: ' + lobsEnHC.length);

    // Paso 2: filtrar solo las que tienen configuracion (salario) en hoja Configuracion
    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hojaConfig = ss.getSheetByName('Configuracion');
    
    if (!hojaConfig) {
        Logger.log('No existe hoja Configuracion — retornando LOBs de H_C sin filtrar');
        return lobsEnHC;
    }

    const dataConfig = hojaConfig.getDataRange().getValues();
    const lobsEnConfig = new Set();

    for (let i = 1; i < dataConfig.length; i++) {
        const lobConfig = String(dataConfig[i][2] || '').trim(); // Col C = LOB
        const salario = String(dataConfig[i][3] || '').trim();   // Col D = Salario_Base

        // Solo contar si tiene salario configurado
        if (lobConfig && salario && salario !== '0' && salario !== '') {
            lobsEnConfig.add(lobConfig);
        }
    }

    Logger.log('LOBs con configuracion: ' + lobsEnConfig.size);

    // Paso 3: interseccion — LOBs que estan en H_C (region) Y tienen configuracion
    const resultado = lobsEnHC.filter(function(lob) {
        // Comparacion tolerante: normalizar ambos lados
        const lobNorm = normalizarTexto(lob);
        for (const lobConf of lobsEnConfig) {
            if (normalizarTexto(lobConf) === lobNorm) return true;
        }
        return false;
    });

    Logger.log('LOBs finales (H_C interseccion Configuracion) para region "' + region + '": ' + resultado.length);
    Logger.log(JSON.stringify(resultado));

    return resultado;
}
 
function REGISTRO_simularTodasLOBsPorRegionConPVE(region, pve, porc) {
    const sim = REGISTRO_simularTodasLOBsPorRegion(region);
    if (!sim.success) return sim;
    const prep = (pve * porc) / 100;
    sim.resultados = sim.resultados.map(r => ({
        ...r,
        pctDelPVE: prep > 0 ? parseFloat((r.totalLOB / prep * 100).toFixed(2)) : 0
    }));
    const inv = sim.resumen.inversionTotal;
    const pct = prep > 0 ? (inv / prep * 100) : 0;
    sim.pveConfig = { pveTotal: pve, porcPVE: porc, presupuestoDisponible: prep };
    sim.estadoPresupuesto = {
        texto: pct > 100 ? '❌ EXCEDE' : '✅ DENTRO',
        color: pct > 100 ? '#e74c3c' : '#27ae60',
        pctUsado: parseFloat(pct.toFixed(2)),
        inversionTotal: inv,
        diferencia: prep - inv
    };
    return sim;
}

    }
}

function REGISTRO_validarKPI1(llave, condicionScore, porcentajePunto, dataCDM, dataUsuario, agentesPermitidos, lobActual, numeroMes = null, fechaSeleccionada = null, esDecrease = false, esProrrateado = true) {
    const resultado = {

        condicionScore, 
        totalCoincidencias: 0, 
        totalCumplenCondicion: 0, 
        empleadosCumplen: [], 
        empleadosNoCumplen: [], 
        empleadosSinDatos: [] 
    };

    if (!condicionScore || condicionScore === 'N/A' || condicionScore.trim() === '') {
        return resultado;
    }

    const normalizar = s => String(s || '')
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase()
        .trim()
        .replace(/\s+/g, ' ');
    
    const llaveNorm = normalizar(llave);
    const lobActualNorm = normalizar(lobActual);

    // ========================================
    // DETECTAR SI ES BONO FIJO (sin operador)
    // ========================================
    const condicionLimpia = condicionScore;

    // ========================================
    // DETECTAR SI ES BONO FIJO
    // ========================================
    const esBonoFijo = !condicionLimpia.match(/(<=|>=|<|>|=)/);
    let valorBonoFijo = 0;
    let operador = null;
    let valorCondicion = 0;
    
    if (esBonoFijo) {
        valorBonoFijo = parseFloat(condicionLimpia.replace(/[^\d.]/g, '')) || 0;
        Logger.log(`💰 Bono Fijo detectado-Limpio: ${llave} = $${valorBonoFijo}`);
    } else {
        const match = condicionLimpia.match(/(<=|>=|<|>|=)\s*(\d+(?:\.\d+)?)/);
        if (!match) {
            Logger.log('❌ Condición de score no válida:', condicionLimpia);
            return resultado;
        }
        operador = match[1];
        valorCondicion = parseFloat(match[2]);
    }

    const todosLosAgentes = obtenerTodosLosAgentesDeLOB(lobActual, agentesPermitidos);
    
    Logger.log(`\n🔍 validarKPI1 - LOB: ${lobActual}`);
    Logger.log(`   KPI: "${llave}"`);
    Logger.log(`   Condición: ${condicionScore}`);
    Logger.log(`   Es Decrease: ${esDecrease ? 'SÍ' : 'NO'}`);
    Logger.log(`   Peso KPI (porcentajePunto): ${(porcentajePunto * 100).toFixed(2)}%`);
    
    const agentesConDatos = new Set();

    // ========================================
    // ⭐ PASO 1: LEER CONFIGURACIÓN DE LA LOB
    // ========================================
    let salarioBaseLOB = 0;
    let porcentajeBonoGlobal = 0;
    
    try {
        const ssConfig = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaConfig = ssConfig.getSheetByName('Configuracion');
        
        if (!hojaConfig) {
            Logger.log('❌ ERROR: No se encontró la hoja Configuracion');
            return resultado;
        }
        
        const dataConfig = hojaConfig.getDataRange().getValues();
        const lobActualNormConfig = normalizar(lobActual);
        let encontrado = false;
        
        for (let c = 1; c < dataConfig.length; c++) {
            const lobConfig = normalizar(String(dataConfig[c][2] || ''));
            
            if (lobConfig === lobActualNormConfig) {
                const salarioConfigRaw = String(dataConfig[c][3] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                const porcentajeConfigRaw = String(dataConfig[c][4] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                
                salarioBaseLOB = parseFloat(salarioConfigRaw);
                porcentajeBonoGlobal = parseFloat(porcentajeConfigRaw);
                
                if (salarioBaseLOB > 0 && porcentajeBonoGlobal > 0) {
                    encontrado = true;
                    Logger.log(`✅ Config LOB: Salario Base=$${salarioBaseLOB}, %Bono=${porcentajeBonoGlobal}%`);
                    break;
                }
            }
        }
        
        if (!encontrado) {
            Logger.log(`❌ ERROR: No se encontró configuración para LOB "${lobActual}"`);
            return resultado;
        }
        
    } catch (error) {
        Logger.log(`❌ ERROR al leer Configuracion: ${error.message}`);
        return resultado;
    }

    // ========================================
    // ⭐⭐⭐ PASO 2: CALCULAR HORAS MÁXIMAS DESDE H_C (PARIDAD OM)
    // Base dinámica: el valor más alto de horasSemanales en H_C
    // ========================================
    const horasMaxima = Math.max(
        ...Object.values(agentesPermitidos).map(a => a.horasSemanales || 0)
    );
    
    Logger.log(`⏱️ Horas máximas en H_C: ${horasMaxima}h (base para prorrateo)`);

    // ========================================
    // BÚSQUEDA DE COLUMNAS EN CDM
    // ========================================
    const headers = dataCDM[0];
    
    let idxBms = headers.indexOf('BMS ID');
    if (idxBms === -1) idxBms = headers.indexOf('CCMSID');
    if (idxBms === -1) idxBms = 1;
    
    let idxValH = headers.indexOf('Columna H');
    if (idxValH === -1) idxValH = headers.indexOf('KPI_Value');
    if (idxValH === -1) idxValH = headers.indexOf('Score');
    if (idxValH === -1) idxValH = 7;
    
    let idxValF = headers.indexOf('Columna F');
    if (idxValF === -1) idxValF = 5;
    
    let idxLlave = headers.indexOf('Llave');
    if (idxLlave === -1) idxLlave = headers.indexOf('KPI');
    if (idxLlave === -1) idxLlave = 8;

    let idxSubLOB = headers.indexOf('SubLOB');
    if (idxSubLOB === -1) idxSubLOB = 2;

    // ========================================
    // ⭐ SI ES BONO FIJO, ASIGNAR A TODOS
    // ========================================
    if (esBonoFijo) {
        Logger.log(`💰 Asignando bono fijo a todos los agentes (CON PRORRATEO POR HORAS H_C)`);
        
        todosLosAgentes.forEach(ag => {
            let salario = salarioBaseLOB;
            let salarioPersonalizado = false;
            
            for (let u = 1; u < dataUsuario.length; u++) {
                if (String(dataUsuario[u][0]).trim() === ag.bmsId) {
                    let sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                    let salarioAgente = parseFloat(sRaw) || 0;
                    
                    if (salarioAgente > 0 && salarioAgente !== salarioBaseLOB) {
                        salario = salarioAgente;
                        salarioPersonalizado = true;
                    } else if (salarioAgente > 0) {
                        salario = salarioAgente;
                    }
                    break;
                }
            }
            
            const bonoBase = salario * (porcentajeBonoGlobal / 100);
            let bonoFinal = bonoBase * porcentajePunto;

            // ⭐⭐⭐ PRORRATEO POR HORAS H_C (PARIDAD OM) ⭐⭐⭐
            const horasAgente = agentesPermitidos[ag.bmsId]?.horasSemanales || horasMaxima;
            const factorHoras = horasMaxima > 0 ? horasAgente / horasMaxima : 1;
            bonoFinal = bonoFinal * factorHoras;

            Logger.log(`   👤 ${ag.nombre} (${ag.bmsId}): ${horasAgente}h / ${horasMaxima}h = ${(factorHoras * 100).toFixed(1)}% → $${bonoFinal.toFixed(2)}`);
            
            if (esDecrease) {
                bonoFinal = -Math.abs(bonoFinal);
            }
            
            resultado.totalCoincidencias++;
            resultado.totalCumplenCondicion++;
            
            resultado.empleadosCumplen.push({
                bmsId: ag.bmsId,
                nombre: ag.nombre,
                valorF: valorBonoFijo,
                bono: bonoFinal,
                horasContrato: horasAgente,
                horasMaxima: horasMaxima,
                factorHoras: factorHoras,
                salarioPersonalizado: salarioPersonalizado
            });
        });
        
        return resultado;
    }

    // ========================================
    // ⭐ PROCESAR FILAS DE CDM (KPI CON CONDICIÓN)
    // ========================================
    const agentesYaProcesados = new Set();
    for (let row = 1; row < dataCDM.length; row++) {
        const llaveCDMRaw = String(dataCDM[row][idxLlave] || '').trim();
        const llaveCDM = normalizar(llaveCDMRaw);
        
        // Extraer KPI
        let kpiExtraidoCDM = '';
        const posLOB = llaveCDM.indexOf(lobActualNorm);
        
        if (posLOB !== -1) {
            kpiExtraidoCDM = llaveCDM.substring(posLOB + lobActualNorm.length).trim();
        } else {
            kpiExtraidoCDM = llaveCDM;
        }
        
        if (kpiExtraidoCDM !== llaveNorm) {
            continue;
        }

        const bmsId = String(dataCDM[row][idxBms]).trim();
        
        if (!agentesPermitidos[bmsId]) {
            continue;
        }

        const lobAgente = normalizar(agentesPermitidos[bmsId].lob);
        
        if (lobAgente !== lobActualNorm) {
            continue;
        }

        const subLobCDM = normalizar(String(dataCDM[row][idxSubLOB] || ''));
        
        if (subLobCDM && subLobCDM !== lobActualNorm) {
            continue;
        }

        // ⭐ FIX: Si ya procesamos este agente, saltar (evita duplicados CDM)
        if (agentesConDatos.has(bmsId)) {
            continue;
        }
        agentesConDatos.add(bmsId);
        resultado.totalCoincidencias++;

        const valorNumerico = parseFloat(String(dataCDM[row][idxValH]).replace('%', '').trim());
        
        if (isNaN(valorNumerico) || valorNumerico === 0) {
            resultado.empleadosSinDatos.push({ 
                bmsId, 
                nombre: agentesPermitidos[bmsId].nombre, 
                valorF: 0,
                bono: 0,
                sinDatos: true,
                salarioPersonalizado: false,
                horasSemanales: 0,
                horasReales: 0,
                horasEsperadas: 0,
                factorProrrateo: 0
            });
            continue;
        }

        // ⭐ VALIDAR CONDICIÓN
        let cumple = false;
        switch (operador) {
            case '>': cumple = valorNumerico > valorCondicion; break;
            case '>=': cumple = valorNumerico >= valorCondicion; break;
            case '<': cumple = valorNumerico < valorCondicion; break;
            case '<=': cumple = valorNumerico <= valorCondicion; break;
            case '=': cumple = valorNumerico === valorCondicion; break;
        }

        if (resultado.totalCoincidencias <= 5) {
            Logger.log(`   ${bmsId}: ${valorNumerico} ${operador} ${valorCondicion} → ${cumple ? '✅' : '❌'}`);
        }

        // ⭐ DETERMINAR SALARIO - USAR COLUMNA 7 (H)
        let salario = salarioBaseLOB;
        let salarioPersonalizado = false;
        
        for (let u = 1; u < dataUsuario.length; u++) {
            if (String(dataUsuario[u][0]).trim() === bmsId) {
                let sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                let salarioAgente = parseFloat(sRaw) || 0;
                
                if (salarioAgente > 0 && salarioAgente !== salarioBaseLOB) {
                    salario = salarioAgente;
                    salarioPersonalizado = true;
                } else if (salarioAgente > 0) {
                    salario = salarioAgente;
                }
                break;
            }
        }
        
        if (salario === 0) {
            salario = salarioBaseLOB;
        }

        // ========================================
        // ⭐⭐⭐ CÁLCULO CON PRORRATEO POR HORAS H_C (PARIDAD OM) ⭐⭐⭐
        // ========================================
        const bonoBase = salario * (porcentajeBonoGlobal / 100);
        let bonoFinal = bonoBase * porcentajePunto;

        // ⭐⭐⭐ PRORRATEO: horasAgente / horasMaxima ⭐⭐⭐
        const horasAgente = agentesPermitidos[bmsId]?.horasSemanales || horasMaxima;
        const factorHoras = horasMaxima > 0 ? horasAgente / horasMaxima : 1;
        bonoFinal = bonoFinal * factorHoras;

        Logger.log(`   👤 ${agentesPermitidos[bmsId].nombre} (${bmsId}): ${horasAgente}h / ${horasMaxima}h = ${(factorHoras * 100).toFixed(1)}% → $${bonoFinal.toFixed(2)}`);
        
        if (esDecrease && cumple) {
            bonoFinal = -Math.abs(bonoFinal);
        }
        
        const item = { 
            bmsId, 
            nombre: agentesPermitidos[bmsId].nombre, 
            valorF: idxValF !== -1 ? dataCDM[row][idxValF] : '-', 
            valorReal: valorNumerico,
            bono: cumple ? bonoFinal : 0,
            horasContrato: horasAgente,
            horasMaxima: horasMaxima,
            factorHoras: factorHoras,
            salarioPersonalizado: salarioPersonalizado
        };

        if (cumple) { 
            resultado.totalCumplenCondicion++; 
            resultado.empleadosCumplen.push(item); 
        } else {
            resultado.empleadosNoCumplen.push(item);
        }
    }

    // ========================================
    // AGENTES SIN DATOS
    // ========================================
    todosLosAgentes.forEach(ag => {
        if (!agentesConDatos.has(ag.bmsId)) {
            resultado.empleadosSinDatos.push({ 
                bmsId: ag.bmsId, 
                nombre: ag.nombre, 
                valorF: '-', 
                bono: 0, 
                sinDatos: true,
                salarioPersonalizado: false,
                horasContrato: agentesPermitidos[ag.bmsId]?.horasSemanales || horasMaxima,
                horasMaxima: horasMaxima,
                factorHoras: 0
            });
        }
    });

    Logger.log(`\n📊 RESUMEN: Coincidencias=${resultado.totalCoincidencias}, Cumplen=${resultado.totalCumplenCondicion}\n`);

    return resultado;
}
function REGISTRO_validarKPIsPorCategoria(registro, lob, dataCDM, dataUsuario, hoja, agentesPermitidos, numeroMes = null, fechaSeleccionada = null) {
    const resultadosBrutos = { 
        keys: [], 
        bonusStructure: [], 
        additionalIncentives: [], 
        decreasedKPIs: [] 
    };

    const procesar = (cat, start, end, inc, conParticipacion) => {
        for (let c = start; c <= end; c += inc) {
            
            // ⭐ FIX CRÍTICO: validación estricta del nombre del KPI
            const valorCelda = registro[c];
            if (valorCelda === null || valorCelda === undefined) continue;
            
            const kpi = String(valorCelda).trim();
            
            // ⭐ Ignorar celdas vacías, solo espacios, guiones, ceros, o textos muy cortos
            if (!kpi || kpi === '' || kpi === '-' || kpi === '0' || kpi.length < 2) continue;
            
            // ⭐ FIX: El score también debe ser válido y no vacío
            const scoreRaw = String(registro[c + 1] || '').trim();
            if (!scoreRaw || scoreRaw === '' || scoreRaw === '-' || scoreRaw === '0') continue;

            let score = scoreRaw;
            
            const scoreRawLower = score.toLowerCase().replace(/\s/g, '');
            const esGate = cat === 'bonusStructure' && scoreRawLower.endsWith(',key');
            if (esGate) {
                score = score.replace(/\s*,\s*key\s*$/i, '').trim();
            }
            score = score.replace(/(\d),(\d)/g, '$1.$2');

            let peso = parseFloat(String(registro[c + 2] || '0').replace('%', '')) || 0;
            if (peso > 1) peso /= 100;

            // ⭐ FIX: Si el peso es 0, no tiene sentido procesar este KPI
            if (peso === 0 && cat !== 'keys') continue;

            let participation = 1;
            
            if (conParticipacion) {
                const participacionRaw = String(registro[c + 3] || '100');
                console.log(`\n📊 Leyendo participación de columna ${c + 3}: "${participacionRaw}"`);
                
                const participacionNum = parseFloat(participacionRaw.replace('%', '').trim()) || 100;
                participation = participacionNum > 1 ? participacionNum / 100 : participacionNum;
                
                console.log(`   Raw: "${participacionRaw}" → Num: ${participacionNum} → Final: ${participation} (${(participation * 100).toFixed(0)}%)`);
            }
            
            if (cat === 'bonusStructure' || cat === 'decreasedKPIs') {
                console.log('\n🔍 ========== PROCESANDO KPI ==========');
                console.log(`Categoría: ${cat}`);
                console.log(`KPI: ${kpi}`);
                console.log(`Score: ${score}`);
                console.log(`Peso procesado: ${peso} (${(peso * 100).toFixed(2)}%)`);
                console.log(`Participación procesada: ${participation} (${(participation * 100).toFixed(2)}%)`);
                console.log('======================================\n');
            }
            
            const esDecrease = cat === 'decreasedKPIs';
            
            const validacion = REGISTRO_validarKPI1(
                kpi, 
                score, 
                peso * participation,
                dataCDM, 
                dataUsuario, 
                agentesPermitidos, 
                lob, 
                numeroMes, 
                fechaSeleccionada,
                esDecrease
            );
            
            resultadosBrutos[cat].push({ 
                kpi, 
                score, 
                porcentaje: peso, 
                participacion: participation,
                validacion,
                esDecrease: esDecrease,
                esGate: esGate
            });
        }
    };

    if (hoja === 'Registro') {
        procesar('keys', 5, 9, 2, false);
        procesar('bonusStructure', 10, 21, 3, false);
        procesar('additionalIncentives', 23, 31, 3, false);
        procesar('decreasedKPIs', 33, 38, 3, false);

        resultadosBrutos.bonusStructure = REGISTRO_aplicarLogicaGate(resultadosBrutos.bonusStructure);

        return resultadosBrutos;
    } else {
        console.log('\n⭐ Procesando Registro_pisos (Multi-Tier)');
        
        procesar('keys', 5, 10, 2, false);
        procesar('bonusStructure', 12, 79, 4, true);
        procesar('additionalIncentives', 81, 100, 4, true);
        procesar('decreasedKPIs', 102, 121, 4, true);
        
        return REGISTRO_consolidarPisos(resultadosBrutos);
    }
}

// ========================================
// ⭐⭐⭐ LÓGICA DE GATE (PARIDAD OM)
// Si un Gate no cumple, bloquea solo ESE KPI para el agente
// ========================================
function REGISTRO_aplicarLogicaGate(bonusStructureKPIs) {
    if (!bonusStructureKPIs || bonusStructureKPIs.length === 0) return bonusStructureKPIs;

    Logger.log(`\n🔒 ========== LÓGICA GATE ==========`);
    Logger.log(`   Total KPIs Bonus Structure: ${bonusStructureKPIs.length}`);

    const gates = bonusStructureKPIs.filter(kpiObj => kpiObj.esGate === true);

    if (gates.length === 0) {
        Logger.log('✅ No hay Gates en Bonus Structure');
        Logger.log('===================================\n');
        return bonusStructureKPIs;
    }

    Logger.log(`   Gates encontrados: ${gates.length}`);
    gates.forEach(g => Logger.log(`   🔒 Gate: "${g.kpi}" score="${g.score}"`));

    gates.forEach(gate => {
        const val = gate.validacion || {};
        const noCumplen = val.empleadosNoCumplen || [];
        const sinDatos  = val.empleadosSinDatos  || [];

        if (noCumplen.length > 0) {
            Logger.log(`\n   🔒 Gate "${gate.kpi}":`);
            noCumplen.forEach(emp => {
                Logger.log(`      ❌ ${emp.nombre || emp.bmsId} NO cumple → bono $0 en este KPI únicamente`);
            });
        }

        if (sinDatos.length > 0) {
            sinDatos.forEach(emp => {
                Logger.log(`      ⚠️ ${emp.nombre || emp.bmsId} SIN DATOS → bono $0 en este KPI únicamente`);
            });
        }

        const cumplen = val.empleadosCumplen || [];
        if (cumplen.length > 0) {
            Logger.log(`   ✅ Gate "${gate.kpi}" — ${cumplen.length} agente(s) cumplen → bonifican este KPI`);
        }
    });

    Logger.log('\n   ℹ️  Los demás KPIs del agente NO se ven afectados por el Gate');
    Logger.log('===================================\n');

    // ✅ Retornar sin modificar — cada KPI ya tiene su propio resultado correcto
    return bonusStructureKPIs;
}

function REGISTRO_consolidarPisos(resultadosBrutos) {
    const resultadosConsolidados = {};

    // ⭐ Función de consolidación reutilizable para cualquier categoría
    const consolidarCategoria = (cat) => {
        const grupos = {};
        resultadosBrutos[cat].forEach(item => {
            if (!grupos[item.kpi]) grupos[item.kpi] = [];
            grupos[item.kpi].push(item);
        });

        return Object.keys(grupos).map(nombreKpi => {
            const tiers = grupos[nombreKpi];

            // Si solo hay 1 tier, retornarlo tal cual
            if (tiers.length === 1) return tiers[0];

            const maxParticipacion = Math.max(...tiers.map(t => t.participacion || 1));

            console.log(`\n📊 Consolidando multi-tier [${cat}]: ${nombreKpi}`);
            console.log(`   Tiers: ${tiers.length}`);
            console.log(`   Participaciones: ${tiers.map(t => ((t.participacion || 1) * 100).toFixed(0) + '%').join(', ')}`);
            console.log(`   Max participación: ${(maxParticipacion * 100).toFixed(0)}%`);

            // ⭐ PARIDAD OM: esGate se preserva desde el primer tier
            const consolidado = {
                kpi: nombreKpi,
                score: "Multi-Escala",
                porcentaje: tiers[0].porcentaje,
                participacion: maxParticipacion,
                esMultiTier: true,
                esGate: tiers[0].esGate === true,   // ⭐ PRESERVAR esGate
                scoresPorTier: tiers.map(t => ({
                    score: t.score,
                    condicion: t.score,
                    participacion: t.participacion
                })),
                validacion: {
                    totalCoincidencias: tiers[0].validacion.totalCoincidencias,
                    totalCumplenCondicion: 0,
                    empleadosCumplen: [],
                    empleadosNoCumplen: [],
                    empleadosSinDatos: tiers[0].validacion.empleadosSinDatos || []
                }
            };

            const mejorResultadoAgente = {};

            // ⭐ PARIDAD OM: SELECCIONAR MEJOR BONO POR AGENTE (bono > comparison)
            tiers.forEach(tier => {
                console.log(`\n   🔍 Procesando tier: ${tier.score} (participación: ${((tier.participacion || 1) * 100).toFixed(0)}%)`);

                (tier.validacion.empleadosCumplen || []).forEach(e => {
                    console.log(`      Agente ${e.bmsId}: bono=$${e.bono.toFixed(2)}`);

                    if (!mejorResultadoAgente[e.bmsId] || e.bono > mejorResultadoAgente[e.bmsId].bono) {
                        mejorResultadoAgente[e.bmsId] = {
                            ...e,
                            scoreCumplido: tier.score,
                            participacionUsada: tier.participacion
                        };

                        console.log(`      ✅ Mejor resultado actualizado: tier="${tier.score}", part=${((tier.participacion || 1) * 100).toFixed(0)}%`);
                    }
                });

                (tier.validacion.empleadosNoCumplen || []).forEach(e => {
                    if (!mejorResultadoAgente[e.bmsId]) {
                        mejorResultadoAgente[e.bmsId] = {
                            ...e,
                            bono: 0,
                            scoreCumplido: tier.score,
                            participacionUsada: tier.participacion
                        };
                    }
                });
            });

            // Clasificar resultados consolidados
            for (let id in mejorResultadoAgente) {
                const resultado = mejorResultadoAgente[id];

                console.log(`\n   👤 Agente ${id}:`);
                console.log(`      scoreCumplido: ${resultado.scoreCumplido}`);
                console.log(`      participacionUsada: ${resultado.participacionUsada ? ((resultado.participacionUsada) * 100).toFixed(0) + '%' : 'undefined'}`);
                console.log(`      bono: $${resultado.bono.toFixed(2)}`);

                if (resultado.bono > 0) {
                    consolidado.validacion.empleadosCumplen.push(resultado);
                    consolidado.validacion.totalCumplenCondicion++;
                } else {
                    consolidado.validacion.empleadosNoCumplen.push(resultado);
                }
            }

            console.log(`✅ Consolidado: ${consolidado.validacion.empleadosCumplen.length} cumplen | esGate=${consolidado.esGate}`);

            return consolidado;
        });
    };

    // Keys: no se consolidan (nunca son multi-tier)
    resultadosConsolidados['keys'] = resultadosBrutos['keys'];

    // ⭐ Las 3 categorías restantes usan la misma lógica de consolidación
    resultadosConsolidados['bonusStructure']      = consolidarCategoria('bonusStructure');
    resultadosConsolidados['additionalIncentives'] = consolidarCategoria('additionalIncentives');
    resultadosConsolidados['decreasedKPIs']        = consolidarCategoria('decreasedKPIs');

    // ⭐ PARIDAD OM: aplicar lógica de Gate después de consolidar
    resultadosConsolidados['bonusStructure'] = REGISTRO_aplicarLogicaGate(resultadosConsolidados['bonusStructure']);

    return resultadosConsolidados;
}



function normalizarTexto(s) {
    return String(s || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim().replace(/\s+/g, ' ');
}

function REGISTRO_contarAgentesPorLob(lob) {
    const agentes = REGISTRO_obtenerAgentesPermitidos();
    const lobNorm = normalizarTexto(lob);
    let count = 0;
    for (const bmsId in agentes) {
        if (normalizarTexto(agentes[bmsId].lob) === lobNorm) {
            count++;
        }
    }
    return count;
}

function REGISTRO_obtenerDiasLaborablesMes(mes, anio) {
  let d = 0, date = new Date(anio, mes - 1, 1);
  while (date.getMonth() === mes - 1) { if (date.getDay() !== 0 && date.getDay() !== 6) d++; date.setDate(date.getDate() + 1); }
  return d;
}

function obtenerTodosLosAgentesDeLOB(lob, agentesPermitidos) {

}

function REGISTRO_obtenerAgentesPermitidos(filtroACCM = null) {
  const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
  const hojaHC = ss.getSheetByName('H_C');
  if (!hojaHC) return {};
  
  const dataHC = hojaHC.getDataRange().getValues();
  const headers = dataHC[0];
  const agentes = {};
  
  const normalizarLocal = s => String(s || '').normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim().replace(/\s+/g, ' ');
  
  // ⭐ DETECCIÓN DINÁMICA DE COLUMNAS (PARIDAD OM)
  let idxBMS = -1, idxLOB = -1, idxNombre = -1, idxRol = -1, idxHoras = -1;
  
  headers.forEach((h, i) => {
    const header = String(h).toLowerCase().trim();
    if (header.includes('bms') || header.includes('ccms') || header.includes('id')) idxBMS = i;
    if (header.includes('lob')) idxLOB = i;
    if (header.includes('nombre') || header.includes('name')) idxNombre = i;
    if (header.includes('rol') || header.includes('role')) idxRol = i;
    if (header.includes('horas') || header.includes('hours') || header === 'h') idxHoras = i;
  });

  // Fallbacks (basados en índices típicos de H_C: A=0, B=1, C=2, D=3, J=9)
  if (idxBMS     === -1) idxBMS     = 0;
  if (idxLOB     === -1) idxLOB     = 1;
  if (idxNombre  === -1) idxNombre  = 2;
  if (idxRol     === -1) idxRol     = 3;
  if (idxHoras   === -1) idxHoras   = 9;

  // PARSEO ROBUSTO DE HORAS
  const parsearHoras = (valor) => {
    if (typeof valor === 'number' && !isNaN(valor) && valor > 0) return valor;
    if (!valor) return 42; 
    const n = parseFloat(String(valor).replace(',', '.'));
    return (!isNaN(n) && n > 0) ? n : 42;
  };

  for (let i = 1; i < dataHC.length; i++) {
    const bms = String(dataHC[i][idxBMS]).trim();
    if (!bms) continue;
    
    // Si hay filtro ACCM, aplicarlo
    if (filtroACCM) {
        // En OM, el ACCM suele estar en una columna específica, 
        // pero aquí solo agregamos la lógica básica de filtrado si se requiere
    }

    agentes[bms] = { 
        nombre: String(dataHC[i][idxNombre]).trim(), 
        lob: dataHC[i][idxLOB], 
        rol: String(dataHC[i][idxRol] || '').trim(),
        horasSemanales: parsearHoras(dataHC[i][idxHoras])
    };
  }
  
  Logger.log(`✅ REGISTRO_obtenerAgentesPermitidos: ${Object.keys(agentes).length} agentes cargados`);
  return agentes;
}

 
function REGISTRO_simularConMetricasReales(lob, mes, numeroMesOpcional = null, esProrrateado = true) {
    try {
        Logger.log(`\n🚀 ========== INICIANDO SIMULACIÓN CDM ==========`);
        Logger.log(`📋 LOB: "${lob}"`);
        Logger.log(`📅 Mes (estructura): "${mes}"`);
        Logger.log(`📅 Número de mes CDM: ${numeroMesOpcional || '(resolución automática)'}`);
 
        if (!lob || !mes) return { success: false, error: 'LOB y Mes son obligatorios' };
 
        const datosUsuarioActivo = REGISTRO_obtenerDatosUsuarioActivo();
        if (!datosUsuarioActivo) return { success: false, error: 'Usuario no encontrado' };
 
        const mesesMap = {
            'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
            'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
            'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
        };
 
        // ⭐ FIX CRÍTICO: Si viene numeroMesOpcional válido (1-12), usarlo para CDM.
        //    NO resolverlo desde el texto del mes (que puede ser un mes pasado).
        let numeroMes;
        if (numeroMesOpcional && numeroMesOpcional >= 1 && numeroMesOpcional <= 12) {
            // Viene de simulación masiva: usar el mes actual que pasó el orquestador
            numeroMes = numeroMesOpcional;
            Logger.log(`✅ Usando numeroMes del orquestador: ${numeroMes} (mes actual)`);
        } else {
            // Viene de simulación individual: resolver desde el texto del mes
            const mesLower = String(mes || '').toLowerCase().trim();
            if (mesLower.includes('-')) {
                // Formato yyyy-MM-dd
                const partes = mesLower.split('-');
                numeroMes = parseInt(partes[1]) || (new Date().getMonth() + 1);
            } else {
                numeroMes = mesesMap[mesLower] || (new Date().getMonth() + 1);
            }
            Logger.log(`📅 Número de mes resuelto del texto "${mes}": ${numeroMes}`);
        }
 
        Logger.log(`📊 Número de mes para CDM: ${numeroMes}`);
 
        // Obtener estructura del registro (usa el mes del registro histórico)
        let est = REGISTRO_obtenerEstructuraRegistrada(lob, mes, datosUsuarioActivo);
        if (!est || !est.encontrado || !est.registroOriginal) {
            return { success: false, error: `No hay estructura registrada para ${lob} en ${mes}` };
        }
        Logger.log(`✅ Estructura: ${est.tipoHoja}, OM: ${est.om}`);
 
        // ⭐ Cargar CDM del mes CORRECTO (actual para masivo, o el del registro para individual)
        Logger.log(`\n📊 ========== CARGANDO CDM_${numeroMes} ==========`);
        let dataCDM = null;
        let lobExisteEnCDM = false;
 
        try {
            dataCDM = REGISTRO_obtenerDataCDMCached(numeroMes);
            if (!dataCDM || dataCDM.length <= 1) {
                Logger.log(`⚠️ CDM_${numeroMes} vacío o sin datos`);
            } else {
                Logger.log(`✅ CDM_${numeroMes} cargado: ${dataCDM.length} filas`);
                const lobNorm = normalizarTexto(lob);
                const headers = dataCDM[0];
                let colSubLOB = -1;
                for (let i = 0; i < headers.length; i++) {
                    const header = String(headers[i]).toLowerCase().trim();
                    if (header === 'sublob' || header === 'sub lob' || header === 'lob') {
                        colSubLOB = i;
                        break;
                    }
                }
                if (colSubLOB === -1) colSubLOB = 2;
                for (let i = 1; i < dataCDM.length; i++) {
                    const subLobNorm = normalizarTexto(String(dataCDM[i][colSubLOB] || '').trim());
                    if (subLobNorm === lobNorm) {
                        lobExisteEnCDM = true;
                        break;
                    }
                }
                Logger.log(`   LOB "${lob}" en CDM_${numeroMes}: ${lobExisteEnCDM ? '✅ SÍ' : '❌ NO'}`);
            }
        } catch (error) {
            Logger.log(`⚠️ Error cargando CDM_${numeroMes}: ${error.message}`);
        }
 
        const hc = REGISTRO_contarAgentesPorLob(lob);
        const agentesPermitidos = REGISTRO_obtenerAgentesPermitidos();
        if (hc === 0) return { success: false, error: `No hay agentes en la LOB ${lob}` };
 
        const ssUsuario = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaUsuario = ssUsuario.getSheetByName('Usuario');
        const dataUsuario = hojaUsuario ? hojaUsuario.getDataRange().getValues() : [];
 
        let resultadosV;
        if (lobExisteEnCDM && dataCDM && dataCDM.length > 1) {
            Logger.log(`\n✅ USANDO: REGISTRO_validarKPIsPorCategoria() - DATOS REALES CDM_${numeroMes}`);
            resultadosV = REGISTRO_validarKPIsPorCategoria(
                est.registroOriginal, lob, dataCDM, dataUsuario,
                est.tipoHoja, agentesPermitidos, numeroMes
            );
        } else {
            Logger.log(`\n⚠️ USANDO: validarKPIsConValoresCero() - LOB no encontrada en CDM_${numeroMes}`);
            resultadosV = validarKPIsConValoresCero(
                est.registroOriginal, lob, hc, est.tipoHoja, agentesPermitidos
            );
        }
        if (!resultadosV) return { success: false, error: 'Error en validación de KPIs' };
 
        // ============================================================
        // CONSOLIDAR TOTALES (sin cambios respecto al original)
        // ============================================================
        let inversionReal = 0;
        const agentesConBono = new Set();
        const todosLosAgentes = obtenerTodosLosAgentesDeLOB(lob, agentesPermitidos);
        const bonosPorAgente = {};
 
        todosLosAgentes.forEach(ag => {
            bonosPorAgente[ag.bmsId] = {
                bmsId: ag.bmsId,
                nombre: ag.nombre,
                totalKeys: 0,
                totalBonus: 0,
                totalAdditional: 0,
                totalDecreased: 0,
                bonoFinal: 0,
                detalleKPIs: []
            };
        });
 
        const procesarMonto = (cat) => {
            if (!resultadosV[cat]) return;
            resultadosV[cat].forEach(kpi => {
                if (!kpi.validacion) return;
                (kpi.validacion.empleadosCumplen || []).forEach(e => {
                    const bono = e.bono || 0;
 
                    if (!bonosPorAgente[e.bmsId]) {
                        bonosPorAgente[e.bmsId] = {
                            bmsId: e.bmsId,
                            nombre: e.nombre,
                            totalKeys: 0,
                            totalBonus: 0,
                            totalAdditional: 0,
                            totalDecreased: 0,
                            bonoFinal: 0,
                            detalleKPIs: []
                        };
                    }
 
                    if (cat === 'keys') {
                        bonosPorAgente[e.bmsId].totalKeys += bono;
                    } else if (cat === 'bonusStructure') {
                        bonosPorAgente[e.bmsId].totalBonus += bono;
                    } else if (cat === 'additionalIncentives') {
                        bonosPorAgente[e.bmsId].totalAdditional += bono;
                    } else if (cat === 'decreasedKPIs') {
                        bonosPorAgente[e.bmsId].totalDecreased += Math.abs(bono);
                    }
 
                    if (bono > 0) agentesConBono.add(e.bmsId);
 
                    bonosPorAgente[e.bmsId].detalleKPIs.push({
                        categoria: cat,
                        kpi: kpi.kpi,
                        score: e.scoreCumplido || kpi.score || '-',
                        participacion: e.participacionUsada || kpi.participacion || 1,
                        valorF: e.valorF || '-',
                        bono: bono,
                        esGate: kpi.esGate || false
                    });
                });
            });
        };
 
        ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(procesarMonto);
 
        // ⭐ Bono final por agente (neto)
        inversionReal = 0;
        Object.keys(bonosPorAgente).forEach(bmsId => {
            const ag = bonosPorAgente[bmsId];
            const sumatoria = ag.totalKeys + ag.totalBonus + ag.totalAdditional - ag.totalDecreased;
            ag.bonoFinal = Math.max(0, sumatoria);
            inversionReal += ag.bonoFinal;
        });
 
        const afectadosKeys = new Set();
        Object.values(bonosPorAgente).forEach(ag => {
            if (ag.detalleKPIs.some(d => d.esGate && d.bono === 0)) {
                afectadosKeys.add(ag.bmsId);
            }
        });
 
        const top10Ganadores = Object.values(bonosPorAgente)
            .sort((a, b) => b.bonoFinal - a.bonoFinal);
 
        // ⭐ Teórico: usa la configuración (salario × % bono × HC) — NO cambia
        const configSalario = REGISTRO_obtenerSalarioPorLob(lob);
        let salarioBase = configSalario.encontrado ? configSalario.salario : 0;
        let porcentajeBono = configSalario.encontrado ? configSalario.porcentaje : 0;
 
        if (salarioBase === 0) {
            Logger.log(`⚠️ Sin salario en Configuracion para ${lob}. Teórico = 0.`);
        }
 
        const inversionTeorica = hc * (salarioBase * (porcentajeBono / 100));
 
        Logger.log(`\n💰 Inversión Teórica: $${inversionTeorica.toLocaleString()} (HC=${hc} × $${salarioBase} × ${porcentajeBono}%)`);
        Logger.log(`💰 Inversión Real CDM_${numeroMes}: $${inversionReal.toLocaleString()}`);
        Logger.log(`👥 Agentes con bono: ${agentesConBono.size} de ${hc}`);
 
        const desglosePorCategoria = {
            keys: (resultadosV.keys || []).map(k => ({
                kpi: k.kpi || 'Desconocido',
                totalCumplen: k.validacion?.totalCumplenCondicion || 0,
                totalNoCumplen: k.validacion?.empleadosNoCumplen?.length || 0,
                totalSinDatos: k.validacion?.empleadosSinDatos?.length || 0
            })),
            bonusStructure: (resultadosV.bonusStructure || []).map(k => ({
                kpi: k.kpi || 'Desconocido',
                porcentaje: k.porcentaje || 0,
                totalCumplen: k.validacion?.totalCumplenCondicion || 0,
                montoTotal: (k.validacion?.empleadosCumplen || []).reduce((s, e) => s + (e.bono || 0), 0)
            })).sort((a, b) => b.montoTotal - a.montoTotal),
            additionalIncentives: (resultadosV.additionalIncentives || []).map(k => ({
                kpi: k.kpi || 'Desconocido',
                porcentaje: k.porcentaje || 0,
                totalCumplen: k.validacion?.totalCumplenCondicion || 0,
                montoTotal: (k.validacion?.empleadosCumplen || []).reduce((s, e) => s + (e.bono || 0), 0)
            })).sort((a, b) => b.montoTotal - a.montoTotal),
            decreasedKPIs: (resultadosV.decreasedKPIs || []).map(k => ({
                kpi: k.kpi || 'Desconocido',
                porcentaje: k.porcentaje || 0,
                totalCumplen: k.validacion?.totalCumplenCondicion || 0,
                montoTotal: (k.validacion?.empleadosCumplen || []).reduce((s, e) => s + Math.abs(e.bono || 0), 0)
            })).sort((a, b) => b.montoTotal - a.montoTotal)
        };
 
        return {
            success: true,
            lob, mes, numeroMes, hc,
            tipoHoja: est.tipoHoja,
            inversionTeorica,   // ⭐ Teórico: sin cambios (HC × Salario × %Bono)
            inversionReal,      // ⭐ Real: basado en CDM del mes actual
            ahorro: inversionTeorica - inversionReal,
            eficienciaReal: inversionTeorica > 0 ? ((inversionReal / inversionTeorica) * 100) : 0,
            configuracion: {
                salario: salarioBase,
                porcentaje: porcentajeBono,
                bonoMaximo: salarioBase * (porcentajeBono / 100)
            },
            estadisticas: {
                totalAgentes: hc,
                agentesCobran: agentesConBono.size,
                agentesNoCobran: hc - agentesConBono.size,
                agentesAfectadosPorKeys: afectadosKeys.size,
                tasaCumplimiento: hc > 0 ? ((agentesConBono.size / hc) * 100) : 0
            },
            desglosePorCategoria,
            resultadosDetallados: resultadosV,
            top10Ganadores,
            fechaSimulacion: new Date().toISOString(),
            esTeorico: !lobExisteEnCDM,
            lobEnCDM: lobExisteEnCDM
        };
 
    } catch (error) {
        Logger.log(`\n❌ ERROR: ${error.message}`);
        Logger.log(`Stack: ${error.stack}`);
        return { success: false, error: error.message || error.toString() };
    }
}
 





/**
 * Función de respaldo cuando no hay datos en CDM.
 * Retorna todos los agentes con bonos en 0 para mantener la integridad del reporte.
 */
function validarKPIsConValoresCero(registro, lob, hc, hojaNombre, agentesPermitidos) {
    const categorizacion = { 'Keys': 'keys', 'Bonus Structure': 'bonusStructure', 'Additional Incentives': 'additionalIncentives', 'Decreased KPIs': 'decreasedKPIs' };
    const resultados = { keys: [], bonusStructure: [], additionalIncentives: [], decreasedKPIs: [] };
    const todosAgentes = obtenerTodosLosAgentesDeLOB(lob, agentesPermitidos);
    const startIdx = (hojaNombre === 'Registro_pisos') ? 4 : 3;

    for (let c = 0; c < 4; c++) {
        const catName = Object.keys(categorizacion)[c];
        const internalCat = categorizacion[catName];
        
        for (let i = 0; i < 5; i++) {
            const kpiIdx = startIdx + (i * 4);
            const kpi = registro[kpiIdx];
            const score = registro[kpiIdx + 1];
            const peso = parseFloat(registro[kpiIdx + 2]) || 0;
            const participation = parseFloat(registro[kpiIdx + 3]) || 1;

            if (kpi && kpi !== '-' && score && score !== '-') {
                const validacion = {
                    llave: kpi,
                    condicionScore: score,
                    totalCoincidencias: todosAgentes.length,
                    totalCumplenCondicion: 0,
                    empleadosCumplen: [],
                    empleadosNoCumplen: [],
                    empleadosSinDatos: todosAgentes.map(ag => ({ bmsId: ag.bmsId, nombre: ag.nombre, valorF: '-', bono: 0, sinDatos: true }))
                };
                resultados[internalCat].push({ kpi, validacion, porcentaje: peso * participation });
            }
        }
    }
    return resultados;
}

/**
 * Orquestador principal para el frontend de OM
 */
function OM_obtenerDatosParaFrontend(lob, numeroMes, fechaSeleccionada) {
  try {
    const usuario = OM_obtenerDatosUsuarioActivo();
    if (!usuario) return { error: "No se identificó el usuario activo" };
    
    // Si numeroMes es nulo, intentar resolverlo
    if (!numeroMes) {
      const mesesMap = {
        'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
        'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
        'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
      };
      const mesActual = new Date().toLocaleString('es-ES', { month: 'long' }).toLowerCase();
      numeroMes = mesesMap[mesActual];
    }

    const resultado = {
      usuario: usuario,
      metricasPorLob: {},
      horasRealesPorAgente: {},
      regiones: OM_obtenerTodasLasRegiones()
    };

    // Determinar LOBs a procesar
    let lobsAProcesar = [];
    let lobsParaDropdown = [];

    if (usuario.rol === 'Reporting Analyst' || usuario.rol === 'Audit') {
      lobsParaDropdown = OM_obtenerTodasLasLobs();
    } else if (usuario.rol === 'OM') {
      lobsParaDropdown = OM_obtenerLobsPorOM(usuario.nombre);
    } else if (usuario.rol === 'ACCM') {
      lobsParaDropdown = OM_obtenerLobsPorACCM(usuario.nombre);
    } else {
      lobsParaDropdown = [usuario.lob];
    }
    
    resultado.lobs = lobsParaDropdown;

    if (!lob) {
      // Si no hay lob seleccionada, solo retornamos los datos iniciales
      return resultado;
    }

    if (lob.startsWith("REGION:")) {
      const regionNombre = lob.replace("REGION:", "").trim();
      const resumenRegion = OM_obtenerResumenLobsPorRegion(regionNombre, numeroMes);
      if (resumenRegion.error) return { error: resumenRegion.error };
      lobsAProcesar = resumenRegion.lobs.filter(l => l.tieneCDM).map(l => l.lob);
      if (lobsAProcesar.length === 0) return { error: `No se encontraron LOBs con datos CDM en la región ${regionNombre}` };
    } else if (lob === "TODAS LAS LOBS") {
      lobsAProcesar = lobsParaDropdown;
    } else {
      lobsAProcesar = [lob];
    }

    // Procesar cada LOB usando la lógica de simulación
    for (const lobActual of lobsAProcesar) {
      const sim = OM_simularConMetricasReales(lobActual, null, numeroMes); // null mes porque simularConMetricasReales lo resuelve
      if (sim.success) {
        // Flatten categories and include important summary metrics for frontend compatibility
        resultado.metricasPorLob[lobActual] = {
          ...sim.desglosePorCategoria,
          success: true,
          inversionTeorica: sim.inversionTeorica,
          inversionReal: sim.inversionReal,
          hc: sim.hc,
          mes: sim.mes,
          numeroMes: sim.numeroMes,
          ahorro: sim.ahorro,
          eficienciaReal: sim.eficienciaReal,
          estadisticas: sim.estadisticas,
          configuracion: sim.configuracion
        };
        // Combinar horas reales para el frontend
        const horas = OM_obtenerHorasPorAgente(lobActual, numeroMes);
        Object.assign(resultado.horasRealesPorAgente, horas);
      } else {
        resultado.metricasPorLob[lobActual] = { error: sim.error };
      }
    }

    return resultado;
  } catch (error) {
    return { error: error.message };
  }
}

/**
 * Función de simulación adaptada del backup
 */
function OM_simularConMetricasReales(lob, mesManual, numeroMesOpcional) {
    try {
        const usuario = OM_obtenerDatosUsuarioActivo();
        if (!usuario) return { success: false, error: 'Usuario no encontrado' };

        // Resolver mes
        const mesesMap = {
            'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
            'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
            'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
        };
        const mesTextosMap = ['enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio', 'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'];
        
        let numeroMes = numeroMesOpcional;
        let mes = mesManual;

        if (!numeroMes && mesManual) {
            numeroMes = mesesMap[mesManual.toLowerCase().trim()];
        }
        if (!numeroMes) {
            numeroMes = new Date().getMonth() + 1;
        }
        if (!mes) {
            mes = mesTextosMap[numeroMes - 1];
        }

        // Obtener estructura mediante la función original del Registro
        // Nota: Estas funciones deben existir en REGISTRO_GS.txt
        let est = REGISTRO_obtenerEstructuraRegistrada(lob, mes, usuario);
        
        if (!est || !est.encontrado || !est.registroOriginal) {
            // Intentar con formato de fecha si el nombre del mes falla
            const fechaFormato = Utilities.formatDate(new Date(new Date().getFullYear(), numeroMes - 1, 1), Session.getScriptTimeZone(), "yyyy-MM-dd");
            est = REGISTRO_obtenerEstructuraRegistrada(lob, fechaFormato, usuario);
            if (!est || !est.encontrado) {
              return { success: false, error: `No hay estructura registrada para ${lob} en ${mes}` };
            }
        }

        let dataCDM = null;
        let lobExisteEnCDM = false;
        
        try {
            dataCDM = OM_obtenerDataCDMCached(numeroMes);
            if (dataCDM && dataCDM.length > 1) {
                const normalizar = s => String(s || '').normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim().replace(/\s+/g, ' ').replace(/[^\w\s]/g, '');
                const lobNorm = normalizar(lob);
                
                // Buscar LOB en CDM
                const headers = dataCDM[0];
                let colSubLOB = headers.findIndex(h => {
                  const hn = String(h).toLowerCase();
                  return hn === 'sublob' || hn === 'sub lob' || hn === 'lob';
                });
                if (colSubLOB === -1) colSubLOB = 2;

                for (let i = 1; i < dataCDM.length; i++) {
                    if (normalizar(dataCDM[i][colSubLOB]) === lobNorm) {
                        lobExisteEnCDM = true;
                        break;
                    }
                }
            }
        } catch (e) {
            Logger.log("⚠️ Error cargando CDM para simulación: " + e.message);
        }

        const hc = REGISTRO_contarAgentesPorLob(lob);
        const agentesPermitidos = OM_obtenerAgentesPermitidos_Dashboard();
        const ssPrincipal = SpreadsheetApp.openById(OM_ID_SS_PRINCIPAL_LOCAL);
        const dataUsuario = ssPrincipal.getSheetByName('Usuario').getDataRange().getValues();
        
        let resultadosV;
        if (lobExisteEnCDM) {
            resultadosV = OM_validarKPIsPorCategoria(est.registroOriginal, lob, dataCDM, dataUsuario, est.tipoHoja, agentesPermitidos, numeroMes);
        } else {
            // Si no hay CDM, podríamos usar valores cero o simplemente fallar según el requerimiento
            return { success: false, error: `No se encontraron datos en CDM para la LOB ${lob} en el mes ${numeroMes}` };
        }

        // Consolidación de totales
        let inversionReal = 0;
        const agentesConBono = new Set();
        const agentesAfectadosKeys = new Set();
        const todosLosAgentes = OM_obtenerTodosLosAgentesDeLOB(lob, agentesPermitidos);
        const bonosPorAgente = {};
        
        todosLosAgentes.forEach(ag => {
            bonosPorAgente[ag.bmsId] = { bmsId: ag.bmsId, nombre: ag.nombre, totalKeys: 0, totalBonus: 0, totalAdditional: 0, totalDecreased: 0, bonoFinal: 0 };
        });




        const procesarMonto = (cat) => {
            if (!resultadosV[cat]) return;
            resultadosV[cat].forEach(kpi => {
                if (!kpi.validacion) return;
                (kpi.validacion.empleadosCumplen || []).forEach(e => {
                    const bono = e.bono || 0;
                    inversionReal += bono;

                    if (!bonosPorAgente[e.bmsId]) {
                        bonosPorAgente[e.bmsId] = { bmsId: e.bmsId, nombre: e.nombre, totalKeys: 0, totalBonus: 0, totalAdditional: 0, totalDecreased: 0, bonoFinal: 0, horasSemanales: 0, horasReales: 0, horasEsperadas: 0 };
                    }

                    // Actualizar horas (usar las de cualquier KPI ya que son las mismas)
                    bonosPorAgente[e.bmsId].horasReales = e.horasReales || 0;
                    bonosPorAgente[e.bmsId].horasEsperadas = e.horasEsperadas || 0;
                    bonosPorAgente[e.bmsId].horasSemanales = e.horasSemanales || 42;

                    // ASIGNAR MONTOS POR CATEGORÍA
                    if (cat === 'keys') {
                        bonosPorAgente[e.bmsId].totalKeys += bono;
                    } else if (cat === 'bonusStructure') {
                        bonosPorAgente[e.bmsId].totalBonus += bono;
                    } else if (cat === 'additionalIncentives') {
                        bonosPorAgente[e.bmsId].totalAdditional += bono;
                    } else if (cat === 'decreasedKPIs') {
                        // Las penalizaciones son negativas, las sumamos como valor absoluto al acumulador de penalizaciones
                        bonosPorAgente[e.bmsId].totalDecreased += Math.abs(bono);
                    }

                    if (bono > 0) {
                        agentesConBono.add(e.bmsId);
                    }

                });
                if (cat === 'keys') {
                    (kpi.validacion.empleadosNoCumplen || []).forEach(e => { agentesAfectadosKeys.add(e.bmsId); });
                }
            });
        };

        ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(procesarMonto);
        
        Object.keys(bonosPorAgente).forEach(bmsId => {
            const agente = bonosPorAgente[bmsId];
            agente.bonoFinal = agente.totalKeys + agente.totalBonus + agente.totalAdditional - agente.totalDecreased;
        });

        const configSalario = OM_obtenerConfiguracionDesdeConfiguracion(lob, est.om);
        const salarioBase = configSalario.encontrado ? configSalario.salario : 2150000;
        const porcentajeBono = configSalario.encontrado ? configSalario.porcentaje : 30;
        const inversionTeorica = hc * (salarioBase * (porcentajeBono / 100));

        return {
            success: true,
            lob: lob,
            mes: mes,
            numeroMes: numeroMes,
            hc: hc,
            tipoHoja: est.tipoHoja,
            inversionTeorica: inversionTeorica,
            inversionReal: inversionReal,
            ahorro: inversionTeorica - inversionReal,
            eficienciaReal: inversionTeorica > 0 ? ((inversionReal / inversionTeorica) * 100) : 0,
            configuracion: { salario: salarioBase, porcentaje: porcentajeBono, bonoMaximo: salarioBase * (porcentajeBono / 100) },
            estadisticas: { totalAgentes: hc, agentesCobran: agentesConBono.size, agentesNoCobran: hc - agentesConBono.size, agentesAfectadosPorKeys: agentesAfectadosKeys.size, tasaCumplimiento: hc > 0 ? ((agentesConBono.size / hc) * 100) : 0 },
            desglosePorCategoria: resultadosV, // Usamos resultadosV para mantener compatibilidad con el frontend
            top10Ganadores: Object.values(bonosPorAgente).filter(a => a.bonoFinal > 0).sort((a, b) => b.bonoFinal - a.bonoFinal).slice(0, 10),
            fechaSimulacion: new Date().toISOString()
        };
    } catch (error) {
        return { success: false, error: error.message };
    }
}

/**
 * ⭐ FUNCIÓN: Orquestador de validaciones por categoría (Alineada con OM)
 */
function OM_validarKPIsPorCategoria(registro, lob, dataCDM, dataUsuario, hoja, agentesPermitidos, numeroMes = null, fechaSeleccionada = null) {
    const resultadosBrutos = { 
        keys: [], 
        bonusStructure: [], 
        additionalIncentives: [], 
        decreasedKPIs: [] 
    };

    const procesar = (cat, start, end, inc, conParticipacion) => {
        for (let c = start; c <= end; c += inc) {
            if (registro[c] && registro[c] !== '-') {
                const kpi = String(registro[c]).trim();
                const scoreRaw = String(registro[c + 1]).trim();

                // ================================================================
                // ✅ DETECTAR GATE: termina en ",key" ignorando espacios/mayúsculas (PARIDAD OM)
                // ================================================================
                const scoreRawLower = scoreRaw.toLowerCase().replace(/\s/g, '');
                const esGate = cat === 'bonusStructure' && scoreRawLower.endsWith(',key');

                let score = scoreRaw;

                if (esGate) {
                    // Limpiar ",key" del score
                    score = scoreRaw.replace(/\s*,\s*key\s*$/i, '').trim();
                    Logger.log(`🔒 Gate detectado: KPI="${kpi}" | scoreRaw="${scoreRaw}" | scoreLimpio="${score}"`);
                }

                // Normalizar coma decimal DESPUÉS de limpiar el gate (PARIDAD OM)
                score = score.replace(/(\d),(\d)/g, '$1.$2');
                let peso = parseFloat(String(registro[c + 2] || '0').replace('%', '')) || 0;
                if (peso > 1) peso /= 100;

                let participation = 1;
                if (conParticipacion) {
                    const pRaw = String(registro[c + 3] || '100');
                    const pNum = parseFloat(pRaw.replace('%', '').trim()) || 100;
                    participation = pNum > 1 ? pNum / 100 : pNum;
                }
                
                const validacion = OM_validarKPI1(
                    kpi, 
                    score, 
                    peso * participation, 
                    dataCDM, 
                    dataUsuario, 
                    agentesPermitidos, 
                    lob, 
                    numeroMes, 
                    cat === 'decreasedKPIs'
                );
                
                resultadosBrutos[cat].push({ 
                    kpi, 
                    score, 
                    porcentaje: peso, 
                    participacion: participation, 
                    validacion,
                    esDecrease: cat === 'decreasedKPIs'
                });
            }
        }
    };

    if (hoja === 'Registro') {
        // Estructura Estándar (Triplets)
        procesar('keys', 5, 8, 3, false); // Forzamos inc 3 porque son Triplets (KPI, Score, Peso)
        procesar('bonusStructure', 10, 21, 3, false);
        procesar('additionalIncentives', 23, 31, 3, false);
        procesar('decreasedKPIs', 33, 38, 3, false);
        return resultadosBrutos;
    } else {
        // Estructura Registro_pisos (Multi-Tier)
        procesar('keys', 5, 10, 4, false); // Multi-tier suele usar triplets o quadruplets. 
        procesar('bonusStructure', 12, 79, 4, true); 
        procesar('additionalIncentives', 81, 100, 4, true);
        procesar('decreasedKPIs', 102, 121, 4, true);
        return OM_consolidarPisos(resultadosBrutos);
    }
}

/**
 * ⭐ FUNCIÓN CORE: validarKPI1 (Alineada con OM)
 * Calcula el bono individual para un KPI específico.
 */
function OM_validarKPI1(llave, condicionScore, porcentajePunto, dataCDM, dataUsuario, agentesPermitidos, lobActual, numeroMes = null, esDecrease = false) {
    const resultado = { 
        llave, 
        condicionScore, 
        totalCoincidencias: 0, 
        totalCumplenCondicion: 0, 
        empleadosCumplen: [], 
        empleadosNoCumplen: [], 
        empleadosSinDatos: [] 
    };

    if (!condicionScore || condicionScore === 'N/A' || condicionScore.trim() === '') {
        return resultado;
    }

    const normalizar = s => String(s || '')
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase()
        .trim()
        .replace(/\s+/g, ' ');
    
    const llaveNorm = normalizar(llave);
    const lobActualNorm = normalizar(lobActual);

    // Detectar si es bono fijo o condición
    const esBonoFijo = !condicionScore.match(/(<=|>=|<|>|=)/);
    let valorBonoFijo = 0, operador = null, valorCondicion = 0;
    
    if (esBonoFijo) {
        valorBonoFijo = parseFloat(condicionScore.replace(/[^\d.]/g, '')) || 0;
    } else {
        const match = condicionScore.match(/(<=|>=|<|>|=)\s*(\d+(?:\.\d+)?)/);
        if (!match) return resultado;
        operador = match[1];
        valorCondicion = parseFloat(match[2]);
    }

    const todosLosAgentes = OM_obtenerTodosLosAgentesDeLOB(lobActual, agentesPermitidos);
    
    // ⭐ PASO 1: CONFIGURACIÓN LOB (Diferenciando entre Salario y % Bono)
    const configLOB = OM_obtenerConfiguracionDesdeConfiguracion(lobActual);
    let salarioBaseLOB = configLOB.encontrado ? configLOB.salario : 2150000;
    let porcentajeBonoGlobal = configLOB.encontrado ? configLOB.porcentaje : 30;

    // ⭐ PASO 2: HORAS SEMANALES (H_C columna J)
    const horasSemanalesPorAgente = {};
    try {
        const ssHC = SpreadsheetApp.openById(OM_ID_SS_PRINCIPAL_LOCAL);
        const hojaHC = ssHC.getSheetByName('H_C');
        if (hojaHC) {
            const dataHC = hojaHC.getDataRange().getValues();
            for (let h = 1; h < dataHC.length; h++) {
              if (normalizar(dataHC[h][1]) === lobActualNorm) {
                horasSemanalesPorAgente[String(dataHC[h][0]).trim()] = parseFloat(dataHC[h][9]) || 42;
              }
            }
        }
    } catch (e) {}

    // ⭐ PASO 3: PARÁMETROS DE MES
    const horasReales = OM_obtenerHorasPorAgente(lobActual, numeroMes);
    const diasLab = OM_obtenerDiasLaborablesMes(numeroMes, new Date().getFullYear());

    // Mapeo dinámico de CDM
    const headers = dataCDM[0];
    const getIdx = (names, def) => {
        const namesNorm = names.map(n => String(n).toLowerCase().trim());
        const idx = headers.findIndex(h => namesNorm.includes(String(h || '').toLowerCase().trim()));
        return idx === -1 ? def : idx;
    };

    const idxBms = getIdx(['BMS ID', 'CCMSID', 'ID'], 1);
    const idxValH = getIdx(['Columna H', 'Score', 'KPI_Value'], 7);
    const idxValF = getIdx(['Columna F', 'Value'], 5);
    const idxLlave = getIdx(['Llave', 'KPI'], 8);

    const agentesConDatos = new Set();

    // ========================================
    // CASO A: BONO FIJO (Sin leer CDM)
    // ========================================
    if (esBonoFijo) {
        todosLosAgentes.forEach(ag => {
            let salario = salarioBaseLOB;
            let salarioPersonalizado = false;
            
            for (let u = 1; u < dataUsuario.length; u++) {
                if (String(dataUsuario[u][0]).trim() === ag.bmsId) {
                    let sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                    let salarioAgente = parseFloat(sRaw) || 0;
                    if (salarioAgente > 0) {
                        salario = salarioAgente;
                        if (salarioAgente !== salarioBaseLOB) salarioPersonalizado = true;
                    }
                    break;
                }
            }
            
            const horasSemanales = horasSemanalesPorAgente[ag.bmsId] || 42;
            const bonoBase = salario * (porcentajeBonoGlobal / 100);
            let bonoParcial = bonoBase * porcentajePunto;
            
            // OM Prorrateo: Auditoría suele usar factor 1.0 si no hay ausencias, pero Registro simula por horas reales.
            // Para paridad absoluta con OM Dashboard, forzamos factorProrrateo = 1 si es simulación masiva,
            // pero mantenemos la lógica de horas para el detalle.
            let factorProrrateo = 1;
            let hReales = (horasReales[ag.bmsId]?.horas) || 0;
            let hEsperadas = (horasSemanales / 5) * diasLab;
            
            if (numeroMes && hReales > 0 && hEsperadas > 0) {
              factorProrrateo = hReales / hEsperadas;
              if (factorProrrateo > 1.2) factorProrrateo = 1;
            }

            let bonoFinal = bonoParcial * factorProrrateo;
            if (esDecrease) bonoFinal = -Math.abs(bonoFinal);
            
            resultado.totalCoincidencias++;
            resultado.totalCumplenCondicion++;
            resultado.empleadosCumplen.push({
                bmsId: ag.bmsId,
                nombre: ag.nombre,
                valorF: valorBonoFijo,
                bono: bonoFinal,
                salarioPersonalizado,
                horasSemanales,
                horasReales: hReales,
                horasEsperadas: hEsperadas,
                factorProrrateo,
                tier: condicionScore
            });
        });
        return resultado;
    }

    // ========================================
    // CASO B: KPI CON CONDICIÓN (Leer CDM)
    // ========================================
    for (let row = 1; row < dataCDM.length; row++) {
        const kpiFull = normalizar(dataCDM[row][idxLlave]);
        const bmsId = String(dataCDM[row][idxBms]).trim();
        
        if (!kpiFull.includes(llaveNorm) || !agentesPermitidos[bmsId] || normalizar(agentesPermitidos[bmsId].lob) !== lobActualNorm) continue;
        if (agentesConDatos.has(bmsId)) continue;
        
        agentesConDatos.add(bmsId);
        resultado.totalCoincidencias++;

        const valorNumerico = parseFloat(String(dataCDM[row][idxValH]).replace('%', ''));
        if (isNaN(valorNumerico)) continue;

        let cumple = false;
        switch (operador) {
            case '>': cumple = valorNumerico > valorCondicion; break;
            case '>=': cumple = valorNumerico >= valorCondicion; break;
            case '<': cumple = valorNumerico < valorCondicion; break;
            case '<=': cumple = valorNumerico <= valorCondicion; break;
            case '=': cumple = valorNumerico === valorCondicion; break;
        }

        let salario = salarioBaseLOB;
        let salarioPersonalizado = false;
        for (let u = 1; u < dataUsuario.length; u++) {
            if (String(dataUsuario[u][0]).trim() === bmsId) {
                let sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                let sNum = parseFloat(sRaw) || 0;
                if (sNum > 0) {
                    salario = sNum;
                    if (sNum !== salarioBaseLOB) salarioPersonalizado = true;
                }
                break;
            }
        }

        const horasSemanales = horasSemanalesPorAgente[bmsId] || 42;
        const bonoBase = salario * (porcentajeBonoGlobal / 100);
        let bonoParcial = bonoBase * porcentajePunto;
        
        let factorProrrateo = 1;
        let hReales = (horasReales[bmsId]?.horas) || 0;
        let hEsperadas = (horasSemanales / 5) * diasLab;
        
        if (numeroMes && hReales > 0 && hEsperadas > 0) {
          factorProrrateo = hReales / hEsperadas;
          if (factorProrrateo > 1.2) factorProrrateo = 1;
        }

        let bonoFinal = bonoParcial * factorProrrateo;
        if (esDecrease && cumple) bonoFinal = -Math.abs(bonoFinal);
        
        const item = { 
            bmsId, 
            nombre: agentesPermitidos[bmsId].nombre, 
            valorF: dataCDM[row][idxValF], 
            bono: cumple ? bonoFinal : 0,
            salarioPersonalizado,
            horasSemanales,
            horasReales: hReales,
            horasEsperadas: hEsperadas,
            factorProrrateo,
            tier: condicionScore
        };
        
        if (cumple) { 
            resultado.totalCumplenCondicion++; 
            resultado.empleadosCumplen.push(item); 
        } else {
            resultado.empleadosNoCumplen.push(item);
        }
    }

    // Agentes sin datos
    todosLosAgentes.forEach(ag => {
        if (!agentesConDatos.has(ag.bmsId)) {
            resultado.empleadosSinDatos.push({ 
                bmsId: ag.bmsId, 
                nombre: ag.nombre, 
                valorF: '-', 
                bono: 0, 
                sinDatos: true,
                horasSemanales: horasSemanalesPorAgente[ag.bmsId] || 42,
                horasReales: 0,
                factorProrrateo: 0
            });
        }
    });

    return resultado;
}

        const cacheKeys = allKeys.filter(k => k.includes('salario') || k.includes('Configuracion'));
        if (cacheKeys.length > 0) cache.removeAll(cacheKeys);
        
        return { success: true, keysEliminadas: keysToDelete.length };
    } catch (e) {
        Logger.log('❌ Error: ' + e.toString());
        return { success: false, error: e.toString() };
    }
}
