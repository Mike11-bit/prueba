const norm = s => s ? s.toString().normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().trim() : '';
const nMap = s => s ? String(s).split(' - ')[0].trim().toLowerCase() : '';

/**
 * 🛠️ DETECTOR MAESTRO DE COLUMNAS (Audit_Detalle)
 * Asegura que el nombre del KPI y la Categoría sean mutuamente excluyentes.
 */
function findAuditDetalleIndices(headers) {
    const h = headers.map(s => norm(String(s)));
    
    // 1. Encontrar Categoría (congelar este índice)
    const iCat = h.indexOf('categoria kpi') !== -1 ? h.indexOf('categoria kpi') : h.findIndex(s => s.includes('categor'));
    
    // 2. Encontrar KPI (NOMBRE) - Priorizar exacto y excluir categoría/fecha
    let iKpi = -1;
    // Intento A: Exacto "kpi"
    const iExact = h.indexOf('kpi');
    if (iExact !== -1 && iExact !== iCat) {
        iKpi = iExact;
    } else {
        // Intento B: Parciales excluyendo categoría
        iKpi = h.findIndex((s, idx) => {
            return idx !== iCat && (s.includes('nombre kpi') || s.includes('kpi name') || s.includes('kpi nombre') || (s === 'kpi')) && !s.includes('categor') && !s.includes('fecha');
        });
    }
    
    // Fallback de emergencia si iKpi sigue siendo iCat o -1
    if (iKpi === -1 || iKpi === iCat) iKpi = 7; 

    return {
        bms:   h.indexOf('bms id') !== -1 ? h.indexOf('bms id') : h.findIndex(s => s.includes('bms')),
        nom:   h.indexOf('nombre agente') !== -1 ? h.indexOf('nombre agente') : h.findIndex(s => s.includes('nombre')),
        lob:   h.indexOf('lob') !== -1 ? h.indexOf('lob') : 3,
        mes:   h.indexOf('mes cdm') !== -1 ? h.indexOf('mes cdm') : h.findIndex(s => s.includes('mes')),
        cat:   iCat !== -1 ? iCat : 6,
        kpi:   iKpi,
        obj:   h.findIndex(s => s.includes('objetivo')),
        val:   h.findIndex(s => s.includes('valor obt')),
        est:   h.indexOf('estado') !== -1 ? h.indexOf('estado') : 10,
        bono:  h.indexOf('bono kpi') !== -1 ? h.indexOf('bono kpi') : h.findIndex(s => s.includes('bono')),
        gate:  h.indexOf('es gate') !== -1 ? h.indexOf('es gate') : h.findIndex(s => s.includes('gate')),
        tipoEstructura: h.findIndex(s => s.includes('estructura') || s.includes('tipo estructura'))
    };
}

  throw new Error(`Usuario ${email} no encontrado en hoja Usuario`);
}

function obtenerTodasLasLobs() {

function _obtenerLobsPorCampo(nombre, campo) {
  try {
    const ss   = SpreadsheetApp.openById(SS_ID);
    const hc   = ss.getSheetByName('H_C');
    if (!hc) return [];
    const data = hc.getDataRange().getValues();
    const headers = data[0].map(h => norm(String(h)));
    
    const idxLOB = headers.indexOf('lob') !== -1 ? headers.indexOf('lob') : headers.findIndex(h => h.includes('lob'));
    const idxFiltro = headers.indexOf(campo) !== -1 ? headers.indexOf(campo) : headers.findIndex(h => h.includes(campo));
    
    if (idxLOB === -1 || idxFiltro === -1) return [];

    const nombreNorm = norm(nombre);
    const lobs = new Set();
    for (let i = 1; i < data.length; i++) {
      if (norm(String(data[i][idxFiltro] || '')) === nombreNorm) {
        const lob = String(data[i][idxLOB] || '').trim();
        if (lob) lobs.add(lob);
      }
    }
    return Array.from(lobs).sort();
  } catch(e) { return []; }
}

function obtenerDatosAuditCompleto() {

  let usuario, permisoDisputas = { permitido: false };

  try {

  } catch (e) {

      status: 'error', 
      usuario: { nombre: 'Error', correo: Session.getActiveUser().getEmail(), rol: '' },
      auditData: [], 
      disputas: [],
      traslados: [], 
      prorrateos: [], 
      horasData: [],
      alertas: [], 
      resumenEjecutivo: {}, 
      comparativoMensual: [],
      permisoDisputas: { permitido: false }, 
      lobsPermitidas: [],
      error: e.toString(), 
      timestamp: new Date().toISOString()
    };
  }

  const lobsPermitidas = obtenerLobsPermitidas(usuario);
  console.log('📋 Rol:', usuario.rol, '| LOBs permitidas:', lobsPermitidas.length);

  let auditData = [], traslados = [], prorrateos = [], horasData = [];
  
  try { auditData  = obtenerDatosAprobaciones(); } catch(e) { console.log('❌ Aprobaciones:', e.toString()); }
  
  // Leer disputas desde hoja separada
  let disputasRaw = [];
  try { disputasRaw = obtenerDatosDisputas(); } catch(e) { console.log('❌ Disputas:', e.toString()); }
  
  // CORREGIDO: let en lugar de const para poder reasignar al filtrar
  let disputasConsolidadas = [...disputasRaw];
  
  // Agregar disputas de Audit que no están en hoja Disputa
  const disputasDeAudit = auditData.filter(r => r.tieneDisputa === true);
  disputasDeAudit.forEach(d => {
    const yaExiste = disputasRaw.some(dr => 
      dr.bmsId === d.bmsId && 
      dr.lob === d.lob && 
      dr.mesCDM === d.mesCDM
    );
    if (!yaExiste) {
      disputasConsolidadas.push({
        ...d,
        tipoAccion: 'Disputa',
        estadoDisputa: d.estado || 'Pendiente',
        detalles: d.comentarioDisputa || 'Disputa desde Audit',
        fila: d.filaHoja || 0
      });
    }
  });
  
  try { traslados  = obtenerDatosTraslados();  } catch(e) { console.log('❌ Traslados:', e.toString()); }
  try { prorrateos = obtenerDatosProrrateos(); } catch(e) { console.log('❌ Prorrateos:', e.toString()); }
  try { horasData  = obtenerDatosHoras();      } catch(e) { console.log('❌ Horas:', e.toString()); }

  // Filtrar por LOBs permitidas
  if (lobsPermitidas.length > 0) {
    const lobsNorm = lobsPermitidas.map(norm);
    const filtrar  = arr => arr.filter(r => lobsNorm.includes(norm(r.lob)));
    auditData             = filtrar(auditData);
    disputasConsolidadas  = filtrar(disputasConsolidadas);  // ← ahora funciona con let
    traslados             = filtrar(traslados);
    prorrateos            = filtrar(prorrateos);
    horasData             = filtrar(horasData);
  }

  const ordenar = arr => arr.sort((a,b) => { 
    try { return new Date(b.timestamp) - new Date(a.timestamp); } 
    catch(e){ return 0; } 
  });
  
  auditData            = ordenar(auditData);
  disputasConsolidadas = ordenar(disputasConsolidadas);
  traslados            = ordenar(traslados);
  prorrateos           = ordenar(prorrateos);

  const alertas            = generarAlertas(auditData, traslados, prorrateos);
  const resumenEjecutivo   = calcularResumenEjecutivo(auditData, traslados, prorrateos, horasData);
  const comparativoMensual = calcularComparativoMensual(auditData, prorrateos);

  console.log(`✅ Audit:${auditData.length} | Disputas:${disputasConsolidadas.length} | Tras:${traslados.length} | Hrs:${horasData.length}`);

  return {
    status: 'success',
    usuario, 
    permisoDisputas, 
    lobsPermitidas,
    auditData, 
    disputas: disputasConsolidadas,
    traslados, 
    prorrateos, 
    horasData,
    alertas, 
    resumenEjecutivo, 
    comparativoMensual,
    timestamp: new Date().toISOString()
  };
}

/**
 * Devuelve un array de LOBs que el usuario puede ver.
 * - Reporting Analyst / Audit → [] (vacío = todas las LOBs)
 * - OM / ACCM / Supervisor → LOBs donde aparecen en H_C
 */
function obtenerLobsPermitidas(usuario) {
  try {
    const rol = (usuario.rol || '').toLowerCase().trim();
    if (rol === 'reporting analyst' || rol === 'audit' || rol === 'admin') return [];

    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hoja = ss.getSheetByName('H_C');
    if (!hoja || hoja.getLastRow() <= 1) return [];

    const data = hoja.getDataRange().getValues();
    const headers = data[0].map(h => norm(String(h)));

    // Buscar columnas de LOB, OM, ACCM, Supervisor
    const iLob  = headers.findIndex(h => h.includes('lob'));
    const iOM   = headers.findIndex(h => h === 'om' || h.includes('operations manager'));
    const iACCM = headers.findIndex(h => h.includes('accm'));
    const iSup  = headers.findIndex(h => h.includes('supervisor'));

    if (iLob === -1) return [];

    const nombreUsuario = norm(usuario.nombre || '');
    const ccmsId = (usuario.ccmsId || '').trim();
    const lobs = new Set();

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      // Verificar si el usuario aparece en OM, ACCM o Supervisor
      const esOM   = iOM   !== -1 && (norm(String(row[iOM]   || '')) === nombreUsuario || String(row[iOM]   || '').trim() === ccmsId);
      const esACCM = iACCM !== -1 && (norm(String(row[iACCM] || '')) === nombreUsuario || String(row[iACCM] || '').trim() === ccmsId);
      const esSup  = iSup  !== -1 && (norm(String(row[iSup]  || '')) === nombreUsuario || String(row[iSup]  || '').trim() === ccmsId);

      if (esOM || esACCM || esSup) {
        const lob = String(row[iLob] || '').trim();
        if (lob) lobs.add(lob);
      }
    }

    return Array.from(lobs);
  } catch (e) {
    Logger.log('❌ obtenerLobsPermitidas: ' + e.message);
    return [];
  }
}

/**
 * Devuelve un Set de BMS IDs que el usuario puede ver.
 * - Reporting Analyst / Audit → null (null = todos los BMS IDs)
 * - OM → BMS de agentes cuyo OM es el usuario (en H_C)
 * - ACCM → BMS de agentes cuyo ACCM es el usuario, o cuyo OM tiene a este ACCM
 * - Supervisor → BMS de agentes cuyo Supervisor es el usuario
 */
function obtenerBmsIdsPermitidos(usuario) {
  try {
    const rol = (usuario.rol || '').toLowerCase().trim();
    if (rol === 'reporting analyst' || rol === 'audit' || rol === 'admin') return null;

    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hoja = ss.getSheetByName('H_C');
    if (!hoja || hoja.getLastRow() <= 1) return null;

    const data = hoja.getDataRange().getValues();
    const headers = data[0].map(h => norm(String(h)));

    const iBms  = headers.findIndex(h => h.includes('bms'));
    const iOM   = headers.findIndex(h => h === 'om' || h.includes('operations manager'));
    const iACCM = headers.findIndex(h => h.includes('accm'));
    const iSup  = headers.findIndex(h => h.includes('supervisor'));

    if (iBms === -1) return null;

    const nombreUsuario = norm(usuario.nombre || '');
    const ccmsId = (usuario.ccmsId || '').trim();
    const bmsSet = new Set();

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const bms = String(row[iBms] || '').trim();
      if (!bms) continue;

      const esOM   = iOM   !== -1 && (norm(String(row[iOM]   || '')) === nombreUsuario || String(row[iOM]   || '').trim() === ccmsId);
      const esACCM = iACCM !== -1 && (norm(String(row[iACCM] || '')) === nombreUsuario || String(row[iACCM] || '').trim() === ccmsId);
      const esSup  = iSup  !== -1 && (norm(String(row[iSup]  || '')) === nombreUsuario || String(row[iSup]  || '').trim() === ccmsId);

      if (esOM || esACCM || esSup) {
        bmsSet.add(bms);
      }
    }

    // Si es ACCM, también incluir agentes de los OMs bajo este ACCM
    if (rol.includes('accm') && iOM !== -1 && iACCM !== -1) {
      // Primero, encontrar los OMs que tienen a este ACCM
      const omsDelAccm = new Set();
      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        const esACCM = norm(String(row[iACCM] || '')) === nombreUsuario || String(row[iACCM] || '').trim() === ccmsId;
        if (esACCM) {
          const omVal = norm(String(row[iOM] || ''));
          if (omVal) omsDelAccm.add(omVal);
        }
      }
      // Luego, incluir todos los BMS de esos OMs
      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        const omVal = norm(String(row[iOM] || ''));
        if (omsDelAccm.has(omVal)) {
          const bms = String(row[iBms] || '').trim();
          if (bms) bmsSet.add(bms);
        }
      }
    }

    Logger.log('🔐 BMS permitidos para ' + usuario.nombre + ': ' + bmsSet.size);
    return bmsSet.size > 0 ? bmsSet : null; // Si no encontró ninguno, devolver null para no bloquear
  } catch (e) {
    Logger.log('❌ obtenerBmsIdsPermitidos: ' + e.message);
    return null; // En caso de error, no bloquear acceso
  }
}

          usuario = {
            ccmsId : String(dU[i][0] || '').trim(),
            nombre : String(dU[i][1] || email.split('@')[0]).trim(),
            correo : email,
            rol    : String(dU[i][4] || dU[i][3] || 'Auditor').trim()
          };
          Logger.log('✅ Usuario: ' + usuario.nombre + ' | Rol: ' + usuario.rol);
        }
      }
    }

    // ── LOBs permitidas según rol ──────────────────────────────
    const lobsPermitidas = obtenerLobsPermitidas(usuario);
    const rolNorm = (usuario.rol || '').toLowerCase().trim();
    const esAdmin = rolNorm === 'reporting analyst' || rolNorm === 'audit';
    Logger.log('🔐 Rol: ' + usuario.rol + ' | LOBs permitidas: ' + lobsPermitidas.length + ' | Admin: ' + esAdmin);

    const lobsNorm = lobsPermitidas.map(norm);

    // ── BMS IDs permitidos (Jerarquía H_C) ──────────────────────
    const bmsPermitidos = obtenerBmsIdsPermitidos(usuario);
    Logger.log('🔐 BMS permitidos: ' + (bmsPermitidos ? bmsPermitidos.size : 'TODOS'));

    function pasaFiltroBms(bms) {
      if (!bmsPermitidos) return true; // null = Admin/Analyst
      return bmsPermitidos.has(String(bms).trim());
    }

    function pasaFiltroLob(lob) {
      if (esAdmin || lobsPermitidas.length === 0) return true;
      return lobsNorm.includes(norm(lob));
    }

    // ── Pre-carga de Respuestas Analyst (Disputa_Respuestas) ────
    const respuestasAnalyst = new Map();
    const hojaRespSet = ss.getSheetByName('Disputa_Respuestas');
    if (hojaRespSet) {
      const dR = hojaRespSet.getDataRange().getValues();
      for (let ir = 1; ir < dR.length; ir++) {
        const bmsR = String(dR[ir][1] || '').trim();
        const mesR = normalizarNombreMes(dR[ir][4]);
        const rolR = String(dR[ir][17] || 'REPORTING').trim().toUpperCase();
        if (bmsR && mesR) respuestasAnalyst.set(bmsR + '|' + mesR, rolR);
      }
    }
    Logger.log('📦 Respuestas Analyst cargadas: ' + respuestasAnalyst.size);

    // ── iCol ROBUSTO — case-insensitive + trim + sin tildes ────
    function makeICol(headerRow) {
      return function(n) {
        const nNorm = norm(n);
        for (let c = 0; c < headerRow.length; c++) {
          if (norm(String(headerRow[c])) === nNorm) return c;
        }
        for (let c = 0; c < headerRow.length; c++) {
          if (norm(String(headerRow[c])).includes(nNorm)) return c;
        }
        Logger.log('⚠️ Columna no encontrada: "' + n + '"');
        return -1;
      };
    }

    // ── parseBool ROBUSTO ──────────────────────────────────────
    function parseBool(val) {
      if (val === true || val === 1) return true;
      const s = norm(String(val || ''));
      return s === 'si' || s === 'true' || s === '1' || s === 'yes' ||
             s === 'x'  || s === 'verdadero';
    }

    // ── parseFecha ROBUSTO ─────────────────────────────────────
    function parseFecha(val) {
      if (!val) return '';

      if (val instanceof Date) {
        return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss");
      }

      const str = String(val).trim();
      if (!str) return '';

      const matchCO = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})(?:\s+(\d{1,2}:\d{2}(?::\d{2})?))?/);
      if (matchCO) {
        const dd   = matchCO[1].padStart(2, '0');
        const mm   = matchCO[2].padStart(2, '0');
        const yyyy = matchCO[3];
        const hora = matchCO[4] || '00:00:00';
        return yyyy + '-' + mm + '-' + dd + 'T' + hora;
      }

      if (str.match(/^\d{4}-\d{2}-\d{2}/)) return str;

      try {
        const d = new Date(str);
        if (!isNaN(d)) {
          return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss");
        }
      } catch(e) {}

      return str;
    }

    // ── Audit principal ────────────────────────────────────────
    const auditData = [];

    if (hojaAud && hojaAud.getLastRow() > 1) {
      const data = hojaAud.getDataRange().getValues();
      const h    = data[0];

      const iCol = makeICol(h);

      Logger.log('📋 Headers Audit: ' + JSON.stringify(h));

      const iBMS    = iCol('BMS ID');
      const iNombre = iCol('Nombre Agente');
      const iLOB    = iCol('LOB');
      const iRole   = iCol('Role');
      const iSup    = iCol('Supervisor');
      const iACCM   = iCol('ACCM');
      const iOM     = iCol('OM');
      const iBono   = iCol('Bono Aprobado');
      const iFecApr = iCol('Fecha Aprobacion');
      const iUsr    = iCol('Usuario Aprobador');
      const iCorr   = iCol('Correo Aprobador');
      const iFecKPI = iCol('Fecha KPI');
      const iMes    = iCol('Mes CDM');
      const iEst    = iCol('Estado');
      const iProrr  = iCol('Fue Prorrateado');
      const iMonOr  = iCol('Monto Original');
      const iMonFin = iCol('Monto Final Prorrateo');
      const iDif    = iCol('Diferencia Prorrateo');
      const iDias   = iCol('Dias Laborados');
      const iHrsHC  = iCol('Horas Semanales HC');
      const iHrsTr  = iCol('Horas Trabajadas');
      const iAus    = iCol('Ausencias');
      const iDisp   = iCol('Tiene Disputa');
      const iCoDisp = iCol('Comentario Disputa');
      const iAdditional = iCol('Additional Incentives');
      const iDecreased  = iCol('Decreased KPIs');
      const iMonDisp    = iCol('Monto Disputado');   // columna AC
      const iID         = (function() {
        const byT = h.findIndex(hdr => {
          const hn = norm(String(hdr));
          return hn === 'id' || hn.includes('id disputa') || hn.includes('disputa id');
        });
        return byT !== -1 ? byT : 27; // AB por defecto
      })();
      Logger.log('📌 iMonDisp (Monto Disputado $) = ' + iMonDisp + ' | iID = ' + iID);

      Logger.log('📌 iLOB=' + iLOB + ' | iProrr=' + iProrr + ' | iDisp=' + iDisp +
                 ' | iBMS=' + iBMS + ' | iMes=' + iMes + ' | iFecApr=' + iFecApr);

      for (let r = 1; r < data.length; r++) {
        const row = data[r];
        if (iBMS === -1 || !row[iBMS]) continue;

        const rowBms  = String(row[iBMS]).trim();
        const lobFila = iLOB !== -1 ? String(row[iLOB] || '').trim() : '';

        if (r === 1 && iLOB === -1) {
          Logger.log('🚨 Columna LOB no encontrada. Headers: ' + JSON.stringify(h));
        }

        if (!pasaFiltroLob(lobFila)) continue;
        if (!pasaFiltroBms(rowBms))  continue;

        const bono    = iBono  !== -1 ? (parseFloat(row[iBono])  || 0) : 0;

        // ✅ FIX: isProrr se calcula ANTES que monOr y monFin
        // Si no fue prorrateado, montoOriginal y montoFinalProrrateo deben ser 0
        const isProrr = iProrr !== -1 ? parseBool(row[iProrr]) : false;
        const monOr   = isProrr ? (iMonOr  !== -1 ? (parseFloat(row[iMonOr])  || bono) : bono) : 0;
        const monFin  = isProrr ? (iMonFin !== -1 ? (parseFloat(row[iMonFin]) || bono) : bono) : 0;

        const tieneD  = iDisp  !== -1 ? parseBool(row[iDisp])  : false;

        const fechaTs = iFecApr !== -1 ? parseFecha(row[iFecApr]) : '';

        const estadoAud = iEst !== -1 ? String(row[iEst] || '').trim() : '';
        const esApelacionAud = estadoAud.toLowerCase().includes('apelac') || estadoAud.toLowerCase().includes('más info') || estadoAud.toLowerCase().includes('mas info');
        const tipoAccionFila = esApelacionAud ? 'Apelación' : 'Aprobación';

        auditData.push({
          timestamp          : fechaTs,
          tipoAccion         : tipoAccionFila,
          lob                : lobFila,
          bmsId              : String(row[iBMS]    || ''),
          nombreAgente       : iNombre !== -1 ? String(row[iNombre] || '') : '',
          rol                : iRole   !== -1 ? String(row[iRole]   || '') : '',
          supervisor         : iSup    !== -1 ? String(row[iSup]    || '') : '',
          accm               : iACCM   !== -1 ? String(row[iACCM]   || '') : '',
          om                 : iOM     !== -1 ? String(row[iOM]     || '') : '',
          monto              : bono,
          montoOriginal      : monOr,
          montoFinalProrrateo: monFin,
          diferenciaProrrateo: iDif    !== -1 ? (parseFloat(row[iDif])  || 0) : 0,
          diasLaborados      : iDias   !== -1 ? (parseFloat(row[iDias]) || 0) : 0,
          horasSemanalesHC   : iHrsHC  !== -1 ? (parseFloat(row[iHrsHC])|| 42) : 42,
          horasTrabajadas    : iHrsTr  !== -1 ? (parseFloat(row[iHrsTr])|| 0) : 0,
          ausencias          : iAus    !== -1 ? (parseFloat(row[iAus])  || 0) : 0,
          fueProrrateado     : isProrr,
          tieneDisputa       : tieneD,
          comentarioDisputa  : iCoDisp !== -1 ? String(row[iCoDisp] || '') : '',
          estado             : iEst    !== -1 ? String(row[iEst]    || '') : '',
          usuario            : iUsr    !== -1 ? String(row[iUsr]    || '') : '',
          correo             : iCorr   !== -1 ? String(row[iCorr]   || '') : '',
          mesCDM             : iMes    !== -1 ? normalizarNombreMes(row[iMes]) : '',
          fechaKPI           : iFecKPI !== -1 ? String(row[iFecKPI] || '') : '',
          additionalMonto    : iAdditional !== -1 ? (parseFloat(row[iAdditional]) || 0) : 0,
          decreasedMonto     : iDecreased  !== -1 ? (parseFloat(row[iDecreased])  || 0) : 0,
          bonoKPIsSelec      : iMonDisp    !== -1 ? (parseFloat(row[iMonDisp])    || 0) : 0,
          id                 : iID !== -1 ? String(row[iID] || '').trim() : ''
        });
      }

      Logger.log('✅ Audit rows leídas: ' + auditData.length +
                 ' | LOBs únicas: ' + [...new Set(auditData.map(r => r.lob))].join(', ') +
                 ' | Ejemplo timestamp: ' + (auditData[0] ? auditData[0].timestamp : 'N/A'));
    }

    // ── Disputas ──────────────────────────────────────────────
    const disputas = [];
    if (hojaDisp && hojaDisp.getLastRow() > 1) {
      const dD = hojaDisp.getDataRange().getValues();
      const hD = dD[0];

      const iDCol = (n) => {
        const nNorm = norm(n);
        const idx = hD.findIndex(h => norm(String(h)).includes(nNorm));
        return idx !== -1 ? idx : -1;
      };

      const dFecha   = iDCol('fecha');
      const dBMS     = iDCol('bms');
      const dNombre  = iDCol('nombre');
      const dLOB     = iDCol('lob');
      const dFecKPI  = iDCol('fecha kpi') !== -1 ? iDCol('fecha kpi') : -1;
      const dMes     = hD.findIndex(h => {
        const hn = norm(String(h));
        return hn === 'mes cdm' || hn === 'mes';
      });
      const dMonto   = iDCol('monto') !== -1 ? iDCol('monto') :
                       iDCol('valor disputado') !== -1 ? iDCol('valor disputado') :
                       iDCol('total bono') !== -1 ? iDCol('total bono') : 9;
      const dComent  = hD.lastIndexOf('Correo Usuario') !== -1 ? hD.lastIndexOf('Correo Usuario') : iDCol('comentario');
      const dEstado  = hD.reduce((acc, cur, i) => norm(cur).includes('estado') ? i : acc, -1);
      const dUsuario = iDCol('usuario');
      const dBonoKPIsSelec = (() => {
        const opciones = ['bono kpis selec', 'bono kpi selec', 'kpis selec', 'bono seleccionado', 'bono selec'];
        for (const op of opciones) {
          const i = hD.findIndex(h => norm(String(h)).includes(op));
          if (i !== -1) return i;
        }
        return -1;
      })();

      for (let i = 1; i < dD.length; i++) {
        const row = dD[i];
        if (!row[dBMS] && !row[dNombre]) continue;

        const rowBms  = String(row[dBMS] || '').trim();
        const lobFila = dLOB !== -1 ? String(row[dLOB] || '').trim() : '';

        if (!pasaFiltroLob(lobFila)) continue;
        if (!pasaFiltroBms(rowBms))  continue;

        let mesCDM = '';
        if (dMes !== -1 && row[dMes]) {
          mesCDM = normalizarNombreMes(row[dMes]);
        } else if (dFecKPI !== -1 && row[dFecKPI]) {
          try {
            const fObj = row[dFecKPI] instanceof Date ? row[dFecKPI] : new Date(row[dFecKPI]);
            if (!isNaN(fObj)) {
              const meses = ['enero','febrero','marzo','abril','mayo','junio',
                             'julio','agosto','septiembre','octubre','noviembre','diciembre'];
              mesCDM = meses[fObj.getMonth()];
            }
          } catch(e) {}
        }

        const estadoRaw = String(row[dEstado] || 'Pendiente').trim();
        const rowMes    = (dMes !== -1 && row[dMes]) ? normalizarNombreMes(row[dMes]) : mesCDM;
        const keyResp   = rowBms + '|' + rowMes;
        const rolHoficial = respuestasAnalyst.get(keyResp);
        const estaRespondidaAnalyst = !!rolHoficial;

        let bonoKPIsSelec = 0;
        if (dBonoKPIsSelec !== -1) {
          bonoKPIsSelec = parseFloat(row[dBonoKPIsSelec]) || 0;
        }

        disputas.push({
          idDisputa        : String(row[0] || '').trim(),
          timestamp        : parseFecha(row[dFecha]),
          bmsId            : rowBms,
          nombreAgente     : String(row[dNombre] || '').trim(),
          lob              : lobFila,
          fechaKPI         : dFecKPI !== -1 ? String(row[dFecKPI] || '') : '',
          mesCDM           : rowMes,
          monto            : dMonto !== -1 ? (parseFloat(row[dMonto]) || 0) : (parseFloat(row[9]) || 0),
          bonoKPIsSelec,
          comentarioDisputa: dComent  !== -1 ? String(row[dComent]  || '') : '',
          estadoDisputa    : estadoRaw,
          estado           : estadoRaw,
          usuario          : dUsuario !== -1 ? String(row[dUsuario] || '') : '',
          fila             : i + 1,
          tieneDisputa     : true,
          respondida       : estaRespondidaAnalyst || (estadoRaw !== 'Pendiente' && estadoRaw !== '' && !rowBms),
          respondidoPor    : estaRespondidaAnalyst ? (rolHoficial.toUpperCase().includes('REPORTING') ? 'REPORTING' : rolHoficial) : (estadoRaw !== 'Pendiente' ? (mapaUserRoles[nMap(row[dUsuario])] || 'OM') : ''),
          origen           : 'Disputa'
        });
      }
    }

    // ── Apelaciones (desde Disputa_Apelaciones) ────────────────
    try {
      const hojaApe = ss.getSheetByName('Disputa_Apelaciones');
      if (hojaApe && hojaApe.getLastRow() > 1) {
        const dataApe = hojaApe.getDataRange().getValues();
        for (let j = 1; j < dataApe.length; j++) {
          const rA     = dataApe[j];
          const bmsApe = String(rA[1] || '').trim();
          const lobApe = String(rA[3] || '').trim();
          const mesApe = normalizarNombreMes(rA[4]);

          if (!pasaFiltroLob(lobApe)) continue;
          if (!pasaFiltroBms(bmsApe)) continue;

          const rolApeOficial      = respuestasAnalyst.get(bmsApe + '|' + mesApe);
          const fueRespondidaOficial = !!rolApeOficial;

          disputas.push({
            idDisputa        : String(rA[0] || '').trim(),
            timestamp        : parseFecha(rA[11]),
            bmsId            : bmsApe,
            nombreAgente     : String(rA[2] || '').trim(),
            lob              : lobApe,
            mesCDM           : mesApe,
            monto            : parseFloat(rA[6]) || 0,
            montoAjustado    : parseFloat(rA[7]) || 0,
            comentarioDisputa: String(rA[8] || '') || String(rA[12] || ''),
            estadoDisputa    : String(rA[9] || 'Pendiente'),
            estado           : String(rA[9] || 'Pendiente'),
            usuario          : String(rA[10] || ''),
            fila             : 0,
            origen           : String(rA[15] || 'Audit'),
            respondida       : fueRespondidaOficial,
            respondidoPor    : fueRespondidaOficial ? (rolApeOficial.toUpperCase().includes('REPORTING') ? 'REPORTING' : rolApeOficial) : (String(rA[9]).toLowerCase().includes('pendiente') ? '' : (String(rA[16]) || mapaUserRoles[nMap(rA[10])] || 'OM')),
            kpisDetalle      : String(rA[13] || '')
          });
        }
      }
    } catch(eApe) { Logger.log('Err Apelaciones Dashboard: ' + eApe); }

    // ── Traslados ──────────────────────────────────────────────
    let traslados = [];
    try {
      traslados = obtenerDatosTraslados();
      if (lobsPermitidas.length > 0 && !esAdmin) {
        traslados = traslados.filter(t => pasaFiltroLob(t.lob) && pasaFiltroBms(t.bmsId));
      } else if (!esAdmin) {
        traslados = traslados.filter(t => pasaFiltroBms(t.bmsId));
      }
    } catch(e) { Logger.log('❌ Traslados: ' + e.toString()); }

    // ── Prorrateos ─────────────────────────────────────────────
    let prorrateos = [];
    try {
      prorrateos = obtenerDatosProrrateos();
      if (lobsPermitidas.length > 0 && !esAdmin) {
        prorrateos = prorrateos.filter(p => pasaFiltroLob(p.lob) && pasaFiltroBms(p.bmsId));
      } else if (!esAdmin) {
        prorrateos = prorrateos.filter(p => pasaFiltroBms(p.bmsId));
      }
    } catch(e) { Logger.log('❌ Prorrateos: ' + e.toString()); }

    // ── Horas ──────────────────────────────────────────────────
    let horasData = [];
    try {
      horasData = obtenerDatosHoras(pasaFiltroBms);
    } catch(e) { Logger.log('❌ Horas: ' + e.toString()); }

    // ── kpiDetalle desde Audit_Detalle ─────────────────────────
    const kpiDetalle = [];
    const hojaAudDet = ss.getSheetByName('Audit_Detalle');

    if (hojaAudDet && hojaAudDet.getLastRow() > 1) {
      const dDet = hojaAudDet.getDataRange().getValues();
      const hDet = dDet[0];
      const idx  = findAuditDetalleIndices(hDet);

      for (let r = 1; r < dDet.length; r++) {
        const row = dDet[r];
        if (idx.bms === -1 || !row[idx.bms]) continue;

        const lobFila = idx.lob !== -1 ? String(row[idx.lob] || '').trim() : '';
        const rowBms  = String(row[idx.bms] || '').trim();
        if (!pasaFiltroLob(lobFila)) continue;
        if (!pasaFiltroBms(rowBms))  continue;

        const esGate = idx.gate !== -1 ? parseBool(row[idx.gate]) : false;

        kpiDetalle.push({
          bmsId         : String(row[idx.bms]    || '').trim(),
          nombreAgente  : idx.nom !== -1 ? String(row[idx.nom] || '').trim() : '',
          lob           : lobFila,
          mesCDM        : idx.mes  !== -1 ? normalizarNombreMes(row[idx.mes]) : '',
          categoria     : idx.cat  !== -1 ? String(row[idx.cat]  || '').trim() : '',
          kpi           : idx.kpi  !== -1 ? String(row[idx.kpi]  || '').trim() : '',
          scoreObjetivo : idx.obj  !== -1 ? String(row[idx.obj]  || '').trim() : '',
          valorObtenido : idx.val  !== -1 ? row[idx.val] : '',
          estado        : idx.est  !== -1 ? String(row[idx.est]  || '').trim() : '',
          bono          : idx.bono !== -1 ? (parseFloat(row[idx.bono]) || 0) : 0,
          tipoEstructura: idx.tipoEstructura !== -1 ? String(row[idx.tipoEstructura] || '').trim() : '',
          esGate        : esGate
        });
      }
    }

    // ── Resumen ejecutivo ──────────────────────────────────────
    const dispPend  = disputas.filter(d => {
      const e = (d.estadoDisputa || '').toLowerCase();
      return e === 'pendiente' || e.includes('pendiente');
    }).length;

    const dispAprob = disputas.filter(d => {
      const e = (d.estadoDisputa || '').toLowerCase();
      return e.includes('aprobada') || e.includes('✅');
    }).length;

    const dispRech  = disputas.filter(d => {
      const e = (d.estadoDisputa || '').toLowerCase();
      return e.includes('rechazada') || e.includes('❌');
    }).length;

    const totalBonosAudit = auditData.reduce((s, r) => s + (r.monto || 0), 0);

    const cantApe = disputas.filter(d => d.origen === 'Audit').length;
    const apePend = disputas.filter(d => d.origen === 'Audit' && (String(d.estadoDisputa).toLowerCase().includes('pendiente'))).length;

    const resumenEjecutivo = {
      cantidadAprobaciones : auditData.length,
      cantidadDisputas     : disputas.length - cantApe,
      cantidadApelaciones  : cantApe,
      disputasPendientes   : dispPend,
      disputasAprobadas    : dispAprob,
      disputasRechazadas   : dispRech,
      apelacionesPendientes: apePend,
      cantidadTraslados    : traslados.length,
      cantidadProrrateos   : prorrateos.length,
      totalBonosAprobados  : totalBonosAudit
    };

    Logger.log('✅ FIN | audit=' + auditData.length +
               ' | disputas=' + disputas.length +
               ' | traslados=' + traslados.length +
               ' | horas=' + horasData.length +
               ' | kpiDetalle=' + kpiDetalle.length);

    // ── Alertas ────────────────────────────────────────────────
    let alertasFinancieras = [];
    try {
      alertasFinancieras = obtenerAlertasFinancierasAdditional(pasaFiltroLob, pasaFiltroBms);
    } catch(e) { Logger.log('❌ Alertas: ' + e.toString()); }

    const alertas       = generarAlertas(auditData, traslados, prorrateos);
    const alertasFinales = [...alertas, ...alertasFinancieras];

    return {
      usuario,
      lobsPermitidas,
      auditData,
      traslados,
      prorrateos,
      horasData,
      alertas         : alertasFinales,
      resumenEjecutivo,
      disputas,
      kpiDetalle,
      configKPIs      : {}
    };

  } catch (e) {
    Logger.log('❌ obtenerDatosAuditSeguro ERROR: ' + e.message + '\n' + e.stack);
    return null;
  }
}

function parsearFechaSegura(fecha) {
  try {
    if (fecha instanceof Date && !isNaN(fecha)) return fecha.toISOString();
    if (typeof fecha === 'string' && fecha.trim()) return new Date(fecha).toISOString();
    if (typeof fecha === 'number') return new Date(fecha).toISOString();
  } catch(e) {}
  return new Date().toISOString();
}

function obtenerDatosAprobaciones() {
  try {
    const ss = SpreadsheetApp.openById(SS_ID);
    const hoja = ss.getSheetByName('Audit');
    if (!hoja || hoja.getLastRow() <= 1) return [];
    const data = hoja.getDataRange().getValues();

    // Detectar índices dinámicamente por header
    const headers = data[0].map(h => String(h).toLowerCase().trim());
    const col = nombre => {
      const idx = headers.findIndex(h => h.includes(nombre));
      return idx !== -1 ? idx : -1;
    };

    const idx = {
      bmsId: col('bms') !== -1 ? col('bms') : 0,
      nombre: col('nombre') !== -1 ? col('nombre') : 1,
      lob: col('lob') !== -1 ? col('lob') : 2,
      role: col('role') !== -1 || col('rol') !== -1 ? Math.max(col('role'), col('rol')) : 3,
      supervisor: col('supervisor') !== -1 ? col('supervisor') : 4,
      accm: col('accm') !== -1 ? col('accm') : 5,
      om: headers.indexOf('om') !== -1 ? headers.indexOf('om') : 6,
      director: col('director') !== -1 || col('roll') !== -1 ? Math.max(col('director'), col('roll')) : 7,
      monto: col('bono') !== -1 ? col('bono') : 8,
      fechaAprobacion: col('aprobaci') !== -1 ? col('aprobaci') : 9,
      usuario: col('usuario') !== -1 ? col('usuario') : 10,
      correo: col('correo') !== -1 ? col('correo') : 11,
      fechaKPI: col('fecha kpi') !== -1 ? col('fecha kpi') : 12,
      mesCDM: headers.findIndex(h => h === 'mes cdm' || h.includes('mes cdm')) !== -1
        ? headers.findIndex(h => h === 'mes cdm' || h.includes('mes cdm'))
        : (col('mes') !== -1 ? col('mes') : 13),
      estado: col('estado') !== -1 ? col('estado') : 14,
      fueProrrateado: col('prorrateado') !== -1 ? col('prorrateado') : 15,
      montoOriginal: col('original') !== -1 ? col('original') : 16,
      montoFinalProrrateo: col('final') !== -1 ? col('final') : 17,
      diferenciaProrrateo: col('diferencia') !== -1 ? col('diferencia') : 18,
      diasLaborados: col('d') !== -1 && headers[col('d')] === 'días laborados' ? col('d') : (col('dias') !== -1 ? col('dias') : 19),
      horasHC: col('horas semanales') !== -1 ? col('horas semanales') : -1,
      horasTrabajadas: col('horas trabajadas') !== -1 ? col('horas trabajadas') : -1,
      ausencias: col('ausencias') !== -1 ? col('ausencias') : -1,
      tieneDisputa: headers.findIndex(h => h === 'tiene disputa' || h.includes('tiene disputa')) !== -1
        ? headers.findIndex(h => h === 'tiene disputa' || h.includes('tiene disputa'))
        : 23,
      comentarioDisputa: col('comentario') !== -1 ? col('comentario') : 21,
      tipoKPI: col('tipo kpi') !== -1 ? col('tipo kpi') : -1,
      categoriaKPI: col('categoria') !== -1 ? col('categoria') : -1,
      tipoEstructura: col('estructura') !== -1 ? col('estructura') : -1,
      esGate: col('gate') !== -1 ? col('gate') : -1,
      bonoKPIsSelec: (function() {
        // Buscar por texto primero; fallback garantizado al índice 28 (columna AC)
        const byText = headers.findIndex(h =>
          h === 'monto disputado ($)' ||
          h.includes('monto disputado') ||
          h.includes('kpis selec')
        );
        return byText !== -1 ? byText : 28;
      })(),
      id: (function() {
        const byHex = headers.findIndex(h => h === 'id' || h.includes('id disputa') || h.includes('disputa id'));
        return byHex !== -1 ? byHex : 27; // AB por defecto
      })()
    };

    console.log('📋 Headers Audit:', data[0].slice(0, 35));
    console.log('📋 idx.mesCDM =', idx.mesCDM, '→ Header:', data[0][idx.mesCDM]);
    console.log('📋 idx.bonoKPIsSelec =', idx.bonoKPIsSelec, '→ Header:', data[0][idx.bonoKPIsSelec]);

    const result = [];
    for (let i = 1; i < data.length; i++) {
      const f = data[i];
      if (!f[idx.bmsId]) continue;

      // Normalizar mesCDM siempre a nombre de mes
      const mesCDMRaw = f[idx.mesCDM];
      const mesCDMNombre = normalizarNombreMes(mesCDMRaw);

      const fueProrrateadoRaw = String(f[idx.fueProrrateado] || '').trim().toLowerCase();
      const esProrrateado = fueProrrateadoRaw === 'sí' || fueProrrateadoRaw === 'si' || fueProrrateadoRaw === 'true' || fueProrrateadoRaw === '1';
      const montoOriginal = parseFloat(f[idx.montoOriginal]) || 0;
      const montoFinalProrrateo = parseFloat(f[idx.montoFinalProrrateo]) || 0;
      const diferenciaProrrateo = parseFloat(f[idx.diferenciaProrrateo]) || 0;
      const diasLaborados = parseFloat(f[idx.diasLaborados]) || 0;
      const tieneDisputaRaw = String(f[idx.tieneDisputa] || '').trim().toLowerCase();
      const tieneDisputa = tieneDisputaRaw === 'sí' || tieneDisputaRaw === 'si' || tieneDisputaRaw === 'true' || tieneDisputaRaw === '1';
      const comentarioDisputa = String(f[idx.comentarioDisputa] || '').trim();
      const monto = parseFloat(f[idx.monto]) || 0;
      
      // Determinar tipo y categoría de KPI
      let tipoKPI = 'keys';
      let categoriaKPI = 'keys';
      
      // Si la hoja tiene columnas específicas, úsalas
      if (idx.tipoKPI !== -1 && f[idx.tipoKPI]) {
        tipoKPI = String(f[idx.tipoKPI]).trim().toLowerCase();
      }
      if (idx.categoriaKPI !== -1 && f[idx.categoriaKPI]) {
        categoriaKPI = String(f[idx.categoriaKPI]).trim().toLowerCase();
      } else {
        // Lógica de clasificación por defecto basada en el LOB y monto
        const lobNorm = String(f[idx.lob] || '').toLowerCase();
        if (lobNorm.includes('adicional') || lobNorm.includes('extra')) {
          categoriaKPI = 'additionalIncentives';
          tipoKPI = 'additional';
        } else if (lobNorm.includes('bono') || lobNorm.includes('estructura')) {
          categoriaKPI = 'bonusStructure';
          tipoKPI = 'bonus';
        } else if (monto < 50000) {
          categoriaKPI = 'decreasedKPIs';
          tipoKPI = 'decreased';
        } else {
          categoriaKPI = 'keys';
          tipoKPI = 'key';
        }
      }

      // Crear un ID único para el KPI (LOB + Mes)
      const kpiId = `${String(f[idx.lob] || '').trim()}_${mesCDMNombre}`;

      // ✅ Evaluar si el estado implica que es una apelación
      const estadoApr = String(f[idx.estado] || 'APROBADO').trim();
      const esApelacionApr = estadoApr.toLowerCase().includes('apelac') || estadoApr.toLowerCase().includes('más info') || estadoApr.toLowerCase().includes('mas info');
      const tipoAccionFilaApr = esApelacionApr ? 'Apelación' : 'Aprobación';

      result.push({
        timestamp: parsearFechaSegura(f[idx.fechaAprobacion]),
        tipoAccion: tipoAccionFilaApr,
        usuario: String(f[idx.usuario] || 'Sistema').trim(),
        correo: String(f[idx.correo] || '').trim(),
        bmsId: String(f[idx.bmsId] || '').trim(),
        nombreAgente: String(f[idx.nombre] || '').trim(),
        lob: String(f[idx.lob] || '').trim(),
        role: String(f[idx.role] || '').trim(),
        supervisor: String(f[idx.supervisor] || '').trim(),
        accm: String(f[idx.accm] || '').trim(),
        om: String(f[idx.om] || '').trim(),
        director: String(f[idx.director] || '').trim(),
        monto,
        mesCDM: mesCDMNombre,
        fechaKPI: f[idx.fechaKPI] || '',
        estado: String(f[idx.estado] || 'APROBADO').trim(),
        fueProrrateado: esProrrateado,
        montoOriginal: montoOriginal || monto,
        montoFinalProrrateo: montoFinalProrrateo || monto,
        diferenciaProrrateo,
        diasLaborados,
        horasTrabajadas: idx.horasTrabajadas !== -1 ? (parseFloat(f[idx.horasTrabajadas]) || 0) : 0,
        ausencias: idx.ausencias !== -1 ? (parseFloat(f[idx.ausencias]) || 0) : 0,
        tieneDisputa,
        comentarioDisputa,
        estadoDisputa: tieneDisputa ? 'Pendiente' : '',
        bonoKPIsSelec: idx.bonoKPIsSelec !== -1 ? (parseFloat(f[idx.bonoKPIsSelec]) || 0) : 0,
        id: idx.id !== -1 ? String(f[idx.id] || '').trim() : '',
        detalles: `Aprobación ${String(f[idx.nombre] || 'N/A')}${esProrrateado ? ' (Prorrateado)' : ''}${tieneDisputa ? ' [DISPUTA]' : ''}`,
        filaHoja: i + 1,
        tipoEstructura: idx.tipoEstructura !== -1 ? String(f[idx.tipoEstructura] || '').trim() : '',
        esGate: idx.esGate !== -1 ? (String(f[idx.esGate]).toLowerCase().trim() === 'sí' || String(f[idx.esGate]).toLowerCase().trim() === 'si' || f[idx.esGate] === true) : false,
        kpiId: kpiId,
        kpiNombre: `${String(f[idx.lob] || '').trim()} - ${mesCDMNombre}`,
        tipoKPI: tipoKPI,
        categoriaKPI: categoriaKPI,
        // Metadatos para análisis
        metadata: {
          tieneProrrateo: esProrrateado,
          tieneDisputa: tieneDisputa,
          esAjustado: esProrrateado || tieneDisputa
        }
      });
    }
    console.log(`✅ ${result.length} aprobaciones cargadas con clasificación KPI`);
    return result;
  } catch (e) {
    console.log('❌ obtenerDatosAprobaciones:', e.toString());
    return [];
  }
}

// DISPUTAS - VERSIÓN COMPLETA Y CORREGIDA
// ============================================================
function obtenerDatosDisputas() {
  try {
    const ss = SpreadsheetApp.openById(SS_ID);
    const hoja = ss.getSheetByName('Disputa');
    if (!hoja || hoja.getLastRow() <= 1) return [];
    const data = hoja.getDataRange().getValues();
    const headers = data[0].map(h => String(h).toLowerCase().trim());

    // Detectar columnas por nombre
    const col = nombre => headers.findIndex(h => h.includes(nombre));

    const idx = {
      fechaDisputa: col('fecha') !== -1 ? col('fecha') : 0,
      bmsId: col('bms') !== -1 ? col('bms') : 1,
      nombre: col('nombre') !== -1 ? col('nombre') : 2,
      lob: col('lob') !== -1 ? col('lob') : 3,
      fechaKPI: col('fecha kpi') !== -1 ? col('fecha kpi') : 4,
      total: col('bono') !== -1 ? col('bono') : (col('total') !== -1 ? col('total') : 5),
      comentario: col('comentario') !== -1 ? col('comentario') : 6,
      estado: col('estado') !== -1 ? col('estado') : 7,
      usuario: col('usuario') !== -1 ? col('usuario') : 8,
      correo: col('correo') !== -1 ? col('correo') : 9,
      // ✅ NUEVO: Columna "Bono KPIs Selec ($)" — Col I en la hoja
      bonoKPIsSelec: (() => {
        // Intentar múltiples nombres posibles para esta columna
        const opciones = ['bono kpis selec', 'bono kpi selec', 'kpis selec', 'bono seleccionado', 'bono selec'];
        for (const op of opciones) {
          const i = headers.findIndex(h => h.includes(op));
          if (i !== -1) return i;
        }
        // Fallback: columna I (índice 8) si las otras columnas coinciden
        return -1;
      })(),
      // Buscar mes CDM específicamente
      mesCDM: headers.findIndex(h => h === 'mes cdm' || h === 'mes' || h.includes('mes cdm')) !== -1
        ? headers.findIndex(h => h === 'mes cdm' || h === 'mes' || h.includes('mes cdm'))
        : -1,
      respuesta: col('respuesta') !== -1 ? col('respuesta') : -1,
      fechaRespuesta: col('fecha respuesta') !== -1 ? col('fecha respuesta') : -1
    };

    console.log('📋 Headers Disputa:', headers);
    console.log('📋 idx.mesCDM Disputa =', idx.mesCDM, '→', idx.mesCDM !== -1 ? data[0][idx.mesCDM] : 'NO ENCONTRADO');
    console.log('📋 idx.bonoKPIsSelec =', idx.bonoKPIsSelec, '→', idx.bonoKPIsSelec !== -1 ? data[0][idx.bonoKPIsSelec] : 'NO ENCONTRADO (se buscará col I)');

    const result = [];
    for (let i = 1; i < data.length; i++) {
      const f = data[i];
      if (!f[idx.bmsId] && !f[idx.nombre]) continue;

      // Obtener mes CDM
      let mesCDM = '';
      if (idx.mesCDM !== -1 && f[idx.mesCDM]) {
        mesCDM = normalizarNombreMes(f[idx.mesCDM]);
      } else if (f[idx.fechaKPI]) {
        try {
          const fechaObj = f[idx.fechaKPI] instanceof Date ? f[idx.fechaKPI] : new Date(f[idx.fechaKPI]);
          if (!isNaN(fechaObj)) {
            const mesNum = fechaObj.getMonth() + 1;
            const meses = ['enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio',
                          'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'];
            mesCDM = meses[mesNum - 1];
          }
        } catch (e) {}
      }

      // Obtener respuesta si existe
      let respuesta = '';
      let fechaRespuesta = null;
      if (idx.respuesta !== -1 && f[idx.respuesta]) {
        respuesta = String(f[idx.respuesta]).trim();
      }
      if (idx.fechaRespuesta !== -1 && f[idx.fechaRespuesta]) {
        fechaRespuesta = parsearFechaSegura(f[idx.fechaRespuesta]);
      }

      // ✅ NUEVO: Leer Bono KPIs Selec ($)
      // Si encontramos la columna por nombre, usarla; si no, intentar col I (índice 8)
      let bonoKPIsSelec = 0;
      if (idx.bonoKPIsSelec !== -1) {
        bonoKPIsSelec = parseFloat(f[idx.bonoKPIsSelec]) || 0;
      } else if (f[8] !== undefined && !isNaN(parseFloat(f[8]))) {
        // Columna I (índice 8) como fallback — solo si el valor es numérico
        const colIHeader = headers[8] || '';
        if (!colIHeader.includes('estado') && !colIHeader.includes('usuario') && !colIHeader.includes('correo')) {
          bonoKPIsSelec = parseFloat(f[8]) || 0;
        }
      }

      result.push({
        timestamp: parsearFechaSegura(f[idx.fechaDisputa]),
        tipoAccion: 'Disputa',
        usuario: String(f[idx.usuario] || 'Sistema').trim(),
        correo: String(f[idx.correo] || '').trim(),
        bmsId: String(f[idx.bmsId] || '').trim(),
        nombreAgente: String(f[idx.nombre] || '').trim(),
        lob: String(f[idx.lob] || '').trim(),
        monto: parseFloat(f[idx.total]) || 0,
        bonoKPIsSelec,  // ✅ NUEVO campo
        fechaKPI: f[idx.fechaKPI] || '',
        mesCDM: mesCDM,
        estado: String(f[idx.estado] || 'Pendiente').trim(),
        estadoDisputa: String(f[idx.estado] || 'Pendiente').trim(),
        detalles: String(f[idx.comentario] || 'Sin comentario').trim(),
        comentarioDisputa: String(f[idx.comentario] || '').trim(),
        respuesta: respuesta,
        fechaRespuesta: fechaRespuesta,
        fila: i + 1,
        respondida: respuesta !== '' && fechaRespuesta !== null
      });
    }
    console.log(`✅ ${result.length} disputas cargadas`);
    return result;
  } catch (e) {
    console.log('❌ obtenerDatosDisputas:', e.toString());
    return [];
  }
}


function obtenerDatosProrrateos() {
  try {
    const ss   = SpreadsheetApp.openById(SS_ID);
    const hoja = ss.getSheetByName('Ajustes_Prorrateo');
    if (!hoja || hoja.getLastRow() <= 1) return [];
    const data    = hoja.getDataRange().getValues();
    const headers = data[0];
    const col     = nombre => headers.findIndex(h => String(h).toLowerCase().trim().includes(nombre.toLowerCase()));
    const idx = {
      fechaProrrateo:   col('fecha prorrateo')   !== -1 ? col('fecha prorrateo')   : 0,
      bmsId:            col('bms id')            !== -1 ? col('bms id')            : 1,
      nombre:           col('nombre')            !== -1 ? col('nombre')            : 2,
      lob:              col('lob')               !== -1 ? col('lob')               : 3,
      mesCDM:           col('mes cdm')           !== -1 ? col('mes cdm')           : 4,
      montoOriginal:    col('monto original')    !== -1 ? col('monto original')    : 10,
      montoProrrateado: col('monto prorrateado') !== -1 ? col('monto prorrateado') : 11,
      diasTrabajados:   col('días trabajados')   !== -1 ? col('días trabajados')   : 7,
      factorProrrateo:  col('factor')            !== -1 ? col('factor')            : 9,
      usuario:          col('usuario')           !== -1 ? col('usuario')           : 12,
      motivo:           col('motivo')            !== -1 ? col('motivo')            : 13
    };
    const result = [];
    for (let i = 1; i < data.length; i++) {
      const f = data[i];
      if (!f[idx.bmsId] && !f[idx.nombre]) continue;
      const montoOriginal    = parseFloat(f[idx.montoOriginal])    || 0;
      const montoProrrateado = parseFloat(f[idx.montoProrrateado]) || 0;
      const diasTrabajados   = parseFloat(f[idx.diasTrabajados])   || 0;
      result.push({
        timestamp:        parsearFechaSegura(f[idx.fechaProrrateo]),
        tipoAccion:       'Prorrateo',
        usuario:          String(f[idx.usuario]  || 'Sistema').trim(),
        bmsId:            String(f[idx.bmsId]    || '').trim(),
        nombreAgente:     String(f[idx.nombre]   || '').trim(),
        lob:              String(f[idx.lob]      || '').trim(),
        mesCDM:           normalizarNombreMes(f[idx.mesCDM]),
        diasTrabajados,
        factorProrrateo:  parseFloat(f[idx.factorProrrateo]) || (diasTrabajados > 0 ? +(diasTrabajados/22).toFixed(4) : 0),
        montoOriginal,
        montoProrrateado,
        diferenciaAjuste: montoOriginal - montoProrrateado,
        monto:            montoProrrateado,
        motivo:           String(f[idx.motivo] || 'Sin motivo').trim(),
        filaHoja:         i + 1
      });
    }
    console.log(`✅ ${result.length} prorrateos`);
    return result;
  } catch(e) { console.log('❌ obtenerDatosProrrateos:', e.toString()); return []; }
}

function obtenerDatosKPIDetalle() {

    if (!hoja || hoja.getLastRow() <= 1) return [];

    const data    = hoja.getDataRange().getValues();
    const headers = data[0].map(function(h) { return String(h).toLowerCase().trim(); });

    // ── DEBUG: ver TODOS los headers y sus índices ────────────────

    headers.forEach(function(h, i) {
      Logger.log('  col ' + i + ' → "' + h + '"  (valor fila 2: "' + data[1][i] + '")');
    });

    const idx = findAuditDetalleIndices(data[0]);

    Logger.log('📋 Mapeo final de columnas:');

    });

    // ── Leer filas ───────────────────────────────────────────────

    for (var i = 1; i < data.length; i++) {
      var f = data[i];

        Logger.log('Fila ' + (i+1) + ' → kpi="' + kpiNombre + '" | scoreObj="' + f[idx.obj] + '" | valorObt="' + f[idx.val] + '"');
      }

      result.push({
        timestamp:     parsearFechaSegura(f[idx.fechaApr]),
        bmsId:         String(f[idx.bms]        || '').trim(),
        nombreAgente:  String(f[idx.nom]        || '').trim(),
        lob:           String(f[idx.lob]           || '').trim(),
        mesCDM:        normalizarNombreMes(f[idx.mes]),
        fechaKPI:      f[idx.fechaKPI]              || '',
        categoria:     String(f[idx.cat]     || '').trim(),
        kpi:           kpiNombre,
        scoreObjetivo: String(f[idx.obj] || '').trim(),
        valorObtenido: f[idx.val]         || '',
        estado:        String(f[idx.est]        || '').trim(),
        bono:          parseFloat(f[idx.bono])      || 0,
        nota:          String(f[idx.nota]          || '').trim(),
        usuario:       String(f[idx.usuario]       || '').trim(),
        esGate:        idx.gate !== -1 ? parseBool(f[idx.gate]) : false,
        tipoEstructura: idx.tipoEstructura !== -1 ? String(f[idx.tipoEstructura] || '').trim() : '',
        kpiId:         kpiNombre + '|' + String(f[idx.cat] || '').trim()
      });
    }

    Logger.log('✅ Audit_Detalle: ' + result.length + ' filas leídas. KPIs únicos: ' +
      [...new Set(result.map(function(r){ return r.kpi; }))].join(', '));

    return result;

  } catch(e) {
    Logger.log('❌ obtenerDatosKPIDetalle: ' + e.toString());
    return [];
  }
}

// RESUMEN EJECUTIVO - VERSIÓN COMPLETA Y CORREGIDA
// ============================================================
function calcularResumenEjecutivo(auditData, traslados, prorrateos, horasData) {
  try {
    const aprobaciones = auditData.filter(r => r.tipoAccion === 'Aprobación');
    const disputas = auditData.filter(r => r.tipoAccion === 'Disputa' || r.tieneDisputa === true);
    
    // Estadísticas de disputas por estado
    const disputasPendientes = disputas.filter(d => {
      const estado = (d.estado || d.estadoDisputa || '').toLowerCase();
      return estado === 'pendiente' || estado.includes('pendiente');
    }).length;
    
    const disputasAprobadas = disputas.filter(d => {
      const estado = (d.estado || '').toLowerCase();
      return estado.includes('aprobada') || estado.includes('aprobado') || estado === '✅ aprobada';
    }).length;
    
    const disputasRechazadas = disputas.filter(d => {
      const estado = (d.estado || '').toLowerCase();
      return estado.includes('rechazada') || estado.includes('rechazado') || estado === '❌ rechazada';
    }).length;
    
    // Totales financieros
    const totalBonosAprobados = aprobaciones.reduce((s, r) => s + (r.monto || 0), 0);
    const totalBonosProrrateados = prorrateos.reduce((s, r) => s + (r.montoProrrateado || 0), 0);
    const totalAhorrosProrrateo = prorrateos.reduce((s, r) => s + (r.diferenciaAjuste || 0), 0);
    
    // Agentes únicos
    const agentesUnicos = new Set(aprobaciones.map(r => r.bmsId).filter(Boolean)).size;
    
    // LOBs activas
    const lobsActivas = new Set(aprobaciones.map(r => r.lob).filter(Boolean)).size;
    
    // Meses activos
    const mesesActivos = new Set(aprobaciones.map(r => r.mesCDM).filter(Boolean));
    const mesesArray = Array.from(mesesActivos).sort();
    
    // Promedio de bono por agente
    const promedioBonoPorAgente = agentesUnicos > 0 ? totalBonosAprobados / agentesUnicos : 0;
    
    // Mes con más bonos
    const porMes = {};
    aprobaciones.forEach(r => {
      if (r.mesCDM) {
        if (!porMes[r.mesCDM]) porMes[r.mesCDM] = 0;
        porMes[r.mesCDM] += r.monto || 0;
      }
    });
    const entriesMes = Object.entries(porMes);
    const mesMasBonos = entriesMes.length > 0 ? entriesMes.sort((a, b) => b[1] - a[1])[0] : null;
    
    // LOB con más bonos
    const porLob = {};
    aprobaciones.forEach(r => {
      if (r.lob) {
        if (!porLob[r.lob]) porLob[r.lob] = 0;
        porLob[r.lob] += r.monto || 0;
      }
    });
    const entriesLob = Object.entries(porLob);
    const lobMasBonos = entriesLob.length > 0 ? entriesLob.sort((a, b) => b[1] - a[1])[0] : null;
    
    // Tasas
    const tasaDisputa = aprobaciones.length > 0 ? +((disputas.length / aprobaciones.length) * 100).toFixed(1) : 0;
    const tasaProrrateo = aprobaciones.length > 0 ? +((prorrateos.length / aprobaciones.length) * 100).toFixed(1) : 0;
    
    // Estadísticas de horas
    const promedioHorasCumplimiento = horasData.length > 0
      ? +(horasData.reduce((s, h) => s + (h.cumplimientoHrs || 0), 0) / horasData.length).toFixed(1)
      : 0;
    
    const agentesConProrrateoHoras = horasData.filter(h => h.aplicaProrrateo).length;
    const totalAusencias = horasData.reduce((s, h) => s + (h.ausencias || 0), 0);
    
    return {
      // Conteos principales
      cantidadAprobaciones: aprobaciones.length,
      cantidadDisputas: disputas.length,
      cantidadTraslados: traslados.length,
      cantidadProrrateos: prorrateos.length,
      
      // Disputas por estado
      disputasPendientes: disputasPendientes,
      disputasAprobadas: disputasAprobadas,
      disputasRechazadas: disputasRechazadas,
      
      // Financiero
      totalBonosAprobados: totalBonosAprobados,
      totalBonosProrrateados: totalBonosProrrateados,
      totalAhorrosProrrateo: totalAhorrosProrrateo,
      promedioBonoPorAgente: promedioBonoPorAgente,
      
      // Agentes y estructura
      agentesUnicos: agentesUnicos,
      lobsActivas: lobsActivas,
      mesesActivos: mesesArray.length,
      
      // Destacados
      mesMasBonos: mesMasBonos ? mesMasBonos[0] : 'N/A',
      mesMasBonos_total: mesMasBonos ? mesMasBonos[1] : 0,
      lobMasBonos: lobMasBonos ? lobMasBonos[0] : 'N/A',
      lobMasBonos_total: lobMasBonos ? lobMasBonos[1] : 0,
      
      // Tasas
      tasaDisputa: tasaDisputa,
      tasaProrrateo: tasaProrrateo,
      
      // Horas
      promedioHorasCumplimiento: promedioHorasCumplimiento,
      agentesConProrrateoHoras: agentesConProrrateoHoras,
      totalAusencias: totalAusencias,
      
      // Meta
      mesActual: mesesArray[mesesArray.length - 1] || 'N/A'
    };
    
  } catch (e) {
    console.log('❌ calcularResumenEjecutivo:', e.toString());
    return {
      cantidadAprobaciones: 0,
      cantidadDisputas: 0,
      cantidadTraslados: 0,
      cantidadProrrateos: 0,
      disputasPendientes: 0,
      disputasAprobadas: 0,
      disputasRechazadas: 0,
      totalBonosAprobados: 0,
      totalBonosProrrateados: 0,
      totalAhorrosProrrateo: 0,
      promedioBonoPorAgente: 0,
      agentesUnicos: 0,
      lobsActivas: 0,
      mesesActivos: 0,
      mesMasBonos: 'N/A',
      mesMasBonos_total: 0,
      lobMasBonos: 'N/A',
      lobMasBonos_total: 0,
      tasaDisputa: 0,
      tasaProrrateo: 0,
      promedioHorasCumplimiento: 0,
      agentesConProrrateoHoras: 0,
      totalAusencias: 0,
      mesActual: 'N/A'
    };
  }
}

function generarAlertas(auditData, traslados, prorrateos) {
  const alertas = [];
  const hoy     = new Date();

  // 1. Disputas pendientes > 3 días
  auditData.filter(r => r.tipoAccion === 'Disputa' && r.estado === 'Pendiente').forEach(d => {
    const dias = Math.floor((hoy - new Date(d.timestamp)) / 86400000);
    if (dias > 3) alertas.push({
      tipo:'warning', icono:'⏰', titulo:'Disputa sin respuesta',
      mensaje:`${d.nombreAgente} (${d.lob}) lleva ${dias} días pendiente`,
      bmsId:d.bmsId, nombreAgente:d.nombreAgente, lob:d.lob, mesCDM:d.mesCDM, diasPendiente:dias, timestamp:d.timestamp
    });
  });

  // 2. Doble pago: trasladado + 2+ aprobaciones mismo mes
  const aprobaciones = auditData.filter(r => r.tipoAccion === 'Aprobación');
  traslados.forEach(t => {
    const aprobMismoMes = aprobaciones.filter(a => a.bmsId === t.bmsId && a.mesCDM === t.mesCDM);
    if (aprobMismoMes.length > 1) alertas.push({
      tipo:'danger', icono:'🚨', titulo:'Riesgo doble pago',
      mensaje:`${t.nombreAgente} fue trasladado Y tiene ${aprobMismoMes.length} aprobaciones en ${t.mesCDM}`,
      bmsId:t.bmsId, nombreAgente:t.nombreAgente, lob:t.lob, mesCDM:t.mesCDM, timestamp:t.timestamp
    });
  });

  // 3. Prorrateo sin motivo
  prorrateos.filter(p => !p.motivo || p.motivo === 'Sin motivo').forEach(p => {
    alertas.push({
      tipo:'info', icono:'📋', titulo:'Prorrateo sin justificación',
      mensaje:`${p.nombreAgente} (${p.lob}) - ${p.mesCDM}: sin motivo registrado`,
      bmsId:p.bmsId, nombreAgente:p.nombreAgente, lob:p.lob, mesCDM:p.mesCDM, timestamp:p.timestamp
    });
  });

  console.log(`⚠️ ${alertas.length} alertas generadas`);
  return alertas;
}


function calcularComparativoMensual(auditData, prorrateos) {
  try {
    const aprobaciones = auditData.filter(r => r.tipoAccion === 'Aprobación');
    const disputas     = auditData.filter(r => r.tipoAccion === 'Disputa');
    const meses = [...new Set([
      ...aprobaciones.map(r=>r.mesCDM), ...disputas.map(r=>r.mesCDM), ...prorrateos.map(r=>r.mesCDM)
    ].filter(Boolean))].sort();
    return meses.map(mes => {
      const aM = aprobaciones.filter(r => r.mesCDM === mes);
      const dM = disputas.filter(r     => r.mesCDM === mes);
      const pM = prorrateos.filter(r   => r.mesCDM === mes);
      return {
        mes, totalBonos: aM.reduce((s,r)=>s+(r.monto||0),0),
        cantidadAprobaciones: aM.length, cantidadDisputas: dM.length, cantidadProrrateos: pM.length,
        tasaDisputa: aM.length > 0 ? +((dM.length/aM.length)*100).toFixed(1) : 0
      };
    });
  } catch(e) { return []; }
}

function responderDisputa(datos) {
  try {
    Logger.log('=== responderDisputa ===');
    Logger.log('BMS: ' + datos.bmsId + ' | Mes: ' + datos.mesCDM);

    const ss       = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const correo   = Session.getActiveUser().getEmail();
    const usuario  = correo.split('@')[0];
    const fecha    = new Date();
    const bmsId    = String(datos.bmsId  || '').trim();
    const mesDatos = String(datos.mesCDM || '').trim().toLowerCase();
    const decision = datos.decision || '';

    // ══════════════════════════════════════════════════════
    // 1. BUSCAR EN HOJA DISPUTA
    // ══════════════════════════════════════════════════════
    const hojaDisputa = ss.getSheetByName('Disputa');
    if (!hojaDisputa) {
      Logger.log('⚠️ No existe hoja Disputa → redirigir a guardarAuditoriaApelacion');
      datos.origen = 'Audit';
      return guardarAuditoriaApelacion(datos);
    }

    const dataDisputa = hojaDisputa.getDataRange().getValues();
    let filaTarget    = -1;

    // Estructura Disputa:
    // B(1)=BMS ID  F(5)=MesCDM (o col 6)  (Estado = dinámico)
      const hD       = dataDisputa[0].map(h => String(h).toLowerCase().trim());
      const idxMes   = hD.lastIndexOf('mes');
      const idxBMS   = hD.findIndex(h => h.includes('bms'));
      const idxEstado = hD.reduce((acc, cur, i) => cur.includes('estado') ? i : acc, -1);

    const idDisputaFront = String(datos.idApelacion || '').trim();

    for (let i = 1; i < dataDisputa.length; i++) {
      const idFila = String(dataDisputa[i][0] || '').trim(); // col A
      const bmsFila = idxBMS !== -1 ? String(dataDisputa[i][idxBMS] || '').trim() : '';
      const mesFila = idxMes !== -1 ? String(dataDisputa[i][idxMes] || '').trim().toLowerCase() : '';
      const estado  = idxEstado !== -1 ? String(dataDisputa[i][idxEstado] || '').trim().toLowerCase() : '';

      // Saltar registros ya cerrados
      if (estado.includes('aprobad') || estado.includes('rechazad')) continue;

      if (idDisputaFront && idFila === idDisputaFront) {
        filaTarget = i + 1;
        break;
      } else if (!idDisputaFront && bmsFila === bmsId && (!mesDatos || mesFila === mesDatos)) {
        filaTarget = i + 1;
        break;
      }
    }

    // ── Si no está en Disputa → es de Audit, redirigir ───
    if (filaTarget === -1) {
      Logger.log('⚠️ BMS [' + bmsId + '] no en Disputa → guardarAuditoriaApelacion');
      datos.origen = 'Audit';
      return guardarAuditoriaApelacion(datos);
    }

    Logger.log('✅ Disputa encontrada en fila: ' + filaTarget);

    // ══════════════════════════════════════════════════════
    // 2. CREAR COLUMNAS DE RESPUESTA SI NO EXISTEN
    // ══════════════════════════════════════════════════════
    const headersRow = dataDisputa[0];

    function getOrCreateCol(nombreHeader, color) {
      let idx = headersRow.indexOf(nombreHeader);
      if (idx >= 0) return idx + 1;
      const nuevaCol = hojaDisputa.getLastColumn() + 1;
      hojaDisputa.getRange(1, nuevaCol)
        .setValue(nombreHeader)
        .setBackground(color || '#1a6b3a')
        .setFontColor('#FFFFFF')
        .setFontWeight('bold');
      headersRow.push(nombreHeader);  // actualizar array local
      return nuevaCol;
    }

    const colDecision      = getOrCreateCol('Decisión',        '#1a6b3a');
    const colRespuesta     = getOrCreateCol('Respuesta',       '#1a6b3a');
    const colMontoAjustado = getOrCreateCol('Monto Ajustado',  '#1a6b3a');
    const colFechaResp     = getOrCreateCol('Fecha Respuesta', '#1a6b3a');
    const colUsuarioResp   = getOrCreateCol('Usuario Resp.',   '#1a6b3a');
    const colComentInt     = getOrCreateCol('Comentario Int.', '#2c3e50');

    // ══════════════════════════════════════════════════════
    // 3. ESCRIBIR RESPUESTA EN LA FILA
    // ══════════════════════════════════════════════════════
    const montoAjustado = parseFloat(datos.montoAjustado) || 0;

    hojaDisputa.getRange(filaTarget, colDecision).setValue(decision);
    hojaDisputa.getRange(filaTarget, colRespuesta).setValue(datos.respuesta || '');
    hojaDisputa.getRange(filaTarget, colMontoAjustado).setValue(montoAjustado)
      .setNumberFormat('$#,##0.00');
    hojaDisputa.getRange(filaTarget, colFechaResp).setValue(fecha)
      .setNumberFormat('dd/mm/yyyy hh:mm:ss');
    hojaDisputa.getRange(filaTarget, colUsuarioResp).setValue(usuario);
    hojaDisputa.getRange(filaTarget, colComentInt).setValue(datos.comentarioInterno || '');

    // ══════════════════════════════════════════════════════
    // 4. ACTUALIZAR ESTADO (col I = índice 8 → col 9)
    // ══════════════════════════════════════════════════════
    const estadoNuevo = decision === 'APROBADA'   ? '✅ Aprobada'
                      : decision === 'RECHAZADA'  ? '❌ Rechazada'
                      : decision === 'EN_PROCESO' ? '⏳ En Proceso'
                      : decision === 'MAS_INFO'   ? '📋 Más Info'
                      : decision;

    hojaDisputa.getRange(filaTarget, idxEstado + 1).setValue(estadoNuevo);

    const bgFila = decision === 'APROBADA'   ? '#dcfce7'
                 : decision === 'RECHAZADA'  ? '#fee2e2'
                 : decision === 'EN_PROCESO' ? '#eff6ff'
                 : '#fef3c7';
    hojaDisputa.getRange(filaTarget, 1, 1, hojaDisputa.getLastColumn())
      .setBackground(bgFila);

    // ══════════════════════════════════════════════════════
    // 5. GUARDAR EN LOG UNIFICADO "Apelaciones"
    // ══════════════════════════════════════════════════════
    const HEADERS_APE_UNIFIED = [
      'ID', 'BMS ID', 'Nombre', 'LOB', 'Mes CDM',
      'Decisión', 'Monto Anterior', 'Monto Ajustado', 'Respuesta',
      'Estado', 'Usuario', 'Fecha', 'Comentario Interno', 'KPIs Detalle',
      'Correo', 'Origen', 'Rol'
    ];

    let hojaApe = ss.getSheetByName('Disputa_Apelaciones');
    if (!hojaApe) {
      hojaApe = ss.insertSheet('Disputa_Apelaciones');
    }
    if (hojaApe.getLastRow() === 0) {
      hojaApe.appendRow(HEADERS_APE_UNIFIED);
      hojaApe.getRange(1, 1, 1, HEADERS_APE_UNIFIED.length)
        .setBackground('#4f46e5').setFontColor('#FFFFFF').setFontWeight('bold');
      hojaApe.setFrozenRows(1);
    }

    // Formato legible para ajustes KPI
    let ajustesResumen = '';
    if (datos.ajustesKPI && datos.ajustesKPI.length > 0) {
      ajustesResumen = datos.ajustesKPI.map(function(k) {
        const ant   = parseFloat(k.valorAnterior) || 0;
        const nvo   = parseFloat(k.valor)         || 0;
        const delta = nvo - ant;
        const flecha = delta > 0 ? '▲' : delta < 0 ? '▼' : '=';
        return '[' + k.categoria + '] ' + k.kpi + ': ' + ant.toFixed(2) + ' ' + flecha + ' ' + nvo.toFixed(2);
      }).join('\n');
    }

    const rolNorm = String(datos.rolUsuario || '').toUpperCase().replace(' ', '_');
    const rolGuardado = rolNorm ? `${rolNorm}_RESPONDIO` : 'SISTEMA_RESPONDIO';

    hojaApe.appendRow([
      datos.idApelacion || ('APE-' + Utilities.getUuid().substring(0, 8).toUpperCase()), // Usar ID original si existe
      bmsId,
      datos.nombre             || String(dataDisputa[filaTarget - 1][2] || ''),
      datos.lob                || String(dataDisputa[filaTarget - 1][3] || ''),
      datos.mesCDM             || String(dataDisputa[filaTarget - 1][5] || ''),
      decision,
      parseFloat(datos.montoAnterior) || parseFloat(dataDisputa[filaTarget - 1][6]) || 0,
      montoAjustado,
      datos.respuesta          || '',
      estadoNuevo,
      rolGuardado, // User wants ROLE instead of email here
      new Date(),
      datos.comentarioInterno  || '',
      ajustesResumen,
      correo,
      'Disputa',
      rolGuardado
    ]);

    const lr = hojaApe.getLastRow();
    hojaApe.getRange(lr, 8, 1, 2).setNumberFormat('$#,##0.00');
    hojaApe.getRange(lr, 1).setNumberFormat('dd/mm/yyyy hh:mm:ss');
    hojaApe.getRange(lr, 1, 1, HEADERS_APE_UNIFIED.length).setBackground(bgFila);

    // ══════════════════════════════════════════════════════
    // 6. GUARDAR DETALLE KPI EN "Apelaciones_KPI" (si aplica)
    // ══════════════════════════════════════════════════════
    if (datos.ajustesKPI && datos.ajustesKPI.length > 0) {
      let hojaKPI = ss.getSheetByName('Apelaciones_KPI');
      if (!hojaKPI) {
        hojaKPI = ss.insertSheet('Apelaciones_KPI');
        hojaKPI.appendRow([
          'Fecha', 'BMS ID', 'Nombre Agente', 'LOB', 'Mes CDM',
          'Categoría', 'KPI', 'Valor Anterior', 'Valor Nuevo', 'Δ Diferencia',
          'Decisión', 'Usuario', 'Correo'
        ]);
        hojaKPI.getRange(1, 1, 1, 13)
          .setBackground('#1e3a5f').setFontColor('#fff').setFontWeight('bold');
        hojaKPI.setFrozenRows(1);
      }

      datos.ajustesKPI.forEach(function(k) {
        const ant   = parseFloat(k.valorAnterior) || 0;
        const nvo   = parseFloat(k.valor)         || 0;
        const delta = nvo - ant;

        hojaKPI.appendRow([
          fecha, bmsId,
          datos.nombre || '', datos.lob || '', datos.mesCDM || '',
          k.categoria || '', k.kpi || '',
          ant, nvo, delta,
          decision, usuario, correo
        ]);

        const lrk = hojaKPI.getLastRow();
        hojaKPI.getRange(lrk, 8, 1, 3).setNumberFormat('0.0000');
        const bgD = delta > 0 ? '#dcfce7' : delta < 0 ? '#fee2e2' : '#f5f5f5';
        hojaKPI.getRange(lrk, 10).setBackground(bgD).setFontWeight('bold');
      });
    }

    Logger.log('✅ responderDisputa completado — fila ' + filaTarget + ' → ' + estadoNuevo);

    // Guardar también en hoja Disputa_Respuestas para consolidar el log
    guardarRespuestaEnHoja({
      fila: filaTarget,
      bmsId: bmsId,
      nombre: datos.nombre || String(dataDisputa[filaTarget - 1][2] || ''),
      lob: datos.lob || String(dataDisputa[filaTarget - 1][3] || ''),
      fechaDisputa: fecha,
      estadoOriginal: String(dataDisputa[filaTarget - 1][idxEstado] || 'Pendiente'),
      respuesta: datos.respuesta || '',
      decision: decision,
      montoAjustado: montoAjustado,
      comentarioInterno: datos.comentarioInterno || '',
      esAjusteOM: false, // Disputa normal
      rolUsuario: rolGuardado
    }, estadoNuevo);

    return {
      success: true,
      fila:    filaTarget,
      estado:  estadoNuevo,
      mensaje: 'Disputa respondida: ' + estadoNuevo
    };

  } catch(err) {
    Logger.log('❌ ERROR responderDisputa: ' + err.message + '\n' + err.stack);
    return { success: false, error: err.message };
  }
}

// ── RUTA DE APELACIONES (Auditoría) ─────────────────────────────────
function guardarAuditoriaApelacion(datos) {
  try {
    Logger.log('=== guardarAuditoriaApelacion ===');
    Logger.log('BMS: ' + datos.bmsId + ' | Origen: ' + datos.origen + ' | Mes: ' + datos.mesCDM);

    const ss       = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const correo   = Session.getActiveUser().getEmail();
    const usuario  = correo.split('@')[0];
    const fecha    = new Date();
    const bmsId    = String(datos.bmsId  || '').trim();
    const mesDatos = String(datos.mesCDM || '').trim().toLowerCase();
    const origen   = datos.origen || 'Audit';
    const decision = datos.decision || '';

    // ══════════════════════════════════════════════════════
    // 1. BUSCAR FILA EN HOJA AUDIT (búsqueda robusta por headers)
    // ══════════════════════════════════════════════════════
    const hojaAudit = ss.getSheetByName('Audit');
    let filaAudit   = -1;
    let idxBMS      = 1;   // fallback col B
    let idxMes      = 13;  // fallback col N
    let idxEstado   = -1;
    let idxBono     = -1;
    let dataAudit   = [];

    let idxID       = 27;  // fallback col AB

    if (hojaAudit) {
      dataAudit = hojaAudit.getDataRange().getValues();
      const headers = dataAudit[0];

      // Detectar columnas dinámicamente
      for (let h = 0; h < headers.length; h++) {
        const hdr = String(headers[h]).trim().toLowerCase();
        if (hdr === 'bms id' || hdr === 'bmsid' || hdr === 'bms')           idxBMS    = h;
        if (hdr === 'mes cdm' || hdr === 'mes')                              idxMes    = h;
        if (hdr === 'estado' || hdr === 'status')                            idxEstado = h;
        if (hdr.includes('bono') || hdr === 'monto' || hdr === 'aprobado')  idxBono   = h;
        if (hdr === 'id' || hdr.includes('id disputa') || hdr.includes('disputa id')) idxID = h;
      }

      Logger.log('Headers Audit: ' + JSON.stringify(headers));
      Logger.log('idxBMS=' + idxBMS + ' idxMes=' + idxMes + ' idxEstado=' + idxEstado + ' idxBono=' + idxBono);

      // Buscar fila
      const idBuscado = String(datos.idDisputa || datos.idApelacion || '').trim();

      Logger.log('🔍 AUDIT MATCH DEBUG (GuardarApelacion): idBuscado=[' + idBuscado + '] bmsId=[' + bmsId + ']');

      for (let i = 1; i < dataAudit.length; i++) {
        const bmsFila = String(dataAudit[i][idxBMS] || '').trim();
        const mesFila = String(dataAudit[i][idxMes] || '').trim().toLowerCase();
        const idFilaAudit = String(dataAudit[i][idxID] !== undefined ? dataAudit[i][idxID] : (dataAudit[i][27] || '')).trim();

        let esMatch = false;
        if (idBuscado && idFilaAudit === idBuscado) {
            esMatch = true;
            Logger.log('✅ Match exacto por ID en Audit fila ' + (i+1));
        } else if (!idBuscado && bmsFila === bmsId && (!mesDatos || mesFila === mesDatos)) {
            esMatch = true;
            Logger.log('✅ Match por BMS en Audit fila ' + (i+1));
        }

        if (esMatch) {
          filaAudit = i + 1;  // base-1 para Sheets
          break;
        } else if (bmsFila === bmsId) {
          Logger.log('⚠️ BMS coincide pero ID no: [' + idFilaAudit + '] vs [' + idBuscado + ']');
        }
      }

      Logger.log('BMS buscado: [' + bmsId + '] mes: [' + mesDatos + '] → fila encontrada: ' + filaAudit);
    }

    // ══════════════════════════════════════════════════════
    // 2. ACTUALIZAR HOJA AUDIT (estado + monto + col Apelación)
    // ══════════════════════════════════════════════════════
    if (hojaAudit && filaAudit > 0) {

      // 2a. Actualizar columna Estado si existe
      if (idxEstado >= 0) {
        hojaAudit.getRange(filaAudit, idxEstado + 1)
          .setValue('🔄 En Apelación')
          .setBackground('#fef3c7')
          .setFontWeight('bold');
      }

      // 2b. Actualizar monto si viene ajuste
      const montoAdj = parseFloat(datos.montoAjustado) || 0;
      if (idxBono >= 0 && montoAdj > 0) {
        hojaAudit.getRange(filaAudit, idxBono + 1)
          .setValue(montoAdj)
          .setNumberFormat('$#,##0.00');
      }

      // 2c. Crear/actualizar columna "Apelación"
      const headersActuales = hojaAudit.getRange(1, 1, 1, hojaAudit.getLastColumn()).getValues()[0];
      let colApe = -1;
      for (let h = 0; h < headersActuales.length; h++) {
        const hdr = String(headersActuales[h]).trim().toLowerCase();
        if (hdr === 'apelación' || hdr === 'apelacion') { colApe = h + 1; break; }
      }
      if (colApe === -1) {
        colApe = hojaAudit.getLastColumn() + 1;
        hojaAudit.getRange(1, colApe)
          .setValue('Apelación')
          .setBackground('#4f46e5')
          .setFontColor('#fff')
          .setFontWeight('bold');
      }

      const emoji    = decision === 'APROBADA' ? '✅' : decision === 'RECHAZADA' ? '❌' : '⏳';
      const fechaStr = Utilities.formatDate(fecha, Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm');
      hojaAudit.getRange(filaAudit, colApe)
        .setValue(emoji + ' ' + decision + ' — ' + usuario + ' (' + fechaStr + ')');

      // 2d. Color de fila completa
      const bgFila = decision === 'APROBADA'  ? '#dcfce7'
                   : decision === 'RECHAZADA' ? '#fee2e2'
                   : '#eff6ff';
      hojaAudit.getRange(filaAudit, 1, 1, hojaAudit.getLastColumn())
        .setBackground(bgFila);

      Logger.log('✅ Hoja Audit actualizada — fila ' + filaAudit + ' → ' + decision);

    } else {
      Logger.log('⚠️ BMS [' + bmsId + '] no encontrado en Audit (mes: [' + mesDatos + ']) — se guarda igual en Apelaciones');
    }

    // ══════════════════════════════════════════════════════
    // 3. GUARDAR EN HOJA "Disputa_Apelaciones" (log unificado)
    // ══════════════════════════════════════════════════════

    const HEADERS_APE_AUDIT = [
      'ID', 'BMS ID', 'Nombre', 'LOB', 'Mes CDM',
      'Decisión', 'Monto Anterior', 'Monto Ajustado', 'Respuesta',
      'Estado', 'Usuario', 'Fecha', 'Comentario Interno', 'KPIs Detalle',
      'Correo', 'Origen', 'Rol', 'Original Audit ID'
    ];

    const rolNorm = String(datos.rolUsuario || '').toUpperCase().replace(' ', '_');
    const rolGuardado = rolNorm ? `${rolNorm}_APELACION` : 'SISTEMA_APELACION';

    let hojaApe = ss.getSheetByName('Disputa_Apelaciones');
    if (!hojaApe) {
      hojaApe = ss.insertSheet('Disputa_Apelaciones');
    }
    if (hojaApe.getLastRow() === 0) {
      hojaApe.appendRow(HEADERS_APE_AUDIT);
      hojaApe.getRange(1, 1, 1, HEADERS_APE_AUDIT.length)
        .setBackground('#4f46e5').setFontColor('#FFFFFF').setFontWeight('bold');
      hojaApe.setFrozenRows(1);
    }

    // Formato legible para ajustes KPI (en lugar de JSON crudo)
    let ajustesResumen = '';
    if (datos.ajustesKPI && datos.ajustesKPI.length > 0) {
      ajustesResumen = datos.ajustesKPI.map(function(k) {
        const ant = parseFloat(k.valorAnterior) || 0;
        const nvo = parseFloat(k.valor)         || 0;
        const delta = nvo - ant;
        const flecha = delta > 0 ? '▲' : delta < 0 ? '▼' : '=';
        return '[' + k.categoria + '] ' + k.kpi + ': ' + ant.toFixed(2) + ' ' + flecha + ' ' + nvo.toFixed(2);
      }).join('\n');
    }

    let estadoLabel = decision === 'APROBADA'   ? '✅ Apelación Aprobada'
                      : decision === 'RECHAZADA'  ? '❌ Apelación Rechazada'
                      : decision === 'EN_PROCESO' ? '⏳ Apelación En Proceso'
                      : decision === 'MAS_INFO'   ? '📋 Apelación - Más Info'
                      : '⚖️ Apelación Registrada';

    // If decision comes with custom text that already contains 'Apelación', don't override
    if (decision && decision !== 'APROBADA' && decision !== 'RECHAZADA' && decision !== 'EN_PROCESO' && decision !== 'MAS_INFO') {
       if (!decision.toLowerCase().includes('apel')) {
           estadoLabel = '⚖️ Apelación ' + decision;
       } else {
           estadoLabel = decision;
       }
    }

    hojaApe.appendRow([
      'APE-' + Utilities.getUuid().substring(0, 8).toUpperCase(), // ID
      bmsId,
      datos.nombre             || (filaAudit > 0 ? String(dataAudit[filaAudit - 1][idxBMS - (idxBMS > 0 ? 1 : 0) + 1] || '') : ''), // Intento nombre
      datos.lob                || (filaAudit > 0 ? String(dataAudit[filaAudit - 1][idxBMS - (idxBMS > 1 ? 1 : 0)] || '') : ''), // Fallback LOB
      datos.mesCDM             || '',
      decision,
      parseFloat(datos.montoAnterior) || 0,
      parseFloat(datos.montoAjustado) || 0,
      datos.respuesta          || '',
      estadoLabel,
      rolGuardado, // User wants ROLE instead of email here
      fecha,
      datos.comentarioInterno  || '',
      ajustesResumen,
      correo,
      origen,
      rolGuardado,
      String(datos.idApelacion || datos.id || '').trim() // Original Audit ID (col AB index)
    ]);

    const lastRow = hojaApe.getLastRow();
    hojaApe.getRange(lastRow, 8, 1, 2).setNumberFormat('$#,##0.00');
    hojaApe.getRange(lastRow, 1).setNumberFormat('dd/mm/yyyy hh:mm:ss');

    const bgApe = decision === 'APROBADA'  ? '#dcfce7'
                : decision === 'RECHAZADA' ? '#fee2e2'
                : '#eff6ff';
    hojaApe.getRange(lastRow, 1, 1, HEADERS_APE_AUDIT.length).setBackground(bgApe);

    // ══════════════════════════════════════════════════════
    // 4. GUARDAR DETALLE KPI EN HOJA "Apelaciones_KPI"
    // ══════════════════════════════════════════════════════
    if (datos.ajustesKPI && datos.ajustesKPI.length > 0) {
      let hojaKPI = ss.getSheetByName('Apelaciones_KPI');
      if (!hojaKPI) {
        hojaKPI = ss.insertSheet('Apelaciones_KPI');
      }
      if (hojaKPI.getLastRow() === 0) {
        hojaKPI.appendRow([
          'Fecha', 'BMS ID', 'Nombre Agente', 'LOB', 'Mes CDM',
          'Categoría', 'KPI', 'Valor Anterior', 'Valor Nuevo', 'Δ Diferencia',
          'Decisión', 'Usuario', 'Correo', 'Rol Usuario'
        ]);
        hojaKPI.getRange(1, 1, 1, 14)
          .setBackground('#1e3a5f').setFontColor('#fff').setFontWeight('bold');
        hojaKPI.setFrozenRows(1);
      }

      datos.ajustesKPI.forEach(function(k) {
        const ant   = parseFloat(k.valorAnterior) || 0;
        const nvo   = parseFloat(k.valor)         || 0;
        const delta = nvo - ant;

        hojaKPI.appendRow([
          fecha,
          bmsId,
          datos.nombre  || '',
          datos.lob     || '',
          datos.mesCDM  || '',
          k.categoria   || '',
          k.kpi         || '',
          ant,
          nvo,
          delta,
          decision,
          usuario,
          correo,
          rolGuardado // Agregando el Rol explícito a Apelaciones_KPI
        ]);

        const lr = hojaKPI.getLastRow();
        hojaKPI.getRange(lr, 8, 1, 3).setNumberFormat('0.0000');
        // Verde si mejora (sube), rojo si baja
        const bgDelta = delta > 0 ? '#dcfce7' : delta < 0 ? '#fee2e2' : '#f5f5f5';
        hojaKPI.getRange(lr, 10).setBackground(bgDelta).setFontWeight('bold');
      });

      Logger.log('✅ ' + datos.ajustesKPI.length + ' ajustes KPI en Apelaciones_KPI');
    }

    // Guardar también en hoja Disputa_Respuestas
    guardarRespuestaEnHoja({
      fila: filaAudit,
      bmsId: bmsId,
      nombre: datos.nombre || (filaAudit > 0 ? String(dataAudit[filaAudit - 1][idxBMS - (idxBMS > 0 ? 1 : 0) + 1] || '') : ''),
      lob: datos.lob || (filaAudit > 0 ? String(dataAudit[filaAudit - 1][idxBMS - (idxBMS > 1 ? 1 : 0)] || '') : ''),
      fechaDisputa: fecha,
      estadoOriginal: 'Auditoría', 
      respuesta: datos.respuesta || '',
      decision: decision,
      montoAjustado: parseFloat(datos.montoAjustado) || 0,
      comentarioInterno: datos.comentarioInterno || '',
      esAjusteOM: true, // Es apelación / ajuste auditoría
      rolUsuario: rolGuardado
    }, estadoLabel);

    Logger.log('✅ guardarAuditoriaApelacion completado');
    return {
      success:    true,
      fila:       filaAudit,
      mensaje:    'Apelación guardada correctamente' + (filaAudit > 0 ? ' (Audit fila ' + filaAudit + ')' : ' (fila no encontrada, guardada en log)')
    };

  } catch(err) {
    Logger.log('❌ ERROR guardarAuditoriaApelacion: ' + err.message + '\n' + err.stack);
    return { success: false, error: err.message };
  }
}
function obtenerHistorialDisputasAgente(bmsId) {
  try {
    const ss      = SpreadsheetApp.openById(SS_ID);
    const result  = [];
    const bmsStr  = String(bmsId).trim();

    // ── 0. Registro "Original" desde Audit ───────────────────
    const hojaAudit = ss.getSheetByName('Audit');
    if (hojaAudit) {
      const dataAud = hojaAudit.getDataRange().getValues();
      const hAud    = dataAud[0].map(h => String(h).toLowerCase().trim());
      const colBMS  = hAud.findIndex(h => h.includes('bms'));
      const colMes  = hAud.findIndex(h => h === 'mes cdm' || h === 'mes');
      const colBono = hAud.findIndex(h => h.includes('bono aprobado') || h === 'bono' || h === 'total bono');
      const colFecha = hAud.findIndex(h => h.includes('fecha aprobac'));
      const colLOB   = hAud.findIndex(h => h === 'lob' || h.includes('lob'));

      if (colBMS !== -1) {
        for (let a = 1; a < dataAud.length; a++) {
          if (String(dataAud[a][colBMS] || '').trim() !== bmsStr) continue;
          result.push({
            id:                'ORIGINAL',
            origen:            'Original',
            bmsId:             bmsStr,
            nombreAgente:      'Aprobación Inicial',
            lob:               colLOB !== -1 ? String(dataAud[a][colLOB] || '').trim() : '',
            mesCDM:            normalizarNombreMes(colMes !== -1 ? dataAud[a][colMes] : ''),
            monto:             parseFloat(colBono !== -1 ? dataAud[a][colBono] : 0) || 0,
            montoAjustado:     0,
            bonoKPIsSelec:     0,  // ✅ campo incluido
            estadoDisputa:     '✅ Aprobado',
            timestamp:         colFecha !== -1 && dataAud[a][colFecha]
                                 ? (dataAud[a][colFecha] instanceof Date
                                    ? dataAud[a][colFecha].toISOString()
                                    : String(dataAud[a][colFecha]))
                                 : null,
            respondida:        true,
            comentarioDisputa: 'Valor aprobado inicialmente'
          });
        }
      }
    }

    // ── 1. Disputas desde hoja Disputa ────────────────────────
    const hojaDisp = ss.getSheetByName('Disputa');
    if (hojaDisp && hojaDisp.getLastRow() > 1) {
      const data    = hojaDisp.getDataRange().getValues();
      const headers = data[0].map(h => String(h).toLowerCase().trim());
      const col     = n => headers.findIndex(h => h.includes(n));

      const idx = {
        bms:        col('bms'),
        nombre:     col('nombre'),
        lob:        col('lob'),
        fecha:      col('fecha'),
        mes:        headers.lastIndexOf('mes'),
        monto:      headers.findIndex(h => h.includes('total bono')) !== -1 ? headers.findIndex(h => h.includes('total bono')) :
                    (col('bono') !== -1 ? col('bono') : col('monto')),
        comentario: headers.lastIndexOf('correo usuario') !== -1 ? headers.lastIndexOf('correo usuario') : col('comentario'),
        estado:     headers.reduce((acc, cur, i) => cur.includes('estado') ? i : acc, -1),
        decision:   headers.findIndex(h => h === 'decisión' || h === 'decision'),
        respuesta:  headers.findIndex(h => h === 'respuesta'),
        montoAjust: headers.findIndex(h => h.includes('monto ajustado')),
        fechaResp:  headers.findIndex(h => h.includes('fecha respuesta')),
        comentInt:  headers.findIndex(h => h.includes('comentario int')),
        usuarioResp:headers.findIndex(h => h.includes('usuario resp')),
        // ✅ NUEVO: Bono KPIs Selec ($)
        bonoKPIsSelec: (() => {
          const opciones = ['bono kpis selec', 'bono kpi selec', 'kpis selec', 'bono seleccionado', 'bono selec'];
          for (const op of opciones) {
            const i = headers.findIndex(h => h.includes(op));
            if (i !== -1) return i;
          }
          return -1;
        })()
      };

      for (let i = 1; i < data.length; i++) {
        const f = data[i];
        if (String(f[idx.bms] || '').trim() !== bmsStr) continue;

        const decision   = idx.decision  !== -1 ? String(f[idx.decision]  || '').trim() : '';
        const respuesta  = idx.respuesta !== -1 ? String(f[idx.respuesta] || '').trim() : '';
        const montoAjust = idx.montoAjust !== -1 ? parseFloat(f[idx.montoAjust]) || 0 : 0;
        const fechaResp  = idx.fechaResp  !== -1 ? f[idx.fechaResp] : null;
        const respondida = decision !== '' || respuesta !== '';
        // ✅ NUEVO
        const bonoKPIsSelec = idx.bonoKPIsSelec !== -1 ? (parseFloat(f[idx.bonoKPIsSelec]) || 0) : 0;

        result.push({
          id:                'DISP-' + (i + 1),
          fila:              i + 1,
          bmsId:             bmsStr,
          nombreAgente:      String(f[idx.nombre]      || '').trim(),
          lob:               String(f[idx.lob]         || '').trim(),
          timestamp:         parsearFechaSegura(f[idx.fecha]),
          mesCDM:            normalizarNombreMes(idx.mes !== -1 ? f[idx.mes] : ''),
          monto:             parseFloat(f[idx.monto])  || 0,
          bonoKPIsSelec,     // ✅ NUEVO
          comentarioDisputa: String(f[idx.comentario]  || '').trim(),
          estadoDisputa:     String(f[idx.estado]      || 'Pendiente').trim(),
          respondida,
          respuesta,
          decision,
          montoAjustado:     montoAjust,
          comentarioInterno: idx.comentInt  !== -1 ? String(f[idx.comentInt]  || '').trim() : '',
          usuarioRespuesta:  idx.usuarioResp !== -1 ? String(f[idx.usuarioResp] || '').trim() : '',
          fechaRespuesta:    fechaResp ? parsearFechaSegura(fechaResp) : null,
          origen:            'Disputa',
          kpisDetalle:       ''
        });
      }
    }

    // ── 2. Apelaciones desde Disputa_Apelaciones ─────────────
    const hojaApe = ss.getSheetByName('Disputa_Apelaciones');
    if (hojaApe && hojaApe.getLastRow() > 1) {
      const dataApe = hojaApe.getDataRange().getValues();
      const headers = dataApe[0].map(h => String(h).toLowerCase().trim());

      const iID      = headers.findIndex(h => h === 'id');
      const iBMS     = headers.findIndex(h => h.includes('bms'));
      const iNombre  = headers.findIndex(h => h.includes('nombre'));
      const iLOB     = headers.findIndex(h => h === 'lob');
      const iMes     = headers.findIndex(h => h.includes('mes'));
      const iDec     = headers.findIndex(h => h.includes('decisi'));
      const iMAnt    = headers.findIndex(h => h.includes('monto ant') || h.includes('monto ori'));
      const iMAj     = headers.findIndex(h => h.includes('monto ajust') || h.includes('monto dis'));
      const iResp    = headers.findIndex(h => h === 'respuesta' || h.includes('motivo'));
      const iEst     = headers.findIndex(h => h === 'estado');
      const iUsr     = headers.findIndex(h => h === 'usuario');
      const iFecha   = headers.findIndex(h => h === 'fecha');
      const iComInt  = headers.findIndex(h => h.includes('comentario'));
      const iKPIs    = headers.findIndex(h => h.includes('kpi'));
      const iOrigen  = headers.findIndex(h => h === 'origen');

      for (let j = 1; j < dataApe.length; j++) {
        const row = dataApe[j];
        if (String(row[iBMS] || '').trim() !== bmsStr) continue;

        const decision   = String(row[iDec   !== -1 ? iDec   : 5] || '').trim();
        const respondida = decision !== '' && decision !== 'En Revisión';

        result.push({
          id:                String(row[iID    !== -1 ? iID    : 0]  || '').trim(),
          fila:              0,
          bmsId:             bmsStr,
          nombreAgente:      String(row[iNombre !== -1 ? iNombre : 2] || '').trim(),
          lob:               String(row[iLOB   !== -1 ? iLOB   : 3]  || '').trim(),
          timestamp:         parsearFechaSegura(row[iFecha !== -1 ? iFecha : 11]),
          mesCDM:            normalizarNombreMes(String(row[iMes !== -1 ? iMes : 4] || '')),
          monto:             parseFloat(row[iMAnt !== -1 ? iMAnt : 6]) || 0,
          montoAjustado:     parseFloat(row[iMAj  !== -1 ? iMAj  : 7]) || 0,
          bonoKPIsSelec:     0,  // ✅ campo incluido (apelaciones no tienen este valor)
          comentarioDisputa: 'Apelación: ' + String(row[iResp !== -1 ? iResp : 8] || ''),
          estadoDisputa:     String(row[iEst !== -1 ? iEst : 9] || 'En Revisión').trim(),
          respondida,
          respuesta:         String(row[iResp   !== -1 ? iResp   : 8]  || '').trim(),
          decision,
          comentarioInterno: String(row[iComInt  !== -1 ? iComInt : 12] || '').trim(),
          usuarioRespuesta:  String(row[iUsr     !== -1 ? iUsr    : 10] || '').trim(),
          fechaRespuesta:    respondida
                               ? parsearFechaSegura(row[iFecha !== -1 ? iFecha : 11])
                               : null,
          origen:            String(row[iOrigen !== -1 ? iOrigen : 15] || 'Audit').trim(),
          kpisDetalle:       String(row[iKPIs !== -1 ? iKPIs : 13] || '').trim()
        });
      }
    }

    // ── 3. Ordenar ────────────────────────────────────────────
    result.sort((a, b) => {
      if (a.origen === 'Original' && b.origen !== 'Original') return -1;
      if (b.origen === 'Original' && a.origen !== 'Original') return  1;
      const tA = a.timestamp ? new Date(a.timestamp).getTime() : 0;
      const tB = b.timestamp ? new Date(b.timestamp).getTime() : 0;
      return tB - tA;
    });

    Logger.log('✅ Historial BMS [' + bmsStr + ']: ' + result.length + ' entradas');
    return result;

  } catch(e) {
    Logger.log('❌ obtenerHistorialDisputasAgente: ' + e.toString());
    return [];
  }
}

function guardarRespuestaDesdeModal(fila, bmsId, mesCDM, montoOriginal) {
    const decision  = document.getElementById('dec-'   + fila)?.value.trim()   || '';
    const respuesta = document.getElementById('resp-'  + fila)?.value.trim()   || '';
    const monto     = parseFloat(document.getElementById('monto-' + fila)?.value) || 0;
    const comentInt = document.getElementById('coment-'+ fila)?.value.trim()   || '';

    // Validar campos obligatorios
    if (!decision) {
        toast('⚠️ Selecciona una decisión', 'warning');
        document.getElementById('dec-' + fila).style.borderColor = 'var(--danger)';
        return;
    }
    if (!respuesta) {
        toast('⚠️ El comentario/justificación es obligatorio', 'warning');
        document.getElementById('resp-' + fila).style.borderColor = 'var(--danger)';
        return;
    }
    if (!monto || monto <= 0) {
        toast('⚠️ El valor en pesos es obligatorio', 'warning');
        document.getElementById('monto-' + fila).style.borderColor = 'var(--danger)';
        return;
    }

    const datosRespuesta = {
        fila,
        bmsId,
        mesCDM,
        decision,
        respuesta,
        montoAjustado:     monto,
        comentarioInterno: comentInt,
        nombre:            document.getElementById('det-nombre')?.textContent || '',
        lob:               document.getElementById('det-lob')?.textContent    || '',
        estadoOriginal:    'Pendiente',
        fechaDisputa:      new Date().toISOString(),
        rolUsuario:        (window.USUARIO_ACTIVO && window.USUARIO_ACTIVO.rol) ? window.USUARIO_ACTIVO.rol : 'Auditor'
    };

    // Deshabilitar botón mientras guarda
    const btn = document.querySelector(`#form-disputa-${fila} button`);
    if (btn) { btn.disabled = true; btn.textContent = '⏳ Guardando...'; }

    toast('Guardando respuesta...', 'info');

    google.script.run
        .withSuccessHandler(function(result) {
            if (result?.sinPermiso) {
                mostrarAlertaSinPermiso('disputas');
                return;
            }
            if (result?.success) {
                toast('✅ Respuesta guardada correctamente', 'success');
                // Recargar detalles del agente para reflejar cambio
                setTimeout(() => {
                    verDetallesAgente(bmsId, document.getElementById('det-nombre')?.textContent || '');
                }, 800);
            } else {
                toast('Error: ' + (result?.error || 'Error desconocido'), 'error');
                if (btn) { btn.disabled = false; btn.textContent = '💾 Guardar Respuesta'; }
            }
        })
        .withFailureHandler(function(err) {
            toast('Error: ' + (err.message || err), 'error');
            if (btn) { btn.disabled = false; btn.textContent = '💾 Guardar Respuesta'; }
        })
        .responderDisputa(datosRespuesta);
}

// Función auxiliar para ajustar bono en Audit
function ajustarBonoEnAudit(bmsId, lob, nuevoMonto, mesCDM) {
  try {
    const hoja = SpreadsheetApp.openById(SS_ID).getSheetByName('Audit');
    if (!hoja) return;
    
    const data = hoja.getDataRange().getValues();
    const headers = data[0].map(h => String(h).toLowerCase().trim());
    
    const cBMS = headers.findIndex(h => h.includes('bms'));
    const cLOB = headers.findIndex(h => h.includes('lob'));
    const cMes = headers.findIndex(h => h.includes('mes cdm') || h.includes('mes'));
    const cMonto = headers.findIndex(h => h.includes('bono'));
    
    if (cBMS === -1 || cMonto === -1) return;
    
    for (let i = 1; i < data.length; i++) {

function obtenerRespuestaDisputa(fila, bmsId, mesCDM) {
  try {
    const ss = SpreadsheetApp.openById(SS_ID);
    const bmsBusq = bmsId ? String(bmsId).trim().toLowerCase() : '';
    const mesBusq = mesCDM ? norm(String(mesCDM)) : '';

    // ── 1. BUSCAR EN Disputa_Respuestas (Disputas de Formulario) ──
    const hojaResp = ss.getSheetByName('Disputa_Respuestas');
    if (hojaResp && hojaResp.getLastRow() > 1) {
      const dataResp = hojaResp.getDataRange().getValues();
      const hR = dataResp[0].map(h => norm(String(h)));
      const iID = hR.indexOf('id');
      const iBMS = hR.indexOf('bms id');
      const iMes = hR.indexOf('mes cdm');

      for (let i = 1; i < dataResp.length; i++) {
        const row = dataResp[i];
        const idFila = iID !== -1 ? parseInt(row[iID]) : -1;
        const bmsFila = iBMS !== -1 ? String(row[iBMS]).toLowerCase().trim() : '';
        const mesFila = iMes !== -1 ? norm(String(row[iMes])) : '';

        if ((fila > 0 && idFila === parseInt(fila)) || (bmsBusq && bmsFila === bmsBusq && mesFila === mesBusq)) {
          return {
            decision: String(row[hR.indexOf('decision')] || row[hR.indexOf('decisión')] || ''),
            respuesta: String(row[hR.indexOf('respuesta')] || ''),
            montoAjustado: parseFloat(String(row[hR.indexOf('monto ajustado')] || '0').replace(/[^0-9.]/g,'')) || 0,
            usuario: String(row[hR.indexOf('usuario')] || ''),
            fechaRespuesta: row[hR.indexOf('fecha respuesta')] instanceof Date ? row[hR.indexOf('fecha respuesta')].toISOString() : String(row[hR.indexOf('fecha respuesta')] || ''),
            comentarioInterno: String(row[hR.indexOf('comentario interno')] || '')
          };
        }
      }
    }

    // ── 2. BUSCAR EN Disputa_Apelaciones (Ajustes de Auditoría) ──
    const hojaApe = ss.getSheetByName('Disputa_Apelaciones');
    if (hojaApe && hojaApe.getLastRow() > 1) {
      const dataApe = hojaApe.getDataRange().getValues();
      // Esquema estándar: [ID, BMS, Nom, LOB, Mes, Dec, M.Ant, M.Aj, Resp, Est, Usr, Fec, ComInt, KPIs]
      for (let i = 1; i < dataApe.length; i++) {
        const rA = dataApe[i];
        const bmsApe = String(rA[1]).toLowerCase().trim();
        const mesApe = norm(String(rA[4]));

        if (bmsBusq && bmsApe === bmsBusq && mesApe === mesBusq) {
          return {
            decision: String(rA[9] || rA[5] || ''), // Estado o Decisión
            respuesta: String(rA[8] || ''),
            montoAjustado: parseFloat(rA[7]) || 0,
            usuario: String(rA[10] || ''),
            fechaRespuesta: rA[11] instanceof Date ? rA[11].toISOString() : String(rA[11] || ''),
            comentarioInterno: String(rA[12] || ''),
            kpisDetalle: String(rA[13] || '')
          };
        }
      }
    }

    // ── 3. FALLBACK: Hoja Disputa (columnas 12-16) ──
    const hojaDisp = ss.getSheetByName('Disputa');
    if (hojaDisp && fila > 0) {
      const data = hojaDisp.getDataRange().getValues();
      if (fila < data.length + 1) {
        const row = data[fila - 1];
        if (row[12]) { // Si hay decisión en col 13
          return {
            decision: String(row[12] || ''),
            respuesta: String(row[13] || ''),
            montoAjustado: parseFloat(row[14]) || 0,
            usuario: String(row[9] || ''),
            fechaRespuesta: row[11] instanceof Date ? row[11].toISOString() : String(row[11] || ''),
            comentarioInterno: String(row[15] || '')
          };
        }
      }
    }

    return null;
  } catch (e) {
    console.log('❌ obtenerRespuestaDisputa:', e.toString());
    return null;
  }
}

function guardarEnHistoricoMax(datosParaHistorico, ssCDM) {
  try {
    console.log('💾 Guardando en Historico_MAX...');
    let hojaCache = ssCDM.getSheetByName('Historico_MAX');
    if (!hojaCache) hojaCache = ssCDM.insertSheet('Historico_MAX');

    // Limpiar columnas extra
    const ultimaColumna = hojaCache.getLastColumn();
    if (ultimaColumna > 19) hojaCache.deleteColumns(20, ultimaColumna - 19);

    const ultimaFila = hojaCache.getLastRow();
    if (ultimaFila === 0) {
      const headers = ['Fecha Generación','Fecha Registro','LOB','OM','Mes','Numero Mes',
                       'Total Bonos','Total Agentes','Keys','Bonus Structure','Additional Incentives',
                       'Decreased KPIs','% Keys','% Bonus','% Additional','% Decreased',
                       'Usuario','Correo Usuario','Año'];
      hojaCache.getRange(1,1,1,19).setValues([headers]);
      const hr = hojaCache.getRange(1,1,1,19);
      hr.setBackground('#8e44ad'); hr.setFontColor('#FFFFFF'); hr.setFontWeight('bold'); hr.setHorizontalAlignment('center');
    }

    const fechaGeneracion = new Date();
    const usuario         = Session.getActiveUser().getEmail();
    const nombreUsuario   = usuario.split('@')[0];
    const anio            = fechaGeneracion.getFullYear();

    const dataExistente   = hojaCache.getLastRow() > 1 ? hojaCache.getRange(2,1,hojaCache.getLastRow()-1,19).getValues() : [];
    const existeMap = {}, filasMap = {};
    dataExistente.forEach((row, i) => {
      const clave = `${String(row[2]||'').trim()}_${String(row[4]||'').trim()}`;
      existeMap[clave] = true;
      filasMap[clave]  = i + 2;
    });

    let nuevos = 0, actualizados = 0;
    datosParaHistorico.forEach(dato => {
      const totalBonos   = dato.totalBonos || 0;
      const porcKeys     = totalBonos > 0 ? ((dato.desglose.keys||0)/totalBonos*100) : 0;
      const porcBonus    = totalBonos > 0 ? ((dato.desglose.bonusStructure||0)/totalBonos*100) : 0;
      const porcAdditional = totalBonos > 0 ? ((dato.desglose.additionalIncentives||0)/totalBonos*100) : 0;
      const porcDecreased  = totalBonos > 0 ? ((dato.desglose.decreasedKPIs||0)/totalBonos*100) : 0;
      const fila = [
        fechaGeneracion, dato.fechaRegistro||fechaGeneracion, dato.lob||'Sin LOB', dato.om||'-',
        dato.mes||'General', dato.numeroMes||0, totalBonos, dato.agentes||0,
        dato.desglose.keys||0, dato.desglose.bonusStructure||0, dato.desglose.additionalIncentives||0,
        dato.desglose.decreasedKPIs||0, porcKeys, porcBonus, porcAdditional, porcDecreased,
        nombreUsuario, usuario, anio
      ];
      const clave = `${dato.lob}_${dato.mes}`;
      if (existeMap[clave]) {
        hojaCache.getRange(filasMap[clave],1,1,19).setValues([fila]);
        actualizados++;
      } else {
        hojaCache.getRange(hojaCache.getLastRow()+1,1,1,19).setValues([fila]);
        nuevos++;
      }
    });

    console.log(`📊 Historico_MAX: ${nuevos} nuevos, ${actualizados} actualizados`);
    return { success:true, count:nuevos+actualizados, nuevos, actualizados,
             mensaje:`${nuevos} nuevos, ${actualizados} actualizados en Historico_MAX` };
  } catch(error) {
    console.error('❌ guardarEnHistoricoMax:', error.toString());
    return { success:false, error:error.toString() };
  }
}

      }
    }

    // ── Historial desde Audit ─────────────────────────────────
    const dataAud = hojaAud.getDataRange().getValues();
    const headAud = dataAud[0];

    const idx = (nombre) => {
      const nNorm = norm(nombre);
      // 1. Exacto
      for (let c = 0; c < headAud.length; c++) if (norm(String(headAud[c])) === nNorm) return c;
      // 2. Parcial
      for (let c = 0; c < headAud.length; c++) if (norm(String(headAud[c])).includes(nNorm)) return c;
      return -1;
    };

    const iBMS    = idx('BMS ID') !== -1 ? idx('BMS ID') : idx('BMS');
    const iNombre = idx('Nombre Agente') !== -1 ? idx('Nombre Agente') : idx('Nombre');
    const iLOB    = idx('LOB') !== -1 ? idx('LOB') : 3;
    const iRole   = idx('Role') !== -1 ? idx('Role') : 4;
    const iRoll   = idx('Roll') !== -1 ? idx('Roll') : 5;
    const iSup    = idx('Supervisor') !== -1 ? idx('Supervisor') : 6;
    const iACCM   = idx('ACCM') !== -1 ? idx('ACCM') : (idx('OAM') !== -1 ? idx('OAM') : 7);
    const iOM     = idx('OM') !== -1 ? idx('OM') : 8;
    const iBono   = idx('Bono Aprobado') !== -1 ? idx('Bono Aprobado') : 9;
    const iFecha  = idx('Fecha Aprobación') !== -1 ? idx('Fecha Aprobación') : 10;
    const iFechaK = idx('Fecha KPI') !== -1 ? idx('Fecha KPI') : 11;
    const iMes    = idx('Mes CDM') !== -1 ? idx('Mes CDM') : 12;
    const iEst    = idx('Estado') !== -1 ? idx('Estado') : 13;
    const iTipo   = idx('Fue Prorrateado') !== -1 ? idx('Fue Prorrateado') : 14;
    const iDet    = idx('Comentario Disputa') !== -1 ? idx('Comentario Disputa') : 15;
    const iHoraHC = idx('Horas Semanales HC') !== -1 ? idx('Horas Semanales HC') : 16;

    const timeline = [];
    let nombreAgente = '', lobAgente = '', role = '', roll = '';
    let supervisor = '', accm = '', om = '';
    let horasSemanales = 42;

    for (let r = 1; r < dataAud.length; r++) {
      const row = dataAud[r];
      if (String(row[iBMS]).trim() !== String(bmsId).trim()) continue;

      // Capturar datos de la primera coincidencia
      if (!nombreAgente) {
        nombreAgente   = row[iNombre] || '';
        lobAgente      = row[iLOB]    || '';
        role           = row[iRole]   || '';
        roll           = row[iRoll]   || '';
        supervisor     = row[iSup]    || '';
        accm           = row[iACCM]   || '';
        om             = row[iOM]     || '';
        horasSemanales = parseFloat(row[iHoraHC]) || 42;
      }

      // Construir item de timeline
      const fechaTs = row[iFecha] instanceof Date
        ? row[iFecha].toISOString()
        : String(row[iFecha] || '');

      // ✅ FIX: usar normalizarNombreMes para que mesCDM siempre sea texto
      const mesCDM = iMes !== -1 ? normalizarNombreMes(row[iMes]) : '';

      timeline.push({
        tipoAccion : row[iEst] || 'Aprobación',
        timestamp  : fechaTs,
        lob        : String(row[iLOB]  || '').trim(),
        monto      : parseFloat(row[iBono]) || 0,
        mesCDM     : mesCDM,
        detalles   : row[iDet]    || '',
        prorrateado: row[iTipo]   || 'No'
      });
    }

    // Completar con H_C si el agente no tiene historial en Audit
    if (!nombreAgente && infoHC) {
      nombreAgente   = String(infoHC[2] || '').trim();
      lobAgente      = String(infoHC[1] || '').trim();
      role           = String(infoHC[3] || '').trim();
      supervisor     = String(infoHC[4] || '').trim();
      accm           = String(infoHC[5] || '').trim();
      om             = String(infoHC[6] || '').trim();
      roll           = role;
      horasSemanales = parseFloat(infoHC[9]) || 42;
    }

    // Ordenar timeline por fecha descendente
    timeline.sort((a, b) => {
      const da = a.timestamp ? new Date(a.timestamp) : 0;
      const db = b.timestamp ? new Date(b.timestamp) : 0;
      return db - da;
    });

    // ✅ FIX PRINCIPAL: obtener historial de disputas del agente
    let historialDisputas = [];
    try {
      historialDisputas = obtenerHistorialDisputasAgente(bmsId);
    } catch(e) {
      Logger.log('⚠️ obtenerHistorialDisputasAgente falló: ' + e.toString());
    }

    Logger.log('✅ Detalles OK | timeline: ' + timeline.length + ' | disputas: ' + historialDisputas.length);

    return {
      bmsId,
      nombreAgente,
      lob            : lobAgente,
      role,
      roll,
      supervisor,
      accm,
      om,
      horasSemanales,
      timeline       : timeline.slice(0, 20),   // máximo 20 entradas
      historialDisputas,                         // ✅ NUEVO — disputas del agente con respondida/decision/respuesta
      kpiDetalle: obtenerKPIsAgente(bmsId)      // ✅ AGREGADO: KPIs del agente
    };

  } catch (e) {
    Logger.log('❌ obtenerDetallesCompletoAgente: ' + e.message);
    return { error: e.message };
  }
}

function validarAprobacionPrevia(lob, fecha, mes) {
  try {
    Logger.log('=== validarAprobacionPrevia v3 ===');
    Logger.log('LOB: ' + lob + ' | Fecha: ' + fecha + ' | Mes: ' + mes);

    const ss = SpreadsheetApp.openById(ID_SS_PRINCIPAL);
    const hojaAudit = ss.getSheetByName('Audit');

    if (!hojaAudit || hojaAudit.getLastRow() < 2) {
      return { yaAprobado: false };
    }

    const data    = hojaAudit.getDataRange().getValues();
    const headers = data[0];

    // Detectar índices de columnas — compatible con 15 o 22 cols
    const idxLOB      = headers.indexOf('LOB');
    const idxBono     = headers.indexOf('Bono Aprobado');
    const idxFechaApr = headers.indexOf('Fecha Aprobación');
    const idxUsuario  = headers.indexOf('Usuario Aprobador');
    const idxFechaKPI = headers.indexOf('Fecha KPI');
    const idxMesCDM   = headers.indexOf('Mes CDM');

    const norm = function(s) {
      return s ? s.toString().trim().toLowerCase() : '';
    };
    const normFecha = function(d) {
      if (!d) return '';
      if (d instanceof Date) {
        return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      }
      const str = d.toString().trim();
      if (str.includes('-')) return str.substring(0, 10);
      if (str.includes('/')) {
        const p = str.split('/');
        if (p.length === 3) {
          return p[0].length === 4
            ? p[0] + '-' + p[1].padStart(2,'0') + '-' + p[2].padStart(2,'0')
            : p[2] + '-' + p[1].padStart(2,'0') + '-' + p[0].padStart(2,'0');
        }
      }
      return str;
    };

    const mesesNombres = {
      1:'enero', 2:'febrero', 3:'marzo', 4:'abril', 5:'mayo', 6:'junio',
      7:'julio', 8:'agosto', 9:'septiembre', 10:'octubre', 11:'noviembre', 12:'diciembre'
    };

    const lobB   = norm(lob);
    const fechaB = normFecha(fecha);
    let mesB = typeof mes === 'number' ? (mesesNombres[mes] || '') : norm(mes);

    // ── Sumar total y contar agentes desde Audit ──
    let totalLOB       = 0;
    let primeraFila    = null;
    let primeraFilaNum = -1;
    let agentesAudit   = 0;

    for (let i = 1; i < data.length; i++) {
      const row       = data[i];
      const lobFila   = norm(row[idxLOB]);
      const fechaFila = normFecha(row[idxFechaKPI]);
      const mesFila   = norm(row[idxMesCDM]);

      if (lobFila === lobB &&
          (fechaB  ? fechaFila === fechaB : true) &&
          (mesB    ? mesFila   === mesB   : true)) {
        totalLOB += parseFloat(row[idxBono]) || 0;
        agentesAudit++;
        if (!primeraFila) { primeraFila = row; primeraFilaNum = i + 1; }
      }
    }

    if (!primeraFila) {
      Logger.log('No hay aprobación previa para esta LOB/Mes');
      return { yaAprobado: false };
    }

    // ── Intentar leer agentes de Historico_MAX columna H ──
    // Historico_MAX estructura asumida:
    //   Col A: LOB  | Col B: Mes | Col C: Fecha | ... | Col G: Total($) | Col H: Agentes(#)
    let agentesHistorico = null;
    try {
      const hojaHist = ss.getSheetByName('Historico_MAX');
      if (hojaHist && hojaHist.getLastRow() > 1) {
        const dataHist = hojaHist.getDataRange().getValues();
        // Encabezados en fila 1
        const headersHist = dataHist[0].map(function(h) { return norm(h); });
        Logger.log('Headers Historico_MAX: ' + JSON.stringify(headersHist));

        // Intentar encontrar columna LOB y Mes en Historico_MAX
        // Si no hay encabezados reconocibles, usamos col 0=LOB, 1=Mes, 6=Total, 7=Agentes (0-indexed)
        let colLobH  = headersHist.indexOf('lob');
        let colMesH  = headersHist.indexOf('mes');
        let colAgH   = headersHist.indexOf('agentes');
        if (colLobH  === -1) colLobH  = 0;
        if (colMesH  === -1) colMesH  = 1;
        if (colAgH   === -1) colAgH   = 7; // columna H = índice 7

        for (let j = 1; j < dataHist.length; j++) {
          const rh = dataHist[j];
          if (norm(rh[colLobH]) === lobB) {
            const mesHist = norm(rh[colMesH]);
            if (!mesB || mesHist === mesB || mesHist.includes(mesB) || mesB.includes(mesHist)) {
              agentesHistorico = rh[colAgH];
              Logger.log('✅ Agentes de Historico_MAX: ' + agentesHistorico);
              break;
            }
          }
        }
      }
    } catch(eH) {
      Logger.log('⚠️ No se pudo leer Historico_MAX: ' + eH.message);
    }

    // Usar Historico_MAX si encontramos el dato, si no usar conteo de Audit
    const agentesFinales = (agentesHistorico !== null && agentesHistorico !== '')
      ? agentesHistorico
      : agentesAudit;

    const fechaAprobStr = primeraFila[idxFechaApr] instanceof Date
      ? Utilities.formatDate(primeraFila[idxFechaApr], Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm:ss')
      : String(primeraFila[idxFechaApr]);

    Logger.log('✅ LOB ya aprobada | Total: $' + totalLOB + ' | Agentes: ' + agentesFinales);

    return {
      yaAprobado: true,
      detalles: {
        fila:           primeraFilaNum,
        lob:            primeraFila[idxLOB],
        fecha:          primeraFila[idxFechaKPI],
        mes:            primeraFila[idxMesCDM],
        total:          '$' + totalLOB.toLocaleString('es-CO', { minimumFractionDigits: 2 }),
        totalNumerico:  totalLOB,
        agentes:        agentesFinales,
        fechaAprobacion: fechaAprobStr,
        usuario:        primeraFila[idxUsuario]
      }
    };

  } catch (e) {
    Logger.log('❌ ERROR validarAprobacionPrevia: ' + e.toString());
    return { yaAprobado: false, error: e.toString() };
  }
}

/**
 * Registra el pago de Additional Incentives en la hoja financiera
 * Se llama desde guardarAprobaciones() cuando hay bonos de categoría "additionalIncentives"
 */
function registrarFinancieroAdditional(datosAprobacion) {
  try {
    const ss = SpreadsheetApp.openById(SS_ID); // ✅ CORREGIDO
    const sheet = ss.getSheetByName('Financiero_Additional_Incentives');
    const sheetUsuario = ss.getSheetByName('Usuario');
    
    if (!sheet) {
      Logger.log('Hoja Financiero_Additional_Incentives no encontrada');
      return { success: false, error: 'Hoja no encontrada' };
    }
    
    let salarioBase = 0;
    if (sheetUsuario) {
      const dataUsuario = sheetUsuario.getDataRange().getValues();
      const headersUsuario = dataUsuario[0];
      const colBMSUsuario = headersUsuario.indexOf('BMS ID');
      const colSalarioUsuario = headersUsuario.indexOf('Salario');
      
      for (let i = 1; i < dataUsuario.length; i++) {
        if (dataUsuario[i][colBMSUsuario] == datosAprobacion.bmsId) {
          salarioBase = parseFloat(dataUsuario[i][colSalarioUsuario]) || 0;
          break;
        }
      }
    }
    
    if (salarioBase === 0) {
      Logger.log('Salario base no encontrado para BMS: ' + datosAprobacion.bmsId);
    }
    
    const limite30 = salarioBase * 0.30;
    const totalAdditional = parseFloat(datosAprobacion.monto) || 0;
    const porcentajeUsado = salarioBase > 0 ? ((totalAdditional / salarioBase) * 100).toFixed(2) : 0;
    const diferencia = limite30 - totalAdditional;
    
    let estado = '✅ OK';
    if (porcentajeUsado >= 30) {
      estado = '🚨 EXCEDE LÍMITE';
    } else if (porcentajeUsado >= 25) {
      estado = '⚠️ CERCA DEL LÍMITE';
    }
    
    const nuevaFila = [
      new Date(),
      datosAprobacion.mesCDM || '',
      datosAprobacion.lob || '',
      datosAprobacion.bmsId || '',
      datosAprobacion.nombreAgente || '',
      salarioBase,
      totalAdditional,
      porcentajeUsado,
      limite30,
      diferencia,
      estado,
      datosAprobacion.desglose || '',
      datosAprobacion.usuario || Session.getActiveUser().getEmail(),
      new Date()
    ];
    
    sheet.appendRow(nuevaFila);
    
    const ultimaFila = sheet.getLastRow();
    let colorFondo = '#d4edda';
    if (estado.includes('EXCEDE')) {
      colorFondo = '#f8d7da';
    } else if (estado.includes('CERCA')) {
      colorFondo = '#fff3cd';
    }
    
    sheet.getRange(ultimaFila, 11).setBackground(colorFondo);
    
    return { success: true, estado: estado };
    
  } catch (error) {
    Logger.log('Error en registrarFinancieroAdditional: ' + error.toString());
    return { success: false, error: error.toString() };
  }
}

        if (!bms) continue;
        
        // Aplicar filtro BMS si existe
        if (pasaFiltroBms && !pasaFiltroBms(bms)) continue;

        mapaHC[bms] = {
          nombre:      String(dataHC[i][cNombre] || '').trim(),
          lob:         String(dataHC[i][cLOB]    || '').trim(),
          horasSemana: parseFloat(dataHC[i][cHoras]) || 42
        };
      }
      console.log('H_C mapa: ' + Object.keys(mapaHC).length + ' agentes');
    }

    // ── Buscar hojas HORAS_* en CDM ──
    const hojasHoras = ssCDM.getSheets().filter(function(h) {
      const n = h.getName().toUpperCase();
      return n.startsWith('HORAS_') || n.startsWith('HORAS ');
    });

    console.log('Hojas HORAS encontradas: ' + hojasHoras.length);
    if (!hojasHoras.length) return [];

    const result = [];

    hojasHoras.forEach(function(hoja) {
      const nombreHoja = hoja.getName();
      const mesCDM     = nombreHoja.replace(/^HORAS[_ ]/i, ''); // e.g. "2"
      if (hoja.getLastRow() <= 1) return;

      const data    = hoja.getDataRange().getValues();
      const headers = data[0];

      // Detectar columnas
      const col = function(buscar) {
        return headers.findIndex(function(h) {
          const hn = String(h).toLowerCase().trim().replace(/\s+/g, '');
          const bn = buscar.toLowerCase().replace(/\s+/g, '');
          return hn === bn || hn.includes(bn) || bn.includes(hn);
        });
      };

      // Igual que OM_obtenerHorasPorAgente: buscar por nombre exacto primero
      let colCCMS      = -1;
      let colHoras     = -1;
      let colAusencias = -1;
      let colDias      = 6; // Columna G = índice 6 (A=0...G=6) ← FIJO igual que en Código 1
      let colLOB       = -1;

      headers.forEach(function(h, i) {
        const hn = String(h).trim();
        if (hn === 'CCMSID' || hn === 'BMS ID' || hn === 'ID') colCCMS      = i;
        if (hn === 'horas'  || hn === 'Horas')                  colHoras     = i;
        if (hn === 'Ausencias' || hn === 'ausencias')           colAusencias = i;
        if (hn === 'SubLOB' || hn === 'LOB')                    colLOB       = i;
      });

      // Si no encontró por nombre exacto, fallback dinámico
      if (colCCMS  === -1) colCCMS  = col('bms');
      if (colHoras === -1) colHoras = col('horas');
      if (colAusencias === -1) colAusencias = col('ausencias');

      if (colCCMS === -1 || colHoras === -1) {
        console.log('⚠️ ' + nombreHoja + ': columnas CCMS/Horas no encontradas');
        return;
      }

      console.log(nombreHoja + ' → CCMS=' + colCCMS + ' Hrs=' + colHoras + ' Aus=' + colAusencias + ' Días(G)=' + colDias);

        for (let i = 1; i < data.length; i++) {
          const fila          = data[i];
          const ccmsCompleto  = String(fila[colCCMS] || '').trim();
          if (!ccmsCompleto) continue;

          // Extraer BMS ID (7 dígitos)
          const matchBMS = ccmsCompleto.match(/(\d{7})/);
          const bmsId    = matchBMS ? matchBMS[1] : null;
          if (!bmsId) continue;
          
          // Aplicar filtro BMS
          if (pasaFiltroBms && !pasaFiltroBms(bmsId)) continue;

        const horasTrabajadas = parseFloat(fila[colHoras])     || 0;
        const ausencias       = colAusencias !== -1 ? (parseFloat(fila[colAusencias]) || 0) : 0;
        const diasLaborados   = parseFloat(fila[colDias])       || 0; // Col G fija
        const lobHoja         = colLOB !== -1 ? String(fila[colLOB] || '').trim() : '';

        const datosHC         = mapaHC[bmsId] || { nombre: '', lob: lobHoja, horasSemana: 42 };
        const horasEsperadas  = +(datosHC.horasSemana / 5 * 22).toFixed(1);
        const cumplimiento    = horasEsperadas > 0 ? +((horasTrabajadas / horasEsperadas) * 100).toFixed(1) : 0;
        const aplicaProrrateo = diasLaborados > 0 && diasLaborados < 20;
        const factorProrrateo = diasLaborados > 0 ? +(diasLaborados / 22).toFixed(4) : 1;

        result.push({
          bmsId,
          mesCDM,
          timestamp:          new Date().toISOString(),
          nombreAgente:       datosHC.nombre || String(fila[1] || '').trim(),
          lob:                datosHC.lob    || lobHoja,
          horasSemanaHC:      datosHC.horasSemana,
          horasEsperadasMes:  horasEsperadas,
          horasTrabajadas,
          diasTrabajados:     diasLaborados,
          ausencias,
          cumplimientoHrs:    cumplimiento,
          aplicaProrrateo,
          factorProrrateo,
          filaHoja:           i + 1
        });
      }
    });

    console.log('✅ Total horas: ' + result.length);
    return result;

  } catch (e) {
    console.log('❌ obtenerDatosHoras: ' + e.toString());
    return [];
  }
}

function obtenerSalariosAudit() {
  try {
    var ss       = SpreadsheetApp.openById(SS_ID);
    var hojaConf = ss.getSheetByName('Configuracion');
    var hojaUsr  = ss.getSheetByName('Usuario');
    var resultado = {};

    // Salarios individuales desde hoja Usuario
    if (hojaUsr) {
      var dataUsr = hojaUsr.getDataRange().getValues();
      for (var i = 1; i < dataUsr.length; i++) {
        var bms = String(dataUsr[i][0] || '').trim();
        var sal = parseFloat(String(dataUsr[i][7] || '0').replace(/\./g,'').replace(',','.')) || 0;
        if (bms && sal > 0) resultado[bms] = sal;
      }
    }

    var lobSalarios = {};
    var lobPctBono  = {};

    // Salario_Base y %_Bono por LOB desde Configuracion
    // Columnas: Mes(0) OM(1) LOB(2) Salario_Base(3) %_Bono(4)
    if (hojaConf) {
      var dataConf = hojaConf.getDataRange().getValues();
      var headers  = dataConf[0].map(function(h){ return String(h).toLowerCase().trim(); });
      var iLOB = 2, iSal = 3, iPct = 4;
      headers.forEach(function(h,i){
        if (h === 'lob') iLOB = i;
        if (h.indexOf('salario') !== -1) iSal = i;
        if (h.indexOf('%') !== -1 || h.indexOf('bono') !== -1) iPct = i;
      });
      for (var j = 1; j < dataConf.length; j++) {
        var lob = String(dataConf[j][iLOB] || '').trim();
        var sal = parseFloat(String(dataConf[j][iSal] || '0').replace(/\./g,'').replace(',','.')) || 0;
        var pct = parseFloat(String(dataConf[j][iPct] || '30').replace(/[^\d,]/g,'').replace(',','.')) || 30;
        if (lob) {
          if (sal > 0) lobSalarios[lob] = sal;
          lobPctBono[lob] = pct;
        }
      }
    }

    resultado._lobSalarios = lobSalarios;
    resultado._lobPctBono  = lobPctBono;
    return resultado;

  } catch(e) {
    Logger.log('Error obtenerSalariosAudit: ' + e.message);
    return {};
  }
}






function normalizarNombreMes(mes) {
  if (!mes) return '';
  
  const meses = [
    'enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio',
    'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'
  ];
  
  // Si es número
  if (typeof mes === 'number') {
    return meses[mes - 1] || '';
  }
  
  // Si es Date
  if (mes instanceof Date) {
    return meses[mes.getMonth()];
  }
  
  // Si es string numérico
  const mesNum = parseInt(mes);
  if (!isNaN(mesNum) && mesNum >= 1 && mesNum <= 12) {
    return meses[mesNum - 1];
  }
  
  // Si ya es nombre de mes
  const mesNorm = String(mes).toLowerCase().trim();
  if (meses.includes(mesNorm)) {
    return mesNorm;
  }
  
  // Buscar parcialmente
  for (let i = 0; i < meses.length; i++) {
    if (mesNorm.includes(meses[i]) || meses[i].includes(mesNorm)) {
      return meses[i];
    }
  }
  
  return mes; // Devolver original si no se puede normalizar
}

function obtenerHistorialAuditoria(bmsId) {
  try {
    const ss      = SpreadsheetApp.openById(SS_ID);
    const historial = [];
    const bmsIdStr  = String(bmsId).trim();

    // ── 1. Aprobaciones / Traslados / Prorrateos desde hoja Audit ──
    const hojaAudit = ss.getSheetByName('Audit');
    if (hojaAudit && hojaAudit.getLastRow() > 1) {
      const dataAudit = hojaAudit.getDataRange().getValues();
      const headAudit = dataAudit[0].map(h => String(h).toLowerCase().trim());

      const iA = {
        bms:       headAudit.findIndex(h => h.includes('bms')),
        nombre:    headAudit.findIndex(h => h.includes('nombre')),
        lob:       headAudit.findIndex(h => h === 'lob'),
        fecha:     headAudit.findIndex(h => h.includes('fecha') || h.includes('timestamp')),
        tipo:      headAudit.findIndex(h => h.includes('tipo') || h.includes('accion')),
        monto:     headAudit.findIndex(h => h.includes('monto') || h.includes('bono')),
        mes:       headAudit.findIndex(h => h === 'mes cdm' || h === 'mes'),
        estado:    headAudit.findIndex(h => h.includes('estado')),
        usuario:   headAudit.findIndex(h => h.includes('usuario') || h.includes('registrado')),
        detalles:  headAudit.findIndex(h => h.includes('detalle') || h.includes('nota'))
      };

      for (let i = 1; i < dataAudit.length; i++) {
        const row = dataAudit[i];
        if (String(row[iA.bms !== -1 ? iA.bms : 3] || '').trim() !== bmsIdStr) continue;

        const tipoAccion = String(row[iA.tipo !== -1 ? iA.tipo : 1] || '').trim();
        const monto      = parseFloat(row[iA.monto !== -1 ? iA.monto : 5]) || 0;
        const fecha      = row[iA.fecha !== -1 ? iA.fecha : 0];

        let icono = '📌';
        if (tipoAccion.includes('probaci') || tipoAccion.includes('Aprobaci')) icono = '✅';
        else if (tipoAccion.includes('Traslado') || tipoAccion.includes('traslado'))   icono = '🔄';
        else if (tipoAccion.includes('Prorrateo') || tipoAccion.includes('prorrateo')) icono = '📊';
        else if (tipoAccion.includes('Disputa')   || tipoAccion.includes('disputa'))   icono = '⚠️';

        historial.push({
          tipo      : tipoAccion || 'Registro',
          tipoAccion: tipoAccion,
          icono,
          titulo    : tipoAccion + ' — ' + String(row[iA.lob !== -1 ? iA.lob : 2] || ''),
          estado    : String(row[iA.estado !== -1 ? iA.estado : 11] || ''),
          monto,
          fecha     : fecha instanceof Date ? fecha.toISOString() : String(fecha || ''),
          usuario   : String(row[iA.usuario  !== -1 ? iA.usuario  : 10] || ''),
          lob       : String(row[iA.lob      !== -1 ? iA.lob      : 2]  || ''),
          mesCDM    : normalizarNombreMes(row[iA.mes !== -1 ? iA.mes : 8]),
          detalle   : String(row[iA.detalles !== -1 ? iA.detalles : 12] || ''),
          fila      : i + 1,
          fuente    : 'Audit'
        });
      }
    }

    // ── 2a. Respuestas desde Disputa_Respuestas (fuente principal) ──
    const hojaResp = ss.getSheetByName('Disputa_Respuestas');

    if (hojaResp && hojaResp.getLastRow() > 1) {
      const dataResp = hojaResp.getDataRange().getValues();
      const headResp = dataResp[0].map(h => String(h).toLowerCase().trim());

      const iBMSr    = headResp.findIndex(h => h.includes('bms'));
      const iDecr    = headResp.findIndex(h => h.includes('decisi'));
      const iResp    = headResp.findIndex(h => h === 'respuesta');
      const iMontR   = headResp.findIndex(h => h.includes('monto'));
      const iFechaR  = headResp.findIndex(h => h.includes('fecha respuesta'));
      const iUsuR    = headResp.findIndex(h => h.includes('usuario'));
      const iComent  = headResp.findIndex(h => h.includes('comentario'));
      const iEstNvo  = headResp.findIndex(h => h.includes('estado nuevo'));

      for (let i = 1; i < dataResp.length; i++) {
        const row = dataResp[i];
        if (String(row[iBMSr !== -1 ? iBMSr : 1] || '').trim() !== bmsIdStr) continue;

        const decision   = String(row[iDecr !== -1 ? iDecr : 8] || '').trim();
        const esAprobado = decision.includes('APROBADA') || decision.includes('✅');
        const fechaR     = row[iFechaR !== -1 ? iFechaR : 11];

        historial.push({
          tipo      : 'DISPUTA',
          icono     : esAprobado ? '✅' : (decision.includes('RECHAZADA') ? '❌' : '📋'),
          titulo    : 'Respuesta Analyst: ' + decision,
          estado    : String(row[iEstNvo !== -1 ? iEstNvo : 6] || decision),
          monto     : parseFloat(String(row[iMontR !== -1 ? iMontR : 9] || '0').replace(/[^0-9.,\-]/g, '').replace(',', '.')) || 0,
          fecha     : fechaR instanceof Date ? fechaR.toISOString() : String(fechaR || ''),
          usuario   : String(row[iUsuR   !== -1 ? iUsuR   : 10] || ''),
          respuesta : String(row[iResp   !== -1 ? iResp   : 7]  || ''),
          comentario: String(row[iComent !== -1 ? iComent : 12] || ''),
          fila      : i + 1,
          detalle   : String(row[iResp   !== -1 ? iResp   : 7]  || ''),
          fuente    : 'Disputa_Respuestas'
        });
      }
    }

    // ── 2b. Fallback: respuestas directo de hoja Disputa cols 12-16 ──
    // Cubre disputas respondidas ANTES del Fix 1 (no guardadas en Disputa_Respuestas)
    const hojaDispFB = ss.getSheetByName('Disputa');
    if (hojaDispFB && hojaDispFB.getLastRow() > 1) {
      const dataFB = hojaDispFB.getDataRange().getValues();
      const headFB = dataFB[0].map(h => String(h).toLowerCase().trim());
      const iBMSfb = headFB.findIndex(h => h.includes('bms'));
      const iEstFb = headFB.findIndex(h => h.includes('estado'));

      // Solo filas que NO están ya en historial (evitar duplicados)
      const filasYaAgregadas = new Set(historial.map(h => h.fila));

      for (let i = 1; i < dataFB.length; i++) {
        if (String(dataFB[i][iBMSfb !== -1 ? iBMSfb : 1] || '').trim() !== bmsIdStr) continue;

        // responderDisputa() escribe en:
        //   col 12 (idx 11) = Fecha Respuesta
        //   col 13 (idx 12) = Decisión
        //   col 14 (idx 13) = Respuesta
        //   col 15 (idx 14) = Monto Ajustado
        //   col 16 (idx 15) = Comentario Interno
        const decision  = String(dataFB[i][12] || '').trim();
        const respuesta = String(dataFB[i][13] || '').trim();

        if (!decision && !respuesta) continue;       // sin respuesta = saltar
        if (filasYaAgregadas.has(i + 1)) continue;  // ya en historial = saltar

        const esAprobado = decision.includes('APROBADA') || decision.includes('✅');
        const fechaFB    = dataFB[i][11];

        historial.push({
          tipo      : 'DISPUTA',
          icono     : esAprobado ? '✅' : (decision.includes('RECHAZADA') ? '❌' : '📋'),
          titulo    : 'Respuesta Analyst: ' + decision,
          estado    : String(dataFB[i][iEstFb !== -1 ? iEstFb : 7] || decision),
          monto     : parseFloat(dataFB[i][14]) || 0,
          fecha     : fechaFB instanceof Date ? fechaFB.toISOString() : String(fechaFB || ''),
          usuario   : String(dataFB[i][9] || ''),
          respuesta,
          comentario: String(dataFB[i][15] || ''),
          fila      : i + 1,
          detalle   : respuesta,
          fuente    : 'Disputa_directa'
        });
      }
    }

    // ── 3. Ordenar cronológicamente (más reciente primero) ──
    historial.sort((a, b) => {
      const fa = a.fecha ? new Date(a.fecha).getTime() : 0;
      const fb = b.fecha ? new Date(b.fecha).getTime() : 0;
      return fb - fa;
    });

    return historial;

  } catch(e) {
    Logger.log('❌ obtenerHistorialAuditoria: ' + e.toString());
    return [];
  }
}


function guardarApelacion(datos) {
  try {
    const ss       = SpreadsheetApp.openById(SS_ID);
    const correo   = Session.getActiveUser().getEmail();
    const ahora    = new Date();
    const bmsId    = String(datos.bmsId  || '').trim();
    const mesDatos = String(datos.mesCDM || datos.mes || '').trim().toLowerCase();
    const decision = datos.decision || '';

    // ══════════════════════════════════════════════════════
    // 0. ROL DEL USUARIO
    // ══════════════════════════════════════════════════════
    let rolUsuario = String(datos.rolUsuario || '').trim();
    if (!rolUsuario) {
      try {
        const hojaUsr = ss.getSheetByName('Usuario');
        if (hojaUsr) {
          const dataUsr = hojaUsr.getDataRange().getValues();
          for (let i = 1; i < dataUsr.length; i++) {
            if (String(dataUsr[i][2] || '').trim().toLowerCase() === correo.toLowerCase()) {
              rolUsuario = String(dataUsr[i][4] || dataUsr[i][3] || '').trim();
              break;
            }
          }
        }
      } catch(eRol) { Logger.log('⚠️ Rol no leído: ' + eRol); }
    }
    const rolNorm     = rolUsuario.toUpperCase().replace(/\s+/g, '_');
    const rolGuardado = rolNorm ? rolNorm + '_APELACION' : 'SISTEMA_APELACION';

    Logger.log('=== guardarApelacion ===');
    Logger.log('BMS: ' + bmsId + ' | Mes: ' + mesDatos + ' | Rol: ' + rolGuardado + ' | Origen: ' + (datos.origen || 'Disputa'));

    // ══════════════════════════════════════════════════════
    // 1. DETECTAR ORIGEN
    // ✅ CLAVE: solo viene de Auditoría HTML cuando origen === 'Audit'
    //           Las disputas normales de agentes NO pasan por aquí
    // ══════════════════════════════════════════════════════
    const esDesdeAudit = String(datos.origen || '').trim() === 'Audit'
                      || String(datos.origen || '').trim() === 'Apelacion';

    // ══════════════════════════════════════════════════════
    // 2. BUSCAR NOMBRE Y LOB — fallback en Audit y H_C
    // ══════════════════════════════════════════════════════
    let nombreAgente  = String(datos.nombre || datos.nombreAgente || '').trim();
    let lobAgente     = String(datos.lob    || '').trim();
    let montoAnterior = parseFloat(datos.montoAnterior) || 0;

    if (!nombreAgente || !lobAgente) {
      try {
        const hojaAudit = ss.getSheetByName('Audit');
        if (hojaAudit) {
          const dataAudit = hojaAudit.getDataRange().getValues();
          const hA = dataAudit[0];
          let idxBMS = -1, idxNom = -1, idxLOB = -1, idxMes = -1, idxBono = -1;
          for (let h = 0; h < hA.length; h++) {
            const hdr = norm(String(hA[h]));
            if (hdr.includes('bms'))                             idxBMS  = h;
            if (hdr.includes('nombre'))                          idxNom  = h;
            if (hdr === 'lob')                                   idxLOB  = h;
            if (hdr === 'mes cdm' || hdr === 'mes')              idxMes  = h;
            if (hdr.includes('bono aprobado') || hdr === 'bono') idxBono = h;
          }
          for (let i = 1; i < dataAudit.length; i++) {
            const bmsFila = String(dataAudit[i][idxBMS] || '').trim();
            const mesFila = norm(String(dataAudit[i][idxMes] || ''));
            if (bmsFila === bmsId && (!mesDatos || mesFila === mesDatos)) {
              if (!nombreAgente  && idxNom  !== -1) nombreAgente  = String(dataAudit[i][idxNom]  || '').trim();
              if (!lobAgente     && idxLOB  !== -1) lobAgente     = String(dataAudit[i][idxLOB]  || '').trim();
              if (!montoAnterior && idxBono !== -1) montoAnterior = parseFloat(dataAudit[i][idxBono]) || 0;
              break;
            }
          }
        }
      } catch(eAudit) { Logger.log('⚠️ Fallback Audit: ' + eAudit); }
    }

    // Fallback H_C
    if (!lobAgente || !nombreAgente) {
      try {
        const hojaHC = ss.getSheetByName('H_C');
        if (hojaHC) {
          const dataHC = hojaHC.getDataRange().getValues();
          const hHC    = dataHC[0].map(h => norm(String(h)));
          const iBmsHC = hHC.findIndex(h => h.includes('bms'));
          const iLobHC = hHC.findIndex(h => h === 'lob');
          const iNomHC = hHC.findIndex(h => h.includes('nombre'));
          for (let i = 1; i < dataHC.length; i++) {
            if (String(dataHC[i][iBmsHC] || '').trim() === bmsId) {
              if (!lobAgente    && iLobHC !== -1) lobAgente    = String(dataHC[i][iLobHC] || '').trim();
              if (!nombreAgente && iNomHC !== -1) nombreAgente = String(dataHC[i][iNomHC] || '').trim();
              break;
            }
          }
        }
      } catch(eHC) { Logger.log('⚠️ Fallback H_C: ' + eHC); }
    }

    Logger.log('✅ Nombre: [' + nombreAgente + '] | LOB: [' + lobAgente + '] | Monto Ant: ' + montoAnterior);

    // ══════════════════════════════════════════════════════
    // 3. ID ÚNICO
    // ══════════════════════════════════════════════════════
    const id = String(datos.idApelacion || '').trim() || ('APE-' + Utilities.getUuid().substring(0, 8).toUpperCase());

    // ══════════════════════════════════════════════════════
    // 4. ESTADO según origen
    // ✅ CLAVE:
    //   - Viene de Auditoría HTML (OM ajusta KPI) → "⏳ Pendiente Reporting"
    //   - Viene de disputa normal de agente       → usar la decisión real
    // ══════════════════════════════════════════════════════
    let estadoLabel, bgFila, solicitudOM;

    if (esDesdeAudit) {
      // OM/ACCM envía desde la hoja de Auditoría → siempre pendiente
      solicitudOM = decision === 'APROBADA' ? 'Solicita Ajuste/Apelación'
                  : decision === 'MAS_INFO' ? 'Agrega Información'
                  : 'Solicitud';
      estadoLabel = '⏳ Pendiente Reporting';
      bgFila      = '#fef3c7';
    } else {
      // Disputa normal de agente → usar la decisión enviada
      solicitudOM = decision;
      estadoLabel = decision === 'APROBADA'   ? '✅ Aprobada'
                  : decision === 'RECHAZADA'  ? '❌ Rechazada'
                  : decision === 'EN_PROCESO' ? '⏳ En Proceso'
                  : decision === 'MAS_INFO'   ? '📋 Más Info'
                  : '⏳ Pendiente';
      bgFila      = decision === 'APROBADA'   ? '#dcfce7'
                  : decision === 'RECHAZADA'  ? '#fee2e2'
                  : decision === 'EN_PROCESO' ? '#eff6ff'
                  : '#fef3c7';
    }

    Logger.log('Estado: ' + estadoLabel + ' | esDesdeAudit: ' + esDesdeAudit);

    // ══════════════════════════════════════════════════════
    // 5. GUARDAR EN Disputa_Apelaciones
    // ══════════════════════════════════════════════════════
    const HEADERS_DA = [
      'ID', 'BMS ID', 'Nombre', 'LOB', 'Mes CDM',
      'Decisión', 'Monto Anterior', 'Monto Ajustado', 'Respuesta',
      'Estado', 'Usuario', 'Fecha', 'Comentario Interno', 'KPIs Detalle',
      'Correo', 'Origen', 'Rol'
    ];

    let hojaDA = ss.getSheetByName('Disputa_Apelaciones');
    if (!hojaDA) hojaDA = ss.insertSheet('Disputa_Apelaciones');

    const celdaA1 = hojaDA.getRange(1, 1).getValue();
    if (!celdaA1 || String(celdaA1).trim() === '') {
      hojaDA.appendRow(HEADERS_DA);
      hojaDA.getRange(1, 1, 1, HEADERS_DA.length)
        .setBackground('#7c3aed').setFontColor('#FFF').setFontWeight('bold');
      hojaDA.setFrozenRows(1);
    }

    // Formato legible KPIs
    let kpisDetalle = '';
    if (datos.ajustesKPI && datos.ajustesKPI.length > 0) {
      kpisDetalle = datos.ajustesKPI.map(function(k) {
        const ant    = parseFloat(k.valorAnterior) || 0;
        const nvo    = parseFloat(k.valor)         || 0;
        const delta  = nvo - ant;
        const flecha = delta > 0 ? '▲' : delta < 0 ? '▼' : '=';
        return '[' + k.categoria + '] ' + k.kpi + ': ' + ant.toFixed(4) + ' ' + flecha + ' ' + nvo.toFixed(4);
      }).join('\n');
    }

    hojaDA.appendRow([
      id,                                      // 0  : ID
      bmsId,                                   // 1  : BMS ID
      nombreAgente,                            // 2  : Nombre
      lobAgente,                               // 3  : LOB
      datos.mesCDM || datos.mes || '',         // 4  : Mes CDM
      solicitudOM,                             // 5  : Decisión / Solicitud OM
      montoAnterior,                           // 6  : Monto Anterior
      parseFloat(datos.montoAjustado) || 0,    // 7  : Monto Ajustado propuesto
      datos.respuesta || '',                   // 8  : Justificación del OM
      estadoLabel,                             // 9  : Estado ← correcto según origen
      rolGuardado,                             // 10 : Rol
      ahora,                                   // 11 : Fecha
      datos.comentarioInterno || '',           // 12 : Comentario Interno
      kpisDetalle,                             // 13 : KPIs Detalle
      correo,                                  // 14 : Correo
      datos.origen || 'Disputa',               // 15 : Origen
      rolGuardado                              // 16 : Rol
    ]);

    const lr = hojaDA.getLastRow();
    hojaDA.getRange(lr, 7, 1, 2).setNumberFormat('$#,##0.00');
    hojaDA.getRange(lr, 12).setNumberFormat('dd/mm/yyyy hh:mm:ss');
    hojaDA.getRange(lr, 1, 1, HEADERS_DA.length).setBackground(bgFila);

    Logger.log('✅ Disputa_Apelaciones fila ' + lr + ' | ID: ' + id + ' | Estado: ' + estadoLabel);

    // ══════════════════════════════════════════════════════
    // 6. ACTUALIZAR HOJA AUDIT (solo si viene de Auditoría)
    // ══════════════════════════════════════════════════════
    const hojaAudit = ss.getSheetByName('Audit');
    let filaAudit   = -1;

    if (hojaAudit) {
      const dataAudit = hojaAudit.getDataRange().getValues();
      const headers   = dataAudit[0];
      let idxBMS = -1, idxMes = -1, idxEstado = -1, idxBono = -1;
      headers.forEach(function(h, i) {
        const hdr = norm(String(h));
        if (hdr.includes('bms'))                             idxBMS    = i;
        if (hdr === 'mes cdm' || hdr === 'mes')              idxMes    = i;
        if (hdr === 'estado' || hdr === 'status')            idxEstado = i;
        if (hdr.includes('bono aprobado') || hdr === 'bono') idxBono   = i;
      });

      for (let i = 1; i < dataAudit.length; i++) {
        const bmsFila = String(dataAudit[i][idxBMS] || '').trim();
        const mesFila = norm(String(dataAudit[i][idxMes] || ''));
        if (bmsFila === bmsId && (!mesDatos || mesFila === mesDatos)) {
          filaAudit = i + 1;
          break;
        }
      }

      if (filaAudit > 0 && esDesdeAudit) {
        // Solo marcar "En Apelación" cuando viene de Auditoría
        if (idxEstado >= 0) {
          hojaAudit.getRange(filaAudit, idxEstado + 1)
            .setValue('🔄 En Apelación')
            .setBackground('#fef3c7')
            .setFontWeight('bold');
        }

        // Columna Apelación
        const hActuales = hojaAudit.getRange(1, 1, 1, hojaAudit.getLastColumn()).getValues()[0];
        let colApe = hActuales.findIndex(h => norm(String(h)) === 'apelacion');
        if (colApe === -1) {
          colApe = hojaAudit.getLastColumn();
          hojaAudit.getRange(1, colApe + 1)
            .setValue('Apelación')
            .setBackground('#4f46e5').setFontColor('#fff').setFontWeight('bold');
          colApe = colApe + 1;
        } else {
          colApe = colApe + 1;
        }
        const fechaStr = Utilities.formatDate(ahora, Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm');
        hojaAudit.getRange(filaAudit, colApe)
          .setValue('⏳ PENDIENTE — ' + id + ' — ' + rolGuardado + ' (' + fechaStr + ')');
        hojaAudit.getRange(filaAudit, 1, 1, hojaAudit.getLastColumn()).setBackground(bgFila);
      }
    }

    // ══════════════════════════════════════════════════════
    // 7. APELACIONES_KPI
    // ══════════════════════════════════════════════════════
    if (datos.ajustesKPI && datos.ajustesKPI.length > 0) {
      let hojaKPI = ss.getSheetByName('Apelaciones_KPI');
      if (!hojaKPI) hojaKPI = ss.insertSheet('Apelaciones_KPI');
      const a1KPI = hojaKPI.getRange(1, 1).getValue();
      if (!a1KPI || String(a1KPI).trim() === '') {
        hojaKPI.appendRow([
          'ID Apelación', 'Fecha', 'BMS ID', 'Nombre', 'LOB', 'Mes CDM',
          'Categoría', 'KPI', 'Valor Anterior', 'Valor Nuevo', 'Δ Diferencia',
          'Solicitud OM', 'Usuario', 'Correo', 'Rol'
        ]);
        hojaKPI.getRange(1, 1, 1, 15)
          .setBackground('#1e3a5f').setFontColor('#fff').setFontWeight('bold');
        hojaKPI.setFrozenRows(1);
      }
      datos.ajustesKPI.forEach(function(k) {
        const ant   = parseFloat(k.valorAnterior) || 0;
        const nvo   = parseFloat(k.valor)         || 0;
        const delta = nvo - ant;
        hojaKPI.appendRow([
          id, ahora, bmsId, nombreAgente, lobAgente,
          datos.mesCDM || '', k.categoria || '', k.kpi || '',
          ant, nvo, delta,
          solicitudOM, rolGuardado, correo, rolGuardado
        ]);
        const lrk = hojaKPI.getLastRow();
        hojaKPI.getRange(lrk, 9, 1, 3).setNumberFormat('0.0000');
        const bgD = delta > 0 ? '#dcfce7' : delta < 0 ? '#fee2e2' : '#f5f5f5';
        hojaKPI.getRange(lrk, 11).setBackground(bgD).setFontWeight('bold');
      });
    }

    // ══════════════════════════════════════════════════════
    // 8. LOG EN Disputa_Respuestas
    // ══════════════════════════════════════════════════════
    try {
      guardarRespuestaEnHoja({
        fila:              filaAudit,
        bmsId:             bmsId,
        nombre:            nombreAgente,
        lob:               lobAgente,
        fechaDisputa:      ahora,
        estadoOriginal:    esDesdeAudit ? 'Auditoría' : 'Disputa',
        respuesta:         datos.respuesta || '',
        decision:          solicitudOM,
        montoAjustado:     parseFloat(datos.montoAjustado) || 0,
        comentarioInterno: datos.comentarioInterno || '',
        esAjusteOM:        esDesdeAudit,
        rolUsuario:        rolGuardado
      }, estadoLabel);
    } catch(eLog) { Logger.log('⚠️ Log Disputa_Respuestas: ' + eLog); }

    Logger.log('✅ guardarApelacion OK | ID: ' + id + ' | Fila Audit: ' + filaAudit + ' | Rol: ' + rolGuardado + ' | Estado: ' + estadoLabel);

    return {
      success : true,
      id      : id,
      fila    : filaAudit,
      mensaje : (esDesdeAudit ? 'Apelación pendiente de Reporting' : 'Apelación registrada') +
                ' | ID: ' + id + ' | Nombre: ' + nombreAgente + ' | LOB: ' + lobAgente
    };

  } catch(e) {
    Logger.log('❌ guardarApelacion: ' + e.message + '\n' + e.stack);
    return { success: false, error: e.message };
  }
}

/**
 * Función unificada para el panel de Disputas del HTML.
 * Devuelve disputas + apelaciones en un solo array normalizado,
 * con el campo 'origenTipo' para que el frontend los distinga.
 */
/**
 * Retorna disputas + apelaciones para el panel HTML.
 * Incluye campo respondidoPor para que el frontend distinga
 * si Reporting ya respondió o no.
 */



function obtenerDisputasParaHTML() {
  try {
    const ss     = SpreadsheetApp.openById(SS_ID);
    const correo = Session.getActiveUser().getEmail();

    // ── Permisos del usuario ──────────────────────────────
    let esAdmin        = false;
    let lobsPermitidas = [];

    const hojaUsr = ss.getSheetByName('Usuario');
    if (hojaUsr) {
      const dataUsr = hojaUsr.getDataRange().getValues();
      for (let i = 1; i < dataUsr.length; i++) {
        if (String(dataUsr[i][2] || '').trim().toLowerCase() === correo.toLowerCase()) {
          const rolUsr = String(dataUsr[i][4] || dataUsr[i][3] || '').toLowerCase().trim();
          esAdmin = rolUsr.includes('audit') || rolUsr.includes('admin') || rolUsr.includes('reporting');
          const lobsRaw = String(dataUsr[i][5] || '').trim();
          if (lobsRaw) lobsPermitidas = lobsRaw.split(',').map(l => l.trim().toLowerCase());
          break;
        }
      }
    }

    function pasaLob(lob) {
      if (esAdmin || lobsPermitidas.length === 0) return true;
      return lobsPermitidas.includes(norm(lob));
    }

    const resultado = [];

    // ════════════════════════════════════════════════════════
    // PRE-CARGA: detectar qué disputas normales ya tienen
    // apelación registrada en Disputa_Apelaciones
    // (Origen = 'Disputa' → el agente apeló la respuesta)
    // ════════════════════════════════════════════════════════
    const bmsYMesApelados = new Set();
    const hojaApeCheck = ss.getSheetByName('Disputa_Apelaciones');
    if (hojaApeCheck && hojaApeCheck.getLastRow() > 1) {
      const dataApeCheck = hojaApeCheck.getDataRange().getValues();
      const hAC      = dataApeCheck[0].map(h => norm(String(h)));
      const acBMS    = hAC.findIndex(h => h.includes('bms'));
      const acMes    = hAC.findIndex(h => h.includes('mes'));
      const acOrigen = hAC.findIndex(h => h === 'origen');
      for (let r = 1; r < dataApeCheck.length; r++) {
        const origenFila = String(dataApeCheck[r][acOrigen !== -1 ? acOrigen : 15] || '').trim();
        // Solo apelaciones que nacieron de una disputa normal de agente
        if (origenFila !== 'Disputa') continue;
        const bmsF = String(dataApeCheck[r][acBMS !== -1 ? acBMS : 1] || '').trim();
        const mesF = norm(String(dataApeCheck[r][acMes !== -1 ? acMes : 4] || ''));
        if (bmsF) bmsYMesApelados.add(bmsF + '|' + mesF);
      }
    }
    Logger.log('🔎 bmsYMesApelados: ' + bmsYMesApelados.size + ' combinaciones BMS+Mes con apelación');

    // ════════════════════════════════════════════════════════
    // 1. HOJA "Disputa" — disputas normales de agentes
    // ════════════════════════════════════════════════════════
    const hojaDisp = ss.getSheetByName('Disputa');
    if (hojaDisp && hojaDisp.getLastRow() > 1) {
      const dataDisp = hojaDisp.getDataRange().getValues();
      const hD       = dataDisp[0].map(h => norm(String(h)));

      const iTS   = hD.findIndex(h => h.includes('fecha') || h.includes('timestamp'));
      const iBMS  = hD.findIndex(h => h.includes('bms'));
      const iNom  = hD.findIndex(h => h.includes('nombre'));
      const iLob  = hD.findIndex(h => h === 'lob');
      const iMes  = hD.findIndex(h => h.includes('mes'));
      const iCom  = hD.findIndex(h => h.includes('comentario') || h.includes('motivo'));
      const iMon  = hD.findIndex(h => h.includes('monto') || h.includes('bono'));
      const iEst  = hD.findIndex(h => h === 'estado' || h.includes('estado'));
      const iResp = hD.findIndex(h => h.includes('respuesta') && !h.includes('fecha'));
      const iFila = hD.findIndex(h => h === 'fila' || h.includes('fila'));

      // Columna Bono KPIs Selec ($)
      const iBonoKPIsSelec = (() => {
        const opciones = ['bono kpis selec', 'bono kpi selec', 'kpis selec', 'bono seleccionado', 'bono selec'];
        for (const op of opciones) {
          const i = hD.findIndex(h => h.includes(op));
          if (i !== -1) return i;
        }
        return -1;
      })();

      Logger.log('📋 obtenerDisputasParaHTML - iBonoKPIsSelec=' + iBonoKPIsSelec +
                 (iBonoKPIsSelec !== -1 ? ' → "' + dataDisp[0][iBonoKPIsSelec] + '"' : ' (no encontrado)'));

      for (let r = 1; r < dataDisp.length; r++) {
        const fila   = dataDisp[r];
        const lobVal = String(fila[iLob !== -1 ? iLob : 3] || '').trim();
        if (!pasaLob(lobVal)) continue;

        const bmsId   = String(fila[iBMS  !== -1 ? iBMS  : 1] || '').trim();
        const nombre  = String(fila[iNom  !== -1 ? iNom  : 2] || '').trim();
        const mesCDM  = String(fila[iMes  !== -1 ? iMes  : 4] || '').trim();
        const estadoV = String(fila[iEst  !== -1 ? iEst  : 5] || 'Pendiente').trim();
        const comDisp = String(fila[iCom  !== -1 ? iCom  : 6] || '').trim();
        const montoV  = parseFloat(fila[iMon !== -1 ? iMon : 7]) || 0;
        const tsVal   = fila[iTS  !== -1 ? iTS  : 0];
        const filaNum = iFila !== -1 ? (parseInt(fila[iFila]) || r + 1) : r + 1;
        if (!bmsId) continue;

        const bonoKPIsSelec = iBonoKPIsSelec !== -1 ? (parseFloat(fila[iBonoKPIsSelec]) || 0) : 0;

        const respondida = estadoV.toLowerCase().includes('aprobada') ||
                           estadoV.toLowerCase().includes('rechazada') ||
                           estadoV.toLowerCase().includes('proceso')   ||
                           (iResp !== -1 && String(fila[iResp] || '').trim() !== '');

        // ✅ FIX: si esta combinación BMS+Mes ya tiene apelación registrada,
        //        cambiar origen a 'Audit' para que el frontend muestre
        //        badge ⚖️ APELACIÓN y mantenga el botón "Ver"
        const claveApe   = bmsId + '|' + norm(mesCDM);
        const yaApelada  = bmsYMesApelados.has(claveApe);

        resultado.push({
          origenTipo:        yaApelada ? 'Apelacion' : 'Disputa',  // ✅ CAMBIADO
          origen:            yaApelada ? 'Audit'     : 'Disputa',  // ✅ CAMBIADO
          timestamp:         tsVal ? new Date(tsVal).toISOString() : new Date().toISOString(),
          bmsId,
          nombreAgente:      nombre,
          lob:               lobVal,
          mesCDM,
          estadoDisputa:     estadoV,
          comentarioDisputa: comDisp,
          monto:             montoV,
          bonoKPIsSelec,
          fila:              filaNum,
          respondida,
          respondidoPor:     respondida ? 'REPORTING' : '',
          rol:               ''
        });
      }
    }

    // ════════════════════════════════════════════════════════
    // 2. HOJA "Disputa_Apelaciones" — apelaciones de OM/ACCM
    // ════════════════════════════════════════════════════════
    const hojaApe = ss.getSheetByName('Disputa_Apelaciones');
    if (hojaApe && hojaApe.getLastRow() > 1) {
      const dataApe = hojaApe.getDataRange().getValues();
      const hA      = dataApe[0].map(h => norm(String(h)));

      const aID     = hA.findIndex(h => h === 'id');
      const aBMS    = hA.findIndex(h => h.includes('bms'));
      const aNom    = hA.findIndex(h => h.includes('nombre'));
      const aLob    = hA.findIndex(h => h === 'lob');
      const aMes    = hA.findIndex(h => h.includes('mes'));
      const aDec    = hA.findIndex(h => h === 'decision' || h === 'solicitud om');
      const aMonAnt = hA.findIndex(h => h.includes('monto anterior'));
      const aMonAdj = hA.findIndex(h => h.includes('monto ajustado'));
      const aRespOM = hA.findIndex(h => h === 'respuesta');
      const aEst    = hA.findIndex(h => h === 'estado');
      const aFecha  = hA.findIndex(h => h === 'fecha');
      const aKPIs   = hA.findIndex(h => h.includes('kpi') && h.includes('detalle'));
      const aCorreo = hA.findIndex(h => h === 'correo');
      const aRol    = hA.findIndex(h => h === 'rol');

      const aDecR    = hA.findIndex(h => h.includes('decision reporting'));
      const aRespR   = hA.findIndex(h => h.includes('respuesta reporting'));
      const aRolR    = hA.findIndex(h => h.includes('rol reporting'));
      const aFechaR  = hA.findIndex(h => h.includes('fecha respuesta'));
      const aMontoR  = hA.findIndex(h => h.includes('monto final'));
      const aComentR = hA.findIndex(h => h.includes('comentario reporting'));

      for (let r = 1; r < dataApe.length; r++) {
        const fila   = dataApe[r];
        const lobVal = String(fila[aLob !== -1 ? aLob : 3] || '').trim();
        if (!pasaLob(lobVal)) continue;

        const idApe    = String(fila[aID     !== -1 ? aID     : 0]  || '').trim();
        const bmsId    = String(fila[aBMS    !== -1 ? aBMS    : 1]  || '').trim();
        const nombre   = String(fila[aNom    !== -1 ? aNom    : 2]  || '').trim();
        const mesCDM   = String(fila[aMes    !== -1 ? aMes    : 4]  || '').trim();
        const solicOM  = String(fila[aDec    !== -1 ? aDec    : 5]  || '').trim();
        const montoAnt = parseFloat(fila[aMonAnt !== -1 ? aMonAnt : 6]) || 0;
        const montoAdj = parseFloat(fila[aMonAdj !== -1 ? aMonAdj : 7]) || 0;
        const respOM   = String(fila[aRespOM !== -1 ? aRespOM : 8]  || '').trim();
        const estadoV  = String(fila[aEst    !== -1 ? aEst    : 9]  || '').trim();
        const rolVal   = String(fila[aRol    !== -1 ? aRol    : 16] || '').trim();
        const tsVal    = fila[aFecha !== -1 ? aFecha : 11];
        const kpisD    = String(fila[aKPIs   !== -1 ? aKPIs   : 13] || '').trim();
        if (!bmsId) continue;

        const decisionReporting  = aDecR   !== -1 ? String(fila[aDecR]   || '').trim() : '';
        const respuestaReporting = aRespR  !== -1 ? String(fila[aRespR]  || '').trim() : '';
        const rolReporting       = aRolR   !== -1 ? String(fila[aRolR]   || '').trim() : '';
        const fechaRespReporting = aFechaR !== -1 ? fila[aFechaR]                      : null;
        const montoFinalReport   = aMontoR !== -1 ? parseFloat(fila[aMontoR] || 0)     : 0;
        const comentReporting    = aComentR!== -1 ? String(fila[aComentR] || '').trim(): '';

        const reportingYaRespondio = decisionReporting !== '' ||
                                     rolReporting.toUpperCase().includes('REPORTING_RESPUESTA');

        let estadoVisible;
        if (reportingYaRespondio && decisionReporting) {
          const emoji = decisionReporting === 'APROBADA'   ? '✅'
                      : decisionReporting === 'RECHAZADA'  ? '❌'
                      : decisionReporting === 'EN_PROCESO' ? '⏳'
                      : decisionReporting === 'MAS_INFO'   ? '📋'
                      : '🔹';
          estadoVisible = emoji + ' ' + decisionReporting + ' (Reporting)';
        } else {
          estadoVisible = estadoV;
        }

        let respondidoPorVal = '';
        if (reportingYaRespondio) {
          respondidoPorVal = 'REPORTING';
        } else if (rolVal) {
          respondidoPorVal = rolVal.replace('_APELACION', '');
        }

        const montoMostrar = montoFinalReport > 0 ? montoFinalReport
                           : montoAdj > 0          ? montoAdj
                           : montoAnt;

        resultado.push({
          origenTipo:          'Apelacion',
          origen:              'Audit',
          timestamp:           tsVal ? new Date(tsVal).toISOString() : new Date().toISOString(),
          id:                  idApe,
          bmsId,
          nombreAgente:        nombre,
          lob:                 lobVal,
          mesCDM,
          estadoDisputa:       estadoVisible,
          comentarioDisputa:   respOM || kpisD || solicOM,
          monto:               montoMostrar,
          montoAjustado:       montoFinalReport > 0 ? montoFinalReport : montoAdj,
          montoAnterior:       montoAnt,
          bonoKPIsSelec:       0,
          kpisDetalle:         kpisD,
          fila:                r + 1,
          respondida:          reportingYaRespondio,
          respondidoPor:       respondidoPorVal,
          rol:                 rolVal,
          decisionReporting,
          respuestaReporting,
          fechaRespuesta:      fechaRespReporting ? new Date(fechaRespReporting).toISOString() : '',
          comentarioReporting: comentReporting
        });
      }
    }

    resultado.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));

    Logger.log('✅ obtenerDisputasParaHTML: ' + resultado.length + ' total | ' +
      resultado.filter(x => x.origenTipo === 'Disputa').length   + ' disputas | ' +
      resultado.filter(x => x.origenTipo === 'Apelacion').length + ' apelaciones');

    return resultado;

  } catch(e) {
    Logger.log('❌ obtenerDisputasParaHTML: ' + e.message + '\n' + e.stack);
    return [];
  }
}

    hojaDA.getRange(filaDA, 1, 1, hojaDA.getLastColumn()).setBackground(bgFila);

// ══════════════════════════════════════════════════════

const hojaAudit = ss.getSheetByName('Audit');
if (hojaAudit) {

    if (hdr === 'bono aprobado')                       idxBonoa   = i;
  });

  // ✅ FIX: normalizar mes antes de comparar
  const mesNormReporting = normalizarNombreMes(String(datos.mesCDM || '').trim());

  for (let i = 1; i < dataAudit.length; i++) {

      else if (idxBMS   === -1 && hdr.includes('bms')) { idxBMS    = h; }

      if (hdr === 'mes cdm' || hdr === 'mes')           { idxMes    = h; }
      if (hdr === 'estado')                             { idxEstado = h; }
      else if (idxEstado === -1 && hdr === 'status')    { idxEstado = h; }

      // ✅ FIX: solo 'bono aprobado' exacto, NUNCA parcial (evita tocar Additional, Decreased, Monto Disputado)

    const mesNorm  = normalizarNombreMes(String(mesDatos || '').trim().toLowerCase());

    let filaAudit = -1;
    for (let i = 1; i < dataAudit.length; i++) {
      const bmsFila = String(dataAudit[i][idxBMS] || '').trim();
      if (bmsFila !== bmsIdStr) continue;

      if (mesNorm && idxMes !== -1) {
        // ✅ FIX: normalizar ambos lados antes de comparar
        const mesFila = normalizarNombreMes(String(dataAudit[i][idxMes] || '').trim());
        if (mesFila !== mesNorm) continue;
      }

      filaAudit = i + 1;
      break;
    }

    if (filaAudit === -1) {
      Logger.log('⚠️ BMS [' + bmsIdStr + '] mes [' + mesNorm + '] NO encontrado en Audit');
      return -1;
    }

    const estadoNuevo = decision === 'APROBADA'   ? '✅ Aprobada'
                      : decision === 'RECHAZADA'  ? '❌ Rechazada'
                      : decision === 'EN_PROCESO' ? '⏳ En Proceso'
                      : decision === 'MAS_INFO'   ? '📋 Más Info'
                      : decision;

    const bgFila = decision === 'APROBADA'   ? '#dcfce7'
                 : decision === 'RECHAZADA'  ? '#fee2e2'
                 : decision === 'EN_PROCESO' ? '#eff6ff'
                 : '#fef3c7';

    if (idxEstado !== -1) {
      hojaAudit.getRange(filaAudit, idxEstado + 1)
        .setValue(estadoNuevo).setBackground(bgFila).setFontWeight('bold');
    }

    // ✅ FIX: solo actualizar monto si fue APROBADA, monto > 0, Y se encontró idxBono exacto
    const monto = parseFloat(montoAjustado) || 0;
    if (decision === 'APROBADA' && monto > 0 && idxBono !== -1) {

    return filaAudit;

  } catch (e) {
    Logger.log('❌ _actualizarEstadoEnAudit: ' + e.message);
    return -1;
  }
}

      'Decisión', 'Monto Anterior', 'Monto Ajustado', 'Respuesta',
      'Estado', 'Usuario', 'Fecha', 'Comentario Interno', 'KPIs Detalle',
      'Correo', 'Origen', 'Rol'
    ];

    let hojaApe = ss.getSheetByName('Disputa_Apelaciones');
    if (!hojaApe) {
      hojaApe = ss.insertSheet('Disputa_Apelaciones');
    }
    if (hojaApe.getLastRow() === 0) {
      hojaApe.appendRow(HEADERS_APE_UNIFIED);
      hojaApe.getRange(1, 1, 1, HEADERS_APE_UNIFIED.length)
        .setBackground('#4f46e5').setFontColor('#FFFFFF').setFontWeight('bold');
      hojaApe.setFrozenRows(1);
    }

    let ajustesResumen = '';
    if (datos.ajustesKPI && datos.ajustesKPI.length > 0) {
      ajustesResumen = datos.ajustesKPI.map(function(k) {
        const ant   = parseFloat(k.valorAnterior) || 0;
        const nvo   = parseFloat(k.valor)         || 0;
        const delta = nvo - ant;
        const flecha = delta > 0 ? '▲' : delta < 0 ? '▼' : '=';
        return '[' + k.categoria + '] ' + k.kpi + ': ' + ant.toFixed(2) + ' ' + flecha + ' ' + nvo.toFixed(2);
      }).join('\n');
    }

    const rolNorm = String(datos.rolUsuario || '').toUpperCase().replace(' ', '_');
    const rolGuardado = rolNorm ? `${rolNorm}_RESPONDIO` : 'SISTEMA_RESPONDIO';

    hojaApe.appendRow([
      datos.idApelacion || ('APE-' + Utilities.getUuid().substring(0, 8).toUpperCase()),
      bmsId,
      datos.nombre             || String(dataDisputa[filaTarget - 1][2] || ''),
      datos.lob                || String(dataDisputa[filaTarget - 1][3] || ''),
      datos.mesCDM             || String(dataDisputa[filaTarget - 1][5] || ''),
      decision,
      parseFloat(datos.montoAnterior) || parseFloat(dataDisputa[filaTarget - 1][6]) || 0,
      montoAjustado,
      datos.respuesta          || '',
      estadoNuevo,
      rolGuardado,
      new Date(),
      datos.comentarioInterno  || '',
      ajustesResumen,
      correo,
      'Disputa',
      rolGuardado
    ]);

    const lr = hojaApe.getLastRow();
    hojaApe.getRange(lr, 8, 1, 2).setNumberFormat('$#,##0.00');
    hojaApe.getRange(lr, 12).setNumberFormat('dd/mm/yyyy hh:mm:ss');
    hojaApe.getRange(lr, 1, 1, HEADERS_APE_UNIFIED.length).setBackground(bgFila);

    // ══════════════════════════════════════════════════════
    // 7. DETALLE KPI EN "Apelaciones_KPI" (si aplica)
    // ══════════════════════════════════════════════════════
    if (datos.ajustesKPI && datos.ajustesKPI.length > 0) {
      let hojaKPI = ss.getSheetByName('Apelaciones_KPI');
      if (!hojaKPI) {
        hojaKPI = ss.insertSheet('Apelaciones_KPI');
        hojaKPI.appendRow([
          'Fecha', 'BMS ID', 'Nombre Agente', 'LOB', 'Mes CDM',
          'Categoría', 'KPI', 'Valor Anterior', 'Valor Nuevo', 'Δ Diferencia',
          'Decisión', 'Usuario', 'Correo'
        ]);
        hojaKPI.getRange(1, 1, 1, 13)
          .setBackground('#1e3a5f').setFontColor('#fff').setFontWeight('bold');
        hojaKPI.setFrozenRows(1);
      }

      datos.ajustesKPI.forEach(function(k) {
        const ant   = parseFloat(k.valorAnterior) || 0;
        const nvo   = parseFloat(k.valor)         || 0;
        const delta = nvo - ant;

        hojaKPI.appendRow([
          fecha, bmsId,
          datos.nombre || '', datos.lob || '', datos.mesCDM || '',
          k.categoria || '', k.kpi || '',
          ant, nvo, delta,
          decision, usuario, correo
        ]);

        const lrk = hojaKPI.getLastRow();
        hojaKPI.getRange(lrk, 8, 1, 3).setNumberFormat('0.0000');
        const bgD = delta > 0 ? '#dcfce7' : delta < 0 ? '#fee2e2' : '#f5f5f5';
        hojaKPI.getRange(lrk, 10).setBackground(bgD).setFontWeight('bold');
      });
    }

    Logger.log('✅ responderDisputa completado — fila Disputa:' + filaTarget +
               ' | fila Audit:' + filaAuditActualizada + ' → ' + estadoNuevo);

    // Log en Disputa_Respuestas
    guardarRespuestaEnHoja({
      fila: filaTarget,
      bmsId: bmsId,
      nombre: datos.nombre || String(dataDisputa[filaTarget - 1][2] || ''),
      lob: datos.lob || String(dataDisputa[filaTarget - 1][3] || ''),
      fechaDisputa: fecha,
      estadoOriginal: String(dataDisputa[filaTarget - 1][idxEstado] || 'Pendiente'),
      respuesta: datos.respuesta || '',
      decision: decision,
      montoAjustado: montoAjustado,
      comentarioInterno: datos.comentarioInterno || '',
      esAjusteOM: false,
      rolUsuario: rolGuardado
    }, estadoNuevo);

    return {
      success: true,
      fila:    filaTarget,
      filaAudit: filaAuditActualizada,
      estado:  estadoNuevo,
      mensaje: 'Disputa respondida: ' + estadoNuevo +
               (filaAuditActualizada > 0 ? ' | Audit fila ' + filaAuditActualizada + ' actualizada' : ' | ⚠️ Audit no actualizada')
    };

  } catch(err) {
    Logger.log('❌ ERROR responderDisputa: ' + err.message + '\n' + err.stack);
    return { success: false, error: err.message };
  }
}
