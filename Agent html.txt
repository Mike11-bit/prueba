function verificarYMostrarAlertaLOB() {
    const alerta = document.getElementById('lobIncorrectaAlerta');
    
    // Lógica para determinar si mostrar la alerta
    // Aquí hay algunas opciones:
    
    // OPÇÃO 1: Mostrar siempre (para testing)
    // alerta.style.display = 'block';
    // return;
    
    // OPÇÃO 2: Verificar en localStorage si el usuario ya actualizó
    const lobActual = datosAgente ? datosAgente.lob : null;
    const ultimaLobGuardada = localStorage.getItem('ultima_lob_conocida');
    const lobActualizadaRecientemente = localStorage.getItem('lob_actualizada_en');
    
    if (lobActual) {
        // Guardar la LOB actual en localStorage
        localStorage.setItem('ultima_lob_conocida', lobActual);
        
        // Si es la primera vez o la LOB cambió, mostrar alerta
        if (!ultimaLobGuardada || ultimaLobGuardada !== lobActual) {
            // Verificar si ya actualizó recientemente (últimos 30 días)
            if (lobActualizadaRecientemente) {
                const fechaActualizacion = new Date(lobActualizadaRecientemente);
                const hoy = new Date();
                const diasDesdeActualizacion = Math.floor((hoy - fechaActualizacion) / (1000 * 60 * 60 * 24));
                
                if (diasDesdeActualizacion > 30) {
                    // Han pasado más de 30 días, mostrar alerta
                    alerta.style.display = 'block';
                } else {
                    alerta.style.display = 'none';
                }
            } else {
                // Nunca ha actualizado, mostrar alerta
                alerta.style.display = 'block';
            }
        } else {
            // Es la misma LOB, ocultar alerta
            alerta.style.display = 'none';
        }
    }
}

        function detectarComparador(comp) {
            const comparadores = {
                '>=': '>=',
                '>': '>',
                '<=': '<=',
                '<': '<',
                '=': '=',
                '==': '='
            };
            return comparadores[comp] || '>=';
        }

        function obtenerIconoComparador(comp) {
            const iconos = {
                '>=': '⬆️',
                '>': '⬆️',
                '<=': '⬇️',
                '<': '⬇️',
                '=': '🎯'
            };
            return iconos[comp] || '⬆️';
        }

        // 🎯 CALCULAR BONO FINAL
        // ============================================================
        function calcularBono() {
            // Recopilar todos los valores
            resultados = { keys: {}, bonus: {}, additional: {}, decreased: {} };
            
            ['keys', 'bonus', 'additional', 'decreased'].forEach(categoria => {
                if (estructuraLOB[categoria]) {
                    estructuraLOB[categoria].forEach((kpi, index) => {
                        const kpiId = `${categoria}_${index}`;
                        const input = document.getElementById(kpiId);
                        const valor = parseFloat(input.value);
                        
                        if (!isNaN(valor) && input.value !== '') {
                            const comparacion = detectarComparador(kpi.comparacion || '>=');
                            const meta = parseFloat(kpi.score);
                            let cumple = false;
                            
                            switch (comparacion) {
                                case '>=': cumple = valor >= meta; break;
                                case '>': cumple = valor > meta; break;
                                case '<=': cumple = valor <= meta; break;
                                case '<': cumple = valor < meta; break;
                                case '=': cumple = valor === meta; break;
                            }
                            
                            resultados[categoria][index] = { cumple, valor };
                        }
                    });
                }
            });
            
            // Calcular totales
            const bonoMaximoTexto = document.getElementById('bonoMaximoDisplay').textContent;
            const bonoMaximo = parseFloat(bonoMaximoTexto.replace(/[^0-9.-]+/g, "")) || 0;
            
            // Verificar Keys
            let keysOK = true;
            if (estructuraLOB.keys && estructuraLOB.keys.length) {
                for (let i = 0; i < estructuraLOB.keys.length; i++) {
                    if (resultados.keys[i] && resultados.keys[i].valor !== null) {
                        if (!resultados.keys[i].cumple) {
                            keysOK = false;
                            break;
                        }
                    }
                }
            }
            
            // Calcular Bonus
            let totalBonus = 0;
            if (estructuraLOB.bonus) {
                estructuraLOB.bonus.forEach((kpi, i) => {
                    if (resultados.bonus[i] && resultados.bonus[i].cumple) {
                        let pct = parseFloat(kpi.porcentaje) || 0;
                        if (pct < 1) pct = pct * 100;
                        totalBonus += (bonoMaximo * pct) / 100;
                    }
                });
            }
            
            // Calcular Additional
            let totalAdditional = 0;
            if (estructuraLOB.additional) {
                estructuraLOB.additional.forEach((kpi, i) => {
                    if (resultados.additional[i] && resultados.additional[i].cumple) {
                        let pct = parseFloat(kpi.porcentaje) || 0;
                        if (pct < 1) pct = pct * 100;
                        totalAdditional += (bonoMaximo * pct) / 100;
                    }
                });
            }
            
            // Calcular Decreased
            let totalDecreased = 0;
            if (estructuraLOB.decreased) {
                estructuraLOB.decreased.forEach((kpi, i) => {
                    if (resultados.decreased[i] && resultados.decreased[i].cumple) {
                        let pct = parseFloat(kpi.porcentaje) || 0;
                        if (pct < 1) pct = pct * 100;
                        totalDecreased += (bonoMaximo * pct) / 100;
                    }
                });
            }
            
            // Calcular total final
            let totalFinal = keysOK ? (totalBonus + totalAdditional - totalDecreased) : 0;
            totalFinal = Math.max(0, totalFinal);
            
            // Mostrar resultado en modal
            mostrarResultadoModal(keysOK, totalFinal, bonoMaximo, totalBonus, totalAdditional, totalDecreased);
        }
