    function agregarFilaKPI(tipo) {
    const tipoHoja = document.getElementById('tipoHoja').value;
    if (!tipoHoja) { alert('Seleccione Tipo de Hoja'); return; }

    const tbody = document.getElementById('tbody' + capitalize(tipo));

    if (tbody.rows.length >= limit) {
        alert('Límite de KPIs alcanzado para esta sección');
        return;
    }

    const row = tbody.insertRow();
    let kpis = kpisDisponibles ? (kpisDisponibles[tipo === 'bonus' ? 'bonusStructure' : tipo === 'additional' ? 'additionalIncentives' : tipo === 'decreased' ? 'decreasedKPIs' : 'keys'] || []) : [];
    let opts = '<option value="">-- Seleccionar --</option>' + kpis.map(k => `<option value="${k}">${k}</option>`).join('');

    let html = `<td><select class="kpi-nombre">${opts}</select></td>
                <td><select class="kpi-comparacion">
                    <option value="<"><</option>
                    <option value="<="><=</option>
                    <option value=">">></option>
                    <option value=">=">>=</option>
                    <option value="=">=</option>
                </select></td>
                <td><input type="number" class="kpi-score" step="0.01"></td>`;

    if (tipo !== 'keys') {
        html += `<td><input type="number" class="kpi-porcentaje" step="0.01" oninput="actualizarIndicadorBono()"></td>`;
        
        // ⭐ AQUÍ ESTÁ EL CAMBIO:
        // Si es Registro_pisos mostramos la celda, si no, la ocultamos (pero la creamos para que exista)

        html += `<td style="display:${displayStyle}"><input type="number" class="kpi-participacion" step="0.01" placeholder="Opcional"></td>`;
    }
    
    html += `<td><button type="button" class="btn-delete" onclick="eliminarFila(this)" style="background:none;border:none;cursor:pointer;">🗑️</button></td>`;

function manejarSubmitRegistro(e) {
    e.preventDefault();

    // ⭐ VALIDAR ANTES DE GUARDAR
    if (!validarAntesDeGuardar()) {
        return; // Detener si la validación falla
    }
    
    // ⭐ USAR VAR EN LUGAR DE CONST

    var mes = document.getElementById('mes').value;

    const datos = {
        mes: mes,
        lob: lob,
        tipoHoja: document.getElementById('tipoHoja').value,
        accm: document.getElementById('accm').value,
        om: document.getElementById('om').value,
        salario: parseFloat(document.getElementById('salario').value),
        porcentaje: parseFloat(document.getElementById('porcentajeBono').value),
        pve: parseFloat(document.getElementById('pvePresupuesto').value) || 0,
        keys: extraerKPIs('keys'),
        bonus: extraerKPIs('bonus'),
        additional: extraerKPIs('additional'),
        decreased: extraerKPIs('decreased')
    };

    if (!confirm(`¿Guardar registro para ${lob}?`)) return;

    mostrarLoading('Guardando...');
    google.script.run.withSuccessHandler(function (r) {

        if (r.success) {

        }
    }).guardarRegistroMetricas(datos);

          function extraerKPIsDeTabla(tipo) {
    const tbody = document.getElementById('tbody' + capitalize(tipo));
    if (!tbody) return [];
    const kpis = [];
    for (let row of tbody.rows) {
        const nombre = row.querySelector('.kpi-nombre')?.value || '';
        const comparacion = row.querySelector('.kpi-comparacion')?.value || '';
        const score = row.querySelector('.kpi-score')?.value || '';
        const porcentaje = row.querySelector('.kpi-porcentaje')?.value || '';
        const participacion = row.querySelector('.kpi-participacion')?.value || '';

        if (nombre) {
            let obj = { nombre, comparacion, score };
            if (tipo !== 'keys') {
                obj.porcentaje = porcentaje;
                if (participacion) obj.participacion = participacion;
            }
            kpis.push(obj);
        }
    }
    return kpis;
}

function extraerKPIs(tipo) {
    const kpis = [];
    document.querySelectorAll(`#tbody${capitalize(tipo)} tr`).forEach(row => {
        const nombre = row.querySelector('.kpi-nombre')?.value;
        if (nombre) {
            const obj = {
                nombre,
                comparacion: row.querySelector('.kpi-comparacion')?.value || '>=',
                score: row.querySelector('.kpi-score')?.value || ''
            };
            
            // ⭐ AGREGAR PORCENTAJE (si existe)
            const porcentajeInput = row.querySelector('.kpi-porcentaje');
            if (porcentajeInput) {
                obj.porcentaje = porcentajeInput.value || '';
            }
            
            // ⭐ AGREGAR PARTICIPACIÓN (si existe)
            const participacionInput = row.querySelector('.kpi-participacion');
            if (participacionInput) {
                const participacion = participacionInput.value;
                if (participacion && participacion !== '') {
                    obj.participacion = participacion;
                }
            }
            
            kpis.push(obj);
        }
    });
    return kpis;
}

        const ClientCache = {
            storage: window.sessionStorage,

            get: function (key) {
                const item = this.storage.getItem(key);
                if (!item) return null;
                try {
                    const record = JSON.parse(item);
                    if (new Date().getTime() > record.expiration) {
                        this.storage.removeItem(key);
                        return null;
                    }
                    console.log(`⚡ Cache Hit: ${key}`);
                    return record.value;
                } catch (e) { return null; }
            },

            set: function (key, value, minutes) {
                const record = {
                    value: value,
                    expiration: new Date().getTime() + (minutes * 60 * 1000)
                };
                try {
                    this.storage.setItem(key, JSON.stringify(record));
                } catch (e) { console.error('Cache Full'); }
            },

            clear: function (prefix) {
                Object.keys(this.storage).forEach(k => {
                    if (k.startsWith(prefix)) this.storage.removeItem(k);
                });
            }
        };

    }

function procesarReporteEjecutivoConCDM(lobs, region, om) {
    console.log(`📊 Procesando ${lobs.length} LOBs para reporte ejecutivo...`);
    
    const meses = ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 
                   'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'];
    const mesActual = meses[new Date().getMonth()];
    const numeroMes = new Date().getMonth() + 1;
    
    let simulacionesPendientes = lobs.length;
    const resultados = [];
    const errores = [];
    
    // ⭐ VALIDAR QUE HAY LOBs
    if (lobs.length === 0) {
        ocultarLoading();
        mostrarToast('❌ No hay LOBs para procesar', 'danger');
        return;
    }
    
    lobs.forEach(function(lob) {
        google.script.run
            .withSuccessHandler(function(resultado) {
                simulacionesPendientes--;
                
                // ⭐ VALIDAR RESPUESTA
                if (!resultado) {
                    console.error(`❌ ${lob}: Respuesta null/undefined`);
                    errores.push(`${lob}: Sin respuesta del servidor`);
                } else if (resultado.success) {
                    resultados.push(resultado);
                    console.log(`✅ ${lob}: $${resultado.inversionReal.toLocaleString()}`);
                } else {
                    console.log(`⚠️ ${lob}: ${resultado.error}`);
                    errores.push(`${lob}: ${resultado.error || 'Error desconocido'}`);
                }
                
                // Cuando todas terminen
                if (simulacionesPendientes === 0) {
                    ocultarLoading();
                    
                    if (resultados.length > 0) {
                        mostrarReporteEjecutivoConCDM(resultados, region, om);
                        
                        // Mostrar errores si los hay
                        if (errores.length > 0) {
                            console.warn('⚠️ Errores en algunas LOBs:', errores);
                            mostrarToast(`⚠️ ${resultados.length} LOBs OK, ${errores.length} con errores`, 'warning');
                        }
                    } else {
                        mostrarToast('❌ No se pudieron procesar las LOBs. Ver consola para detalles.', 'danger');
                        console.error('Errores:', errores);
                    }
                }
            })
            .withFailureHandler(function(err) {
                simulacionesPendientes--;
                console.error(`❌ Error en ${lob}:`, err);
                errores.push(`${lob}: ${err.message || err}`);
                
                if (simulacionesPendientes === 0) {
                    ocultarLoading();
                    if (resultados.length > 0) {
                        mostrarReporteEjecutivoConCDM(resultados, region, om);
                        mostrarToast(`⚠️ ${resultados.length} LOBs OK, ${errores.length} fallaron`, 'warning');
                    } else {
                        mostrarToast('❌ Error procesando LOBs. Ver consola.', 'danger');
                        console.error('Errores:', errores);
                    }
                }
            })
            .simularConMetricasReales(lob, mesActual, numeroMes);
    });
}

function exportarReporteEjecutivoExcel(data) {
    if (!data) {
        mostrarToast('⚠️ No hay datos para exportar', 'warning');
        return;
    }
    
    mostrarLoading('Generando Excel ejecutivo...');
    
    const datosExportar = {
        encabezados: ['LOB', 'HC', 'Cobran', '$/Agente', 'Inv. Teórica', 'Inv. Real', 'Ahorro', 'Efic. Real %'],
        filas: data.resultados.map(function(r) {
            const promAgente = r.hc > 0 ? r.inversionReal / r.hc : 0;
            const ahorro = r.inversionTeorica - r.inversionReal;
            return [
                r.lob, 
                r.hc, 
                r.estadisticas.agentesCobran,
                promAgente,
                r.inversionTeorica,
                r.inversionReal,
                ahorro,
                r.eficienciaReal.toFixed(1)
            ];
        }),
        resumen: [
            ['TOTAL LOBs', data.resumen.totalLOBs],
            ['TOTAL AGENTES', data.resumen.totalAgentes],
            ['INVERSIÓN TEÓRICA', data.resumen.inversionTeorica],
            ['INVERSIÓN REAL', data.resumen.inversionReal],
            ['AHORRO TOTAL', data.resumen.ahorro],
            ['EFICIENCIA GLOBAL', data.resumen.eficienciaGlobal.toFixed(1) + '%']
        ]
    };
    
    google.script.run.withSuccessHandler(function (url) {
        ocultarLoading();
        if (url) {
            window.open(url, '_blank');
            mostrarToast('✅ Excel exportado exitosamente', 'success');
        } else {
            mostrarToast('✅ Exportación simulada', 'success');
        }
    }).exportarSimulacionAExcel(datosExportar);
}

function generarHTMLSimulacionConDetalle(data, salario, porcentaje, bonoMaximo, hcCount) {
    let totalBonus = 0;
    let totalAdditional = 0;

    if (data.kpis?.bonusStructure) {
        data.kpis.bonusStructure.forEach(item => {
            // ⭐ CORREGIR: Normalizar porcentaje antes de calcular
            let pct = parseFloat(item.porcentaje) || 0;
            if (pct > 0 && pct < 1) {
                pct = pct * 100;
            }
            totalBonus += (bonoMaximo * pct) / 100;
        });
    }

    if (data.kpis?.additionalIncentives) {
        data.kpis.additionalIncentives.forEach(item => {
            // ⭐ CORREGIR: Normalizar porcentaje antes de calcular
            let pct = parseFloat(item.porcentaje) || 0;
            if (pct > 0 && pct < 1) {
                pct = pct * 100;
            }
            totalAdditional += (bonoMaximo * pct) / 100;
        });
    }

    const granTotal = totalBonus + totalAdditional;
    const eficiencia = bonoMaximo > 0 ? (granTotal / bonoMaximo) * 100 : 0;

    let html = `<div class="simulation-container">`;

    // Resumen Principal con SALARIO VISIBLE
    html += `
    <div class="summary-card" style="background: #f8f9fa; padding: 20px; border-radius: 10px; margin-bottom: 20px;">

            <div class="summary-item"><strong>Salario Base:</strong> ${formatearMoneda(salario)}</div>  <!-- ⭐ AGREGADO -->

            <div class="summary-item"><strong>Bono Máx/Agente:</strong> ${formatearMoneda(bonoMaximo)}</div>
            <div class="summary-item"><strong>Total Potencial:</strong> ${formatearMoneda(bonoMaximo * hcCount)}</div>
        </div>
    </div>

            // Indicador de Eficiencia
            html += `

                        padding: 20px; border-radius: 10px; margin: 20px 0; text-align: center;">
                <div style="font-size: 0.9em; color: #666;">📊 Eficiencia de Asignación</div>
                <div style="font-size: 2.5em; font-weight: bold; 
                            color: ${eficiencia >= 90 ? '#27ae60' : eficiencia >= 70 ? '#f39c12' : '#e74c3c'};">
                    ${eficiencia.toFixed(1)}%
                </div>
            </div>
        `;

            // Detalle de KPIs
            html += `
            <div style="background: white; padding: 25px; border-radius: 10px; margin-bottom: 20px; box-shadow: 0 2px 8px rgba(0,0,0,0.1);">
                <h3 style="color: #2c3e50; margin-bottom: 20px; border-bottom: 2px solid #eee; padding-bottom: 10px;">📋 Desglose Detallado</h3>
                
                <div style="margin-bottom: 25px;">
                    <h4 style="color: #27ae60;">💰 Bonus Structure</h4>
                    ${data.kpis?.bonusStructure?.length > 0 ? generarTablaKPIsDetallada(data.kpis.bonusStructure, bonoMaximo, hcCount, 'bonus') : '<p>Sin KPIs</p>'}
                </div>
                
                <div>
                    <h4 style="color: #f39c12;">⭐ Additional Incentive</h4>
                    ${data.kpis?.additionalIncentives?.length > 0 ? generarTablaKPIsDetallada(data.kpis.additionalIncentives, bonoMaximo, hcCount, 'additional') : '<p>Sin KPIs</p>'}
                </div>
            </div>
        `;

            // Resumen Totales

            <div style="background: linear-gradient(135deg, #667eea, #764ba2); color: white; padding: 25px; border-radius: 10px; margin-bottom: 20px;">
                <h3 style="text-align: center;">📊 Resumen Financiero</h3>
                <div style="display: grid; grid-template-columns: repeat(3, 1fr); gap: 15px; text-align: center;">

function generarTablaKPIsDetallada(kpis, bonoMaximo, hcCount, tipo) {
    const colorBorde = tipo === 'bonus' ? '#27ae60' : '#f39c12';
    let html = `<table class="kpi-table" style="border-left: 4px solid ${colorBorde};">
    <thead><tr><th>KPI</th><th>Meta</th><th>% Bono</th><th>Valor/Agente</th><th>Total x HC</th></tr></thead><tbody>`;

    let totalVal = 0;
    kpis.forEach(kpi => {
        // ⭐ CORREGIR: Si el porcentaje es menor a 1, multiplicar por 100
        let pct = parseFloat(kpi.porcentaje) || 0;
        
        // Si el porcentaje está en formato decimal (ej: 0.30), convertir a porcentaje (30)
        if (pct > 0 && pct < 1) {
            pct = pct * 100;
        }
        
        const val = (bonoMaximo * pct) / 100;
        totalVal += val * hcCount;
        
        html += `<tr>
            <td>${kpi.nombre}</td>
            <td>${kpi.score || '-'}</td>
            <td>${pct.toFixed(2)}%</td>
            <td>${formatearMoneda(val)}</td>
            <td>${formatearMoneda(val * hcCount)}</td>
        </tr>`;
    });
    html += `</tbody></table>`;
    return html;
}

        function agregarDestinatario() {
            const input = document.getElementById('inputNuevoCorreo');
            const correo = input.value.trim();

            // Validar formato de correo
            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (!emailRegex.test(correo)) {
                mostrarToast('⚠️ Ingrese un correo válido', 'warning');
                return;
            }

            // Verificar que no esté duplicado
            if (window.destinatariosCorreo.includes(correo)) {
                mostrarToast('⚠️ Este correo ya está en la lista', 'warning');
                return;
            }

            window.destinatariosCorreo.push(correo);
            input.value = '';
            actualizarListaDestinatarios();
            mostrarToast('✅ Destinatario agregado', 'success');
        }

   function abrirModalSimulacion() {
    const lob = document.getElementById('simSelectLob').value;
    const mes = document.getElementById('simSelectMes').value;

    if (!lob || !mes) {
        mostrarToast('⚠️ Seleccione LOB y Mes', 'warning');
        return;
    }

    mostrarLoading('Cargando simulación...');

    // ⭐ PASO 1: Obtener configuración oficial
    google.script.run
        .withSuccessHandler(function(configOficial) {
            // ⭐ PASO 2: Obtener estructura registrada
            google.script.run
                .withSuccessHandler(function(estructura) {
                    ocultarLoading();
                    
                    if (!estructura.encontrado) {
                        mostrarToast('❌ No hay estructura registrada', 'danger');
                        return;
                    }
                    
                    // ⭐ JERARQUÍA DE SALARIO
                    let salario, porcentaje;
                    
                    // 1️⃣ Primero: Input manual (para simulaciones)
                    const inputSalario = parseFloat(document.getElementById('simInputSalario').value);
                    const inputPorcentaje = parseFloat(document.getElementById('simInputPorcentaje').value);
                    
                    if (inputSalario && inputPorcentaje) {
                        salario = inputSalario;
                        porcentaje = inputPorcentaje;
                        console.log('💡 Usando valores manuales del simulador');
                    }
                    // 2️⃣ Segundo: Estructura registrada (tiene salario directo)
                    else if (estructura.salario && estructura.porcentaje) {
                        salario = estructura.salario;
                        porcentaje = estructura.porcentaje;
                        console.log('✅ Usando salario de estructura registrada');
                    }
                    // 3️⃣ Tercero: Configuracion (fallback)
                    else if (configOficial.encontrado) {
                        salario = configOficial.salario;
                        porcentaje = configOficial.porcentaje;
                        console.log('⚠️ Usando configuración oficial (fallback)');
                    }
                    // ❌ Error: No hay salario
                    else {
                        mostrarToast('❌ No se encontró salario configurado', 'danger');
                        return;
                    }
                    
                    const bonoMaximo = (salario * porcentaje) / 100;
                    const hc = estructura.hc || 1;
                    
                    const html = generarHTMLSimulacionConDetalle(
                        estructura, salario, porcentaje, bonoMaximo, hc
                    );
                    
                    document.getElementById('contenidoSimulacion').innerHTML = html;
                    document.getElementById('simulacionModal').style.display = 'block';
                })
                .obtenerEstructuraRegistrada(lob, mes, datosUsuario);
        })
        .obtenerSalarioPorLob(lob);
}

        function llenarKPIsEnFormulario(kpis) {
            // Limpiar tablas existentes
            ['Keys', 'Bonus', 'Additional', 'Decreased'].forEach(tipo => {
                const tbody = document.getElementById('tbody' + tipo);
                if (tbody) tbody.innerHTML = '';
            });

            // Llenar Keys
            if (kpis?.keys) {
                kpis.keys.forEach(kpi => agregarFilaKPIConDatos('keys', kpi));
            }

            // Llenar Bonus
            if (kpis?.bonusStructure) {
                kpis.bonusStructure.forEach(kpi => agregarFilaKPIConDatos('bonus', kpi));
            }

            // Llenar Additional
            if (kpis?.additionalIncentives) {
                kpis.additionalIncentives.forEach(kpi => agregarFilaKPIConDatos('additional', kpi));
            }

            // Llenar Decreased
            if (kpis?.decreasedKPIs) {
                kpis.decreasedKPIs.forEach(kpi => agregarFilaKPIConDatos('decreased', kpi));
            }
        }

function agregarFilaKPI(tipo) {

        alert('⚠️ Seleccione Tipo de Hoja primero');
        return;

    if (tbody.rows.length >= limite) {
        alert(`⚠️ Límite alcanzado: máximo ${limite} para ${tipoHoja}`);
        return;
    }

                <option value="<=">&lt;=</option>
                <option value=">">&gt;</option>
                <option value=">=">&gt;=</option>

function calcularMontoPorcentaje(pct, max) {
    // ⭐ NORMALIZAR: Si está en decimal (0.30), convertir a porcentaje (30)
    let porcentaje = parseFloat(pct) || 0;
    
    if (porcentaje > 0 && porcentaje < 1) {
        porcentaje = porcentaje * 100;
    }
    
    return (max * porcentaje) / 100;
}        
        function calcularBonoMaximo() {
            const s = parseFloat(document.getElementById('salario').value) || 0;

function actualizarIndicadorBono() {
    const salario = parseFloat(document.getElementById('salario').value) || 0;
    const porcentajeBono = parseFloat(document.getElementById('porcentajeBono').value) || 0;
    const bonoMaximo = (salario * porcentajeBono) / 100;
    
    // ⭐ NUEVA LÓGICA: Agrupar por KPI para evitar duplicar porcentajes
    const kpisPorNombre = {};
    
    // Procesar Bonus Structure
    document.querySelectorAll('#tbodyBonus tr').forEach(row => {
        const nombreKPI = row.querySelector('.kpi-nombre')?.value;
        const porcentaje = parseFloat(row.querySelector('.kpi-porcentaje')?.value) || 0;
        const participacion = parseFloat(row.querySelector('.kpi-participacion')?.value) || 0;
        
        if (nombreKPI && porcentaje > 0) {
            if (!kpisPorNombre[nombreKPI]) {
                kpisPorNombre[nombreKPI] = {
                    porcentaje: porcentaje, // Tomar el % del primer registro
                    pisos: []
                };
            }
            
            kpisPorNombre[nombreKPI].pisos.push({
                participacion: participacion,
                porcentaje: porcentaje
            });
        }
    });
    
    // Procesar Additional Incentives
    document.querySelectorAll('#tbodyAdditional tr').forEach(row => {
        const nombreKPI = row.querySelector('.kpi-nombre')?.value;
        const porcentaje = parseFloat(row.querySelector('.kpi-porcentaje')?.value) || 0;
        const participacion = parseFloat(row.querySelector('.kpi-participacion')?.value) || 0;
        
        if (nombreKPI && porcentaje > 0) {
            if (!kpisPorNombre[nombreKPI]) {
                kpisPorNombre[nombreKPI] = {
                    porcentaje: porcentaje,
                    pisos: []
                };
            }
            
            kpisPorNombre[nombreKPI].pisos.push({
                participacion: participacion,
                porcentaje: porcentaje
            });
        }
    });
    
    // ⭐ CALCULAR TOTAL: Sumar solo UNA VEZ cada KPI
    let totalAsignado = 0;
    Object.keys(kpisPorNombre).forEach(nombreKPI => {
        const kpi = kpisPorNombre[nombreKPI];
        totalAsignado += kpi.porcentaje; // ⭐ SOLO UNA VEZ POR KPI
    });
    
    // ⭐ CALCULAR MONTO ASIGNADO (tomar la participación más alta de cada KPI)
    let montoAsignadoTotal = 0;
    Object.keys(kpisPorNombre).forEach(nombreKPI => {
        const kpi = kpisPorNombre[nombreKPI];
        
        // Encontrar la participación más alta
        const participacionMaxima = Math.max(...kpi.pisos.map(p => p.participacion || 100));
        
        // Calcular monto con la participación máxima
        const montoPorKPI = (bonoMaximo * kpi.porcentaje / 100) * (participacionMaxima / 100);
        montoAsignadoTotal += montoPorKPI;
    });
    
    const disponible = 100 - totalAsignado;
    
    // ⭐ ACTUALIZAR UI
    const progressBar = document.getElementById('progressBar');
    if (progressBar) {
        progressBar.style.width = Math.min(totalAsignado, 100) + '%';
        progressBar.textContent = totalAsignado.toFixed(1) + '%';
        
        // Cambiar color según el estado
        progressBar.className = 'progress-bar';
        if (totalAsignado > 100) {
            progressBar.classList.add('danger');
        } else if (totalAsignado > 90) {
            progressBar.classList.add('warning');
        }
    }
    
    const porcentajeAsignado = document.getElementById('porcentajeAsignadoTexto');
    const porcentajeDisponible = document.getElementById('porcentajeDisponibleTexto');
    const montoAsignadoTexto = document.getElementById('montoAsignadoTexto');
    
    if (porcentajeAsignado) porcentajeAsignado.textContent = totalAsignado.toFixed(1) + '%';
    if (porcentajeDisponible) {
        porcentajeDisponible.textContent = Math.max(0, disponible).toFixed(1) + '%';
        
        // Cambiar color si está en negativo
        if (disponible < 0) {
            porcentajeDisponible.style.color = '#e74c3c';
        } else {
            porcentajeDisponible.style.color = '#27ae60';
        }
    }
    
    if (montoAsignadoTexto) montoAsignadoTexto.textContent = formatearMoneda(montoAsignadoTotal);
    
    // ⭐ VALIDAR SI EXCEDE 100%
    if (totalAsignado > 100) {
        console.warn(`⚠️ ADVERTENCIA: Total asignado (${totalAsignado.toFixed(1)}%) excede 100%`);
    }
    
    console.log(`📊 Indicador actualizado:`);
    console.log(`   KPIs únicos: ${Object.keys(kpisPorNombre).length}`);
    console.log(`   Total asignado: ${totalAsignado.toFixed(1)}%`);
    console.log(`   Monto asignado: $${montoAsignadoTotal.toFixed(2)}`);
    console.log(`   Disponible: ${disponible.toFixed(1)}%`);
}

        function cambiarTipoAccesoUsuario() {
            const select = document.getElementById('tipoAccesoSelect');
            const statusSpan = document.getElementById('tipoAccesoStatus');

            if (!select) {
                console.error('❌ No se encontró el select de tipo de acceso');
                mostrarToast('❌ Error: No se encontró el selector', 'danger');
                return;
            }

            const tipoAcceso = select.value;
            console.log('🔄 Cambiando tipo de acceso a:', tipoAcceso);

            // Validar selección
            if (!tipoAcceso || tipoAcceso === '') {
                mostrarToast('⚠️ Por favor selecciona un tipo de acceso', 'warning');
                return;
            }

            // Deshabilitar botón temporalmente
            const boton = event.target;
            if (boton && boton.tagName === 'BUTTON') {
                boton.disabled = true;
                boton.textContent = '⏳ Procesando...';
            }

            // Usar correo del usuario actual
            const email = datosUsuario.correo || (document.getElementById('correoUsuario') ? document.getElementById('correoUsuario').textContent : '');

            mostrarLoading(`Cambiando acceso a ${tipoAcceso}...`);

            google.script.run
                .withSuccessHandler(function (response) {
                    ocultarLoading();

                    if (response.success) {
                        mostrarToast('✅ Tipo de acceso cambiado a: ' + tipoAcceso + '. Recargando...', 'success');

                        setTimeout(function () {
                            // Recargar para aplicar cambios
                            google.script.run.withSuccessHandler(function (url) {
                                if (url) window.top.location.href = url + '?page=' + tipoAcceso;
                                else window.top.location.reload();
                            }).getUrl();
                        }, 1500);

                    } else {
                        mostrarToast('❌ Error: ' + response.error, 'danger');
                        if (boton && boton.tagName === 'BUTTON') {
                            boton.disabled = false;
                            boton.textContent = '🔄 Cambiar';
                        }
                    }
                })
                .withFailureHandler(function (error) {
                    ocultarLoading();
                    if (boton && boton.tagName === 'BUTTON') {
                        boton.disabled = false;
                        boton.textContent = '🔄 Cambiar';
                    }
                    mostrarToast('❌ Error de conexión: ' + error.message, 'danger');
                })
                .cambiarTipoAcceso(email, tipoAcceso);
        }

                    <td style="font-weight: 600;">${kpi.kpi}</td>
                    <td style="text-align: center;">-</td>
                    <td style="text-align: center; font-weight: 600; color: #f39c12;">${porcentajeDisplay.toFixed(1)}%</td>
                    <td style="text-align: center; color: #27ae60; font-weight: bold;">${kpi.totalCumplen} / ${data.hc}</td>
                    <td style="text-align: right;">${formatearMoneda(valorPorAgente)}</td>
                    <td style="text-align: right; font-weight: bold;">${formatearMoneda(kpi.montoTotal)}</td>
                </tr>
            `;
        });
        
        html += `
                    <tr style="background: #fef5e7; font-weight: bold;">
                        <td colspan="5" style="text-align: right;">SUBTOTAL ADDITIONAL:</td>
                        <td style="text-align: right;">${formatearMoneda(totalAdditional)}</td>
                    </tr>
                </tbody>
            </table>
        </div>
        `;
    }

        // ⭐ DECREASED KPIs (PENALIZACIONES)
    if (data.desglosePorCategoria.decreasedKPIs && data.desglosePorCategoria.decreasedKPIs.length > 0) {
        const totalDecreased = data.desglosePorCategoria.decreasedKPIs.reduce((sum, k) => sum + k.montoTotal, 0);
        
        html += `
        <div style="margin-bottom: 30px;">
            <h4 style="color: #e74c3c;">🔴 Decreased (Penalizaciones)</h4>
            <table class="kpi-table">
                <thead>
                    <tr>
                        <th>KPI</th>
                        <th>% Penalización</th>
                        <th>Afectados</th>
                        <th>Penalización Total</th>
                    </tr>
                </thead>
                <tbody>
        `;
        
        data.desglosePorCategoria.decreasedKPIs.forEach(kpi => {
            let porcentajeDisplay = parseFloat(kpi.porcentaje) || 0;
            if (porcentajeDisplay > 0 && porcentajeDisplay < 1) {
                porcentajeDisplay = porcentajeDisplay * 100;
            }
            
            html += `
                <tr>
                    <td style="font-weight: 600;">${kpi.kpi}</td>
                    <td style="text-align: center; color: #e74c3c; font-weight: bold;">${porcentajeDisplay.toFixed(1)}%</td>
                    <td style="text-align: center;">${kpi.totalCumplen}</td>
                    <td style="text-align: right; font-weight: bold; color: #e74c3c;">-${formatearMoneda(kpi.montoTotal)}</td>
                </tr>
            `;
        });
        
        html += `
                    <tr style="background: #ffebee; font-weight: bold;">
                        <td colspan="3" style="text-align: right;">TOTAL PENALIZACIONES:</td>
                        <td style="text-align: right; color: #e74c3c;">-${formatearMoneda(totalDecreased)}</td>
                    </tr>
                </tbody>
            </table>
        </div>
        `;
    }
    
    html += `</div>`; // Cierre desglose
    
    // ========================================
    // 💰 RESUMEN FINANCIERO
    // ========================================
    const totalBonus = data.desglosePorCategoria.bonusStructure.reduce((sum, k) => sum + k.montoTotal, 0);
    const totalAdditional = data.desglosePorCategoria.additionalIncentives.reduce((sum, k) => sum + k.montoTotal, 0);
    
    html += `
    <div style="background: linear-gradient(135deg, #667eea, #764ba2); color: white; padding: 30px; border-radius: 10px; margin-bottom: 20px;">
        <h3 style="text-align: center; margin-bottom: 20px;">💰 Resumen Financiero REAL</h3>
        <div style="display: grid; grid-template-columns: repeat(3, 1fr); gap: 20px; text-align: center;">
            <div>
                <div style="font-size: 0.9rem; opacity: 0.9;">Bonus Structure</div>
                <div style="font-size: 2rem; font-weight: bold;">${formatearMoneda(totalBonus)}</div>
            </div>
            <div>
                <div style="font-size: 0.9rem; opacity: 0.9;">Additional</div>
                <div style="font-size: 2rem; font-weight: bold;">${formatearMoneda(totalAdditional)}</div>
            </div>
            <div style="border-left: 2px solid rgba(255,255,255,0.3);">
                <div style="font-size: 1rem; opacity: 0.9;">TOTAL A PAGAR</div>
                <div style="font-size: 2.5rem; font-weight: bold;">${formatearMoneda(data.inversionReal)}</div>
            </div>
        </div>
    </div>
    `;
    
    // ========================================
    // 🏆 TOP 10 GANADORES
    // ========================================
    if (data.top10Ganadores && data.top10Ganadores.length > 0) {
        html += `
        <div style="background: white; border: 2px solid #f39c12; border-radius: 10px; padding: 20px; margin-bottom: 20px;">
            <h3 style="color: #f39c12; text-align: center; margin-bottom: 15px;">🏆 Top 10 Ganadores del Mes</h3>
            <table class="kpi-table">
                <thead>
                    <tr>
                        <th>#</th>
                        <th>BMS ID</th>
                        <th>Nombre</th>
                        <th>Keys</th>
                        <th>Bonus</th>
                        <th>Additional</th>
                        <th>TOTAL</th>
                    </tr>
                </thead>
                <tbody>
        `;
        
        data.top10Ganadores.forEach((agente, idx) => {
            const medalla = idx === 0 ? '🥇' : idx === 1 ? '🥈' : idx === 2 ? '🥉' : (idx + 1);
            
            const totalKeys = parseFloat(agente.totalKeys) || 0;
            const totalBonus = parseFloat(agente.totalBonus) || 0;
            const totalAdditional = parseFloat(agente.totalAdditional) || 0;
            const bonoFinal = parseFloat(agente.bonoFinal) || 0;
            
            const bonoMaximo = data.configuracion?.bonoMaximo || 0;
            const maxEsperado = bonoMaximo * 1.5;
            
            let estiloFila = '';
            let colorTotal = '#27ae60';
            
            if (bonoFinal > maxEsperado && bonoMaximo > 0) {
                estiloFila = 'background: #ffebee; border-left: 3px solid #e74c3c;';
                colorTotal = '#e74c3c';
                console.warn(`⚠️ BONO SOSPECHOSO: BMS ${agente.bmsId} = ${formatearMoneda(bonoFinal)} (Max esperado: ${formatearMoneda(maxEsperado)})`);
            }
            
            html += `
                <tr style="${estiloFila}">
                    <td style="text-align: center; font-size: 1.2rem;">${medalla}</td>
                    <td style="text-align: center;">${agente.bmsId}</td>
                    <td style="font-weight: 600;">${agente.nombre}</td>
                    <td style="text-align: right;">${formatearMoneda(totalKeys)}</td>
                    <td style="text-align: right;">${formatearMoneda(totalBonus)}</td>
                    <td style="text-align: right;">${formatearMoneda(totalAdditional)}</td>
                    <td style="text-align: right; font-weight: bold; color: ${colorTotal};">${formatearMoneda(bonoFinal)}</td>
                </tr>
            `;
        });
        
        html += `
                </tbody>
            </table>
        </div>
        `;
    }
    
    // ========================================
    // 🎯 BOTÓN EXPORTAR
    // ========================================
    html += `
    <div style="text-align: center; margin-top: 30px;">
        <button onclick="exportarSimulacionReal()" class="btn btn-success" style="padding: 15px 40px; font-size: 1.1rem;">
            📥 EXPORTAR REPORTE COMPLETO
        </button>
    </div>
    `;
    
    content.innerHTML = html;
    modal.style.display = 'block';
    
    mostrarToast('✅ Simulación real completada con éxito', 'success');
}

function obtenerNumeroMesDesdeNombre(nombreMes) {
    const mesesMap = {
        'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
        'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
        'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
    };
    
    const mesLower = nombreMes.toLowerCase().trim();
    return mesesMap[mesLower] || null;
}
