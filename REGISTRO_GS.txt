function obtenerLobsPorOM(nombreOM) {
    const cacheKey = `lobs_om_${normalizarKey(nombreOM)}`;
    return getCached(cacheKey, () => filtrarLobsPorColumna('H_C', 6, nombreOM), 3600);
}

function obtenerLobsPorACCM(nombreACCM) {
    const cacheKey = `lobs_accm_${normalizarKey(nombreACCM)}`;
    return getCached(cacheKey, () => filtrarLobsPorColumna('H_C', 5, nombreACCM), 3600);
}

function obtenerLobsPorSupervisor(nombreSupervisor) {
    const cacheKey = `lobs_sup_${normalizarKey(nombreSupervisor)}`;
    return getCached(cacheKey, () => filtrarLobsPorColumna('H_C', 4, nombreSupervisor), 3600);

function obtenerLobsPorRegion(region) {
    const cacheKey = `lobs_region_${normalizarKey(region)}`;
    return getCached(cacheKey, function () {
        console.log(`🔍 Obteniendo LOBs para región: ${region}`);
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaHC = ss.getSheetByName('H_C');
        if (!hojaHC) return [];

        const data = hojaHC.getDataRange().getValues();
        const lobsSet = new Set();
        const regionNorm = normalizarKey(region);

        for (let i = 1; i < data.length; i++) {
            const regionFila = normalizarKey(data[i][8]); // Col I
            const lobFila = String(data[i][1] || '').trim(); // Col B
            if (regionFila === regionNorm && lobFila) {
                lobsSet.add(lobFila);
            }
        }

        return Array.from(lobsSet).sort();
    }, 3600);
}

function contarAgentesPorLob(lob) {
    const cacheKey = `count_agentes_${normalizarKey(lob)}`;
    return getCached(cacheKey, function () {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaHC = ss.getSheetByName('H_C');
        const data = hojaHC.getDataRange().getValues();
        const lobNorm = normalizarKey(lob);
        let count = 0;

        const headers = data[0].map(h => String(h).toLowerCase());
        const colStatus = headers.findIndex(h => h.includes('status') || h.includes('estado'));

        for (let i = 1; i < data.length; i++) {
            if (normalizarKey(data[i][1]) === lobNorm) {
                if (colStatus > -1) {
                    const estado = String(data[i][colStatus]).toLowerCase();
                    if (estado === 'activo' || estado === 'active' || estado === 'alta') count++;
                } else {
                    count++;
                }
            }
        }
        return count;
    }, 3600);
}

function filtrarLobsPorColumna(nombreHoja, colIndex, valorBusqueda) {
    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hoja = ss.getSheetByName(nombreHoja);
    const data = hoja.getDataRange().getValues();
    const lobs = new Set();
    const valNorm = normalizarKey(valorBusqueda);

    for (let i = 1; i < data.length; i++) {
        const valFila = normalizarKey(data[i][colIndex]);
        if (valFila === valNorm) {
            const lob = String(data[i][1]).trim();
            if (lob) lobs.add(lob);
        }
    }
    return Array.from(lobs).sort();
}

function normalizarKey(str) {
    return String(str || '').toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-z0-9]/g, '');
}

function obtenerResponsablesPorLob(lob) {
    try {
        console.log(`🔍 [Optimizado] Buscando responsables para LOB: ${lob}`);

        const cacheKey = `resp_lob_${normalizarKey(lob)}`;
        const datosLOB = getCached(cacheKey, function () {
            const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
            const hojaHC = ss.getSheetByName('H_C');
            const dataHC = hojaHC.getDataRange().getValues();
            const lobNorm = normalizarKey(lob);

            for (let i = 1; i < dataHC.length; i++) {
                if (normalizarKey(dataHC[i][1]) === lobNorm) {
                    return {
                        encontrado: true,
                        supervisor: String(dataHC[i][4] || '').trim(),
                        accm: String(dataHC[i][5] || '').trim(),
                        om: String(dataHC[i][6] || '').trim()
                    };
                }
            }
            return { encontrado: false };
        }, 3600);

        if (!datosLOB.encontrado) {
            return { supervisor: '', accm: '', om: '', error: 'LOB no encontrada en H_C' };
        }

        const usuario = obtenerDatosUsuarioActivo();
        if (usuario.rol === 'OM') {
            const omAsignadoNorm = normalizarKey(datosLOB.om);
            const usuarioNombreNorm = normalizarKey(usuario.nombre);

            if (omAsignadoNorm !== usuarioNombreNorm) {
                return {
                    supervisor: '', accm: '', om: '',
                    error: `Esta LOB está asignada a otro OM (${datosLOB.om}).`
                };
            }
        }
        return datosLOB;

    } catch (error) {
        return { supervisor: '', accm: '', om: '', error: error.toString() };
    }
}

function obtenerDatosParaFormularioRegistro() {
    try {
        console.log('🔍 [Optimizado] Obteniendo datos iniciales...');
        const usuario = obtenerDatosUsuarioActivo();
        if (!usuario) return { success: false, error: 'Usuario no encontrado' };

        console.log(`👤 Usuario: ${usuario.nombre}, Rol: ${usuario.rol}`);

        // ⭐ OBTENER LOBs SEGÚN ROL
        let lobs = [];
        
        if (usuario.rol === 'Reporting Analyst' || usuario.rol === 'Audit') {
            lobs = obtenerTodasLasLobs();
            console.log(`📊 Reporting/Audit - Total LOBs: ${lobs.length}`);
        } else if (usuario.rol === 'OM') {
            lobs = obtenerLobsPorOM(usuario.nombre);
            console.log(`📊 OM - LOBs asignadas: ${lobs.length}`);
            console.log(`📋 LOBs: ${lobs.join(', ')}`);
        } else if (usuario.rol === 'ACCM') {
            lobs = obtenerLobsPorACCM(usuario.nombre);
            console.log(`📊 ACCM - LOBs asignadas: ${lobs.length}`);
        } else if (usuario.rol === 'Supervisor') {
            lobs = obtenerLobsPorSupervisor(usuario.nombre);
            console.log(`📊 Supervisor - LOBs asignadas: ${lobs.length}`);
        }

        if (lobs.length === 0) {
            console.log(`⚠️ No se encontraron LOBs para ${usuario.rol}`);
        }

        return {
            success: true,
            usuario: usuario,
            lobs: lobs
        };
    } catch (error) {
        console.log(`❌ Error: ${error.toString()}`);
        return { success: false, error: error.toString() };
    }
}

function obtenerSalarioPorLob(lob) {
    const cacheKey = `salario_lob_${normalizarKey(lob)}`;
    return getCached(cacheKey, function () {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaConfig = ss.getSheetByName('Configuracion');
        
        if (!hojaConfig) {
            Logger.log('❌ Hoja Configuracion NO EXISTE');
            return { encontrado: false };
        }

        const data = hojaConfig.getDataRange().getValues();
        const lobNorm = normalizarKey(lob);

        Logger.log(`\n🔍 ========== BÚSQUEDA EN CONFIGURACION ==========`);
        Logger.log(`📋 LOB buscada: "${lob}"`);
        Logger.log(`🔑 LOB normalizada: "${lobNorm}"`);
        Logger.log(`📊 Total filas en Configuracion: ${data.length - 1}`);
        
        // ⭐ MOSTRAR TODAS LAS LOBs EN CONFIGURACION
        Logger.log(`\n📋 LOBs disponibles en Configuracion:`);
        const lobsEncontradas = [];
        for (let i = 1; i < data.length; i++) {
            const lobFila = String(data[i][2] || '').trim(); // Columna C
            const lobFilaNorm = normalizarKey(lobFila);
            
            if (lobFila) {
                lobsEncontradas.push({
                    fila: i + 1,
                    original: lobFila,
                    normalizada: lobFilaNorm,
                    coincide: lobFilaNorm === lobNorm
                });
                
                Logger.log(`   Fila ${i+1}: "${lobFila}" → normalizada: "${lobFilaNorm}" ${lobFilaNorm === lobNorm ? '✅ COINCIDE' : ''}`);
            }
        }
        
        // ⭐ BUSCAR COINCIDENCIA (de atrás hacia adelante)
        Logger.log(`\n🔍 Buscando coincidencia...`);
        for (let i = data.length - 1; i >= 1; i--) {
            const lobFila = String(data[i][2] || '').trim();
            const lobFilaNorm = normalizarKey(lobFila);
            
            if (lobFilaNorm === lobNorm) {
                const salario = parseFloat(data[i][3]) || 0;
                const porcentaje = parseFloat(data[i][4]) || 0;
                const mes = String(data[i][0] || '').trim();
                const om = String(data[i][1] || '').trim();
                
                Logger.log(`\n✅ ========== CONFIG ENCONTRADA ==========`);
                Logger.log(`📍 Fila: ${i+1}`);
                Logger.log(`📋 LOB: "${lobFila}"`);
                Logger.log(`📅 Mes: "${mes}"`);
                Logger.log(`👤 OM: "${om}"`);
                Logger.log(`💰 Salario (Col D): ${data[i][3]} → parseado: ${salario}`);
                Logger.log(`📊 % Bono (Col E): ${data[i][4]} → parseado: ${porcentaje}`);
                Logger.log(`⏰ Timestamp (Col F): ${data[i][5]}`);
                Logger.log(`💵 PVE (Col G): ${data[i][6]}`);
                Logger.log(`=========================================\n`);
                
                return {
                    encontrado: true,
                    salario: salario,
                    porcentaje: porcentaje,
                    mes: mes,
                    om: om
                };
            }
        }
        
        Logger.log(`\n❌ ========== NO SE ENCONTRÓ CONFIG ==========`);
        Logger.log(`⚠️ La LOB "${lob}" NO existe en Configuracion`);
        Logger.log(`\n💡 Verifica:`);
        Logger.log(`   1. ¿El nombre es EXACTAMENTE igual?`);
        Logger.log(`   2. ¿Hay espacios extra al inicio/final?`);
        Logger.log(`   3. ¿Hay caracteres especiales ocultos?`);
        Logger.log(`\n📋 LOBs similares encontradas:`);
        
        // Buscar LOBs similares
        const similares = lobsEncontradas.filter(l => 
            l.normalizada.includes(lobNorm.substring(0, 5)) || 
            lobNorm.includes(l.normalizada.substring(0, 5))
        );
        
        if (similares.length > 0) {
            similares.forEach(s => {
                Logger.log(`   - "${s.original}" (Fila ${s.fila})`);
            });
        } else {
            Logger.log(`   (ninguna)`);
        }
        Logger.log(`=============================================\n`);
        
        return { encontrado: false };
    }, 600);
}

// 4. GUARDADO SEGURO (CON LOCK SERVICE)
// ========================================

function guardarRegistroMetricas(datos) {
    const lock = LockService.getScriptLock();
    try {
        if (!lock.tryLock(30000)) {
            return { success: false, error: '⚠️ Servidor ocupado. Intente nuevamente.' };
        }
        console.log('🔒 Guardado seguro iniciado...');

        // Validaciones Básicas
        if (!datos || !datos.mes || !datos.lob || !datos.tipoHoja) {
            return { success: false, error: 'Faltan campos obligatorios' };
        }

        // ⭐ VALIDACIÓN DE PORCENTAJES (CORREGIDA PARA PISOS)
        if (datos.bonus) {
            let totalPct = datos.bonus.reduce((sum, kpi) => sum + (parseFloat(kpi.porcentaje) || 0), 0);
            
            // Si NO es 'Registro_pisos', mantenemos la validación estricta del 100%
            // Si ES 'Registro_pisos', permitimos >100% porque hay múltiples filas (pisos) para el mismo KPI
            if (datos.tipoHoja !== 'Registro_pisos') {
                if (totalPct > 100) return { success: false, error: `Bonus (${totalPct}%) excede 100%` };
            }
        }

        // Guardar configuración inicial si se envía (Salario y %)
        if (datos.salario && datos.porcentaje) {
            guardarConfiguracion(datos.mes, datos.om, datos.lob, datos.salario, datos.porcentaje);
        }

        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hoja = ss.getSheetByName(datos.tipoHoja);
        if (!hoja) return { success: false, error: 'Hoja no encontrada' };

        // Buscar si ya existe el registro para sobrescribir
        const data = hoja.getDataRange().getValues();
        const mesNorm = normalizarKey(datos.mes);
        const lobNorm = normalizarKey(datos.lob);
        let filaIndex = -1;

        for (let i = 1; i < data.length; i++) {
            if (normalizarKey(data[i][0]) === mesNorm && normalizarKey(data[i][1]) === lobNorm) {
                filaIndex = i + 1;
                break;
            }
        }

        // Construir fila según tipo de hoja
        const fila = _construirFila(datos, hoja);

        if (filaIndex > 0) {
            hoja.getRange(filaIndex, 1, 1, fila.length).setValues([fila]);
            return { success: true, mensaje: 'Registro actualizado correctamente' };
        } else {
            hoja.appendRow(fila);
            return { success: true, mensaje: 'Nuevo registro creado exitosamente' };
        }

    } catch (error) {
        console.error('❌ Error guardando:', error);
        return { success: false, error: error.toString() };
    } finally {
        lock.releaseLock();
        console.log('🔓 Bloqueo liberado');
    }
}
function _construirFila(datos, hoja) {
    let fila = [];
    
    // ⭐ FUNCIÓN HELPER MEJORADA - AHORA INCLUYE PARTICIPACIÓN
    const norm = (arr, i) => {
        const k = (arr && arr[i]) ? arr[i] : {};
        let sc = k.score || '';
        
        // Normalizar score con comparación
        if (k.comparacion && sc) {
            sc = k.comparacion + sc.replace('%', '');
        }
        
        return { 
            n: k.nombre || '',
            s: sc,
            p: (k.porcentaje || '').toString().replace('%', ''),
            part: (k.participacion || '').toString().replace('%', '')
        };
    };

    if (datos.tipoHoja === 'Registro') {
        fila.push(datos.mes, datos.lob, datos.accm || '', datos.om || '');
        fila.push('');
        
        for (let i = 0; i < 2; i++) { 
            const k = norm(datos.keys, i); 
            fila.push(k.n, k.s); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 4; i++) { 
            const k = norm(datos.bonus, i); 
            fila.push(k.n, k.s, k.p); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 3; i++) { 
            const k = norm(datos.additional, i); 
            fila.push(k.n, k.s, k.p); 
        }
        
        fila.push('');
        
        for (let i = 0; i < 2; i++) { 
            const k = norm(datos.decreased, i); 
            fila.push(k.n, k.s, k.p); 
        }
    } 
    else if (datos.tipoHoja === 'Registro_pisos') {
        fila.push(datos.mes, datos.lob, datos.accm || '', datos.om || '');
        
        const numAgentes = contarAgentesPorLob(datos.lob);
        fila.push(numAgentes || '');
        
        // Keys: 6 keys * 2 columns = 12 columns.
        for (let i = 0; i < 6; i++) { const k = norm(datos.keys, i); fila.push(k.n, k.s); }
        
        fila.push('Bonus Structure');
        
        // Bonus: 17 KPIs * 4 columns (Name, Score, %, Part)
        for (let i = 0; i < 17; i++) { 
            const k = norm(datos.bonus, i); 
            fila.push(k.n, k.s, k.p, k.part); 
        }
        
        fila.push('Additional Incentive');
        
        // Additional: 5 KPIs * 4 columns
        for (let i = 0; i < 5; i++) { 
            const k = norm(datos.additional, i); 
            fila.push(k.n, k.s, k.p, k.part); 
        }
        
        fila.push('Decreased');
        
        // Decreased: 5 KPIs * 4 columns
        for (let i = 0; i < 5; i++) { 
            const k = norm(datos.decreased, i); 
            fila.push(k.n, k.s, k.p, k.part); 
        }
    }
    
    return fila;
}

function simularTodasLasLOBsDelUsuario() {
    try {
        console.log('🚀 Iniciando simulación de todas las LOBs...');
        const usuario = obtenerDatosUsuarioActivo();
        if (!usuario) return { success: false, error: 'Usuario no encontrado' };

        const lobsRegistradas = obtenerLobsRegistradas();
        if (lobsRegistradas.length === 0) return { success: false, error: 'No se encontraron LOBs registradas' };

        const lobsSimuladas = [];
        let totalGeneral = 0;
        let totalAgentesGeneral = 0;
        let totalBonoMaximo = 0;

        lobsRegistradas.forEach(lob => {
            const meses = obtenerMesesRegistrados(lob, usuario);
            if (meses.length === 0) return;
            const mesReciente = meses[meses.length - 1];
            const estructura = obtenerEstructuraRegistrada(lob, mesReciente, usuario);

            if (!estructura.encontrado) return;

            const resConfig = obtenerSalarioPorLob(lob);
            const salario = parseFloat(resConfig.salario) || 0;
            const porcentaje = parseFloat(resConfig.porcentaje) || 0;
            const hc = parseInt(estructura.hc) || contarAgentesPorLob(lob);

            if (salario === 0 || porcentaje === 0) return;

            const bonoMaximo = (salario * porcentaje) / 100;
            let totalBonus = 0;
            let totalAdditional = 0;

            if (estructura.kpis.bonusStructure) {
                estructura.kpis.bonusStructure.forEach(kpi => {
                    const pct = parseFloat(kpi.porcentaje) || 0;
                    totalBonus += (bonoMaximo * pct) / 100;
                });
            }
            if (estructura.kpis.additionalIncentives) {
                estructura.kpis.additionalIncentives.forEach(kpi => {
                    const pct = parseFloat(kpi.porcentaje) || 0;
                    totalAdditional += (bonoMaximo * pct) / 100;
                });
            }

            const totalPorAgente = totalBonus + totalAdditional;
            const totalLOB = totalPorAgente * hc;

            lobsSimuladas.push({
                lob: lob,
                mes: mesReciente,
                hc: hc,
                bonoPorAgente: totalPorAgente,
                totalBonus: totalBonus * hc,
                totalAdditional: totalAdditional * hc,
                totalLOB: totalLOB
            });

            totalGeneral += totalLOB;
            totalAgentesGeneral += hc;
            totalBonoMaximo += (bonoMaximo * hc);
        });

        return {
            success: true,
            lobsSimuladas: lobsSimuladas,
            totalGeneral: totalGeneral,
            resumen: {
                totalAgentes: totalAgentesGeneral,
                bonoMaximoTotal: totalBonoMaximo,
                totalBonus: lobsSimuladas.reduce((sum, l) => sum + l.totalBonus, 0),
                totalAdditional: lobsSimuladas.reduce((sum, l) => sum + l.totalAdditional, 0)
            },
            usuario: { nombre: usuario.nombre, rol: usuario.rol }
        };
    } catch (error) { return { success: false, error: error.toString() }; }
}

                    });
                });
            }
            
            // Calcular Additional
            if (estructura.kpis.additionalIncentives) {
                estructura.kpis.additionalIncentives.forEach(k => {
                    const pctRaw = parseFloat(k.porcentaje) || 0;
                    const pct = pctRaw > 1 ? pctRaw / 100 : pctRaw;
                    const monto = bonoMaximo * pct;
                    totalAdditional += monto;
                    kpisAdditional.push({
                        nombre: k.nombre,
                        score: k.score,
                        porcentaje: pct * 100,
                        montoPorAgente: monto
                    });
                });
            }
            
            const totalPorAgente = totalBonus + totalAdditional;
            const totalLOB = totalPorAgente * hc;
            const eficiencia = bonoMaximo > 0 ? ((totalPorAgente / bonoMaximo) * 100) : 0;
            
            Logger.log(`  💰 Total LOB: $${totalLOB.toLocaleString()}`);
            
            // ✅ Guardar con FECHA
            resultadosMap.set(lob, {
                lob: lob,
                mes: mesReciente,
                fecha: estructura.fecha || mesReciente, // ✅ FECHA DE REGISTRO/REGISTRO_PISOS
                hc: hc,
                salarioBase: salario,
                porcentajeBono: porcentaje,
                bonoMaximoPorAgente: bonoMaximo,
                bonoPorAgente: totalPorAgente,
                totalBonus: totalBonus * hc,
                totalAdditional: totalAdditional * hc,
                totalLOB: totalLOB,
                eficiencia: eficiencia.toFixed(1) + '%',
                kpisBonus: kpisBonus,
                kpisAdditional: kpisAdditional,
                kpisKeys: estructura.kpis.keys || [],
                kpisDecreased: estructura.kpis.decreasedKPIs || []
            });
            
            totalAgentesGeneral += hc;
            totalBonoMaximo += (bonoMaximo * hc);
        }
        
        const resultados = Array.from(resultadosMap.values());
        const totalGeneral = resultados.reduce((sum, lob) => sum + lob.totalLOB, 0);
        
        // ✅ LOGS FINALES
        Logger.log(`\n🎯 ========== RESULTADO FINAL SIMULACIÓN ==========`);
        Logger.log(`📊 OM: ${nombreOM}`);
        Logger.log(`📊 LOBs en Configuracion: ${lobsDelOM.length}`);
        Logger.log(`📊 LOBs procesadas (con registros): ${resultados.length}`);
        Logger.log(`👥 Total Agentes: ${totalAgentesGeneral}`);
        Logger.log(`💰 Inversión Total: $${totalGeneral.toLocaleString()}`);
        
        Logger.log(`\n📅 FECHAS DE CADA LOB:`);
        resultados.forEach((r, idx) => {
            Logger.log(`\n  ${idx + 1}. LOB: ${r.lob}`);
            Logger.log(`     📅 Fecha: "${r.fecha}" (Tipo: ${typeof r.fecha})`);
            Logger.log(`     📋 Mes: "${r.mes}"`);
            Logger.log(`     💰 Total: $${r.totalLOB.toLocaleString()}`);
        });
        
        Logger.log(`\n==================================================\n`);
        
        return {
            success: true,
            om: nombreOM,
            totalLOBs: lobsDelOM.length,
            lobsProcesadas: resultados.length,
            resultados: resultados.sort((a, b) => b.totalLOB - a.totalLOB),
            resumen: {
                totalAgentes: totalAgentesGeneral,
                inversionTotal: totalGeneral,
                bonoMaximoTotal: totalBonoMaximo,
                promedioPorAgente: totalAgentesGeneral > 0 ? totalGeneral / totalAgentesGeneral : 0
            },
            fechaSimulacion: new Date().toISOString()
        };
        
    } catch (error) {
        Logger.log(`❌ Error: ${error.message}`);
        return { success: false, error: error.toString() };
    }
}

function simularTodasLOBsPorRegion(region) {
    try {
        const lobsDeRegion = obtenerLobsPorRegion(region);
        if (!lobsDeRegion.length) return { success: false, error: 'Sin LOBs en región' };

        // El resto lógica es idéntica a porOM pero iterando lobsDeRegion
        // Replicando lógica simplificada
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const usuario = obtenerDatosUsuarioActivo();
        const resultadosMap = new Map();
        let totalAgentes = 0;
        let totalInv = 0;

        for (const lob of lobsDeRegion) {
            const meses = obtenerMesesRegistrados(lob, usuario);
            if (!meses.length) continue;
            const mes = meses[meses.length - 1];
            const est = obtenerEstructuraRegistrada(lob, mes, usuario);
            if (!est.encontrado) continue;

            const conf = obtenerSalarioPorLob(lob);
            const sal = parseFloat(conf.salario) || 0;
            const por = parseFloat(conf.porcentaje) || 0;
            const hc = contarAgentesPorLob(lob);

            if (!sal || !por || !hc) continue;

            const maxBono = (sal * por) / 100;
            let totalBonus = 0;
            let totalAdditional = 0;

            if (est.kpis.bonusStructure) est.kpis.bonusStructure.forEach(k => totalBonus += (maxBono * (parseFloat(k.porcentaje) || 0)) / 100);
            if (est.kpis.additionalIncentives) est.kpis.additionalIncentives.forEach(k => totalAdditional += (maxBono * (parseFloat(k.porcentaje) || 0)) / 100);

            const totBono = totalBonus + totalAdditional;
            const totLOB = totBono * hc;

            resultadosMap.set(lob, {
                lob, mes, hc, salarioBase: sal, porcentajeBono: por,
                totalLOB: totLOB,
                totalBonus: totalBonus * hc, // 🔥 FIX: Agregado
                totalAdditional: totalAdditional * hc, // 🔥 FIX: Agregado
                bonoPorAgente: totBono,
                eficiencia: (maxBono > 0 ? (totBono / maxBono * 100) : 0).toFixed(2)
            });
            totalAgentes += hc;
            totalInv += totLOB;
        }

        const resultados = Array.from(resultadosMap.values());
        return {
            success: true,
            region,
            resultados: resultados.sort((a, b) => b.totalLOB - a.totalLOB),
            lobsProcesadas: resultados.length, // 🔥 FIX: Agregado para que no salga undefined
            resumen: { totalAgentes, inversionTotal: totalInv }
        };
    } catch (e) { return { success: false, error: e.toString() }; }
}

function exportarSimulacionAExcel(datos) {
    try {
        const ss = SpreadsheetApp.getActiveSpreadsheet();
        const nombreHoja = 'Exportacion_Simulacion_' + new Date().toISOString().slice(0, 10);
        let hoja = ss.getSheetByName(nombreHoja);
        if (!hoja) hoja = ss.insertSheet(nombreHoja);
        else hoja.clear();

        if (datos.encabezados && datos.encabezados.length > 0) {
            hoja.getRange(1, 1, 1, datos.encabezados.length).setValues([datos.encabezados])
                .setBackground('#667eea').setFontColor('white').setFontWeight('bold');
        }
        if (datos.filas && datos.filas.length > 0) {
            hoja.getRange(2, 1, datos.filas.length, datos.filas[0].length).setValues(datos.filas);
        }
        hoja.autoResizeColumns(1, datos.encabezados ? datos.encabezados.length : 10);
        return ss.getUrl() + '#gid=' + hoja.getSheetId();
    } catch (error) { return null; }
}

/**
 * Obtiene las LOBs donde el OM está asignado en Registro/Registro_pisos
 * Busca en columna D (OM)
 */
function obtenerLobsRegistradasPorUsuario() {
    try {
        const usuario = obtenerDatosUsuarioActivo();
        if (!usuario) {
            Logger.log('❌ Usuario no encontrado');
            return [];
        }
        
        Logger.log(`\n🔍 ========== OBTENIENDO LOBs REGISTRADAS ==========`);
        Logger.log(`👤 Usuario: ${usuario.nombre}`);
        Logger.log(`📋 Rol: ${usuario.rol}`);
        
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojas = [
            ss.getSheetByName('Registro'),
            ss.getSheetByName('Registro_pisos')
        ];
        
        const lobsSet = new Set();
        const normalizar = s => s ? s.toString().normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim() : "";
        const usuarioNorm = normalizar(usuario.nombre);
        
        // ⭐ BUSCAR SEGÚN ROL
        hojas.forEach(hoja => {
            if (!hoja) return;
            
            const data = hoja.getDataRange().getValues();
            Logger.log(`\n📊 Buscando en ${hoja.getName()}: ${data.length - 1} registros`);
            
            for (let i = 1; i < data.length; i++) {
                const lobFila = String(data[i][1] || '').trim();   // Columna B - LOB
                const accmFila = String(data[i][2] || '').trim();  // Columna C - ACCM
                const omFila = String(data[i][3] || '').trim();    // Columna D - OM
                
                if (!lobFila) continue;
                
                let incluir = false;
                
                // ⭐ FILTRAR SEGÚN ROL
                if (usuario.rol === 'Reporting Analyst' || usuario.rol === 'Audit') {
                    incluir = true;
                    
                } else if (usuario.rol === 'OM') {
                    // ⭐ COMPARACIÓN EXACTA NORMALIZADA
                    const omFilaNorm = normalizar(omFila);
                    
                    if (omFilaNorm === usuarioNorm) {
                        incluir = true;
                        Logger.log(`✅ Fila ${i+1}: LOB="${lobFila}", OM="${omFila}" (Match: ${usuario.nombre})`);
                    } else {
                        Logger.log(`❌ Fila ${i+1}: LOB="${lobFila}", OM="${omFila}" (No match con ${usuario.nombre})`);
                    }
                    
                } else if (usuario.rol === 'ACCM') {
                    const accmFilaNorm = normalizar(accmFila);
                    
                    if (accmFilaNorm === usuarioNorm) {
                        incluir = true;
                        Logger.log(`✅ Fila ${i+1}: LOB="${lobFila}", ACCM="${accmFila}" ✓`);
                    }
                    
                } else if (usuario.rol === 'Supervisor') {
                    const lobsPermitidas = obtenerLobsPorSupervisor(usuario.nombre);
                    if (lobsPermitidas.includes(lobFila)) {
                        incluir = true;
                    }
                }
                
                if (incluir) {
                    lobsSet.add(lobFila);
                }
            }
        });
        
        const resultado = Array.from(lobsSet).sort();
        
        Logger.log(`\n📊 ========== RESUMEN ==========`);
        Logger.log(`✅ Total LOBs encontradas: ${resultado.length}`);
        Logger.log(`📋 LOBs: ${resultado.join(', ')}`);
        Logger.log(`================================\n`);
        
        return resultado;
        
    } catch (error) {
        Logger.log(`❌ Error en obtenerLobsRegistradasPorUsuario: ${error.message}`);
        return [];
    }
}

            return { encontrado: false };
        }
        
        const data = hojaConfig.getDataRange().getValues();
        const normalizar = s => s ? s.toString().normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim() : "";
        
        const lobNorm = normalizar(lob);
        const omNorm = normalizar(nombreOM);
        
        Logger.log(`🔍 Buscando config: LOB="${lob}", OM="${nombreOM}"`);
        
        for (let i = 1; i < data.length; i++) {
            const omFila = String(data[i][1] || '').trim();      // Columna B - OM
            const lobFila = String(data[i][2] || '').trim();     // Columna C - LOB
            
            const omFilaNorm = normalizar(omFila);
            const lobFilaNorm = normalizar(lobFila);
            
            // ⭐ BUSCAR POR OM Y LOB
            if (omFilaNorm === omNorm && lobFilaNorm === lobNorm) {
                const config = {
                    encontrado: true,
                    mes: String(data[i][0] || '').trim(),        // Columna A
                    om: omFila,                                   // Columna B
                    lob: lobFila,                                 // Columna C
                    salario: parseFloat(data[i][3]) || 0,        // Columna D
                    porcentaje: parseFloat(data[i][4]) || 0      // Columna E
                };
                
                Logger.log(`✅ Config encontrada en fila ${i+1}:`);
                Logger.log(`   Mes: ${config.mes}`);
                Logger.log(`   OM: ${config.om}`);
                Logger.log(`   LOB: ${config.lob}`);
                Logger.log(`   Salario: $${config.salario.toLocaleString()}`);
                Logger.log(`   % Bono: ${config.porcentaje}%`);
                
                return config;
            }
        }
        
        Logger.log(`⚠️ No se encontró config para LOB="${lob}" y OM="${nombreOM}"`);
        return { encontrado: false };
        
    } catch (error) {
        Logger.log(`❌ Error en obtenerConfiguracionDesdeCofiguracion: ${error.message}`);
        return { encontrado: false, error: error.message };
    }
}



/**

// 🎯 FUNCIÓN BACKEND: SIMULAR CON MÉTRICAS REALES DEL CDM (CON SOPORTE PISOS)
// ============================================================
function simularConMetricasReales(lob, mes, numeroMes) {
  try {

    console.log(`   Número Mes: ${numeroMes}`);

    // ========================================

    if (!estructura || !estructura.encontrado) {
      return {
        success: false,
        error: 'No se encontró estructura registrada para esta LOB y mes'
      };
    }
    
    console.log('✅ Estructura encontrada:', estructura.lob);
    
    // ========================================
    // 2️⃣ VALIDAR CONFIGURACIÓN
    // ========================================
    if (!estructura.salario || !estructura.porcentaje) {
      return {
        success: false,
        error: 'Estructura sin configuración de salario/porcentaje'
      };
    }
    
    const bonoMaximo = (estructura.salario * estructura.porcentaje) / 100;
    
    console.log(`💰 Configuración:`);
    console.log(`   Salario: $${estructura.salario}`);
    console.log(`   % Bono: ${estructura.porcentaje}%`);
    console.log(`   Bono Máximo: $${bonoMaximo}`);
    
    // ========================================
    // 3️⃣ OBTENER AGENTES DE LA LOB
    // ========================================
    const ssCDM = SpreadsheetApp.openById('1PtYuAb5C2kIoX8sE7dOvt1nVS-aLamZLOJQUSbTBBCw');
    const hojaHC = ssCDM.getSheetByName('H_C');
    
    if (!hojaHC) {
      return { success: false, error: 'No se encontró la hoja H_C' };
    }
    
    const dataHC = hojaHC.getDataRange().getValues();
    const agentesLOB = [];

      const lobFila = String(dataHC[i][1] || '').trim();
      if (lobFila === lob) {
        agentesLOB.push({
          bmsId: String(dataHC[i][0]).trim(),
          nombre: String(dataHC[i][2] || '').trim(),
          lob: lobFila
        });
      }
    }
    
    console.log(`👥 Total agentes en LOB: ${agentesLOB.length}`);
    
    if (agentesLOB.length === 0) {
      return { success: false, error: 'No se encontraron agentes para esta LOB' };
    }

    // 4️⃣ OBTENER MÉTRICAS DEL CDM
    // ========================================
    const hojaCDM = ssCDM.getSheetByName('CDM');
    
    if (!hojaCDM) {
      return { success: false, error: 'No se encontró la hoja CDM' };
    }
    
    const dataCDM = hojaCDM.getDataRange().getValues();
    const headers = dataCDM[0];
    
    const bmsIdIndex = headers.indexOf('BMS ID');
    const mesIndex = headers.indexOf('Mes');

      return { success: false, error: 'Estructura de CDM inválida' };
    }
    
    // Crear mapa de métricas por agente
    const metricasPorAgente = {};
    
    for (let i = 1; i < dataCDM.length; i++) {
      const fila = dataCDM[i];
      const bmsId = String(fila[bmsIdIndex]).trim();
      const mesCDM = parseInt(fila[mesIndex]) || 0;

          if (nombreMetrica !== 'BMS ID' && nombreMetrica !== 'Mes') {

    // 5️⃣ AGRUPAR KPIs POR NOMBRE (DETECTAR PISOS)

      decreasedKPIs: {}

    // Agrupar Bonus Structure
    if (estructura.kpis.bonusStructure) {

        if (!kpisPorNombre.additionalIncentives[nombre]) {
          kpisPorNombre.additionalIncentives[nombre] = [];

            if (!cumple) {
              cumpleKeys = false;
            }
          } else {
            cumpleKeys = false;
          }
        });
      }
      
      if (!cumpleKeys) {
        agentesAfectadosPorKeys++;
      }
      
      // ⭐ CALCULAR BONUS STRUCTURE CON PISOS
      Object.keys(kpisPorNombre.bonusStructure).forEach(nombreKPI => {

          return; // Sin datos
        }

        for (let i = 0; i < kpis.length; i++) {
          const kpi = kpis[i];
          const cumple = evaluarCondicion(scoreAgente, kpi.comparacion, kpi.score);
          
          if (cumple) {
            pisoCobrado = kpi;
            break; // ⭐ Toma el primero que cumpla (el más alto)
          }
        }
        
        if (pisoCobrado && cumpleKeys) {
          let porcentajeKPI = parseFloat(pisoCobrado.porcentaje) || 0;
          let participacion = parseFloat(pisoCobrado.participacion) || 100;
          
          // Normalizar si están en formato decimal
          if (porcentajeKPI > 0 && porcentajeKPI <= 1) {
            porcentajeKPI = porcentajeKPI * 100;
          }
          
          if (participacion > 0 && participacion <= 1) {
            participacion = participacion * 100;
          }

          const montoPiso = (bonoMaximo * porcentajeKPI / 100) * (participacion / 100);
          
          totalBonus += montoPiso;

          let kpiExistente = desglosePorCategoria.bonusStructure.find(k => k.kpi === nombreKPI);
          if (!kpiExistente) {
            kpiExistente = {
              kpi: nombreKPI,
              porcentaje: porcentajeKPI,
              participacion: participacion,
              totalCumplen: 0,
              montoTotal: 0
            };
            desglosePorCategoria.bonusStructure.push(kpiExistente);
          }
          
          kpiExistente.totalCumplen++;
          kpiExistente.montoTotal += montoPiso;
        }
      });
      
      // ⭐ CALCULAR ADDITIONAL CON PISOS
      Object.keys(kpisPorNombre.additionalIncentives).forEach(nombreKPI => {
        const kpis = kpisPorNombre.additionalIncentives[nombreKPI];
        const scoreAgente = metricasAgente[nombreKPI];
        
        if (scoreAgente === undefined || scoreAgente === null || scoreAgente === '') {
          return;
        }
        
        let pisoCobrado = null;
        
        for (let i = 0; i < kpis.length; i++) {
          const kpi = kpis[i];
          const cumple = evaluarCondicion(scoreAgente, kpi.comparacion, kpi.score);
          
          if (cumple) {
            pisoCobrado = kpi;
            break;
          }
        }
        
        if (pisoCobrado && cumpleKeys) {
          let porcentajeKPI = parseFloat(pisoCobrado.porcentaje) || 0;
          let participacion = parseFloat(pisoCobrado.participacion) || 100;
          
          if (porcentajeKPI > 0 && porcentajeKPI <= 1) {
            porcentajeKPI = porcentajeKPI * 100;
          }
          
          if (participacion > 0 && participacion <= 1) {
            participacion = participacion * 100;
          }
          
          const montoPiso = (bonoMaximo * porcentajeKPI / 100) * (participacion / 100);
          
          totalAdditional += montoPiso;
          
          let kpiExistente = desglosePorCategoria.additionalIncentives.find(k => k.kpi === nombreKPI);
          if (!kpiExistente) {
            kpiExistente = {
              kpi: nombreKPI,
              porcentaje: porcentajeKPI,
              participacion: participacion,
              totalCumplen: 0,
              montoTotal: 0
            };
            desglosePorCategoria.additionalIncentives.push(kpiExistente);
          }
          
          kpiExistente.totalCumplen++;
          kpiExistente.montoTotal += montoPiso;
        }
      });
      
      // ⭐ CALCULAR DECREASED CON PISOS
      Object.keys(kpisPorNombre.decreasedKPIs).forEach(nombreKPI => {
        const kpis = kpisPorNombre.decreasedKPIs[nombreKPI];
        const scoreAgente = metricasAgente[nombreKPI];
        
        if (scoreAgente === undefined || scoreAgente === null || scoreAgente === '') {
          return;
        }
        
        let pisoCobrado = null;
        
        for (let i = 0; i < kpis.length; i++) {
          const kpi = kpis[i];
          const cumple = evaluarCondicion(scoreAgente, kpi.comparacion, kpi.score);
          
          if (cumple) {
            pisoCobrado = kpi;
            break;
          }
        }
        
        if (pisoCobrado) {
          let porcentajeKPI = parseFloat(pisoCobrado.porcentaje) || 0;
          let participacion = parseFloat(pisoCobrado.participacion) || 100;
          
          if (porcentajeKPI > 0 && porcentajeKPI <= 1) {
            porcentajeKPI = porcentajeKPI * 100;
          }
          
          if (participacion > 0 && participacion <= 1) {

              participacion: participacion,
              totalCumplen: 0,
              montoTotal: 0
            };
            desglosePorCategoria.decreasedKPIs.push(kpiExistente);
          }
          
          kpiExistente.totalCumplen++;
          kpiExistente.montoTotal += montoPiso;
        }

      // ⭐ CALCULAR BONO FINAL

        resultadosPorAgente.push({

      }
    });
    
    // ⭐ TOP 10 GANADORES
    const top10 = resultadosPorAgente
      .sort((a, b) => b.bonoFinal - a.bonoFinal)
      .slice(0, 10);
    
    // ========================================
    // 8️⃣ RESULTADO FINAL
    // ========================================
    const inversionTeorica = bonoMaximo * agentesLOB.length;

    const eficienciaReal = inversionTeorica > 0 ? (inversionTotal / inversionTeorica) * 100 : 0;
    
    console.log(`✅ Simulación completada:`);
    console.log(`   Inversión teórica: $${inversionTeorica}`);
    console.log(`   Inversión real: $${inversionTotal}`);
    console.log(`   Ahorro: $${ahorro}`);
    console.log(`   Eficiencia: ${eficienciaReal.toFixed(2)}%`);
    
    return {
      success: true,
      lob: lob,
      mes: mes,
      hc: agentesLOB.length,
      inversionTeorica: inversionTeorica,
      inversionReal: inversionTotal,
      ahorro: ahorro,
      eficienciaReal: eficienciaReal,
      estadisticas: {
        agentesCobran: agentesCobran,
        agentesNoCobran: agentesNoCobran,
        agentesAfectadosPorKeys: agentesAfectadosPorKeys,
        tasaCumplimiento: agentesLOB.length > 0 ? (agentesCobran / agentesLOB.length * 100) : 0
      },
      desglosePorCategoria: desglosePorCategoria,
      top10Ganadores: top10,
      configuracion: {
        salario: estructura.salario,
        porcentajeBono: estructura.porcentaje,
        bonoMaximo: bonoMaximo
      }
    };
    
  } catch (error) {
    console.error('❌ Error en simularConMetricasReales:', error);
    return {
      success: false,
      error: error.toString()
    };
  }
}

/**
 * Calcula el bono escalonado según participación
 * Ejemplo: CSAT con 20% base, pero 3 niveles de participación (5%, 10%, 15%)
 */
function calcularBonoEscalonado(kpisCategoria, bonoMaximo, scoreAgente) {
    const kpisAgrupados = {};
    
    // ⭐ AGRUPAR KPIs CON MISMO NOMBRE
    kpisCategoria.forEach(kpi => {
        const nombre = kpi.nombre;
        if (!kpisAgrupados[nombre]) {
            kpisAgrupados[nombre] = [];
        }
        kpisAgrupados[nombre].push(kpi);
    });
    
    let bonoTotal = 0;
    
    // ⭐ PROCESAR CADA GRUPO
    Object.keys(kpisAgrupados).forEach(nombreKPI => {
        const grupo = kpisAgrupados[nombreKPI];
        
        // Si solo hay 1 KPI (sin escalones), usar lógica normal
        if (grupo.length === 1) {
            const kpi = grupo[0];
            if (cumpleCondicion(scoreAgente[nombreKPI], kpi.score, kpi.comparacion)) {
                const pct = parseFloat(kpi.porcentaje) || 0;
                bonoTotal += (bonoMaximo * pct) / 100;
            }
            return;
        }
        
        // ⭐ LÓGICA ESCALONADA: Ordenar por score descendente
        const grupoOrdenado = grupo.sort((a, b) => {
            const scoreA = parseFloat(a.score.replace(/[<>=]/g, '')) || 0;
            const scoreB = parseFloat(b.score.replace(/[<>=]/g, '')) || 0;
            return scoreB - scoreA; // Mayor a menor
        });
        
        // Buscar el nivel más alto que cumple
        for (const kpi of grupoOrdenado) {
            if (cumpleCondicion(scoreAgente[nombreKPI], kpi.score, kpi.comparacion)) {
                const pctBase = parseFloat(kpi.porcentaje) || 0;
                const participacion = parseFloat(kpi.participacion) || 100; // Si no tiene, paga 100%
                
                const bonoParcial = (bonoMaximo * pctBase * participacion) / 10000; // (% base * % participación) / 100^2
                bonoTotal += bonoParcial;
                
                Logger.log(`✅ ${nombreKPI}: Score=${scoreAgente[nombreKPI]}, Nivel=${kpi.score}, Base=${pctBase}%, Part=${participacion}% → $${bonoParcial}`);
                break; // Solo pagar el nivel más alto alcanzado
            }
        }
    });
    
    return bonoTotal;
}

function cumpleCondicion(valorAgente, score, comparacion) {
    const val = parseFloat(valorAgente) || 0;
    const meta = parseFloat(score.replace(/[<>=]/g, '')) || 0;
    
    switch(comparacion) {
        case '<': return val < meta;
        case '<=': return val <= meta;
        case '>': return val > meta;
        case '>=': return val >= meta;
        case '=': return val === meta;
        default: return false;
    }
}
