// 🗄️ BUSCAR AGENTE CON CACHÉ
// ============================================================
function buscarAgentePorCorreo(correo) {
  const cache = CacheService.getUserCache();
  const cacheKey = `agente_${correo.toLowerCase()}`;
  
  // Intentar obtener del caché
  const cached = cache.get(cacheKey);
  if (cached) {
    console.log('✅ Agente obtenido del caché');
    return JSON.parse(cached);
  }
  
  // Si no está en caché, buscar en la hoja
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const hojaUsuario = ss.getSheetByName('Usuario');
  
  if (!hojaUsuario) {
    throw new Error('No se encuentra la hoja "Usuario"');
  }
  
  const datosUsuario = hojaUsuario.getDataRange().getValues();
  const correoNormalizado = correo.toLowerCase().trim();
  
  // Buscar por correo (columna C)
  for (let i = 1; i < datosUsuario.length; i++) {
    const correoFila = String(datosUsuario[i][2] || '').trim().toLowerCase();
    if (correoFila === correoNormalizado) {
      const agente = {
        bmsId: String(datosUsuario[i][0] || '').trim(),
        nombre: String(datosUsuario[i][1] || '').trim(),
        correo: String(datosUsuario[i][2] || '').trim(),
        lob: String(datosUsuario[i][4] || '').trim()
      };
      
      // Guardar en caché
      cache.put(cacheKey, JSON.stringify(agente), CACHE_TIME_USUARIOS);
      console.log('✅ Agente guardado en caché');
      
      return agente;
    }
  }
  
  return null;
}

function obtenerEstructuraLOB(ss, lob) {
  console.log(`🔍 Buscando estructura para LOB: ${lob}`);
  
  // Intentar primero en Registro_pisos
  const hojaPisos = ss.getSheetByName('Registro_pisos');
  if (hojaPisos) {
    const estructura = buscarEstructuraEnHoja(hojaPisos, lob, 'pisos');
    if (estructura) {
      console.log('✅ Estructura encontrada en Registro_pisos');
      return estructura;
    }
  }
  
  // Si no, buscar en Registro normal
  const hojaRegistro = ss.getSheetByName('Registro');
  if (hojaRegistro) {
    const estructura = buscarEstructuraEnHoja(hojaRegistro, lob, 'normal');
    if (estructura) {
      console.log('✅ Estructura encontrada en Registro');
      return estructura;
    }
  }
  
  console.log('❌ No se encontró estructura para esta LOB');
  return null;
}

function buscarEstructuraEnHoja(hoja, lobBuscado, tipo) {
  const datos = hoja.getDataRange().getValues();
  const lobNormalizado = lobBuscado.toLowerCase().trim();
  
  // Buscar la última fila para esta LOB
  let ultimaFila = -1;
  
  for (let i = 1; i < datos.length; i++) {
    const lobFila = String(datos[i][1] || '').toLowerCase().trim();
    if (lobFila === lobNormalizado) {
      ultimaFila = i;
    }
  }
  
  if (ultimaFila === -1) return null;
  
  const fila = datos[ultimaFila];
  
  // Parsear según tipo de hoja
  if (tipo === 'normal') {
    return parsearEstructuraNormal(fila);
  } else {
    return parsearEstructuraPisos(fila);
  }
}

// 🔧 PARSEAR ESTRUCTURA NORMAL
// ============================================================
function parsearEstructuraNormal(fila) {
  const estructura = {
    tipo: 'normal',
    mes: String(fila[0] || ''),
    keys: [],
    bonus: [],
    additional: [],
    decreased: []
  };
  
  // Keys (columnas 5-8) - pares de nombre/score
  for (let i = 5; i <= 8; i += 2) {
    if (fila[i]) {
      const kpi = parsearKPI(fila[i], fila[i + 1]);
      if (kpi) estructura.keys.push(kpi);
    }
  }
  
  // Bonus (columnas 10-21) - tríos de nombre/score/porcentaje
  for (let i = 10; i <= 21; i += 3) {
    if (fila[i]) {
      const kpi = parsearKPI(fila[i], fila[i + 1], fila[i + 2]);
      if (kpi) estructura.bonus.push(kpi);
    }
  }
  
  // Additional (columnas 23-31)
  for (let i = 23; i <= 31; i += 3) {
    if (fila[i]) {
      const kpi = parsearKPI(fila[i], fila[i + 1], fila[i + 2]);
      if (kpi) estructura.additional.push(kpi);
    }
  }
  
  // Decreased (columnas 33-38)
  for (let i = 33; i <= 38; i += 3) {
    if (fila[i]) {
      const kpi = parsearKPI(fila[i], fila[i + 1], fila[i + 2]);
      if (kpi) estructura.decreased.push(kpi);
    }
  }
  
  return estructura;
}

// ============================================================
// 🔧 PARSEAR ESTRUCTURA PISOS
// ============================================================
function parsearEstructuraPisos(fila) {
  const estructura = {
    tipo: 'pisos',
    mes: String(fila[0] || ''),
    keys: [],
    bonus: [],
    additional: [],
    decreased: []
  };
  
  // Keys (columnas 5-10)
  for (let i = 5; i <= 10; i += 2) {
    if (fila[i] && String(fila[i]) !== 'Bonus Structure') {
      const kpi = parsearKPI(fila[i], fila[i + 1]);
      if (kpi) estructura.keys.push(kpi);
    }
  }
  
  // Bonus (columnas 12-50)
  for (let i = 12; i <= 50; i += 3) {
    if (fila[i] && String(fila[i]) !== 'Additional Incentive') {
      const kpi = parsearKPI(fila[i], fila[i + 1], fila[i + 2]);
      if (kpi) estructura.bonus.push(kpi);
    }
  }
  
  // Additional (columnas 52-60)
  for (let i = 52; i <= 60; i += 3) {
    if (fila[i] && String(fila[i]) !== 'Decreased') {
      const kpi = parsearKPI(fila[i], fila[i + 1], fila[i + 2]);
      if (kpi) estructura.additional.push(kpi);
    }
  }
  
  // Decreased (columnas 62-73)
  for (let i = 62; i <= 73; i += 3) {
    if (fila[i]) {
      const kpi = parsearKPI(fila[i], fila[i + 1], fila[i + 2]);
      if (kpi) estructura.decreased.push(kpi);
    }
  }
  
  return estructura;
}

// ============================================================
// 🎯 PARSEAR KPI INDIVIDUAL (CON DETECCIÓN DE COMPARADOR)
// ============================================================
function parsearKPI(nombre, scoreRaw, porcentajeRaw) {
  if (!nombre) return null;
  
  const nombreStr = String(nombre).trim();
  const scoreStr = String(scoreRaw || '').trim();
  
  // Detectar comparador en el score
  const resultado = detectarComparadorYValor(scoreStr);
  
  const kpi = {
    nombre: nombreStr,
    comparacion: resultado.comparador,
    score: resultado.valor,
    scoreOriginal: scoreStr
  };
  
  // Si hay porcentaje, agregarlo
  if (porcentajeRaw !== undefined) {
    kpi.porcentaje = String(porcentajeRaw || '0').trim();
  }
  
  return kpi;
}

// ============================================================
// 🔍 DETECTAR COMPARADOR Y VALOR
// ============================================================
function detectarComparadorYValor(scoreStr) {
  if (!scoreStr) {
    return { comparador: '>=', valor: '' };
  }
  
  // Limpiar espacios
  const str = scoreStr.trim();
  
  // Patrones de comparadores
  const patrones = [
    { regex: /^>=\s*(.+)$/, comparador: '>=' },
    { regex: /^<=\s*(.+)$/, comparador: '<=' },
    { regex: /^>\s*(.+)$/, comparador: '>' },
    { regex: /^<\s*(.+)$/, comparador: '<' },
    { regex: /^=\s*(.+)$/, comparador: '=' },
    { regex: /^==\s*(.+)$/, comparador: '=' }
  ];
  
  // Intentar cada patrón
  for (const patron of patrones) {
    const match = str.match(patron.regex);
    if (match) {
      return {
        comparador: patron.comparador,
        valor: match[1].trim()
      };
    }
  }
  
  // Si empieza con número o %, asumir >= por defecto
  if (/^[\d%]/.test(str)) {
    return { comparador: '>=', valor: str };
  }
  
  // Por defecto
  return { comparador: '>=', valor: str };
}

// 💰 OBTENER CONFIGURACIÓN SALARIAL
// ============================================================
function obtenerConfiguracionSalarial(ss, lob) {
  const hojaConfig = ss.getSheetByName('Configuracion');
  if (!hojaConfig) {
    console.warn('⚠️ No se encuentra hoja Configuracion');
    return { salario: 0, porcentaje: 0 };
  }
  
  const datos = hojaConfig.getDataRange().getValues();
  const lobNormalizado = lob.toLowerCase().trim();
  
  for (let i = 1; i < datos.length; i++) {
    const lobFila = String(datos[i][2] || '').toLowerCase().trim();
    if (lobFila === lobNormalizado) {
      let porcentaje = parseFloat(datos[i][4]) || 0;
      
      const config = {
        salario: parseFloat(datos[i][3]) || 0,
        porcentaje: porcentaje
      };
      console.log('✅ Configuración salarial encontrada:', config);
      return config;
    }
  }
  
  console.warn('⚠️ No se encontró configuración salarial para', lob);
  return { salario: 0, porcentaje: 0 };
}

    const startTime = new Date().getTime();
    
    const correo = Session.getActiveUser().getEmail();
    console.log('📧 Correo de sesión:', correo);
    
    if (!correo) {
      throw new Error('No se pudo obtener el correo de sesión');
    }
    
    // Buscar agente (primero en caché)
    let agenteEncontrado = buscarAgentePorCorreo(correo);
    
    if (!agenteEncontrado) {
      throw new Error('No estás registrado en el sistema');
    }
    
    console.log('✅ Agente encontrado:', agenteEncontrado.nombre, '- LOB:', agenteEncontrado.lob);
    
    // Obtener estructura
    let estructura = obtenerEstructuraLOBOptimizada(agenteEncontrado.lob);
    
    // Si no hay estructura, limpiar caché y reintentar UNA VEZ
    if (!estructura) {
      console.log('⚠️ No se encontró estructura, limpiando caché...');
      
      // Limpiar caché del agente y de la estructura
      const userCache = CacheService.getUserCache();
      const scriptCache = CacheService.getScriptCache();
      
      userCache.remove(`agente_${correo.toLowerCase()}`);
      scriptCache.remove(`estructura_${agenteEncontrado.lob.toLowerCase()}`);
      scriptCache.remove(`config_${agenteEncontrado.lob.toLowerCase()}`);
      
      // Reintentar búsqueda desde Sheets
      agenteEncontrado = buscarAgentePorCorreo(correo);
      estructura = obtenerEstructuraLOBOptimizada(agenteEncontrado.lob);
      
      if (!estructura) {
        throw new Error(`No hay estructura definida para tu LOB: ${agenteEncontrado.lob}`);
      }
    }
    
    // Obtener configuración
    const configuracion = obtenerConfiguracionSalarialOptimizada(agenteEncontrado.lob);
    
    const endTime = new Date().getTime();
    console.log(`⏱️ Tiempo de ejecución: ${endTime - startTime}ms`);
    
    return {
      success: true,
      agente: agenteEncontrado,
      estructura: estructura,
      configuracion: configuracion
    };
    
  } catch (error) {
    console.error('❌ Error:', error.message);
    return {
      success: false,
      error: error.message
    };
  }
}

    console.log('🎉 CACHÉ COMPLETAMENTE LIMPIADO');

    return {

      mensaje: 'Caché completamente limpiado',
