function doPost(e) {

  try {
    if (!e || !e.parameter) {
      throw new Error('No se recibieron parámetros.');
    }

    var userIdInput = e.parameter.id ? e.parameter.id.trim() : '';
    var passwordInput = e.parameter.password ? e.parameter.password.trim() : '';

    if (!userIdInput || !passwordInput) {
      return renderLoginWithMessage('Por favor, ingrese BMS ID y contraseña.');
    }

    var userData = ss.getDataRange().getValues();
    var activeUserEmail = Session.getActiveUser().getEmail();
    var userFound = false;

    for (var i = 1; i < userData.length; i++) {
      if (userData[i].length < 4) continue;

      var userId = String(userData[i][0]).trim();
      var userEmail = String(userData[i][2]).trim();
      var userProfile = String(userData[i][3]).trim();

      if (userId === userIdInput && userEmail === activeUserEmail && userId === passwordInput) {
        userFound = true;
        var role = userProfile;
        var username = userData[i][1];
        return renderUserPage(role, username, role);
      }
    }

    return renderLoginWithMessage('BMS ID incorrecto, el correo no coincide o la contraseña es incorrecta.');
  } catch (error) {
    Logger.log('Error en doPost: ' + error.message);
    return HtmlService.createHtmlOutput('Error en el inicio de sesión: ' + error.message);
  }
}

function cambiarTipoAcceso(email, tipoAcceso) {
  try {
    const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
    const hojaUsuario = ss.getSheetByName('Usuario');
    const data = hojaUsuario.getDataRange().getValues();
    
    Logger.log(`🔄 Intentando cambiar tipo de acceso para: ${email} a ${tipoAcceso}`);
    
    // Validar tipo de acceso
    const tiposValidos = ['Registro', 'Audit', 'OM'];
    if (!tiposValidos.includes(tipoAcceso)) {
      return {
        success: false,
        error: 'Tipo de acceso inválido. Debe ser: Registro, Audit o OM'
      };
    }
    
    for (let i = 1; i < data.length; i++) {
      const emailFila = String(data[i][2]).toLowerCase().trim();
      
      if (emailFila === email.toLowerCase().trim()) {
        // Columna D (índice 3) es Rol
        hojaUsuario.getRange(i + 1, 4).setValue(tipoAcceso);
        
        Logger.log(`✅ Tipo de acceso actualizado: ${tipoAcceso} para ${email}`);
        
        return {
          success: true,
          mensaje: `Tipo de acceso cambiado a: ${tipoAcceso}`
        };
      }
    }
    
    Logger.log(`⚠️ Usuario no encontrado: ${email}`);
    return {
      success: false,
      error: 'Usuario no encontrado'
    };
    
  } catch (error) {
    Logger.log(`❌ Error en cambiarTipoAcceso: ${error.message}`);
    return {
      success: false,
      error: error.message
    };
  }
}

function obtenerAgentesPermitidos() {
  const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
  const hojaHC = ss.getSheetByName('H_C');
  
  if (!hojaHC) {
    Logger.log('❌ ERROR: No se encontró la hoja H_C');
    return {};
  }
  
  const dataHC = hojaHC.getDataRange().getValues();
  const agentes = {};
  
  const normalizar = s => {
    if (!s) return "";
    return s.toString()
      .normalize("NFD")
      .replace(/[\u0300-\u036f]/g, "")
      .toLowerCase()
      .trim()
      .replace(/\s+/g, ' ');
  };
  
  Logger.log("📋 ========== CARGANDO AGENTES DE H_C ==========");
  Logger.log("Headers H_C:", dataHC[0]);
  
  for (let i = 1; i < dataHC.length; i++) {
    const bms = String(dataHC[i][0]).trim();           // Columna A: BMS ID
    const lobOriginal = dataHC[i][1];                  // Columna B: LOB
    const nombreHC = String(dataHC[i][2]).trim();      // ⭐ Columna C: NOMBRE
    const rolHC = String(dataHC[i][3]).trim();         // ⭐⭐⭐ Columna D: ROL
    const horasSemanalesRaw = dataHC[i][9];            // ⭐ Columna J: HORAS SEMANALES
    
    // ⭐ DEBUG: Primeras 5 filas
    if (i <= 5) {
      Logger.log(`\n🔍 Fila ${i+1}:`);
      Logger.log(`  A (BMS): "${bms}"`);
      Logger.log(`  B (LOB): "${lobOriginal}"`);
      Logger.log(`  C (Nombre): "${nombreHC}"`);
      Logger.log(`  D (ROL): "${rolHC}"   ⭐⭐⭐ DEBE CARGARSE`);
      Logger.log(`  J (Horas): ${horasSemanalesRaw}h`);
    }
    
    if (!bms || !lobOriginal) {
      if (i <= 10) {
        Logger.log(`⚠️ Fila ${i+1} ignorada - BMS o LOB vacío`);
      }
      continue;
    }
    
    // ⭐ Validar que nombreHC no esté vacío
    if (!nombreHC || nombreHC === '') {
      if (i <= 10) {
        Logger.log(`⚠️ Fila ${i+1}: Nombre vacío, usando fallback "Agente ${bms}"`);
      }
    }
    
    // ⭐ Validar que rolHC no esté vacío
    if (!rolHC || rolHC === '') {
      if (i <= 10) {
        Logger.log(`⚠️ Fila ${i+1}: ROL vacío para BMS ${bms}`);
      }
    }
    
    const lobNormalizada = normalizar(lobOriginal);
    const horasSemanales = parseFloat(horasSemanalesRaw) || 42;
    
    // ⭐⭐⭐ GUARDAR: NOMBRE (col C) Y ROL (col D)
    agentes[bms] = { 
      nombre: nombreHC || `Agente ${bms}`,       // ⭐ Columna C
      rol: rolHC || 'Sin rol',                   // ⭐⭐⭐ Columna D (ROL)
      lob: lobNormalizada,
      lobOriginal: String(lobOriginal).trim(),
      horasSemanales: horasSemanales             // ⭐ Columna J
    };
    
    // ⭐ DEBUG: Mostrar primeros 3 agentes agregados
    if (Object.keys(agentes).length <= 3) {
      Logger.log(`\n✅ Agente ${Object.keys(agentes).length} agregado:`);
      Logger.log(`   BMS: ${bms}`);
      Logger.log(`   NOMBRE: "${agentes[bms].nombre}"`);
      Logger.log(`   ROL: "${agentes[bms].rol}"   ⭐⭐⭐ VERIFICAR`);
      Logger.log(`   LOB: "${agentes[bms].lobOriginal}"`);
      Logger.log(`   Horas: ${agentes[bms].horasSemanales}h`);
    }
  }
  
  Logger.log(`\n📊 TOTAL AGENTES CARGADOS: ${Object.keys(agentes).length}`);
  Logger.log("========================================\n");
  
  return agentes;
}

  lobs.forEach((lob, i) => {
    Logger.log(`   ${i+1}. "${lob}"`);

function obtenerRegistroPorUsuarioYLob(nombre, lob, rol = null) {
  const ssUsuario = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
  const hojasBuscar = ['Registro', 'Registro_pisos'];
  
  const normalizeString = (str) => {
    return (str && typeof str === 'string') ? str.replace(/\s+/g, ' ').trim() : '';
  };

  for (const hojaName of hojasBuscar) {
    const hoja = ssUsuario.getSheetByName(hojaName);
    if (!hoja) continue;
    
    const data = hoja.getDataRange().getValues();
    
    for (let i = 1; i < data.length; i++) {
      let nombreRegistroRaw;
      const lobRegistroRaw = data[i][1];
      
      if (rol === 'ACCM') {
        nombreRegistroRaw = data[i][2];
      } else {
        nombreRegistroRaw = data[i][3];
      }
      
      if (nombreRegistroRaw == null || lobRegistroRaw == null) {
        continue;
      }
      
      const nombreRegistro = normalizeString(nombreRegistroRaw);
      const lobRegistro = normalizeString(lobRegistroRaw);
      
      if (normalizeString(nombre) === nombreRegistro && normalizeString(lob) === lobRegistro) {
        Logger.log(`✅ ${rol || 'Usuario'} encontrado: ${nombre} en LOB: ${lob}, Hoja: ${hojaName}`);
        return { registro: data[i], hoja: hojaName };
      }
    }
  }
  
  Logger.log(`⚠️ No se encontró registro para ${rol || 'usuario'}: ${nombre} en LOB: ${lob}`);
  return null;
}

// ========================================

function obtenerFechasPorLob(lobSeleccionada) {
  const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
  const hojasBuscar = ['Registro', 'Registro_pisos'];
  const fechasSet = new Set();
  
  if (!lobSeleccionada) return [];
  
  const normalizar = s => s ? s.toString().normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim() : "";
  const lobNorm = normalizar(lobSeleccionada);
  
  Logger.log("✅ Buscando fechas para LOB:", lobNorm);
  
  hojasBuscar.forEach(nombreHoja => {
    const hoja = ss.getSheetByName(nombreHoja);
    if (!hoja) return;
    
    const data = hoja.getDataRange().getValues();
    
    for (let i = 1; i < data.length; i++) {
      const fila = data[i];
      const fecha = fila[0];
      const lob = fila[1];
      
      // ⭐ VALIDAR: Si es Registro_pisos, verificar que tenga OM (columna D)
      if (nombreHoja === 'Registro_pisos') {
        const om = fila[3]; // Columna D (índice 3)
        
        if (!om || om.toString().trim() === '' || om.toString().trim() === '-') {
          Logger.log(`⚠️ Fila ${i+1} ignorada en Registro_pisos: No tiene OM asignado`);
          continue; // Saltar esta fila
        }
      }
      
      if (!fecha || !lob) continue;
      
      const lobFilaNorm = normalizar(lob);
      if (lobFilaNorm !== lobNorm) continue;
      
      let fechaTxt;
      if (fecha instanceof Date) {
        fechaTxt = Utilities.formatDate(fecha, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      } else {
        fechaTxt = fecha.toString().trim();
      }
      
      if (fechaTxt) fechasSet.add(fechaTxt);
    }
  });
  
  const fechas = Array.from(fechasSet).sort();
  Logger.log("📅 Fechas encontradas:", fechas);
  return fechas;
}

        empleadosSinDatos: [] 
    };

    if (!condicionScore || condicionScore === 'N/A' || condicionScore.trim() === '') {
        return resultado;
    }

    const normalizar = s => String(s || '')
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase()
        .trim()
        .replace(/\s+/g, ' ');
    
    const llaveNorm = normalizar(llave);
    const lobActualNorm = normalizar(lobActual);

    // ========================================
    // DETECTAR SI ES BONO FIJO
    // ========================================
    const esBonoFijo = !condicionScore.match(/(<=|>=|<|>|=)/);
    let valorBonoFijo = 0;
    let operador = null;
    let valorCondicion = 0;
    
    if (esBonoFijo) {
        valorBonoFijo = parseFloat(condicionScore.replace(/[^\d.]/g, '')) || 0;
        Logger.log(`💰 Bono Fijo detectado: ${llave} = $${valorBonoFijo}`);
    } else {
        const match = condicionScore.match(/(<=|>=|<|>|=)\s*(\d+(?:\.\d+)?)/);
        if (!match) {
            Logger.log('❌ Condición de score no válida:', condicionScore);
            return resultado;
        }
        operador = match[1];
        valorCondicion = parseFloat(match[2]);
    }

    const todosLosAgentes = obtenerTodosLosAgentesDeLOB(lobActual, agentesPermitidos);
    
    Logger.log(`\n🔍 validarKPI1 - LOB: ${lobActual}`);
    Logger.log(`   KPI: "${llave}"`);
    Logger.log(`   Condición: ${condicionScore}`);
    Logger.log(`   Es Decrease: ${esDecrease ? 'SÍ' : 'NO'}`);
    Logger.log(`   Peso KPI (porcentajePunto): ${(porcentajePunto * 100).toFixed(2)}%`);
    
    const agentesConDatos = new Set();

    // ========================================
    // ⭐ PASO 1: LEER CONFIGURACIÓN DE LA LOB
    // ========================================
    let salarioBaseLOB = 0;
    let porcentajeBonoGlobal = 0;
    
    try {
        const ssConfig = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaConfig = ssConfig.getSheetByName('Configuracion');
        
        if (!hojaConfig) {
            Logger.log('❌ ERROR: No se encontró la hoja Configuracion');
            return resultado;
        }
        
        const dataConfig = hojaConfig.getDataRange().getValues();
        const lobActualNormConfig = normalizar(lobActual);
        let encontrado = false;
        
        for (let c = 1; c < dataConfig.length; c++) {
            const lobConfig = normalizar(String(dataConfig[c][2] || ''));
            
            if (lobConfig === lobActualNormConfig) {
                const salarioConfigRaw = String(dataConfig[c][3] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                const porcentajeConfigRaw = String(dataConfig[c][4] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                
                salarioBaseLOB = parseFloat(salarioConfigRaw);
                porcentajeBonoGlobal = parseFloat(porcentajeConfigRaw);
                
                if (salarioBaseLOB > 0 && porcentajeBonoGlobal > 0) {
                    encontrado = true;
                    Logger.log(`✅ Config LOB: Salario Base=$${salarioBaseLOB}, %Bono=${porcentajeBonoGlobal}%`);
                    break;
                }
            }
        }
        
        if (!encontrado) {
            Logger.log(`❌ ERROR: No se encontró configuración para LOB "${lobActual}"`);
            return resultado;
        }
        
    } catch (error) {
        Logger.log(`❌ ERROR al leer Configuracion: ${error.message}`);
        return resultado;
    }

    // ========================================
    // ⭐ PASO 2: LEER HORAS SEMANALES DESDE H_C (COLUMNA J)
    // ========================================
    const horasSemanalesPorAgente = {};
    
    try {
        const ssHC = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hojaHC = ssHC.getSheetByName('H_C');
        
        if (hojaHC) {
            const dataHC = hojaHC.getDataRange().getValues();
            const lobNormHC = normalizar(lobActual);
            
            for (let h = 1; h < dataHC.length; h++) {
                const bmsHC = String(dataHC[h][0]).trim();
                const lobHC = normalizar(String(dataHC[h][1] || ''));
                const horasSemanalesRaw = dataHC[h][9]; // Columna J (índice 9)
                
                if (lobHC === lobNormHC && bmsHC) {
                    const horasSemanales = parseFloat(horasSemanalesRaw);
                    
                    if (horasSemanales && horasSemanales > 0) {
                        horasSemanalesPorAgente[bmsHC] = horasSemanales;
                    }
                }
            }
            
            Logger.log(`✅ Horas semanales cargadas para ${Object.keys(horasSemanalesPorAgente).length} agentes`);
        }
    } catch (error) {
        Logger.log(`⚠️ Error al leer H_C: ${error.message}`);
    }

    // ========================================
    // BÚSQUEDA DE COLUMNAS EN CDM
    // ========================================
    const headers = dataCDM[0];
    
    let idxBms = headers.indexOf('BMS ID');
    if (idxBms === -1) idxBms = headers.indexOf('CCMSID');
    if (idxBms === -1) idxBms = 1;
    
    let idxValH = headers.indexOf('Columna H');
    if (idxValH === -1) idxValH = headers.indexOf('KPI_Value');
    if (idxValH === -1) idxValH = headers.indexOf('Score');
    if (idxValH === -1) idxValH = 7;
    
    let idxValF = headers.indexOf('Columna F');
    if (idxValF === -1) idxValF = 5;
    
    let idxLlave = headers.indexOf('Llave');
    if (idxLlave === -1) idxLlave = headers.indexOf('KPI');
    if (idxLlave === -1) idxLlave = 8;

    let idxSubLOB = headers.indexOf('SubLOB');
    if (idxSubLOB === -1) idxSubLOB = 2;

    // ========================================
    // ⭐ SI ES BONO FIJO, ASIGNAR A TODOS
    // ========================================
    if (esBonoFijo) {
        Logger.log(`💰 Asignando bono fijo a todos los agentes`);
        
        todosLosAgentes.forEach(ag => {
            // Determinar salario (LOB o personalizado)
            let salario = salarioBaseLOB;
            let salarioPersonalizado = false;
            
            for (let u = 1; u < dataUsuario.length; u++) {
                if (String(dataUsuario[u][0]).trim() === ag.bmsId) {
                    let sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                    let salarioAgente = parseFloat(sRaw) || 0;
                    
                    if (salarioAgente > 0 && salarioAgente !== salarioBaseLOB) {
                        salario = salarioAgente;
                        salarioPersonalizado = true;
                    } else if (salarioAgente > 0) {
                        salario = salarioAgente;
                    }
                    break;
                }
            }
            
            // ⭐⭐⭐ CÁLCULO CORRECTO CON PROPORCIÓN DE HORAS ⭐⭐⭐
            
            // 1. Obtener horas semanales
            const horasSemanales = horasSemanalesPorAgente[ag.bmsId] || 42;
            
            // 2. Calcular salario proporcional
            const salarioProporcional = salario * (horasSemanales / 42);
            
            // 3. Bono base (% global sobre salario proporcional)
            const bonoBase = salarioProporcional * (porcentajeBonoGlobal / 100);
            
            // 4. Aplicar peso del KPI
            let bonoFinal = bonoBase * porcentajePunto;
            
            // 5. Si es Decrease, debe ser negativo
            if (esDecrease) {
                bonoFinal = -Math.abs(bonoFinal);
            }
            
            Logger.log(`💰 ${ag.nombre} (BMS ${ag.bmsId}):`);
            Logger.log(`   Salario base: $${salario}${salarioPersonalizado ? ' (personalizado)' : ''}`);
            Logger.log(`   Horas semanales: ${horasSemanales}h`);
            Logger.log(`   Salario proporcional: $${salarioProporcional.toFixed(2)}`);
            Logger.log(`   Bono base (${porcentajeBonoGlobal}%): $${bonoBase.toFixed(2)}`);
            Logger.log(`   Con peso KPI (${(porcentajePunto * 100).toFixed(2)}%): $${bonoFinal.toFixed(2)}`);
            
            resultado.totalCoincidencias++;
            resultado.totalCumplenCondicion++;
            
            resultado.empleadosCumplen.push({
                bmsId: ag.bmsId,
                nombre: ag.nombre,
                valorF: valorBonoFijo,
                bono: bonoFinal,
                salarioPersonalizado: salarioPersonalizado
            });
        });
        
        return resultado;
    }

    // ========================================
    // ⭐ PROCESAR FILAS DE CDM (KPI CON CONDICIÓN)
    // ========================================
    for (let row = 1; row < dataCDM.length; row++) {
        const llaveCDMRaw = String(dataCDM[row][idxLlave] || '').trim();
        const llaveCDM = normalizar(llaveCDMRaw);
        
        // Extraer KPI
        let kpiExtraidoCDM = '';
        const posLOB = llaveCDM.indexOf(lobActualNorm);
        
        if (posLOB !== -1) {
            kpiExtraidoCDM = llaveCDM.substring(posLOB + lobActualNorm.length).trim();
        } else {
            kpiExtraidoCDM = llaveCDM;
        }
        
        if (kpiExtraidoCDM !== llaveNorm) {
            continue;
        }

        const bmsId = String(dataCDM[row][idxBms]).trim();
        
        if (!agentesPermitidos[bmsId]) {
            continue;
        }

        const lobAgente = normalizar(agentesPermitidos[bmsId].lob);
        
        if (lobAgente !== lobActualNorm) {
            continue;
        }

        const subLobCDM = normalizar(String(dataCDM[row][idxSubLOB] || ''));
        
        if (subLobCDM && subLobCDM !== lobActualNorm) {
            continue;
        }

        agentesConDatos.add(bmsId);
        resultado.totalCoincidencias++;

        const valorNumerico = parseFloat(String(dataCDM[row][idxValH]).replace('%', '').trim());
        
        if (isNaN(valorNumerico) || valorNumerico === 0) {
            resultado.empleadosSinDatos.push({ 
                bmsId, 
                nombre: agentesPermitidos[bmsId].nombre, 
                valorF: 0,
                bono: 0,
                sinDatos: true,
                salarioPersonalizado: false
            });
            continue;
        }

        // ⭐ VALIDAR CONDICIÓN
        let cumple = false;
        switch (operador) {
            case '>': cumple = valorNumerico > valorCondicion; break;
            case '>=': cumple = valorNumerico >= valorCondicion; break;
            case '<': cumple = valorNumerico < valorCondicion; break;
            case '<=': cumple = valorNumerico <= valorCondicion; break;
            case '=': cumple = valorNumerico === valorCondicion; break;
        }

        if (resultado.totalCoincidencias <= 5) {
            Logger.log(`   ${bmsId}: ${valorNumerico} ${operador} ${valorCondicion} → ${cumple ? '✅' : '❌'}`);
        }

        // ⭐ DETERMINAR SALARIO
        let salario = salarioBaseLOB;
        let salarioPersonalizado = false;
        
        for (let u = 1; u < dataUsuario.length; u++) {
            if (String(dataUsuario[u][0]).trim() === bmsId) {
                let sRaw = String(dataUsuario[u][7] || '0').replace(/[^\d.,]/g, '').replace(',', '.');
                let salarioAgente = parseFloat(sRaw) || 0;
                
                if (salarioAgente > 0 && salarioAgente !== salarioBaseLOB) {
                    salario = salarioAgente;
                    salarioPersonalizado = true;
                } else if (salarioAgente > 0) {
                    salario = salarioAgente;
                }
                break;
            }
        }
        
        if (salario === 0) {
            salario = salarioBaseLOB;
        }

        // ========================================
        // ⭐⭐⭐ CÁLCULO CORRECTO CON PROPORCIÓN DE HORAS ⭐⭐⭐
        // ========================================
        
        // 1. Obtener horas semanales
        const horasSemanales = horasSemanalesPorAgente[bmsId] || 42;
        
        // 2. Calcular salario proporcional
        const salarioProporcional = salario * (horasSemanales / 42);
        
        // 3. Bono base (% global sobre salario proporcional)
        const bonoBase = salarioProporcional * (porcentajeBonoGlobal / 100);
        
        // 4. Aplicar peso del KPI
        let bonoParcial = bonoBase * porcentajePunto;
        
        Logger.log(`\n💰 ${bmsId} (${cumple ? 'CUMPLE' : 'NO CUMPLE'}):`);
        Logger.log(`   Salario base: $${salario}${salarioPersonalizado ? ' (personalizado)' : ''}`);
        Logger.log(`   Horas semanales: ${horasSemanales}h`);
        Logger.log(`   Salario proporcional: $${salarioProporcional.toFixed(2)}`);
        Logger.log(`   Bono base (${porcentajeBonoGlobal}%): $${bonoBase.toFixed(2)}`);
        Logger.log(`   Con peso KPI (${(porcentajePunto * 100).toFixed(2)}%): $${bonoParcial.toFixed(2)}`);
        
        // ⭐ PASO 5: Prorrateo SOLO por horas trabajadas (Horas_N - vacaciones/ausencias)
        let factorProrrateo = 1;
        const horasRealesPorAgente = (numeroMes && typeof obtenerHorasPorAgente === 'function') ? 
            obtenerHorasPorAgente(lobActual, numeroMes) : {};
            
        if (numeroMes && horasRealesPorAgente[bmsId]) {
            const hReales = horasRealesPorAgente[bmsId].horas || 0;
            
            // Calcular días laborables del mes
            let diasLaborables = 22;
            if (fechaSeleccionada && typeof obtenerDiasLaborablesMes === 'function') {
                const anio = (fechaSeleccionada instanceof Date) ? 
                    fechaSeleccionada.getFullYear() : 
                    new Date(fechaSeleccionada).getFullYear();
                diasLaborables = obtenerDiasLaborablesMes(numeroMes, isNaN(anio) ? new Date().getFullYear() : anio);
            }
            
            const hEsperadas = (horasSemanales / 5) * diasLaborables;
            factorProrrateo = hEsperadas > 0 ? hReales / hEsperadas : 0;
            if (factorProrrateo > 1.2) factorProrrateo = 1;
            
            Logger.log(`   📊 Prorrateo Horas_N: ${hReales}h / ${hEsperadas.toFixed(0)}h = ${(factorProrrateo * 100).toFixed(1)}%`);
        }

        // ⭐ PASO 6: Aplicar prorrateo (solo si hubo ausencias)
        let bonoFinal = bonoParcial * factorProrrateo;
        
        Logger.log(`   ✅ Bono FINAL: $${bonoFinal.toFixed(2)}`);
        
        // ⭐ Si es Decrease Y cumple → negativo
        if (esDecrease && cumple) {
            bonoFinal = -Math.abs(bonoFinal);
        }
        
        const item = { 
            bmsId, 
            nombre: agentesPermitidos[bmsId].nombre, 
            valorF: idxValF !== -1 ? dataCDM[row][idxValF] : '-', 
            bono: cumple ? bonoFinal : 0,
            salarioPersonalizado: salarioPersonalizado
        };

        if (cumple) { 
            resultado.totalCumplenCondicion++; 
            resultado.empleadosCumplen.push(item); 
        } else {
            resultado.empleadosNoCumplen.push(item);
        }
    }

    // ========================================
    // AGENTES SIN DATOS
    // ========================================
    todosLosAgentes.forEach(ag => {
        if (!agentesConDatos.has(ag.bmsId)) {
            resultado.empleadosSinDatos.push({ 
                bmsId: ag.bmsId, 
                nombre: ag.nombre, 
                valorF: '-', 
                bono: 0, 
                sinDatos: true,
                salarioPersonalizado: false
            });
        }
    });

    Logger.log(`\n📊 RESUMEN: Coincidencias=${resultado.totalCoincidencias}, Cumplen=${resultado.totalCumplenCondicion}\n`);

    return resultado;
}

function obtenerNombreMes(n) {
    if (!n) return '';
    const meses = ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 
                   'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'];
    return meses[n - 1] || '';
}


function consolidarPisos(resultadosBrutos) {
    const resultadosConsolidados = {};
    
    for (let cat in resultadosBrutos) {
        if (cat === 'keys') {
            resultadosConsolidados[cat] = resultadosBrutos[cat];
            continue;
        }

        const grupos = {};
        resultadosBrutos[cat].forEach(item => {
            if (!grupos[item.kpi]) grupos[item.kpi] = [];
            grupos[item.kpi].push(item);
        });

        resultadosConsolidados[cat] = Object.keys(grupos).map(nombreKpi => {
            const tiers = grupos[nombreKpi];
            
            // Si solo hay 1 tier, retornarlo tal cual
            if (tiers.length === 1) return tiers[0];

            // ⭐ MULTI-TIER: Usar la MAYOR participación entre los tiers
            const maxParticipacion = Math.max(...tiers.map(t => t.participacion || 1));
            
            console.log(`\n📊 Consolidando multi-tier: ${nombreKpi}`);
            console.log(`   Tiers: ${tiers.length}`);
            console.log(`   Participaciones: ${tiers.map(t => (t.participacion * 100).toFixed(0) + '%').join(', ')}`);
            console.log(`   Max participación: ${(maxParticipacion * 100).toFixed(0)}%`);

            const consolidado = {
                kpi: nombreKpi,
                score: "Multi-Escala",
                porcentaje: tiers[0].porcentaje,  // ⭐ PESO BASE
                participacion: maxParticipacion,   // ⭐ MÁXIMA PARTICIPACIÓN
                esMultiTier: true,
                scoresPorTier: tiers.map(t => ({ 
                    score: t.score, 
                    condicion: t.score,
                    participacion: t.participacion  // ⭐ GUARDAR participación de cada tier
                })),
                validacion: {
                    totalCoincidencias: tiers[0].validacion.totalCoincidencias,
                    totalCumplenCondicion: 0,
                    empleadosCumplen: [],
                    empleadosNoCumplen: [],
                    empleadosSinDatos: tiers[0].validacion.empleadosSinDatos || []
                }
            };

            const mejorResultadoAgente = {};
            
            // ⭐⭐⭐ SELECCIONAR MEJOR BONO POR AGENTE (ya viene calculado correctamente)
            tiers.forEach(tier => {
                console.log(`\n   🔍 Procesando tier: ${tier.score} (participación: ${(tier.participacion * 100).toFixed(0)}%)`);
                
                (tier.validacion.empleadosCumplen || []).forEach(e => {
                    console.log(`      Agente ${e.bmsId}: bono=$${e.bono.toFixed(2)}`);
                    
                    if (!mejorResultadoAgente[e.bmsId] || e.bono > mejorResultadoAgente[e.bmsId].bono) {
                        mejorResultadoAgente[e.bmsId] = { 
                            ...e, 
                            scoreCumplido: tier.score,
                            participacionUsada: tier.participacion  // ⭐⭐⭐ CRÍTICO: GUARDAR participación usada
                        };
                        
                        console.log(`      ✅ Mejor resultado actualizado: tier="${tier.score}", part=${(tier.participacion * 100).toFixed(0)}%`);
                    }
                });
                
                (tier.validacion.empleadosNoCumplen || []).forEach(e => {
                    if (!mejorResultadoAgente[e.bmsId]) {
                        mejorResultadoAgente[e.bmsId] = { 
                            ...e, 
                            bono: 0,
                            scoreCumplido: tier.score,
                            participacionUsada: tier.participacion
                        };
                    }
                });
            });

            // Clasificar en cumplen/no cumplen
            for (let id in mejorResultadoAgente) {
                const resultado = mejorResultadoAgente[id];
                
                // ⭐ LOG CRÍTICO: Verificar que participacionUsada esté presente
                console.log(`\n   👤 Agente ${id}:`);
                console.log(`      scoreCumplido: ${resultado.scoreCumplido}`);
                console.log(`      participacionUsada: ${resultado.participacionUsada ? (resultado.participacionUsada * 100).toFixed(0) + '%' : 'undefined'}`);
                console.log(`      bono: $${resultado.bono.toFixed(2)}`);
                
                if (resultado.bono > 0) {
                    consolidado.validacion.empleadosCumplen.push(resultado);
                    consolidado.validacion.totalCumplenCondicion++;
                } else {
                    consolidado.validacion.empleadosNoCumplen.push(resultado);
                }
            }
            
            console.log(`✅ Consolidado: ${consolidado.validacion.empleadosCumplen.length} cumplen`);
            
            return consolidado;
        });
    }
    
    return resultadosConsolidados;
}

// ⭐ FUNCIÓN AUXILIAR: Calcular días laborables del mes
// ========================================
function obtenerDiasLaborablesMes(numeroMes, anio) {
    const primerDia = new Date(anio, numeroMes - 1, 1);
    const ultimoDia = new Date(anio, numeroMes, 0);
    
    let diasLaborables = 0;
    let dia = new Date(primerDia);
    
    while (dia <= ultimoDia) {
        const diaSemana = dia.getDay();
        if (diaSemana !== 0 && diaSemana !== 6) {
            diasLaborables++;
        }
        dia.setDate(dia.getDate() + 1);
    }
    
    Logger.log(`📅 Mes ${numeroMes}/${anio}: ${diasLaborables} días laborables`);
    return diasLaborables;
}


function validarKPIsPorCategoria(registro, lob, dataCDM, dataUsuario, hoja, agentesPermitidos, numeroMes = null, fechaSeleccionada = null) {
    const resultadosBrutos = { 
        keys: [], 
        bonusStructure: [], 
        additionalIncentives: [], 
        decreasedKPIs: [] 
    };

    const procesar = (cat, start, end, inc, conParticipacion) => {
        for (let c = start; c <= end; c += inc) {
            if (registro[c]) {
                const kpi = String(registro[c]).trim();
                const score = String(registro[c + 1]).trim();
                let peso = parseFloat(String(registro[c + 2] || '0').replace('%', '')) || 0;
                if (peso > 1) peso /= 100;

                // ⭐⭐⭐ CORRECCIÓN CRÍTICA: LEER PARTICIPACIÓN CORRECTAMENTE
                let participation = 1; // Default 100%
                
                if (conParticipacion) {
                    const participacionRaw = String(registro[c + 3] || '100');
                    console.log(`\n📊 Leyendo participación de columna ${c + 3}: "${participacionRaw}"`);
                    
                    // Eliminar cualquier símbolo de % y parsear
                    const participacionNum = parseFloat(participacionRaw.replace('%', '').trim()) || 100;
                    
                    // Si el valor es > 1, está en porcentaje (ej: 50), dividir entre 100
                    // Si el valor es <= 1, ya está en decimal (ej: 0.5)
                    participation = participacionNum > 1 ? participacionNum / 100 : participacionNum;
                    
                    console.log(`   Raw: "${participacionRaw}" → Num: ${participacionNum} → Final: ${participation} (${(participation * 100).toFixed(0)}%)`);
                }
                
                // ⭐ LOG DE DEBUG PARA DIAGNOSTICAR PARTICIPACIÓN
                if (cat === 'bonusStructure' || cat === 'decreasedKPIs') {
                    console.log('\n🔍 ========== PROCESANDO KPI ==========');
                    console.log(`Categoría: ${cat}`);
                    console.log(`KPI: ${kpi}`);
                    console.log(`Score: ${score}`);
                    console.log(`Columna Peso [${c + 2}]: "${registro[c + 2]}"`);
                    console.log(`Peso procesado: ${peso} (${(peso * 100).toFixed(2)}%)`);
                    console.log(`Columna Participación [${c + 3}]: "${registro[c + 3]}"`);
                    console.log(`Participación procesada: ${participation} (${(participation * 100).toFixed(2)}%)`);
                    console.log(`porcentajePunto = ${peso} × ${participation} = ${peso * participation}`);
                    console.log(`Como %: ${(peso * participation * 100).toFixed(2)}%`);
                    console.log('======================================\n');
                }
                
                const esDecrease = cat === 'decreasedKPIs';
                
                const validacion = validarKPI1(
                    kpi, 
                    score, 
                    peso * participation,  // ⭐ Este es el peso FINAL con participación
                    dataCDM, 
                    dataUsuario, 
                    agentesPermitidos, 
                    lob, 
                    numeroMes, 
                    fechaSeleccionada,
                    esDecrease
                );
                
                // ⭐⭐⭐ GUARDAR PESO Y PARTICIPACIÓN POR SEPARADO
                resultadosBrutos[cat].push({ 
                    kpi, 
                    score, 
                    porcentaje: peso,           // ⭐ PESO BASE (sin participación)
                    participacion: participation, // ⭐ PARTICIPACIÓN DEL TIER
                    validacion,
                    esDecrease: esDecrease
                });
            }
        }
    };

    if (hoja === 'Registro') {
        procesar('keys', 5, 8, 1, false);
        procesar('bonusStructure', 10, 21, 3, false);
        procesar('additionalIncentives', 23, 31, 3, false);
        procesar('decreasedKPIs', 33, 38, 3, false);
        return resultadosBrutos;
    } else {
        // Registro_pisos (Multi-Tier)
        console.log('\n⭐ Procesando Registro_pisos (Multi-Tier)');
        
        procesar('keys', 5, 10, 2, false);
        procesar('bonusStructure', 12, 79, 4, true);  // ⭐ inc=4 porque son 4 columnas
        procesar('additionalIncentives', 81, 100, 4, true);
        procesar('decreasedKPIs', 102, 121, 4, true);
        
        return consolidarPisos(resultadosBrutos);
    }
}

function obtenerRegistrosPorLobYFecha(lob, fecha, nombreUsuario = null, rol = null) {
  const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
  const hojasBuscar = ['Registro', 'Registro_pisos'];
  const registros = [];
  
  const normalizar = s => s ? s.toString().normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim() : "";
  const lobNorm = normalizar(lob);
  const fechaNorm = normalizar(fecha);
  
  Logger.log(`📋 Buscando registros para LOB=${lobNorm}, Fecha=${fechaNorm}, Usuario=${nombreUsuario}, Rol=${rol}`);
  
  hojasBuscar.forEach(nombreHoja => {
    const hoja = ss.getSheetByName(nombreHoja);
    if (!hoja) return;
    
    const data = hoja.getDataRange().getValues();
    
    for (let i = 1; i < data.length; i++) {
      const fila = data[i];
      const fechaFila = fila[0];
      const lobFila = fila[1];
      
      let fechaTxt = '';
      if (fechaFila instanceof Date) {
        fechaTxt = Utilities.formatDate(fechaFila, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      } else {
        fechaTxt = fechaFila.toString().trim();
      }
      
      if (normalizar(lobFila) !== lobNorm || normalizar(fechaTxt) !== fechaNorm) {
        continue;
      }
      
      // ⭐ CAMBIO CRÍTICO: ACCM, OM y Supervisor NO filtran por nombre
      // Solo Reporting Analyst NO filtra (porque ve todo)
      // Los demás roles (ACCM, OM, Supervisor) ya fueron filtrados por LOB en obtenerDatosParaFrontend
      
      registros.push({
        registro: fila,
        hoja: nombreHoja
      });
      Logger.log(`✅ Encontrado registro en ${nombreHoja}, fila ${i+1}`);
    }
  });
  
  Logger.log(`📊 Total registros encontrados: ${registros.length}`);
  return registros;
}

function obtenerDatosParaFrontend(lobSeleccionada = null, fechaSeleccionada = null, mesComparativo = null) {
    try {
        Logger.log("🔵 ========== INICIO obtenerDatosParaFrontend ==========");
        Logger.log(`Parámetros: LOB="${lobSeleccionada}", Fecha="${fechaSeleccionada}", Mes=${mesComparativo}`);

        const usuario = obtenerDatosUsuarioActivo();
        if (!usuario) {
            Logger.log("❌ Usuario no encontrado");
            return { error: "Usuario no encontrado" };
        }

        Logger.log(`👤 Usuario: ${usuario.nombre}, Rol: ${usuario.rol}`);

        let lobs = [];
        let todasLasLobs = [];

        // ⭐ Determinar LOBs según rol
        if (usuario.rol === 'Reporting Analyst') {
            todasLasLobs = obtenerTodasLasLobs();
            lobs = lobSeleccionada ? [lobSeleccionada] : todasLasLobs;
            Logger.log(`📊 Reporting Analyst - Todas las LOBs: ${todasLasLobs.length}`);

        } else if (usuario.rol === 'OM') {
            todasLasLobs = obtenerLobsPorOM(usuario.nombre);
            lobs = lobSeleccionada ? [lobSeleccionada] : todasLasLobs;
            Logger.log(`📊 OM - LOBs asignadas: ${todasLasLobs.length}`);
            if (lobs.length === 0) return { error: "OM sin LOBs asignadas" };

        } else if (usuario.rol === 'ACCM') {
            todasLasLobs = obtenerLobsPorACCM(usuario.nombre);
            lobs = lobSeleccionada ? [lobSeleccionada] : todasLasLobs;
            Logger.log(`📊 ACCM - LOBs asignadas: ${todasLasLobs.length}`);
            if (lobs.length === 0) {
                Logger.log("⚠️ ACCM sin LOBs en H_C, usando LOB de Usuario");
                lobs = [usuario.lob];
                todasLasLobs = [usuario.lob];
            }

        } else if (usuario.rol === 'Supervisor') {
            todasLasLobs = obtenerLobsPorSupervisor(usuario.nombre);
            lobs = lobSeleccionada ? [lobSeleccionada] : todasLasLobs;
            Logger.log(`📊 Supervisor - LOBs asignadas: ${todasLasLobs.length}`);
            if (lobs.length === 0) return { error: "Supervisor sin LOBs asignadas" };

        } else {
            return { error: `Rol "${usuario.rol}" no soportado` };
        }

        // Si no hay mes ni fecha, solo retornar info básica
        if (!mesComparativo && !fechaSeleccionada) {
            Logger.log("⚠️ Sin mes ni fecha - retornando solo info básica");
            return {
                usuario,
                lobs,
                todasLasLobs,
                fechasPorLob: {},
                metricasPorLob: {},
                mesSeleccionado: null,
                esperandoMes: true
            };
        }

        // ⭐ Cargar agentes permitidos
        const agentesPermitidos = obtenerAgentesPermitidos();
        Logger.log(`✅ Agentes permitidos cargados: ${Object.keys(agentesPermitidos).length}`);

        // Determinar qué CDM usar
        const numeroMes = mesComparativo || (fechaSeleccionada ? obtenerNumeroMesDeFecha(fechaSeleccionada) : null);
        const dataCDM = obtenerDatosCDMPorMes(numeroMes);
        Logger.log(`📋 Usando CDM del mes: ${numeroMes || 'general'}`);

        const ssUsuario = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const dataUsuario = ssUsuario.getSheetByName('Usuario').getDataRange().getValues();

        const resultado = {
            usuario,
            lobs,
            todasLasLobs,
            fechasPorLob: {},
            metricasPorLob: {},
            mesSeleccionado: numeroMes,
            agentesPermitidos: agentesPermitidos // ⭐ CRÍTICO: Enviar esto al frontend
        };

        // ⭐ PROCESAR TODAS LAS LOBs
        const lobsAProcesar = todasLasLobs.length > 0 ? todasLasLobs : lobs;

        for (const lob of lobsAProcesar) {
            Logger.log(`\n🔹 ========== Procesando LOB: ${lob} ==========`);

            const fechas = obtenerFechasPorLob(lob);
            resultado.fechasPorLob[lob] = fechas;
            Logger.log(`📅 Fechas disponibles: ${fechas.length}`);

            if (fechaSeleccionada && fechas.includes(fechaSeleccionada)) {
                Logger.log(`📆 Buscando registros para fecha: ${fechaSeleccionada}`);

                const registrosFiltrados = obtenerRegistrosPorLobYFecha(lob, fechaSeleccionada, usuario.nombre, usuario.rol);

                if (!registrosFiltrados || registrosFiltrados.length === 0) {
                    Logger.log(`⚠️ No hay registros para esta fecha`);
                    resultado.metricasPorLob[lob] = { error: "No hay registros para esta fecha" };
                } else {
                    Logger.log(`✅ Encontrados ${registrosFiltrados.length} registros`);

                    const resultadosCombinados = {
                        keys: [],
                        bonusStructure: [],
                        additionalIncentives: [],
                        decreasedKPIs: []
                    };

                    registrosFiltrados.forEach((regData, idx) => {
                        Logger.log(`\n🔸 Procesando registro ${idx + 1}/${registrosFiltrados.length}`);

                        const validacion = validarKPIsPorCategoria(
                            regData.registro,
                            lob,
                            dataCDM,
                            dataUsuario,
                            regData.hoja,
                            agentesPermitidos,
                            numeroMes,
                            fechaSeleccionada
                        );

                        ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(cat => {
                            if (validacion[cat] && validacion[cat].length > 0) {
                                resultadosCombinados[cat] = resultadosCombinados[cat].concat(validacion[cat]);
                            }
                        });
                    });

                    resultado.metricasPorLob[lob] = resultadosCombinados;
                }
            } else {
                Logger.log(`📋 Buscando registro por usuario y LOB`);

                const registroData = obtenerRegistroPorUsuarioYLob(usuario.nombre, lob, usuario.rol);

                if (!registroData) {
                    Logger.log(`⚠️ No se encontró registro`);
                    resultado.metricasPorLob[lob] = { error: "No se encontró registro para esta LOB" };
                } else {
                    Logger.log(`✅ Registro encontrado en hoja: ${registroData.hoja}`);

                    resultado.metricasPorLob[lob] = validarKPIsPorCategoria(
                        registroData.registro,
                        lob,
                        dataCDM,
                        dataUsuario,
                        registroData.hoja,
                        agentesPermitidos,
                        numeroMes,
                        fechaSeleccionada
                    );
                }
            }
        }

        // ⭐ CARGAR HORAS POR AGENTE
        if (lobSeleccionada && numeroMes) {
            Logger.log('🔄 Cargando horas para LOB: ' + lobSeleccionada + ', Mes: ' + numeroMes);
            resultado.horasPorAgente = obtenerHorasPorAgente(lobSeleccionada, numeroMes);
            Logger.log('✅ Horas cargadas: ' + Object.keys(resultado.horasPorAgente).length + ' agentes');
        } else {
            Logger.log('⚠️ No se cargaron horas (LOB o mes no especificado)');
            resultado.horasPorAgente = {};
        }

        // ⭐ NUEVO: CARGAR PRORRATEOS APLICADOS
        if (lobSeleccionada && fechaSeleccionada && numeroMes) {
            Logger.log('🔄 Cargando prorrateos aplicados...');
            resultado.agentesProrrateo = (typeof obtenerAgentesConProrrateo === 'function') ?
                obtenerAgentesConProrrateo(lobSeleccionada, fechaSeleccionada, numeroMes) : {};
            Logger.log('✅ Prorrateos cargados: ' + Object.keys(resultado.agentesProrrateo).length + ' agentes');
        } else {
            resultado.agentesProrrateo = {};
        }

        Logger.log("\n🔵 ========== FIN obtenerDatosParaFrontend ==========");
        Logger.log(`📦 Resultado: ${Object.keys(resultado.metricasPorLob).length} LOBs procesadas`);

        return resultado;

    } catch (error) {
        Logger.log('❌ Error en obtenerParaFrontend: ' + error.message);
        return { error: error.message };
    }
}

function obtenerHorasPorAgente(lob, numMes) {
    try {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const h = ss.getSheetByName(`Horas_${numMes}`);
        
        if (!h) {
            Logger.log(`⚠️ No se encontró la hoja Horas_${numMes}`);
            return {};
        }
        
        const data = h.getDataRange().getValues();
        
        if (data.length < 2) {
            Logger.log(`⚠️ Hoja Horas_${numMes} está vacía`);
            return {};
        }
        
        // ⭐ VALIDAR HEADERS
        const headers = data[0];
        Logger.log(`📋 Headers de Horas_${numMes}:`, headers);
        
        // Buscar índices de columnas
        let idxCCMS = -1;
        let idxHoras = -1;
        let idxAusencias = -1;
        let idxDias = -1;
        
        // Buscar por nombre de columna (más robusto)
        for (let i = 0; i < headers.length; i++) {
            const header = String(headers[i]).toLowerCase().trim();
            
            if (header.includes('ccms') || header.includes('bms')) {
                idxCCMS = i;
            } else if (header === 'horas') {
                idxHoras = i;
            } else if (header.includes('ausencia')) {
                idxAusencias = i;
            } else if (header.includes('dia')) {
                idxDias = i;
            }
        }
        
        // Si no encontró por nombre, usar posiciones por defecto
        if (idxCCMS === -1) idxCCMS = 1;  // Columna B
        if (idxHoras === -1) idxHoras = 4;  // Columna E
        if (idxAusencias === -1) idxAusencias = 5;  // Columna F
        if (idxDias === -1) idxDias = 6;  // Columna G
        
        Logger.log(`📊 Índices detectados:`);
        Logger.log(`   CCMS ID: columna ${String.fromCharCode(65 + idxCCMS)} (índice ${idxCCMS})`);
        Logger.log(`   Horas: columna ${String.fromCharCode(65 + idxHoras)} (índice ${idxHoras})`);
        Logger.log(`   Ausencias: columna ${String.fromCharCode(65 + idxAusencias)} (índice ${idxAusencias})`);
        Logger.log(`   Días: columna ${String.fromCharCode(65 + idxDias)} (índice ${idxDias})`);
        
        const map = {};
        let registrosCargados = 0;
        
        // ⭐ LEER DATOS DESDE FILA 1 (saltando headers)
        for (let i = 1; i < data.length; i++) {
            const fila = data[i];
            
            // Leer CCMS ID desde la columna correcta
            const ccmsId = String(fila[idxCCMS] || '').trim();
            
            if (!ccmsId) {
                if (i <= 5) {
                    Logger.log(`⚠️ Fila ${i+1}: CCMS ID vacío, saltando...`);
                }
                continue;
            }
            
            // Leer horas (puede venir como número o texto)
            let horas = 0;
            const horasRaw = fila[idxHoras];
            
            if (horasRaw !== null && horasRaw !== undefined && horasRaw !== '') {
                horas = parseFloat(horasRaw) || 0;
            }
            
            // Leer ausencias y días (opcional)
            const ausencias = parseFloat(fila[idxAusencias]) || 0;
            const dias = parseFloat(fila[idxDias]) || 0;
            
            // Guardar en el mapa
            map[ccmsId] = {
                horas: horas,
                ausencias: ausencias,
                dias: dias
            };
            
            registrosCargados++;
            
            // ⭐ LOG de las primeras 5 filas para debug
            if (i <= 5) {
                Logger.log(`\n✅ Fila ${i+1} cargada:`);
                Logger.log(`   CCMS ID: "${ccmsId}"`);
                Logger.log(`   Horas: ${horas}h`);
                Logger.log(`   Ausencias: ${ausencias}`);
                Logger.log(`   Días: ${dias}`);
            }
        }
        
        Logger.log(`\n📊 RESUMEN:`);
        Logger.log(`   Total registros en Horas_${numMes}: ${registrosCargados}`);
        Logger.log(`   Primeros IDs: ${Object.keys(map).slice(0, 5).join(', ')}`);
        
        return map;
        
    } catch (e) {
        Logger.log(`❌ Error en obtenerHorasPorAgente: ${e.message}`);
        Logger.log(`Stack: ${e.stack}`);
        return {};
    }
}

// Calcula cuántos días habia que trabajar (L-V)
function obtenerDiasLaborablesMes(mes, anio) {
  let d = 0, date = new Date(anio, mes-1, 1);
  while (date.getMonth() === mes-1) {
    if (date.getDay() !== 0 && date.getDay() !== 6) d++;
    date.setDate(date.getDate() + 1);
  }
  return d;
}

/**
 * Obtiene el Top 3 de agentes con mayor bono para un KPI específico
 */
function obtenerTop3PorKPI(lob, mes, kpi) {
  try {
    const ss = SpreadsheetApp.openById('1PtYuAb5C2kIoX8sE7dOvt1nVS-aLamZLOJQUSbTBBCw');
    const sheetName = 'CDM_' + mes;
    const sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      Logger.log(`⚠️ No existe la hoja ${sheetName}`);
      return [];
    }
    
    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    
    // Encontrar índices de columnas
    const lobIndex = headers.indexOf('LOB');
    const bmsIndex = headers.indexOf('BMS ID');
    const nombreIndex = headers.indexOf('Nombre');
    const llaveIndex = headers.indexOf('Llave');
    const valorFIndex = headers.indexOf('Columna F');
    const bonoIndex = headers.indexOf('Bono');
    
    if (lobIndex === -1 || bmsIndex === -1 || llaveIndex === -1) {
      Logger.log('❌ No se encontraron las columnas necesarias');
      return [];
    }
    
    Logger.log(`🏆 Calculando Top 3 para KPI="${kpi}" en LOB="${lob}"`);
    
    // Recopilar datos de agentes
    const agentes = [];
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      
      if (row[lobIndex] !== lob) continue;
      
      const llave = String(row[llaveIndex]).trim();
      
      // Verificar si esta fila corresponde al KPI buscado
      if (!llave.endsWith(kpi)) continue;
      
      const bmsId = String(row[bmsIndex]).trim();
      const nombre = String(row[nombreIndex]).trim();
      const valorF = row[valorFIndex];
      const bono = parseFloat(row[bonoIndex]) || 0;
      
      if (bono > 0 && bmsId && nombre) {
        agentes.push({
          bmsId: bmsId,
          nombre: nombre,
          valor: valorF,
          bono: bono
        });
      }
    }
    
    // Ordenar por bono descendente
    agentes.sort((a, b) => b.bono - a.bono);
    
    // Retornar top 3
    const top3 = agentes.slice(0, 3);
    
    Logger.log(`✅ Top 3 calculado: ${top3.length} agentes`);
    top3.forEach((a, i) => {
      Logger.log(`   ${i+1}. ${a.nombre} (${a.bmsId}) - $${a.bono.toFixed(2)}`);
    });
    
    return top3;
    
  } catch (error) {
    Logger.log(`❌ Error en obtenerTop3PorKPI: ${error.message}`);
    return [];
  }
}

/**

 */
function obtenerHistoricoMaximos() {
  try {
    const ss = SpreadsheetApp.openById('1PtYuAb5C2kIoX8sE7dOvt1nVS-aLamZLOJQUSbTBBCw');
    const sheets = ss.getSheets();

    const cdmSheets = sheets.filter(s => {
      const nombre = s.getName();
      const match = nombre.match(/^CDM_(\d+)$/);
      if (match) {
        Logger.log('✅ Hoja CDM encontrada: ' + nombre);
        return true;
      }
      return false;
    });
    
    Logger.log(`📊 Hojas CDM numeradas encontradas: ${cdmSheets.length}`);
    
    if (cdmSheets.length === 0) {
      Logger.log('⚠️ No se encontraron hojas CDM numeradas');
      return [];
    }
    
    // Estructura para almacenar máximos por KPI
    const maximos = {};
    
    const mesesNombres = {
      1: 'Enero', 2: 'Febrero', 3: 'Marzo', 4: 'Abril',
      5: 'Mayo', 6: 'Junio', 7: 'Julio', 8: 'Agosto',
      9: 'Septiembre', 10: 'Octubre', 11: 'Noviembre', 12: 'Diciembre'
    };
    
    cdmSheets.forEach(sheet => {
      const sheetName = sheet.getName();
      const numeroMes = parseInt(sheetName.replace('CDM_', ''));
      const nombreMes = mesesNombres[numeroMes] || sheetName;
      
      Logger.log(`\n📊 Procesando ${sheetName} (${nombreMes})...`);
      
      const data = sheet.getDataRange().getValues();
      
      if (data.length < 2) {
        Logger.log('⚠️ Hoja vacía o sin datos');
        return;
      }
      
      const headers = data[0];
      Logger.log('📋 Headers: ' + headers.join(', '));
      
      const lobIndex = headers.indexOf('LOB');
      const fechaIndex = headers.indexOf('Fecha');
      const llaveIndex = headers.indexOf('Llave');
      const bonoIndex = headers.indexOf('Bono');
      
      Logger.log(`Índices - LOB: ${lobIndex}, Fecha: ${fechaIndex}, Llave: ${llaveIndex}, Bono: ${bonoIndex}`);
      
      if (lobIndex === -1 || llaveIndex === -1 || bonoIndex === -1) {
        Logger.log(`⚠️ Columnas faltantes en ${sheetName}`);
        return;
      }
      
      let registrosProcesados = 0;
      
      // Procesar cada fila
      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        
        const lob = String(row[lobIndex] || '').trim();
        const llave = String(row[llaveIndex] || '').trim();
        const bono = parseFloat(row[bonoIndex]) || 0;
        const fecha = row[fechaIndex];
        
        if (!lob || !llave || bono <= 0) continue;
        
        // Extraer KPI de la llave
        let kpiName = llave;
        const lobIndexPos = kpiName.indexOf(lob);
        if (lobIndexPos !== -1) {
          kpiName = kpiName.substring(lobIndexPos + lob.length).trim();
        }
        
        if (!kpiName) continue;
        
        registrosProcesados++;
        
        // Determinar categoría
        let categoria = 'bonusStructure';
        
        const kpiLower = kpiName.toLowerCase();
        if (kpiLower.includes('aht') || 
            kpiLower.includes('fcr') ||
            kpiLower.includes('csat')) {
          categoria = 'keys';
        } else if (kpiLower.includes('adicional')) {
          categoria = 'additionalIncentives';
        } else if (kpiLower.includes('penalizacion')) {
          categoria = 'decreasedKPIs';
        }
        
        // Crear clave única por KPI y LOB
        const key = `${kpiName}|${lob}`;
        
        // Inicializar o actualizar máximo
        if (!maximos[key]) {
          maximos[key] = {
            kpi: kpiName,
            lob: lob,
            categoria: categoria,
            maxBono: bono,
            mesMaximo: nombreMes,
            fechaMaximo: fecha,
            totalBonos: bono,
            conteo: 1
          };
        } else {
          maximos[key].totalBonos += bono;
          maximos[key].conteo++;
          
          if (bono > maximos[key].maxBono) {
            maximos[key].maxBono = bono;
            maximos[key].mesMaximo = nombreMes;
            maximos[key].fechaMaximo = fecha;
          }
        }
      }
      
      Logger.log(`✅ ${sheetName}: ${registrosProcesados} registros procesados`);
    });
    
    Logger.log(`\n📊 Total de KPIs únicos encontrados: ${Object.keys(maximos).length}`);
    
    // Convertir a array y calcular promedios
    const resultado = Object.values(maximos).map(record => ({
      kpi: record.kpi,
      lob: record.lob,
      categoria: record.categoria,
      maxBono: record.maxBono,
      mesMaximo: record.mesMaximo,
      fechaMaximo: formatearFecha(record.fechaMaximo),
      promedio: record.totalBonos / record.conteo
    }));
    
    // Ordenar por bono máximo descendente
    resultado.sort((a, b) => b.maxBono - a.maxBono);
    
    Logger.log(`\n✅ Histórico generado: ${resultado.length} registros únicos`);
    
    if (resultado.length > 0) {
      Logger.log('📊 Primeros 3 registros:');
      resultado.slice(0, 3).forEach((r, i) => {
        Logger.log(`  ${i+1}. ${r.kpi} (${r.lob}) - $${r.maxBono}`);
      });
    }
    
    return resultado;
    
  } catch (error) {
    Logger.log(`❌ Error en obtenerHistoricoMaximos: ${error.message}`);
    Logger.log(`Stack: ${error.stack}`);
    return [];
  }
}

/**
 * Helper para comparar fechas de forma segura (Formato yyyy-MM-dd)
 */
function normalizarFechaComparacion(fecha) {
    if (!fecha) return "";
    if (fecha instanceof Date) return Utilities.formatDate(fecha, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    const str = fecha.toString().trim();
    
    // Si ya viene como yyyy-MM-dd (ej. 2026-02-15)
    if (str.includes('-')) return str.substring(0, 10);
    
    // Si viene como dd/MM/yyyy (ej. 15/02/2026) -> convertir a yyyy-MM-dd
    if (str.includes('/')) {
        const parts = str.split('/');
        if (parts.length === 3) {
            if (parts[0].length === 4) return `${parts[0]}-${parts[1].padStart(2, '0')}-${parts[2].padStart(2, '0')}`; // yyyy/MM/dd
            return `${parts[2]}-${parts[1].padStart(2, '0')}-${parts[0].padStart(2, '0')}`; // dd/MM/yyyy
        }
    }
    return str;
}

function obtenerTodasLasLobsDelSistema() {
    return getCached('todas_lobs_sistema', function () {
        const ss = SpreadsheetApp.openById('1nzXCGsp8emRdDXphPWU75Ms5c_GfXiC7_TMODBS2ed4');
        const hoja = ss.getSheetByName('H_C');
        if (!hoja) return [];

        const data = hoja.getDataRange().getValues();
        const lobs = new Set();

        // Asumiendo que LOB está en la columna B (índice 1)
        for (let i = 1; i < data.length; i++) {
            const lob = String(data[i][1]).trim();
            if (lob && lob.length > 2) {
                lobs.add(lob);
            }
        }

        return Array.from(lobs).sort();
    }, 21600); // Cache por 6 horas
}
