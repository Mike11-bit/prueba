function cambiarTipoAccesoUsuario() {
    const select = document.getElementById('tipoAccesoSelect');
    const statusSpan = document.getElementById('tipoAccesoStatus');
    
    if (!select) {
        console.error('❌ No se encontró el select de tipo de acceso');
        alert('❌ Error: No se encontró el selector');
        return;
    }
    
    const tipoAcceso = select.value;
    console.log('🔄 Cambiando tipo de acceso a:', tipoAcceso);
    
    // Validar selección
    if (!tipoAcceso || tipoAcceso === '') {
        alert('⚠️ Por favor selecciona un tipo de acceso');
        return;
    }
    
    // Actualizar status (si existe)
    if (statusSpan) {
        statusSpan.textContent = '⏳ Actualizando...';
        statusSpan.style.color = '#f39c12';
    }
    
    // Deshabilitar botón temporalmente
    const boton = event.target;
    boton.disabled = true;
    boton.textContent = '⏳ Procesando...';
    
    const email = document.getElementById('correoUsuario').textContent;
    
    google.script.run
        .withSuccessHandler(function(response) {
            console.log('✅ Respuesta:', response);
            
            // Rehabilitar botón
            boton.disabled = false;
            boton.textContent = '🔄 Cambiar';
            
            if (response.success) {
                if (statusSpan) {
                    statusSpan.textContent = '✓ ' + response.mensaje;
                    statusSpan.style.color = '#27ae60';
                }
                
                // Mostrar alerta de éxito
                alert('✅ Tipo de acceso cambiado exitosamente a: ' + tipoAcceso + '\n\n🔄 La página se recargará para aplicar los cambios.');
                
                // Recargar página después de 1 segundo
                setTimeout(function() {
                    location.reload();
                }, 1000);
                
            } else {
                if (statusSpan) {
                    statusSpan.textContent = '✗ Error: ' + response.error;
                    statusSpan.style.color = '#e74c3c';
                }
                alert('❌ Error al cambiar tipo de acceso: ' + response.error);
            }
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error:', error);
            
            // Rehabilitar botón
            boton.disabled = false;
            boton.textContent = '🔄 Cambiar';
            
            if (statusSpan) {
                statusSpan.textContent = '✗ Error de conexión';
                statusSpan.style.color = '#e74c3c';
            }
            alert('❌ Error de conexión: ' + error.message);
        })
        .cambiarTipoAcceso(email, tipoAcceso);
}

  function dibujarRuleta() {
      const centerX = canvas.width / 2;
      const centerY = canvas.height / 2;
      const radius = 250;
      
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      
      const anglePerSegment = (2 * Math.PI) / participantesRuleta.length;
      
      // Colores vibrantes para cada segmento
      const colores = [
          '#e74c3c', '#3498db', '#2ecc71', '#f39c12', 
          '#9b59b6', '#1abc9c', '#e67e22', '#34495e',
          '#16a085', '#c0392b', '#8e44ad', '#27ae60'
      ];
      
      participantesRuleta.forEach((participante, i) => {
          const startAngle = anguloActual + (i * anglePerSegment);
          const endAngle = startAngle + anglePerSegment;
          
          // Dibujar segmento
          ctx.beginPath();
          ctx.moveTo(centerX, centerY);
          ctx.arc(centerX, centerY, radius, startAngle, endAngle);
          ctx.closePath();
          ctx.fillStyle = colores[i % colores.length];
          ctx.fill();
          ctx.strokeStyle = '#fff';
          ctx.lineWidth = 3;
          ctx.stroke();
          
          // Dibujar texto
          ctx.save();
          ctx.translate(centerX, centerY);
          ctx.rotate(startAngle + anglePerSegment / 2);
          ctx.textAlign = 'center';
          ctx.fillStyle = '#fff';
          ctx.font = 'bold 14px Arial';
          ctx.fillText(participante.nombre.split(',')[0].substring(0, 15), radius * 0.65, 5);
          ctx.font = '12px Arial';
          ctx.fillText(participante.lob, radius * 0.65, 20);
          ctx.restore();
      });
      
      // Indicador (flecha apuntando arriba)
      ctx.beginPath();
      ctx.moveTo(centerX, 30);
      ctx.lineTo(centerX - 20, 70);
      ctx.lineTo(centerX + 20, 70);
      ctx.closePath();
      ctx.fillStyle = '#e74c3c';
      ctx.fill();
      ctx.strokeStyle = '#fff';
      ctx.lineWidth = 3;
      ctx.stroke();
      
      // Centro de la ruleta
      ctx.beginPath();
      ctx.arc(centerX, centerY, 40, 0, 2 * Math.PI);
      ctx.fillStyle = '#2c3e50';
      ctx.fill();
      ctx.strokeStyle = '#fff';
      ctx.lineWidth = 4;
      ctx.stroke();

  function girarRuleta() {
      if (girando) return;
      
      girando = true;
      document.getElementById('ruleta-ganador').style.display = 'none';
      
      // Velocidad inicial y desaceleración
      let velocidad = 0.5;
      const desaceleracion = 0.99;
      const minimoVelocidad = 0.005;
      
      // Rotaciones aleatorias (3-6 vueltas completas)
      const vueltasExtra = Math.random() * 3 * Math.PI + 6 * Math.PI;
      let anguloTotal = 0;
      
      function animar() {
          anguloTotal += velocidad;
          anguloActual += velocidad;
          velocidad *= desaceleracion;
          
          dibujarRuleta();
          
          if (velocidad > minimoVelocidad) {
              animacionRuleta = requestAnimationFrame(animar);
          } else {
              girando = false;
              seleccionarGanador();
          }
      }
      
      animar();
  }

      let diferencia = (anguloIndicador - anguloNormalizado + 2 * Math.PI) % (2 * Math.PI);

      ganadorActual = participantesRuleta[indiceGanador];

function registrarGanador() {
    if (!ganadorActual) {
        mostrarAlerta('⚠️ No hay ganador seleccionado', 'warning');
        return;
    }
    
    const porcentaje = parseFloat(document.getElementById('porcentajeAdicional').value) || 0;
    const mensaje = document.getElementById('mensajeGanador').value.trim(); // ⭐ CAPTURAR mensaje
    
    if (porcentaje < 0 || porcentaje > 100) {
        mostrarAlerta('⚠️ El porcentaje debe estar entre 0 y 100', 'warning');
        return;
    }
    
    // ⭐ VALIDACIÓN: Advertir si no hay mensaje
    if (!mensaje) {
        const confirmar = confirm('⚠️ No has escrito un mensaje personalizado.\n\n¿Deseas continuar sin mensaje?');
        if (!confirmar) return;
    }
    
    const fecha = document.getElementById('ruletaSelectFecha').value;
    const kpi = document.getElementById('ruletaSelectKpi').value;
    
    console.log('💾 Registrando ganador:', ganadorActual, 'Porcentaje:', porcentaje, 'Mensaje:', mensaje);
    
    mostrarLoading('Registrando ganador...');
    
    google.script.run
        .withSuccessHandler(function(response) {
            ocultarLoading();
            
            if (response.success) {
                mostrarAlerta('✅ Ganador registrado exitosamente. Bono total: $' + response.bonoTotal.toLocaleString('es-CO', {minimumFractionDigits: 2}), 'success');
                
                // ⭐ 1. ELIMINAR GANADOR DE LA LISTA DE PARTICIPANTES
                participantesRuleta = participantesRuleta.filter(p => p.bmsId !== ganadorActual.bmsId);
                console.log(`🗑️ Ganador eliminado. Participantes restantes: ${participantesRuleta.length}`);
                
                // ⭐ 2. ACTUALIZAR INTERFAZ
                if (participantesRuleta.length > 0) {
                    // Hay más participantes: actualizar ruleta y lista
                    anguloActual = 0; // Resetear ángulo
                    dibujarRuleta();
                    mostrarListaParticipantes();
                    document.getElementById('ruleta-ganador').style.display = 'none';
                    ganadorActual = null;
                } else {
                    // No quedan participantes
                    document.getElementById('ruleta-canvas-container').innerHTML = 
                        '<div style="background:#f39c12;color:white;padding:30px;border-radius:12px;text-align:center;font-size:1.2rem;">' +
                        '🏁 ¡Todos los participantes han sido registrados!<br>' +
                        '<small style="font-size:0.9rem;opacity:0.9;margin-top:10px;display:block;">Selecciona otra fecha o KPI para continuar</small>' +
                        '</div>';
                    document.getElementById('ruleta-participantes').style.display = 'none';
                    document.getElementById('ruleta-ganador').style.display = 'none';
                }
                
                // Guardar participantes
                google.script.run
                    .withSuccessHandler(function(resp) {
                        console.log('✅ Participantes actualizados');
                    })
                    .withFailureHandler(function(error) {
                        console.error('❌ Error al guardar participantes:', error);
                    })
                    .guardarParticipantesRuleta(participantesRuleta, fecha, kpi);
                
                // Recargar lista de ganadores
                cargarGanadoresDelMes(fecha);
                
            } else {
                mostrarAlerta('❌ Error al registrar: ' + response.error, 'danger');
            }
        })
        .withFailureHandler(function(error) {
            ocultarLoading();
            mostrarAlerta('❌ Error: ' + error.message, 'danger');
            console.error('Error:', error);
        })
        .guardarGanadorRuleta(ganadorActual, fecha, kpi, porcentaje, mensaje); // ⭐ AGREGAR mensaje
}

  function cargarParticipantesRuleta(fecha, kpi, callback) {
      // Primero obtener ganadores previos
      google.script.run
          .withSuccessHandler(function(ganadoresPrevios) {
              console.log('📋 Ganadores previos:', ganadoresPrevios);
              
              // Crear Set de BMS IDs de ganadores
              const ganadoresSet = new Set(ganadoresPrevios.map(g => g.bmsId));
              
              // Continuar con la carga normal pero filtrar
              google.script.run
                  .withSuccessHandler(function(res) {
                      if (!res.lobs || res.lobs.length === 0) return;
                      
                      participantesRuleta = [];
                      let procesadas = 0;
                      
                      res.lobs.forEach(lob => {
                          google.script.run
                              .withSuccessHandler(function(datosRecibidos) {
                                  if (!datosRecibidos.error && datosRecibidos.metricasPorLob && datosRecibidos.metricasPorLob[lob]) {
                                      const metricas = datosRecibidos.metricasPorLob[lob];
                                      
                                      if (!metricas.error) {
                                          const categorias = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
                                          
                                          for (const cat of categorias) {
                                              const kpis = metricas[cat] || [];
                                              const kpiMatch = kpis.find(k => k.kpi === kpi);
                                              
                                              if (kpiMatch && kpiMatch.validacion) {
                                                  const empleadosCumplen = kpiMatch.validacion.empleadosCumplen || [];
                                                  
                                                  const top3 = empleadosCumplen
                                                      .sort((a, b) => (b.bono || 0) - (a.bono || 0))
                                                      .slice(0, 3);
                                                  
                                                  // ⭐ FILTRAR: Excluir ganadores previos
                                                  top3.forEach(agente => {
                                                      if (!ganadoresSet.has(agente.bmsId)) {
                                                          participantesRuleta.push({
                                                              nombre: agente.nombre,
                                                              bmsId: agente.bmsId,
                                                              lob: lob,
                                                              bono: agente.bono || 0
                                                          });
                                                      } else {
                                                          console.log('🚫 Excluido (ya ganó):', agente.nombre);
                                                      }
                                                  });
                                                  
                                                  break;
                                              }
                                          }
                                      }
                                  }
                                  
                                  procesadas++;
                                  
                                  if (procesadas === res.lobs.length) {
                                      if (participantesRuleta.length === 0) {
                                          mostrarAlerta('⚠️ No hay participantes disponibles (todos ya ganaron)', 'warning');
                                          return;
                                      }
                                      
                                      console.log(`✅ ${participantesRuleta.length} participantes (${ganadoresPrevios.length} excluidos)`);
                                      
                                      if (callback) callback();
                                  }
                              })
                              .withFailureHandler(function(error) {
                                  console.error('❌ Error:', error);
                                  procesadas++;
                              })
                              .obtenerDatosParaFrontend(lob, fecha, null);
                      });
                  })
                  .withFailureHandler(function(error) {
                      console.error('❌ Error:', error);
                  })
                  .obtenerDatosParaFrontend();
          })
          .withFailureHandler(function(error) {
              console.error('❌ Error al obtener ganadores previos:', error);
              // Continuar sin filtrar
              if (callback) callback();
          })
          .obtenerGanadoresPrevios(fecha, kpi);
  }

    const stepTimes = [3000, 3500, 3200, 3800, 4500, 4200, 3800, 3500]; // 3-4.5 segundos por paso
    
    // Función para mover la tortuga MUY LENTAMENTE
    function moveTurtleSlowly(targetPercentage, duration = 20000) { // 20 segundos para ir de 0 a 100%
        if (!turtle || !turtleProgress || !turtleStatus) return;
        
        const containerWidth = document.getElementById('walking-turtle-container').offsetWidth;
        const startPercentage = turtlePercentage;
        const percentageDiff = targetPercentage - startPercentage;
        const startTime = Date.now();
        
        function animateTurtle() {
            const elapsed = Date.now() - startTime;
            const progress = Math.min(elapsed / duration, 1);
            const currentPercentage = startPercentage + (percentageDiff * progress);
            
            const turtlePosition = (currentPercentage / 100) * containerWidth;
            
            // Mover tortuga y progreso
            turtle.style.left = `${turtlePosition}px`;
            turtleProgress.style.width = `${currentPercentage}%`;
            
            // Actualizar porcentaje de la tortuga
            turtlePercentage = currentPercentage;
            
            // Aplicar animación según dirección
            if (isGoingForward) {
                turtle.style.animation = 'turtleWalkSlow 3s infinite';
                turtle.style.transform = 'translate(-50%, -50%)';
            } else {
                turtle.style.animation = 'turtleWalkBackSlow 3s infinite';
                turtle.style.transform = 'translate(-50%, -50%) scaleX(-1)';
            }
            
            // Actualizar estado y velocidad
            updateTurtleStatus();
            updateTurtleSpeed(progress);
            
            if (progress < 1) {
                requestAnimationFrame(animateTurtle);
            } else {
                // Cuando termina el movimiento
                turtlePercentage = targetPercentage;
                onTurtleMovementComplete();
            }
        }
        
        animateTurtle();
    }
    
    // Función para actualizar la velocidad mostrada
    function updateTurtleSpeed(progress) {
        if (!turtleSpeed) return;
        
        const speeds = [
            "Extremadamente lenta",
            "Muy lenta", 
            "Lenta",
            "Paso de tortuga",
            "Casi llegando",
            "Último esfuerzo"
        ];
        
        const speedIndex = Math.floor(progress * (speeds.length - 1));
        turtleSpeed.textContent = speeds[speedIndex];
    }
    
    // Función para actualizar el estado de la tortuga
    function updateTurtleStatus() {
        if (!turtleStatus) return;
        
        const messagesForward = [
            "La tortuga está empezando... muy lentamente...",
            "La tortuga avanza... paso a paso...",
            "Revisando cuidadosamente...",
            "Verificando cada detalle...",
            "Poco a poco... avanzando...",
            "Casi ahí... pero sin prisa...",
            "Últimos metros... con calma...",
            "¡Llegando! Pero sin correr..."
        ];
        
        const messagesBackward = [
            "Regresando... tranquilamente...",
            "De vuelta... revisando de nuevo...",
            "Verificando por segunda vez...",
            "Sin prisa... todo debe estar bien...",
            "Regreso lento pero seguro...",
            "Casi de vuelta al inicio...",
            "Último tramo del regreso...",
            "¡Listo para empezar de nuevo!"
        ];
        
        const messages = isGoingForward ? messagesForward : messagesBackward;
        const messageIndex = Math.floor((turtlePercentage / 100) * (messages.length - 1));
        
        if (messages[messageIndex]) {
            turtleStatus.textContent = messages[messageIndex];
        }
    }
    
    // Función cuando termina un movimiento
    function onTurtleMovementComplete() {
        if (isGoingForward) {
            // Llegó al 100%
            turtle.style.animation = 'turtleRest 2s ease infinite';
            turtleStatus.textContent = "🐢 ¡Llegó! Descansando un momento...";
            
            // Esperar y regresar
            setTimeout(() => {
                isGoingForward = false;
                turtleStatus.textContent = "Empezando el regreso... muy lentamente...";
                setTimeout(() => moveTurtleSlowly(0, 18000), 1500);
            }, 2000);
        } else {
            // Regresó al 0%
            turtle.style.animation = 'turtleRest 2s ease infinite';
            turtleStatus.textContent = "🐢 ¡De vuelta! Descansando antes del siguiente ciclo...";
            
            // Incrementar ciclo
            cycleCount++;
            
            if (cycleCount <= maxCycles) {
                setTimeout(() => {
                    cycleNumber.textContent = cycleCount;
                    isGoingForward = true;
                    turtleStatus.textContent = "Iniciando nuevo ciclo... con paciencia...";
                    setTimeout(() => moveTurtleSlowly(100, 22000), 1500);
                }, 2500);
            } else {
                // Terminar proceso
                setTimeout(completeProcess, 2000);
            }
        }
    }
    
    // Función para completar el proceso
    function completeProcess() {
        // Detener animación de caminar
        turtle.style.animation = 'none';
        
        // Cambiar tortuga por checkmark
        turtle.textContent = '✅';
        turtle.style.transform = 'translate(-50%, -50%)';
        
        // Animación final lenta
        turtle.style.animation = 'turtleRest 3s ease infinite';
        
        // Actualizar mensajes finales
        turtleStatus.textContent = '✅ ¡Proceso completado! Verificación minuciosa finalizada';
        turtleStatus.style.color = '#27ae60';
        turtleStatus.style.fontWeight = 'bold';
        
        timeRemainingEl.textContent = '✅ Todas las ausencias verificadas y horas prorrateadas';
        timeRemainingEl.style.color = '#27ae60';
        
        // Cambiar el contador de ciclos
        document.getElementById('cycle-counter').innerHTML = '✅ <strong>Verificación completa</strong>';
        turtleSpeed.textContent = "Tarea completada";
        turtleSpeed.style.color = '#27ae60';
    }
    
    // Función principal de progreso de pasos (también lenta)
    function advanceProgress() {
        currentStep++;
        
        if (currentStep > totalSteps) {
            clearTimeout(window.currentProgressTimeout);
            return;
        }
        
        // Actualizar pasos LENTAMENTE
        if (currentStep > 1) {
            const prevStep = document.querySelector(`.progress-step[data-step="${currentStep-1}"]`);
            if (prevStep) {
                setTimeout(() => {
                    prevStep.classList.remove('active');
                    prevStep.classList.add('completed');
                    prevStep.querySelector('.progress-icon').textContent = '✅';
                }, 500);
            }
        }
        
        const currentStepEl = document.querySelector(`.progress-step[data-step="${currentStep}"]`);
        if (currentStepEl) {
            setTimeout(() => {
                currentStepEl.classList.add('active');
                currentStepEl.querySelector('.progress-icon').textContent = '🔄';
            }, 300);
        }
        
        // Calcular porcentaje general
        const percentage = Math.round((currentStep / totalSteps) * 100);
        const progressBar = document.getElementById('progress-bar');
        const progressPercentage = document.getElementById('progress-percentage');
        
        // Transición lenta para la barra de progreso
        if (progressBar) {
            progressBar.style.transition = 'width 3s ease';
            setTimeout(() => {
                progressBar.style.width = percentage + '%';
            }, 100);
        }
        
        if (progressPercentage) {
            progressPercentage.textContent = percentage + '%';
            
            // Cambiar color cuando llega al 100%
            if (percentage === 100) {
                setTimeout(() => {
                    progressPercentage.textContent = '100% - Verificación completa';
                    progressPercentage.style.color = '#27ae60';
                    progressPercentage.style.fontWeight = 'bold';
                }, 3500);
            }
        }
        
        // Programar siguiente paso con tiempos LARGOS
        if (currentStep < totalSteps) {
            window.currentProgressTimeout = setTimeout(advanceProgress, stepTimes[currentStep-1]);
        }
    }
    
    // Iniciar todo
    window.currentProgressTimeout = setTimeout(advanceProgress, 2000);
    
    // Iniciar movimiento de la tortuga MUY LENTAMENTE después de un delay
    setTimeout(() => {
        turtleStatus.textContent = "Preparándose... la tortuga se estira...";
        
        setTimeout(() => {
            moveTurtleSlowly(100, 25000); // 25 segundos para el primer viaje
        }, 2000);
    }, 1000);
    
    // Actualizar tiempo restante periódicamente
    let totalTime = stepTimes.reduce((a, b) => a + b, 0);
    let elapsedTime = 0;
    
    const timeUpdateInterval = setInterval(() => {
        elapsedTime += 1000;
        const remaining = Math.max(0, totalTime - elapsedTime);
        const remainingSeconds = Math.ceil(remaining / 1000);
        
        if (timeRemainingEl && remainingSeconds > 0) {
            timeRemainingEl.textContent = `⏱️ Tiempo restante: ~${remainingSeconds} segundos`;
        }
        
        if (remaining <= 0) {
            clearInterval(timeUpdateInterval);
        }
    }, 1000);
}

function cargarDatosCompletos(lob, fecha, mes) {
    console.log('🔍 Cargando datos completos:', { lob, fecha, mes });
    
    limpiarComponentesDuplicados();
mostrarLoadingConProgreso();
    limpiarAlertas();
    limpiarExtras();
    
    let intentos = 0;
    const maxIntentos = 2;
    
    function ejecutarCarga() {
        intentos++;
        console.log(`🔄 Intento ${intentos} de ${maxIntentos}`);
        
        google.script.run
            .withSuccessHandler(function(datosRecibidos) {
                console.log('✅ Respuesta recibida del servidor en intento', intentos);
                console.log('📦 Datos completos recibidos:', datosRecibidos);
                
                if (!datosRecibidos) {
                    console.error('❌ datosRecibidos es null/undefined en intento', intentos);
                    
                    if (intentos < maxIntentos) {
                        console.log('🔄 Reintentando...');
                        setTimeout(ejecutarCarga, 2000);
                        return;
                    }
                    
                    mostrarError('No se pudieron cargar los datos después de ' + maxIntentos + ' intentos.');
                    return;
                }
                
                if (datosRecibidos.error) {
                    mostrarError('Error del servidor: ' + datosRecibidos.error);
                    return;
                }
                
                datos = datosRecibidos;
                
                // Cargar horas por agente
                if (datosRecibidos.horasPorAgente) {
                    horasPorAgente = datosRecibidos.horasPorAgente;
                    console.log('⏱️ Horas cargadas:', Object.keys(horasPorAgente).length, 'agentes');
                } else {
                    horasPorAgente = {};
                    console.warn('⚠️ No se recibieron horas por agente');
                }
                
                console.log('✅ Datos cargados exitosamente en intento', intentos);
                validarYMostrarAgentes(lob);
                procesarYMostrarDatos(lob);
                
                // ⭐ APLICAR DISPUTAS DEL CACHE DESPUÉS DE CARGAR TODO
                setTimeout(() => {
                    aplicarDisputasDesdeCache();
                }, 1000);
                
                ocultarLoading();
            })
            .withFailureHandler(function(err) {
                console.error('❌ Error en intento', intentos, ':', err);
                
                if (intentos < maxIntentos) {
                    console.log('🔄 Reintentando en 2 segundos...');
                    setTimeout(ejecutarCarga, 2000);
                    return;
                }
                
                mostrarError('Error de conexión después de ' + maxIntentos + ' intentos: ' + err.message);
            })
            .obtenerDatosParaFrontend(lob, fecha, mes);
    }
    
    ejecutarCarga();
}


  function validarYMostrarAgentes(lob) {
      console.log('🔍 Validando agentes para LOB:', lob);
      
      // ⭐ VALIDACIÓN: Verificar que datos exista
      if (!datos || !datos.metricasPorLob) {
          mostrarAlerta('❌ No se encontraron datos de métricas', 'danger');
          return;
      }
      
      if (!datos.metricasPorLob[lob]) {
          mostrarAlerta(`❌ No se encontraron métricas para esta LOB: ${lob}`, 'danger');
          return;
      }
      
      const metricas = datos.metricasPorLob[lob];
      
      // ⭐ VALIDACIÓN SEGURA para error
      if (metricas.error) {
          mostrarAlerta(`⚠️ ${metricas.error}`, 'warning');
          return;
      }
      
      let totalAgentes = 0;
      const agentesUnicos = new Set();
      const categorias = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
      
      categorias.forEach(cat => {
          if (metricas[cat]) {
              metricas[cat].forEach(kpiObj => {
                  const val = kpiObj.validacion || {};
                  
                  (val.empleadosCumplen || []).forEach(emp => {
                      agentesUnicos.add(emp.bmsId);
                      totalAgentes++;
                  });
                  
                  (val.empleadosNoCumplen || []).forEach(emp => {
                      agentesUnicos.add(emp.bmsId);
                      totalAgentes++;
                  });
                  
                  (val.empleadosSinDatos || []).forEach(emp => {
                      agentesUnicos.add(emp.bmsId);
                      totalAgentes++;
                  });
              });
          }
      });
      
      console.log(`📊 Agentes: ${agentesUnicos.size} únicos, ${totalAgentes} registros`);
      
      if (agentesUnicos.size === 0) {
          mostrarAlerta('⚠️ <strong>NO se encontraron agentes para esta LOB</strong>', 'danger');
      } else {
          mostrarAlerta(`✅ Se encontraron <strong>${agentesUnicos.size} agentes únicos</strong> (${totalAgentes} evaluaciones) para: <strong>${lob}</strong>`, 'success');
      }
  }

function generarMetricas(lobSeleccionada) {
    const empleados = empleadosPorLob[lobSeleccionada];
    
    if (!empleados || Object.keys(empleados).length === 0) {
        console.warn('⚠️ No hay empleados');
        document.getElementById('metrics-grid').style.display = 'none';
        return;
    }
    
    const keys = Object.keys(empleados);
    let totalBonos = 0;
    let totalKPIs = 0;
    let kpisCumplidos = 0;
    let topBono = 0;
    let topNombre = '';
    
    console.log('📊 Calculando métricas para LOB:', lobSeleccionada);
    
    keys.forEach(key => {
        const emp = empleados[key];
        let bonoEmpleado = 0;
        let keysDelEmpleado = [];
        let otrasCategoriasDelEmpleado = 0;
        let totalDecreases = 0;
        
        // ========================================
        // ⭐ PASO 1: CALCULAR KEYS PRIMERO
        // ========================================
        Object.keys(emp.resultados).forEach(resKey => {
            if (resKey.startsWith('keys_')) {
                const res = emp.resultados[resKey];
                totalKPIs++;
                
                if (!res.sinDatos) {
                    if (res.cumple) {
                        kpisCumplidos++;
                        const bono = res.bono || 0;
                        if (bono > 0) {
                            keysDelEmpleado.push({ cumple: true, bono: bono });
                        }
                    } else {
                        keysDelEmpleado.push({ cumple: false, bono: 0 });
                    }
                }
            }
        });
        
        // ========================================
        // ⭐ PASO 2: CALCULAR OTRAS CATEGORÍAS
        // ========================================
        Object.keys(emp.resultados).forEach(resKey => {
            const res = emp.resultados[resKey];
            
            if (resKey.startsWith('bonusStructure_') || resKey.startsWith('additionalIncentives_')) {
                totalKPIs++;
                
                if (res.cumple && !res.sinDatos) {
                    kpisCumplidos++;
                    const bono = res.bono || 0;
                    if (bono > 0) {
                        otrasCategoriasDelEmpleado += bono;
                    }
                }
            } else if (resKey.startsWith('decreasedKPIs_')) {
                totalKPIs++;
                
                if (res.cumple && !res.sinDatos) {
                    kpisCumplidos++;
                    const bono = res.bono || 0;
                    totalDecreases += bono;
                }
            }
        });
        
        // ========================================
        // ⭐ PASO 3: APLICAR LÓGICA DE KEYS
        // ========================================
        if (keysDelEmpleado.length > 0) {
            // Tiene keys: validar que todos cumplan
            const todosKeysCumplen = keysDelEmpleado.every(k => k.cumple);
            
            if (todosKeysCumplen) {
                // ✅ Todos los keys OK: sumar keys + otras
                const totalKeys = keysDelEmpleado.reduce((sum, k) => sum + k.bono, 0);
                bonoEmpleado = Math.max(0, totalKeys) + Math.max(0, otrasCategoriasDelEmpleado) + totalDecreases;
                console.log(`✅ ${emp.nombre}: Keys OK ($${totalKeys}) + Otras ($${otrasCategoriasDelEmpleado}) + Decreases ($${totalDecreases}) = $${bonoEmpleado}`);
            } else {
                // ❌ Falló algún key: solo decreases
                bonoEmpleado = totalDecreases;
                console.log(`❌ ${emp.nombre}: Falló key → $${totalDecreases} (perdió $${otrasCategoriasDelEmpleado})`);
            }
        } else {
            // Sin keys: solo otras categorías + decreases
            bonoEmpleado = Math.max(0, otrasCategoriasDelEmpleado) + totalDecreases;
            console.log(`ℹ️ ${emp.nombre}: Sin keys → $${bonoEmpleado}`);
        }
        
        // ========================================
        // ⭐ PASO 4: GARANTIZAR QUE NUNCA SEA NEGATIVO
        // ========================================
        if (bonoEmpleado < 0) {
            console.warn(`⚠️ ${emp.nombre}: Bono negativo detectado (${bonoEmpleado}), forzando a 0`);
            bonoEmpleado = 0;
        }
        
        // ⭐ SOLO SUMAR SI ES POSITIVO
        if (bonoEmpleado > 0) {
            totalBonos += bonoEmpleado;
        }
        
        // Actualizar top performer
        if (bonoEmpleado > topBono) {
            topBono = bonoEmpleado;
            topNombre = emp.nombre;
        }
    });
    
    const porcentajeCumplimiento = totalKPIs > 0 ? ((kpisCumplidos / totalKPIs) * 100).toFixed(1) : 0;
    
    console.log('📊 Resumen de métricas:');
    console.log(`   Total Bonos: $${totalBonos.toFixed(2)}`);
    console.log(`   Total KPIs: ${totalKPIs}`);
    console.log(`   KPIs Cumplidos: ${kpisCumplidos}`);
    console.log(`   Cumplimiento: ${porcentajeCumplimiento}%`);
    console.log(`   Top Performer: ${topNombre} - $${topBono.toFixed(2)}`);
    
    const metricsGrid = document.getElementById('metrics-grid');
    metricsGrid.innerHTML = `
        <div class="metric-card success">
            <div class="metric-icon">💰</div>
            <div class="metric-label">Total Bonos</div>
            <div class="metric-value">$${totalBonos.toLocaleString('es-CO', {minimumFractionDigits: 2})}</div>
            <div class="metric-subtitle">Suma total pagada</div>
        </div>
        
        <div class="metric-card">
            <div class="metric-icon">📊</div>
            <div class="metric-label">Cumplimiento</div>
            <div class="metric-value">${porcentajeCumplimiento}%</div>
            <div class="metric-subtitle">${kpisCumplidos} de ${totalKPIs} KPIs</div>
        </div>
        
        <div class="metric-card warning">
            <div class="metric-icon">👥</div>
            <div class="metric-label">Empleados</div>
            <div class="metric-value">${keys.length}</div>
            <div class="metric-subtitle">Total evaluados</div>
        </div>
        
        <div class="metric-card danger">
            <div class="metric-icon">🏆</div>
            <div class="metric-label">Top Performer</div>
            <div class="metric-value">$${topBono.toLocaleString('es-CO', {minimumFractionDigits: 2})}</div>
            <div class="metric-subtitle">${topNombre.split(',')[0]}</div>
        </div>
    `;
    
    metricsGrid.style.display = 'grid';
    console.log(`✅ Métricas generadas: ${keys.length} empleados`);
}







  function generarTop10(lobSeleccionada) {
      // ⭐ PRIMERO: Limpiar Top 10 existente
      const top10Existente = document.querySelector('.top10-container');
      if (top10Existente) {
          top10Existente.remove();
      }
      
      const empleados = empleadosPorLob[lobSeleccionada];
      
      if (!empleados || Object.keys(empleados).length === 0) return;
      
      const ranking = [];
      
      Object.keys(empleados).forEach(key => {
          const emp = empleados[key];
          let totalBono = 0;
          
          Object.values(emp.resultados).forEach(res => {
              if (res.cumple && !res.sinDatos) {
                  totalBono += res.bono || 0;
              }
          });
          
          if (totalBono > 0) {
              ranking.push({ bmsId: emp.bmsId, nombre: emp.nombre, bono: totalBono });
          }
      });
      
      ranking.sort((a, b) => b.bono - a.bono);
      const top10 = ranking.slice(0, 10);
      
      if (top10.length === 0) return;
      
      let html = '<div class="top10-container" style="background:white;border-radius:12px;padding:25px;margin-bottom:25px;box-shadow:0 5px 15px rgba(0,0,0,0.08);"><button class="accordion-header" onclick="toggleAccordion(this)"><h3 style="color:var(--primary);margin:0;">🏆 Top 10 Performers - ' + lobSeleccionada + '</h3><span class="accordion-arrow">▼</span></button><div class="accordion-content" style="display:none;grid-template-columns:1fr;gap:10px;margin-top:20px;">';
      
      top10.forEach((emp, i) => {
          const medalla = i === 0 ? '🥇' : i === 1 ? '🥈' : i === 2 ? '🥉' : (i + 1) + '.';
          const bgColor = i === 0 ? '#ffd700' : i === 1 ? '#c0c0c0' : i === 2 ? '#cd7f32' : '#f8f9fa';
          
          html += '<div style="display:flex;align-items:center;justify-content:space-between;padding:12px 15px;background:' + bgColor + '15;border-radius:8px;border-left:4px solid ' + bgColor + ';"><div style="display:flex;align-items:center;gap:10px;"><span style="font-size:1.5rem;">' + medalla + '</span><div><div style="font-weight:600;color:var(--primary);">' + emp.nombre.split(',')[0] + '</div><div style="font-size:0.85rem;color:#7f8c8d;">BMS ' + emp.bmsId + '</div></div></div><div style="text-align:right;"><div style="font-size:1.3rem;font-weight:700;color:var(--success);">$' + emp.bono.toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</div></div></div>';
      });
      
      html += '</div></div>';
      
      // ⭐ INSERTAR en posición específica (después de indicadores, antes de tabla)
      const indicadores = document.querySelector('.indicadores-container');
      const tableContainer = document.getElementById('table-container');
      
      if (indicadores && tableContainer) {
          indicadores.insertAdjacentHTML('afterend', html);
      } else if (tableContainer) {
          tableContainer.insertAdjacentHTML('beforebegin', html);
      }
      
      console.log(`✅ Top 10 generado`);
  }




  function generarIndicadoresPerformance(lobSeleccionada) {
      // ⭐ PRIMERO: Limpiar indicadores existentes
      const indicadoresExistente = document.querySelector('.indicadores-container');
      if (indicadoresExistente) {
          indicadoresExistente.remove();
      }
      
      const empleados = empleadosPorLob[lobSeleccionada];
      if (!empleados || Object.keys(empleados).length === 0) return;
      
      const metricas = datos.metricasPorLob[lobSeleccionada];
      if (!metricas) return;
      
      const categorias = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
      const indicadores = {};
      
      // ⭐ TOTAL DE AGENTES ÚNICOS
      const totalAgentesUnicos = Object.keys(empleados).length;
      
      categorias.forEach(cat => {
          const kpis = metricas[cat] || [];
          
          // ⭐ Contar cuántos agentes ÚNICOS cumplieron AL MENOS 1 KPI de esta categoría
          const agentesCumplen = new Set();
          
          kpis.forEach(kpiObj => {
              const val = kpiObj.validacion || {};
              (val.empleadosCumplen || []).forEach(emp => {
                  agentesCumplen.add(emp.bmsId);
              });
          });
          
          const cumplidos = agentesCumplen.size;
          const total = totalAgentesUnicos;
          const porcentaje = total > 0 ? ((cumplidos / total) * 100).toFixed(1) : 0;
          const color = porcentaje >= 80 ? '#27ae60' : porcentaje >= 50 ? '#f39c12' : '#e74c3c';
          
          indicadores[cat] = { cumplidos, total, porcentaje, color };
      });
      
      const nombres = { 
          keys: 'Keys', 
          bonusStructure: 'Bonus Structure', 
          additionalIncentives: 'Additional Incentives', 
          decreasedKPIs: 'Decreased KPIs' 
      };
      
      let html = '<div class="indicadores-container" style="background:white;border-radius:12px;padding:25px;margin-bottom:25px;box-shadow:0 5px 15px rgba(0,0,0,0.08);"><h3 style="color:var(--primary);margin-bottom:20px;">📈 Indicadores por Categoría</h3><div style="display:grid;grid-template-columns:repeat(auto-fit,minmax(250px,1fr));gap:15px;">';

      });
      
      html += '</div></div>';
      
      // ⭐ INSERTAR en posición específica (antes de la tabla)
      const tableContainer = document.getElementById('table-container');
      if (tableContainer) {
          tableContainer.insertAdjacentHTML('beforebegin', html);

  function crearGraficoKPIs() {
      const kpisCumplimiento = {
          keys: { cumplidos: 0, noCumplidos: 0, sinDatos: 0, valorPagado: 0 },
          bonusStructure: { cumplidos: 0, noCumplidos: 0, sinDatos: 0, valorPagado: 0 },
          additionalIncentives: { cumplidos: 0, noCumplidos: 0, sinDatos: 0, valorPagado: 0 },
          decreasedKPIs: { cumplidos: 0, noCumplidos: 0, sinDatos: 0, valorPagado: 0 }
      };
      
      const lobActual = selectLob.value;
      const empleados = empleadosPorLob[lobActual];
      
      if (!empleados || Object.keys(empleados).length === 0) return;
      
      Object.keys(empleados).forEach(empKey => {
          const emp = empleados[empKey];
          Object.keys(emp.resultados).forEach(resKey => {
              const res = emp.resultados[resKey];
              const categoria = resKey.split('_')[0];
              
              if (!kpisCumplimiento[categoria]) return;
              
              if (res.sinDatos) {
                  kpisCumplimiento[categoria].sinDatos++;
              } else if (res.cumple) {
                  kpisCumplimiento[categoria].cumplidos++;
                  kpisCumplimiento[categoria].valorPagado += res.bono || 0;
              } else {
                  kpisCumplimiento[categoria].noCumplidos++;
              }
          });
      });
      
      const ctx = document.getElementById('chartKPIs');
      if (chartKPIs) chartKPIs.destroy();
      
      const categoriasLabels = ['Keys', 'Bonus Structure', 'Additional Incentives', 'Decreased KPIs'];
      const categoriasKeys = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
      
      chartKPIs = new Chart(ctx, {
          type: 'bar',
          data: {
              labels: categoriasLabels,
              datasets: [
                  { label: 'Cumplidos', data: categoriasKeys.map(k => kpisCumplimiento[k].cumplidos), backgroundColor: 'rgba(39,174,96,0.8)', borderColor: 'rgba(39,174,96,1)', borderWidth: 2 },
                  { label: 'No Cumplidos', data: categoriasKeys.map(k => kpisCumplimiento[k].noCumplidos), backgroundColor: 'rgba(231,76,60,0.8)', borderColor: 'rgba(231,76,60,1)', borderWidth: 2 },
                  { label: 'Sin Datos', data: categoriasKeys.map(k => kpisCumplimiento[k].sinDatos), backgroundColor: 'rgba(149,165,166,0.6)', borderColor: 'rgba(149,165,166,1)', borderWidth: 2 }
              ]
          },
          options: {
              responsive: true,
              maintainAspectRatio: true,
              plugins: {
                  tooltip: { callbacks: { afterLabel: function(context) {
                      const catIndex = context.dataIndex;
                      const catKey = categoriasKeys[catIndex];
                      const valor = kpisCumplimiento[catKey].valorPagado;
                      return 'Valor: $' + valor.toLocaleString('es-CO', {minimumFractionDigits: 2});
                  }}},
                  legend: { display: true, position: 'top' }
              },
              scales: { y: { beginAtZero: true, title: { display: true, text: 'Cantidad' }, ticks: { stepSize: 1 }}}
          }
      });
  }

  function crearGraficoLOBs() {
      const bonosPorLob = {};

      // Iterar a través de todas las LOBs que tienen métricas procesadas en datos.metricasPorLob
      for (const lob in datos.metricasPorLob) {
          bonosPorLob[lob] = 0;
          const metricasDeLob = datos.metricasPorLob[lob];

          // Si hay un error para esta LOB, la saltamos
          if (metricasDeLob.error) {
              continue;
          }

          // Iterar a través de las categorías de KPIs para sumar los bonos
          ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(cat => {
              if (metricasDeLob[cat]) {
                  metricasDeLob[cat].forEach(kpiObj => {
                      const validacion = kpiObj.validacion;
                      if (validacion && validacion.empleadosCumplen) {
                          validacion.empleadosCumplen.forEach(emp => {
                              bonosPorLob[lob] += emp.bono || 0;
                          });
                      }
                  });
              }
          });
      }

      // >>> INICIO DEL CAMBIO: Filtrar LOBs con bonos mayores a 0
      const lobsConBonos = Object.keys(bonosPorLob).filter(lob => bonosPorLob[lob] > 0);
      // <<< FIN DEL CAMBIO

      const ctx = document.getElementById('chartLOBs');
      if (chartLOBs) chartLOBs.destroy();

      // Usar las LOBs filtradas para el gráfico
      const lobsLabels = lobsConBonos;
      const lobsData = lobsConBonos.map(lob => bonosPorLob[lob]);

      if (lobsLabels.length === 0) {
          // Si no hay LOBs con bonos, no se crea el gráfico
          return;
      }

      chartLOBs = new Chart(ctx, {
          type: 'bar',
          data: {
              labels: lobsLabels,
              datasets: [{ label: 'Total Bonos ($)', data: lobsData, backgroundColor: 'rgba(52,152,219,0.8)', borderColor: 'rgba(52,152,219,1)', borderWidth: 2 }]
          },
          options: {
              responsive: true,
              maintainAspectRatio: true,
              plugins: { tooltip: { callbacks: { label: function(context) { return '$' + context.parsed.y.toLocaleString('es-CO', {minimumFractionDigits: 2}); }}}, legend: { display: false }},
              scales: { y: { beginAtZero: true, title: { display: true, text: 'Total ($)' }, ticks: { callback: function(value) { return '$' + value.toLocaleString('es-CO'); }}}, x: { ticks: { autoSkip: false, maxRotation: 45, minRotation: 45 }}}
          }
      });
  }

  function crearGraficoDistribucion() {
      const lobActual = selectLob.value;
      const empleados = empleadosPorLob[lobActual];
      
      if (!empleados || Object.keys(empleados).length === 0) return;
      
      let cumplidos = 0, noCumplidos = 0, sinDatos = 0;
      
      Object.keys(empleados).forEach(empKey => {
          const emp = empleados[empKey];
          Object.values(emp.resultados).forEach(res => {
              if (res.sinDatos) sinDatos++;
              else if (res.cumple) cumplidos++;
              else noCumplidos++;
          });
      });
      
      const ctx = document.getElementById('chartDistribucion');
      if (chartDistribucion) chartDistribucion.destroy();
      
      chartDistribucion = new Chart(ctx, {
          type: 'doughnut',
          data: {
              labels: ['✅ Cumplidos', '❌ No Cumplidos', '⚠️ Sin Datos'],
              datasets: [{ data: [cumplidos, noCumplidos, sinDatos], backgroundColor: ['rgba(39,174,96,0.8)', 'rgba(231,76,60,0.8)', 'rgba(149,165,166,0.6)'], borderColor: ['rgba(39,174,96,1)', 'rgba(231,76,60,1)', 'rgba(149,165,166,1)'], borderWidth: 2 }]
          },
          options: { responsive: true, maintainAspectRatio: true, plugins: { legend: { position: 'bottom' }, tooltip: { callbacks: { label: function(context) { const total = cumplidos + noCumplidos + sinDatos; const porcentaje = ((context.parsed / total) * 100).toFixed(1); return context.label + ': ' + context.parsed + ' (' + porcentaje + '%)'; }}}}}
      });
  }

  function crearGraficoKPIPagos() {
      const lobActual = selectLob.value;
      const metricas = datos.metricasPorLob ? datos.metricasPorLob[lobActual] : null;
      
      if (!metricas || metricas.error) return;
      
      const categorias = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
      const kpisPagos = {};
      
      // Recopilar todos los pagos por KPI
      categorias.forEach(cat => {
          const kpis = metricas[cat] || [];
          kpis.forEach(kpiObj => {
              const kpiNombre = kpiObj.kpi;
              const validacion = kpiObj.validacion || {};
              
              if (!kpisPagos[kpiNombre]) {
                  kpisPagos[kpiNombre] = {
                      nombre: kpiNombre,
                      totalPagado: 0,
                      categoria: cat,
                      agentesQueCobraron: 0
                  };
              }
              
              // Sumar bonos de empleados que cumplieron
              (validacion.empleadosCumplen || []).forEach(emp => {
                  kpisPagos[kpiNombre].totalPagado += emp.bono || 0;
                  kpisPagos[kpiNombre].agentesQueCobraron++;
              });
          });
      });
      
      // Convertir a array y ordenar por monto pagado (descendente)
      const kpisArray = Object.values(kpisPagos).sort((a, b) => b.totalPagado - a.totalPagado);
      
      if (kpisArray.length === 0) {
          document.getElementById('kpi-payment-chart-container').style.display = 'none';
          return;
      }
      
      // Preparar datos para el gráfico
      const labels = kpisArray.map(k => k.nombre);
      const data = kpisArray.map(k => k.totalPagado);
      const agentes = kpisArray.map(k => k.agentesQueCobraron);
      
      // Colores según categoría
      const coloresPorCategoria = {
          'keys': 'rgba(52, 152, 219, 0.8)',
          'bonusStructure': 'rgba(39, 174, 96, 0.8)',

          'decreasedKPIs': 'rgba(231, 76, 60, 0.8)'
      };
      
      const backgroundColors = kpisArray.map(k => coloresPorCategoria[k.categoria] || 'rgba(149, 165, 166, 0.8)');
      const borderColors = kpisArray.map(k => coloresPorCategoria[k.categoria]?.replace('0.8', '1') || 'rgba(149, 165, 166, 1)');
      
      const ctx = document.getElementById('chartKPIPagos');
      if (chartKPIPagos) chartKPIPagos.destroy();
      
      chartKPIPagos = new Chart(ctx, {
          type: 'bar',
          data: {
              labels: labels,
              datasets: [{
                  label: 'Total Pagado ($)',
                  data: data,
                  backgroundColor: backgroundColors,
                  borderColor: borderColors,
                  borderWidth: 2
              }]
          },
          options: {
              responsive: true,
              maintainAspectRatio: true,
              indexAxis: 'y', // Gráfico horizontal
              plugins: {
                  tooltip: {
                      callbacks: {
                          label: function(context) {
                              const valor = context.parsed.x;
                              const index = context.dataIndex;
                              const agentesCount = agentes[index];
                              return [
                                  'Total: $' + valor.toLocaleString('es-CO', {minimumFractionDigits: 2}),
                                  'Agentes que cobraron: ' + agentesCount
                              ];
                          }
                      }
                  },
                  legend: {
                      display: false
                  },
                  title: {
                      display: true,
                      text: 'Monto total pagado por cada KPI - LOB: ' + lobActual,
                      font: {
                          size: 14,
                          weight: 'bold'
                      }
                  }
              },
              scales: {
                  x: {
                      beginAtZero: true,
                      title: {
                          display: true,
                          text: 'Monto Total Pagado ($)'
                      },
                      ticks: {
                          callback: function(value) {
                              return '$' + value.toLocaleString('es-CO');
                          }
                      }
                  },
                  y: {
                      title: {
                          display: true,
                          text: 'KPIs'
                      },
                      ticks: {
                          autoSkip: false,
                          font: {
                              size: 10
                          }
                      }
                  }
              }
          }
      });
      
      // Mostrar el contenedor del gráfico
      document.getElementById('kpi-payment-chart-container').style.display = 'block';
      
      console.log('✅ Gráfico de KPI Pagos generado:', kpisArray.length, 'KPIs');
  }

function mostrarTablaPorLob(lobSeleccionada) {
    const metricas = datos.metricasPorLob ? datos.metricasPorLob[lobSeleccionada] : null;
    
    if (!metricas || metricas.error) {
        const container = document.getElementById('table-container');
        container.innerHTML = '<div class="info-message">' + (metricas?.error || 'No hay datos') + '</div>';
        container.style.display = 'block';
        return;
    }
    
    const categorias = ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'];
    const empleados = {};
    
    // ⭐⭐⭐ OBTENER AGENTES PERMITIDOS DEL OBJETO DATOS
    const agentesPermitidos = datos.agentesPermitidos || {};
    
    console.log('📋 Agentes permitidos recibidos:', Object.keys(agentesPermitidos).length);
    
    // ⭐ DEBUG: Primeros 3 agentes
    let debugCount = 0;
    for (const bms in agentesPermitidos) {
        if (debugCount >= 3) break;
        const ag = agentesPermitidos[bms];
        console.log(`  ${bms}: Nombre="${ag.nombre}", Role="${ag.rol}", Horas=${ag.horasSemanales}h`);
        debugCount++;
    }
    
    categorias.forEach(cat => {
        const kpis = metricas[cat] || [];
        kpis.forEach((kpiObj, idx) => {
            const valid = kpiObj.validacion || {};
            const tipoValidacion = kpiObj.tipo || 'mayor';
            
            // ⭐ EMPLEADOS QUE CUMPLEN
            (valid.empleadosCumplen || []).forEach(emp => {
                const key = emp.bmsId + '_' + emp.nombre;
                if (!empleados[key]) {
                    // ⭐⭐⭐ OBTENER ROLE Y HORAS DESDE agentesPermitidos
                    const datosAgente = agentesPermitidos[emp.bmsId] || {};
                    
                    empleados[key] = {
                        bmsId: emp.bmsId, 
                        nombre: emp.nombre, 
                        rol: datosAgente.rol || 'Sin rol',           // ⭐⭐⭐ NUEVO
                        horasSemanales: datosAgente.horasSemanales || 42, // ⭐⭐⭐
                        resultados: {}
                    };
                    
                    console.log(`✅ Empleado agregado: ${emp.bmsId} - Role: ${empleados[key].rol}`);
                }
                
                const esKey = cat === 'keys';
                
                empleados[key].resultados[cat + '_' + idx] = {
                    ...emp,
                    cumple: true,
                    sinDatos: false,
                    esKey: esKey,
                    tipoValidacion: tipoValidacion,
                    categoria: cat
                };
            });
            
            // ⭐ EMPLEADOS QUE NO CUMPLEN
            (valid.empleadosNoCumplen || []).forEach(emp => {
                const key = emp.bmsId + '_' + emp.nombre;
                if (!empleados[key]) {
                    const datosAgente = agentesPermitidos[emp.bmsId] || {};
                    
                    empleados[key] = {
                        bmsId: emp.bmsId, 
                        nombre: emp.nombre, 
                        rol: datosAgente.rol || 'Sin rol',           // ⭐⭐⭐ NUEVO
                        horasSemanales: datosAgente.horasSemanales || 42,
                        resultados: {}
                    };
                }
                
                const esKey = cat === 'keys';
                
                empleados[key].resultados[cat + '_' + idx] = {
                    ...emp,
                    cumple: false,
                    sinDatos: false,
                    esKey: esKey,
                    tipoValidacion: tipoValidacion,
                    categoria: cat
                };
            });
            
            // ⭐ EMPLEADOS SIN DATOS
            (valid.empleadosSinDatos || []).forEach(emp => {
                const key = emp.bmsId + '_' + emp.nombre;
                if (!empleados[key]) {
                    const datosAgente = agentesPermitidos[emp.bmsId] || {};
                    
                    empleados[key] = {
                        bmsId: emp.bmsId, 
                        nombre: emp.nombre, 
                        rol: datosAgente.rol || 'Sin rol',           // ⭐⭐⭐ NUEVO
                        horasSemanales: datosAgente.horasSemanales || 42,
                        resultados: {}
                    };
                }
                
                empleados[key].resultados[cat + '_' + idx] = {
                    ...emp,
                    cumple: false,
                    sinDatos: true,
                    esKey: cat === 'keys',
                    tipoValidacion: tipoValidacion,
                    categoria: cat
                };
            });
        });
    });
    
    empleadosPorLob[lobSeleccionada] = empleados;
    
    console.log(`✅ ${Object.keys(empleados).length} empleados procesados`);
    
    // ⭐ CARGAR TODAS LAS LOBS DEL SISTEMA
    mostrarLoading('Cargando LOBs disponibles...');
    
    google.script.run
        .withSuccessHandler(function(todasLasLobs) {
            console.log('✅ Todas las LOBs del sistema:', todasLasLobs);
            generarTablaConLobs(lobSeleccionada, empleados, metricas, categorias, todasLasLobs);
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error al cargar LOBs:', error);
            const todasLasLobs = datos.lobs || [lobSeleccionada];
            generarTablaConLobs(lobSeleccionada, empleados, metricas, categorias, todasLasLobs);
        })
        .obtenerTodasLasLobsDelSistema();
}

function generarTablaConLobs(lob, empleados, metricas, categorias, todasLasLobs) {
    if (!empleados || !metricas || !categorias || !todasLasLobs) {
        console.error('❌ Parámetros inválidos en generarTablaConLobs');
        mostrarError('Datos incompletos para generar la tabla');
        return;
    }
    
    console.log('📊 Generando tabla para LOB:', lob);
    ocultarLoading();
    
    const traslados = datos.trasladosPorLob && datos.trasladosPorLob[lob] ? datos.trasladosPorLob[lob] : { salientes: [], entrantes: [] };
    const agentesTrasladados = traslados.salientes.map(t => t.bmsId);
    
    // ========================================
    // ⭐ OBTENER HORAS SEMANALES DESDE agentesPermitidos (H_C columna J)
    // ========================================
    const horasSemanalesPorAgente = {};
    const agentesPermitidos = datos.agentesPermitidos || {};
    
    console.log('📊 Extrayendo horas semanales de agentesPermitidos...');
    
    Object.keys(agentesPermitidos).forEach(bmsId => {
        const agente = agentesPermitidos[bmsId];
        
        const normalizar = s => {
            if (!s) return "";
            return s.toString()
                .normalize("NFD")
                .replace(/[\u0300-\u036f]/g, "")
                .toLowerCase()
                .trim()
                .replace(/\s+/g, ' ');
        };
        
        if (normalizar(agente.lob) === normalizar(lob)) {
            horasSemanalesPorAgente[bmsId] = agente.horasSemanales || 42;
        }
    });
    
    console.log(`✅ Horas semanales cargadas: ${Object.keys(horasSemanalesPorAgente).length} agentes de ${lob}`);
    
    let totalGeneralTabla = 0;
    
    // ========================================
    // ⭐⭐⭐ THEAD - FILA 1 (CON COLUMNA ROLE)
    // ========================================
    let theadHTML = '<tr><th rowspan="2" onclick="ordenarTabla(0)" style="cursor:pointer;" title="Ordenar">#</th>';
    theadHTML += '<th rowspan="2" style="background:#3498db;color:white;min-width:200px;">🔄 Trasladar</th>';
    theadHTML += '<th rowspan="2" onclick="ordenarTabla(1)" style="cursor:pointer;" title="Ordenar">CCMS ID</th>';
    theadHTML += '<th rowspan="2" onclick="ordenarTabla(2)" style="cursor:pointer;" title="Ordenar">Nombre</th>';
    theadHTML += '<th rowspan="2" onclick="ordenarTabla(3)" style="cursor:pointer;background:#667eea;color:white;" title="Ordenar">Role</th>';
    theadHTML += '<th rowspan="2" style="background:#16a085;color:white;">⏱️ Horas Semanales</th>';
        
    categorias.forEach(cat => {
        const kpis = metricas[cat] || [];
        theadHTML += kpis.length > 0 ? '<th colspan="' + (kpis.length * 2) + '">' + cat + '</th>' : '<th colspan="2">' + cat + '</th>';
    });
    
    theadHTML += '<th rowspan="2" onclick="ordenarTabla(-1)" style="cursor:pointer;background:#27ae60;" title="Ordenar">Total $</th>';
    theadHTML += '<th rowspan="2" style="background:#16a085;color:white;">📊 Asistencia<br><small style="font-size:0.65rem;font-weight:normal;">(Auto)</small></th>';
    theadHTML += '<th rowspan="2" style="background:#28a745;color:white;min-width:100px;">✅ Aprobar<br><label style="font-size:0.75rem;font-weight:normal;cursor:pointer;margin-top:5px;display:inline-block;"><input type="checkbox" id="select-all-aprobar" onchange="seleccionarTodosAprobar(this)" style="width:16px;height:16px;cursor:pointer;margin-right:5px;">Todos</label></th>';
    theadHTML += '<th rowspan="2" style="background:#9b59b6;color:white;">Disputar</th>';
    theadHTML += '<th rowspan="2" style="background:#9b59b6;color:white;">Comentario</th></tr><tr>';
    
    // ========================================
    // ⭐ THEAD - FILA 2 (Headers de KPIs)
    // ========================================
    categorias.forEach(cat => {
        const kpis = metricas[cat] || [];
        if (kpis.length === 0) {
            theadHTML += '<th colspan="2">-</th>';
        } else {
            kpis.forEach(kpiObj => {
                const esMultiTier = kpiObj.esMultiTier === true;
                
                let headerKPI = kpiObj.kpi;
                let headerScore = '';
                
                if (esMultiTier) {
                    headerScore = 'Multi-Escala<br><small style="font-size:0.6rem;color:#7f8c8d;font-weight:normal;">';
                    
                    if (kpiObj.scoresPorTier && kpiObj.scoresPorTier.length > 0) {
                        kpiObj.scoresPorTier.forEach((tier, idx) => {
                            const part = tier.participacion ? (tier.participacion * 100).toFixed(0) + '%' : '100%';
                            const sep = idx < kpiObj.scoresPorTier.length - 1 ? ' | ' : '';
                            
                            headerScore += `${tier.score} <span style="color:#f39c12;">(${part})</span>${sep}`;
                        });
                    }
                    
                    headerScore += '</small>';
                } else {
                    headerScore = kpiObj.score || '-';
                }
                
                theadHTML += `<th style="font-size:0.8rem;line-height:1.4;">${headerKPI}<br>${headerScore}</th>`;
                theadHTML += '<th style="font-size:0.8rem;background:#34495e;">Bono $</th>';
            });
        }
    });
    
    theadHTML += '</tr>';
    
    // ========================================
    // ⭐⭐⭐ TBODY
    // ========================================
    let tbodyHTML = '';
    const keys = Object.keys(empleados);
    
    if (keys.length === 0) {
        tbodyHTML = '<tr><td colspan="100" class="info-message">No hay empleados</td></tr>';
    } else {
        let contadorVisible = 0;
        
        // ⭐⭐⭐ LOOP DE EMPLEADOS ⭐⭐⭐
        keys.forEach((empKey, i) => {
            const emp = empleados[empKey];
            const fueTrasladado = agentesTrasladados.includes(emp.bmsId);
            
            let tieneSalarioPersonalizado = false;
            Object.values(emp.resultados).forEach(res => {
                if (res.salarioPersonalizado === true) {
                    tieneSalarioPersonalizado = true;
                }
            });

            let estiloFila = '';
            let claseFila = '';

            if (fueTrasladado) {
                estiloFila = 'style="background:linear-gradient(135deg, #fff3cd 0%, #ffeaa7 100%);border-left:4px solid #f39c12;"';
                claseFila = 'class="agente-trasladado"';
            } else if (tieneSalarioPersonalizado) {
                estiloFila = 'style="background:linear-gradient(135deg, #fff9c4 0%, #ffeb3b 100%);border-left:4px solid #fbc02d;"';
                claseFila = 'class="agente-salario-personalizado"';
            }
            
            tbodyHTML += `<tr data-bms="${emp.bmsId}" data-nombre="${emp.nombre.toLowerCase()}" id="row-${emp.bmsId}" ${estiloFila} ${claseFila}>`;
            tbodyHTML += '<td>' + (++contadorVisible) + '</td>';

            tbodyHTML += '<td style="padding:8px;"><div style="display:flex;align-items:center;gap:8px;justify-content:center;">';
            
            if (fueTrasladado) {
                const trasladoInfo = traslados.salientes.find(t => t.bmsId === emp.bmsId);
                tbodyHTML += `<span style="background:#f39c12;color:white;padding:4px 8px;border-radius:10px;font-size:0.7rem;font-weight:bold;white-space:nowrap;">📤 Trasladado a ${trasladoInfo ? trasladoInfo.lobDestino : 'otra LOB'}</span>`;
            } else {
                tbodyHTML += '<input type="checkbox" class="recalc-check" data-bms="' + emp.bmsId + '" onchange="toggleRecalcLob(this)" style="width:18px;height:18px;cursor:pointer;">';
                tbodyHTML += '<select class="recalc-lob-select" data-bms="' + emp.bmsId + '" disabled style="padding:5px;border:1px solid #ddd;border-radius:4px;font-size:0.85rem;width:110px;"><option value="">LOB destino</option>';

            // ⭐⭐⭐ NOMBRE
            tbodyHTML += '<td style="text-align:left;">' + emp.nombre;

                tbodyHTML += '<br><small style="color:#e67e22;font-weight:bold;">🚨 Trasladado temporalmente</small>';
            }
            
            if (tieneSalarioPersonalizado) {
                tbodyHTML += '<br><small style="background:#fbc02d;color:#000;padding:2px 8px;border-radius:10px;font-size:0.7rem;font-weight:bold;">💰 Salario Personalizado</small>';
            }
            
            tbodyHTML += '</td>';

            // ⭐⭐⭐ COLUMNA: ROLE
            tbodyHTML += '<td style="text-align:center;">';
            const roleAgente = agentesPermitidos[emp.bmsId]?.rol || emp.rol || '-';
            tbodyHTML += '<span style="background:#667eea15;color:#5b5fc7;padding:6px 12px;border-radius:8px;font-size:0.85rem;font-weight:700;display:inline-block;box-shadow:0 2px 4px rgba(102,126,234,0.2);">' + roleAgente + '</span>';

            // ========================================

            // ========================================
            tbodyHTML += '<td style="text-align:center;">';
            const horasSemanales = horasSemanalesPorAgente[emp.bmsId] || 42;
            tbodyHTML += `<span style="background:#16a085;color:white;padding:4px 10px;border-radius:12px;font-weight:600;">${horasSemanales}h/sem</span>`;
            tbodyHTML += '</td>';

            // ========================================
            // ⭐⭐⭐ CALCULAR TOTAL CON LÓGICA DE KEYS
            // ========================================
            let totalKeys = 0;
            let todosKeysOK = true;
            let tieneAlgunKey = false;
            
            const kpisKeys = metricas['keys'] || [];
            kpisKeys.forEach((kpiObj, idx) => {
                const res = emp.resultados['keys_' + idx] || {};
                const bono = res.bono || 0;
                
                if (!res.sinDatos) {
                    tieneAlgunKey = true;
                    if (res.cumple) {
                        totalKeys += bono;
                    } else {
                        todosKeysOK = false;
                    }
                }
            });
            
            if (tieneAlgunKey && !todosKeysOK) {
                totalKeys = 0;
            }
            
            let totalOtrasCategorias = 0;
            let totalDecreases = 0;

            // ========================================
            // ⭐⭐⭐ GENERAR CELDAS DE KPIs ⭐⭐⭐
            // ========================================
            categorias.forEach(cat => {
                const kpis = metricas[cat] || [];
                
                if (kpis.length === 0) {

                    kpis.forEach((kpiObj, idx) => {
                        const res = emp.resultados[cat + '_' + idx] || {};
                        const bono = res.bono || 0;
                        const esKey = cat === 'keys';
                        const esDecrease = cat === 'decreasedKPIs';
                        
                        // ACUMULAR EN CATEGORÍAS
                        if (!esKey && res.cumple && !res.sinDatos) {
                            if (esDecrease) {
                                totalDecreases += bono;
                            } else {

                        const esMultiTier = kpiObj.esMultiTier === true;
                        
                        if (esMultiTier) {
                            const tieneScoreCumplido = res.scoreCumplido && 
                                                      res.scoreCumplido !== null && 
                                                      res.scoreCumplido !== undefined &&
                                                      res.scoreCumplido !== '';
                            
                            if (tieneScoreCumplido && res.cumple) {
                                scoreAMostrar = res.scoreCumplido;
                                
                                if (res.participacionUsada !== undefined && res.participacionUsada !== null) {
                                    participacionAMostrar = (res.participacionUsada * 100).toFixed(0) + '%';
                                } else {
                                    const tierMatch = kpiObj.scoresPorTier?.find(t => {
                                        const tierScore = String(t.score).trim().toLowerCase();
                                        const cumplido = String(res.scoreCumplido).trim().toLowerCase();
                                        return tierScore === cumplido;
                                    });
                                    
                                    if (tierMatch && tierMatch.participacion) {
                                        participacionAMostrar = (tierMatch.participacion * 100).toFixed(0) + '%';
                                    }
                                }
                            }
                        }
                        
                        // FORMATEAR valorF
                        let valorFormateado = '-';
                        if (res.valorF !== undefined && res.valorF !== null) {
                            const valor = parseFloat(res.valorF);
                            if (!isNaN(valor)) {
                                if (Number.isInteger(valor)) {
                                    valorFormateado = valor.toLocaleString('es-CO', {
                                        minimumFractionDigits: 0,
                                        maximumFractionDigits: 0
                                    });
                                } else {
                                    valorFormateado = valor.toLocaleString('es-CO', {
                                        minimumFractionDigits: 2,
                                        maximumFractionDigits: 2
                                    });
                                }
                            } else {
                                valorFormateado = res.valorF;
                            }
                        }
                        
                        // ========================================
                        // ⭐ RENDERIZADO SEGÚN TIPO DE KPI
                        // ========================================
                        if (esKey) {
                            // KEYS
                            if (res.sinDatos) {
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">Sin datos</td>';
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">$0.00</td>';
                            } else if (res.cumple) {
                                tbodyHTML += '<td style="background:#d4edda;color:#155724;font-weight:bold;">' + valorFormateado + '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small></td>';
                                tbodyHTML += '<td style="background:#d4edda;color:#155724;font-weight:bold;">$' + bono.toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</td>';
                            } else {
                                tbodyHTML += '<td style="background:#f8d7da;color:#721c24;font-weight:bold;">' + valorFormateado + '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small></td>';
                                tbodyHTML += '<td style="background:#f8d7da;color:#e74c3c;font-weight:bold;">$0.00</td>';
                            }
                        } else if (esDecrease) {
                            // DECREASED KPIs
                            if (res.sinDatos) {
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">Sin datos</td>';
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">$0.00</td>';
                            } else if (res.cumple) {
                                const bonoFormateado = bono < 0 
                                    ? '-$' + Math.abs(bono).toLocaleString('es-CO', {minimumFractionDigits: 2})
                                    : '$' + bono.toLocaleString('es-CO', {minimumFractionDigits: 2});
                                
                                let infoDecrease = '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small>';
                                
                                if (esMultiTier && participacionAMostrar) {
                                    infoDecrease += `<br><small style="font-size:0.65rem;color:#f39c12;">📊 ${participacionAMostrar}</small>`;
                                }
                                
                                infoDecrease += '<br><small style="font-size:0.65rem;color:#e74c3c;">❌ Penalización</small>';
                                
                                tbodyHTML += '<td style="background:#f8d7da;color:#721c24;font-weight:bold;">' + valorFormateado + infoDecrease + '</td>';
                                tbodyHTML += '<td style="background:#f8d7da;color:#e74c3c;font-weight:bold;">' + bonoFormateado + '</td>';
                            } else {
                                let infoDecrease = '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small>';
                                
                                if (esMultiTier && participacionAMostrar) {
                                    infoDecrease += `<br><small style="font-size:0.65rem;color:#f39c12;">📊 ${participacionAMostrar}</small>`;
                                }
                                
                                infoDecrease += '<br><small style="font-size:0.65rem;color:#27ae60;">✅ Sin penalización</small>';
                                
                                tbodyHTML += '<td style="background:#d4edda;color:#155724;font-weight:bold;">' + valorFormateado + infoDecrease + '</td>';
                                tbodyHTML += '<td style="background:#d4edda;color:#155724;font-weight:bold;">$0.00</td>';
                            }
                        } else {
                            // ⭐⭐⭐ BONUS STRUCTURE Y ADDITIONAL INCENTIVES ⭐⭐⭐
                            if (res.sinDatos) {
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">Sin datos</td>';
                                tbodyHTML += '<td style="background:#f0f0f0;">$0.00</td>';
                            } else {
                                const claseEstilo = res.cumple ? 'cumple' : 'no-cumple';
                                
                                let infoAdicional = '';
                                
                                if (esMultiTier && res.cumple && participacionAMostrar) {
                                    infoAdicional = `
                                        <br><small style="font-size:0.65rem;color:#3498db;font-weight:600;">
                                            ⭐ Tier: ${scoreAMostrar}
                                        </small>
                                        <br><small style="font-size:0.65rem;color:#f39c12;font-weight:600;background:rgba(243,156,18,0.15);padding:2px 6px;border-radius:8px;display:inline-block;margin-top:2px;">
                                            📊 ${participacionAMostrar}
                                        </small>
                                    `;
                                } else if (!esMultiTier) {
                                    infoAdicional = '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small>';
                                }
                                
                                tbodyHTML += '<td class="' + claseEstilo + '">' 
                                    + valorFormateado 
                                    + infoAdicional
                                    + '</td>';
                                tbodyHTML += '<td>$' + Math.max(0, bono).toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</td>';
                            }
                        }
                    });
                }
            });
            
            // ========================================
            // ⭐⭐⭐ CALCULAR TOTAL FINAL
            // ========================================
            let totalFinal = 0;
            let mensajeTotal = '';
            let estiloTotal = 'background:#d5f4e6;font-weight:bold;';
            
            if (tieneAlgunKey) {
                if (todosKeysOK) {
                    totalFinal = Math.max(0, totalKeys) + Math.max(0, totalOtrasCategorias) + totalDecreases;
                } else {
                    totalFinal = totalDecreases;
                    if (totalOtrasCategorias > 0) {
                        estiloTotal = 'background:#f8d7da;font-weight:bold;border-left:4px solid #e74c3c;';

                    }
                }
            } else {
                totalFinal = Math.max(0, totalOtrasCategorias) + totalDecreases;
            }
            
            totalFinal = Math.max(0, totalFinal);

            // ⭐⭐⭐ PRORRATEO AUTOMÁTICO POR AUSENCIAS (Horas_N)
            // ========================================
            let totalAjustado = totalFinal;
            let mensajeAjuste = '';
            
            const horasData = datos.horasPorAgente && datos.horasPorAgente[emp.bmsId] 
                ? datos.horasPorAgente[emp.bmsId] 
                : null;
            
            if (horasData && horasData.horas !== undefined && totalFinal > 0) {
                const horasReales = horasData.horas || 0;
                
                // Calcular horas esperadas del mes
                let diasLaborables = 22;
                if (datos.mesSeleccionado && datos.fechaSeleccionada) {
                    const fecha = new Date(datos.fechaSeleccionada);
                    const anio = fecha.getFullYear();
                    const mes = datos.mesSeleccionado;
                    const primerDia = new Date(anio, mes - 1, 1);
                    const ultimoDia = new Date(anio, mes, 0);
                    let dias = 0;
                    let d = new Date(primerDia);
                    while (d <= ultimoDia) {
                        const diaSemana = d.getDay();
                        if (diaSemana !== 0 && diaSemana !== 6) dias++;
                        d.setDate(d.getDate() + 1);
                    }
                    if (dias > 0) diasLaborables = dias;
                }
                
                const horasEsperadas = (horasSemanales / 5) * diasLaborables;
                
                // ⭐ APLICAR PRORRATEO SOLO SI TRABAJÓ MENOS
                if (horasReales < horasEsperadas && horasEsperadas > 0) {
                    const factorProrrateo = horasReales / horasEsperadas;
                    totalAjustado = totalFinal * factorProrrateo;
                    
                    mensajeAjuste = `<br><small style="font-size:0.65rem;color:#e74c3c;font-weight:600;background:rgba(231,76,60,0.1);padding:3px 8px;border-radius:8px;display:inline-block;margin-top:4px;">
                        📊 Ausencias: ${horasReales.toFixed(1)}h / ${horasEsperadas.toFixed(1)}h = ${(factorProrrateo * 100).toFixed(1)}%
                    </small>`;
                    
                    console.log(`⚙️ Prorrateo automático: ${emp.nombre}`);
                    console.log(`   Horas semanales (H_C): ${horasSemanales}h`);
                    console.log(`   Días laborables: ${diasLaborables}`);
                    console.log(`   Horas esperadas: ${horasEsperadas.toFixed(1)}h`);
                    console.log(`   Horas reales (Horas_${datos.mesSeleccionado}): ${horasReales.toFixed(1)}h`);
                    console.log(`   Factor: ${(factorProrrateo * 100).toFixed(1)}%`);
                    console.log(`   Bono: $${totalFinal.toFixed(2)} → $${totalAjustado.toFixed(2)}`);
                }
            }
            
            totalGeneralTabla += totalAjustado;
            
            // CELDA TOTAL
            tbodyHTML += `<td class="celda-total" data-bms="${emp.bmsId}" data-bono-original="${totalAjustado}" style="${estiloTotal}">$${totalAjustado.toLocaleString('es-CO', {minimumFractionDigits: 2})}${mensajeTotal}${mensajeAjuste}</td>`;
            
            // CELDA PRORRATEO AUTOMÁTICO (Indicador visual)
            tbodyHTML += '<td style="text-align:center;padding:8px;">';
            if (horasData && horasData.horas !== undefined && totalFinal > 0) {
                const horasReales = horasData.horas || 0;
                let diasLaborables = 22;
                
                if (datos.mesSeleccionado && datos.fechaSeleccionada) {
                    const fecha = new Date(datos.fechaSeleccionada);
                    const anio = fecha.getFullYear();
                    const mes = datos.mesSeleccionado;
                    const primerDia = new Date(anio, mes - 1, 1);
                    const ultimoDia = new Date(anio, mes, 0);
                    let dias = 0;
                    let d = new Date(primerDia);
                    while (d <= ultimoDia) {
                        const diaSemana = d.getDay();
                        if (diaSemana !== 0 && diaSemana !== 6) dias++;
                        d.setDate(d.getDate() + 1);
                    }
                    if (dias > 0) diasLaborables = dias;
                }
                
                const horasEsperadas = (horasSemanales / 5) * diasLaborables;
                
                if (horasReales < horasEsperadas && horasEsperadas > 0) {
                    const ausenciasPct = ((1 - (horasReales / horasEsperadas)) * 100).toFixed(1);
                    tbodyHTML += `<span style="background:#e74c3c;color:white;padding:4px 10px;border-radius:12px;font-weight:600;font-size:0.75rem;" title="Prorrateo aplicado: ${horasReales.toFixed(1)}h de ${horasEsperadas.toFixed(1)}h esperadas">
                        ⚠️ -${ausenciasPct}%
                    </span>`;
                } else {
                    tbodyHTML += `<span style="background:#27ae60;color:white;padding:4px 10px;border-radius:12px;font-weight:600;font-size:0.75rem;" title="Sin ausencias: ${horasReales.toFixed(1)}h / ${horasEsperadas.toFixed(1)}h">
                        ✓ 100%
                    </span>`;
                }
            } else {
                tbodyHTML += '<span style="color:#95a5a6;font-size:0.85rem;">-</span>';
            }
            tbodyHTML += '</td>';
            
            // CELDA APROBAR
            tbodyHTML += '<td style="text-align:center;padding:8px;background:#f0fff4;">';
            tbodyHTML += `<input type="checkbox" class="aprobado-check" data-bms="${emp.bmsId}" data-nombre="${emp.nombre}" data-total="${totalAjustado}" onchange="actualizarTotalAprobar()" style="width:22px;height:22px;cursor:pointer;accent-color:#28a745;">`;
            tbodyHTML += '</td>';
            
            // DISPUTAR Y COMENTARIO
            tbodyHTML += '<td style="text-align:center;">';
            tbodyHTML += '<input type="checkbox" class="dispute-check" data-bms="' + emp.bmsId + '" data-nombre="' + emp.nombre + '" data-total="' + totalAjustado + '" onchange="toggleDisputeComment(this)" style="width:20px;height:20px;cursor:pointer;">';
            tbodyHTML += '</td>';
            tbodyHTML += '<td class="celda-comentario-disputa" data-bms="' + emp.bmsId + '" style="padding:8px;font-size:0.85rem;color:#7f8c8d;font-style:italic;">';
            tbodyHTML += '<span class="texto-comentario">Sin comentario</span>';
            tbodyHTML += '</td>';
            tbodyHTML += '</tr>';
        });
    }
    
    // FILA DE TOTALES
    if (keys.length > 0) {
        let colspanAntes = 6;
        categorias.forEach(cat => {
            const kpis = metricas[cat] || [];
            colspanAntes += kpis.length > 0 ? (kpis.length * 2) : 2;
        });
        
        tbodyHTML += '<tr style="background:#27ae60;color:white;font-weight:bold;font-size:1.1rem;">';
        tbodyHTML += '<td colspan="' + colspanAntes + '" style="text-align:right;padding:15px;">TOTAL GENERAL (CON AJUSTES AUTOMÁTICOS):</td>';
        tbodyHTML += '<td style="text-align:center;padding:15px;font-size:1.3rem;">$' + totalGeneralTabla.toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</td>';
        tbodyHTML += '<td colspan="4"></td>';
        tbodyHTML += '</tr>';
    }
    
    // CONSTRUCCIÓN HTML FINAL
    const container = document.getElementById('table-container');
    let html = '<div class="table-header"><h3 class="table-title">📋 Agentes - ' + lob + '</h3>';
    html += '<div style="display:flex;align-items:center;gap:15px;flex-wrap:wrap;">';
    html += '<input type="text" id="search-input" placeholder="🔍 Buscar..." onkeyup="filtrarTabla()" style="padding:10px 15px;border:2px solid #dfe6e9;border-radius:8px;font-size:1rem;width:300px;">';
    html += '<span id="search-count" style="color:#7f8c8d;font-size:0.9rem;"></span>';
    html += '<div id="total-aprobar-indicator" style="background:#28a745;color:white;padding:10px 20px;border-radius:8px;font-weight:bold;display:none;"><span>Total a aprobar: </span><span id="total-aprobar-monto">$0</span></div>';
    html += '<div class="btn-group">';
    html += '<button class="btn-success" onclick="exportarAExcel()">📥 Excel</button>';
    html += '<button class="btn-primary" onclick="aplicarTraslados()" style="background:#e67e22;">🔄 Aplicar Traslados</button>';
    html += '<button class="btn-primary" onclick="enviaraprobados()" style="background:#28a745;">✅ Enviar y Aprobar</button>';
    html += '<button class="btn-primary" onclick="actualizarSoloTabla()" style="background:#3498db;">🔄 Actualizar Tabla</button>';
    html += '<button class="btn-primary" onclick="toggleTabla()">👁️ Ocultar</button>';
    html += '</div></div></div>';
    html += '<div style="background:#d1ecf1;border-left:4px solid #0c5460;padding:12px;margin-bottom:15px;border-radius:6px;"><strong>ℹ️ Prorrateo Automático:</strong> <span style="font-size:0.9rem;">El ajuste por ausencias se aplica automáticamente basado en las horas de la hoja Horas_' + (datos.mesSeleccionado || 'N') + '. Los agentes con ausencias muestran un indicador rojo con el porcentaje de reducción.</span></div>';
    
    if (traslados.salientes.length > 0 || traslados.entrantes.length > 0) {
        html += '<div style="margin-bottom:20px;">';
        if (traslados.salientes.length > 0) {
            html += `<div style="background:#fff3cd;border-left:4px solid #f39c12;padding:12px;margin-bottom:10px;border-radius:6px;"><strong>📤 Agentes Trasladados:</strong> <span style="font-size:0.9rem;">${traslados.salientes.length} agente(s) salieron</span></div>`;

    }
    
    html += '<div id="tabla-detalle" class="table-scroll"><table id="tabla-agentes"><thead>' + theadHTML + '</thead><tbody id="tbody-agentes">' + tbodyHTML + '</tbody></table>';
    html += '<div style="text-align:center;padding:20px;border-top:3px solid #ecf0f1;"><button class="btn-dispute" onclick="enviarDisputas()" style="padding:15px 40px;font-size:1.1rem;">📨 Enviar Disputas</button></div></div>';
    
    container.innerHTML = html;
    container.style.display = 'block';
    
    setTimeout(function() {
        actualizarTotalAprobar();
        setTimeout(() => {
            aplicarDisputasDesdeCache();
        }, 500);
    }, 1000);
}

    valor = Number(valor) || 0;
    return '$' + valor.toLocaleString('es-CO', {
        minimumFractionDigits: 0,
        maximumFractionDigits: 0
    });
}

// ⭐ FUNCIÓN: FORMATEAR NÚMERO SIN DECIMALES
// ========================================
function formatearNumero(valor) {
    if (typeof valor !== 'number') {
        valor = parseFloat(valor) || 0;
    }
    
    return valor.toLocaleString('en-US', {
        minimumFractionDigits: 0,
        maximumFractionDigits: 0
    });
}

  function ordenarTabla(columnIndex) {
      const tbody = document.getElementById('tbody-agentes');
      const rows = Array.from(tbody.getElementsByTagName('tr'));
      
      if (ordenActual.columna === columnIndex) {
          ordenActual.ascendente = !ordenActual.ascendente;
      } else {
          ordenActual.columna = columnIndex;
          ordenActual.ascendente = true;
      }
      
      rows.sort((a, b) => {
          let valA, valB;
          
          if (columnIndex === -1) {
              const celdasA = a.getElementsByTagName('td');
              const celdasB = b.getElementsByTagName('td');
              valA = parseFloat(celdasA[celdasA.length - 1].textContent.replace(/[^0-9.-]/g, ''));
              valB = parseFloat(celdasB[celdasB.length - 1].textContent.replace(/[^0-9.-]/g, ''));
          } else {
              valA = a.getElementsByTagName('td')[columnIndex].textContent.trim().toLowerCase();
              valB = b.getElementsByTagName('td')[columnIndex].textContent.trim().toLowerCase();
              if (!isNaN(valA) && !isNaN(valB)) {
                  valA = parseFloat(valA);
                  valB = parseFloat(valB);
              }
          }
          
          if (valA < valB) return ordenActual.ascendente ? -1 : 1;
          if (valA > valB) return ordenActual.ascendente ? 1 : -1;
          return 0;
      });
      
      rows.forEach(row => tbody.appendChild(row));
  }

  function filtrarTabla() {
      const input = document.getElementById('search-input');
      const filter = input.value.toLowerCase().trim();
      const tbody = document.getElementById('tbody-agentes');
      const rows = tbody.getElementsByTagName('tr');
      let visibleCount = 0;
      
      for (let i = 0; i < rows.length; i++) {
          const row = rows[i];
          const bms = row.getAttribute('data-bms') || '';
          const nombre = row.getAttribute('data-nombre') || '';
          
          if (bms.includes(filter) || nombre.includes(filter)) {
              row.style.display = '';
              visibleCount++;
          } else {
              row.style.display = 'none';
          }
      }
      
      const countSpan = document.getElementById('search-count');
      countSpan.textContent = filter ? 'Mostrando ' + visibleCount + ' de ' + rows.length : '';
  }

document.getElementById('top3SelectFecha').addEventListener('change', function() {
    const lob = document.getElementById('top3SelectLob').value;
    const fecha = this.value;
    const selectKpi = document.getElementById('top3SelectKpi');
    
    selectKpi.innerHTML = '<option value="">Cargando KPIs...</option>';
    selectKpi.disabled = true;
    document.getElementById('top3-all-months-results').innerHTML = '';
    
    if (!lob || !fecha) return;
    
    console.log('📅 Cargando KPIs para:', { lob, fecha });
    mostrarAlerta('⏳ Cargando KPIs disponibles...', 'info');
    
    google.script.run
        .withSuccessHandler(function(datosRecibidos) {
            console.log('📦 Datos recibidos:', datosRecibidos);
            
            if (!datosRecibidos.metricasPorLob || !datosRecibidos.metricasPorLob[lob]) {
                selectKpi.innerHTML = '<option value="">No hay KPIs</option>';
                mostrarAlerta('⚠️ No hay KPIs disponibles para esta LOB', 'warning');
                return;
            }
            
            const metricas = datosRecibidos.metricasPorLob[lob];
            
            if (metricas.error) {
                selectKpi.innerHTML = '<option value="">Error: ' + metricas.error + '</option>';
                mostrarAlerta('❌ Error: ' + metricas.error, 'danger');
                return;
            }
            
            const kpisUnicos = new Set();
            
            // ⭐ PROCESAR TODAS LAS CATEGORÍAS
            ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(cat => {
                console.log(`📂 Procesando categoría: ${cat}`);
                
                if (metricas[cat] && Array.isArray(metricas[cat])) {
                    console.log(`   ${metricas[cat].length} KPIs en ${cat}`);
                    
                    metricas[cat].forEach(kpiObj => {
                        const kpiNombre = kpiObj.kpi;
                        
                        // ⭐ VALIDAR: Solo agregar si tiene validación con empleados
                        if (kpiNombre && kpiObj.validacion) {
                            const val = kpiObj.validacion;
                            const totalEmpleados = (val.empleadosCumplen || []).length + 
                                                  (val.empleadosNoCumplen || []).length;
                            
                            if (totalEmpleados > 0) {
                                kpisUnicos.add(kpiNombre);
                                console.log(`      ✅ KPI: "${kpiNombre}" - ${totalEmpleados} empleados`);
                            } else {
                                console.log(`      ⚠️ KPI: "${kpiNombre}" - SIN empleados`);
                            }
                        }
                    });
                } else {
                    console.log(`   ⚠️ Categoría ${cat} vacía o no es array`);
                }
            });
            
            console.log(`📊 Total KPIs únicos encontrados: ${kpisUnicos.size}`);
            
            selectKpi.innerHTML = '<option value="">Selecciona un KPI</option>';
            
            if (kpisUnicos.size === 0) {
                selectKpi.innerHTML = '<option value="">No hay KPIs con datos</option>';
                mostrarAlerta('⚠️ No hay KPIs con datos de empleados', 'warning');
                return;
            }
            
            // ⭐ AGREGAR KPIs AL SELECT
            Array.from(kpisUnicos).sort().forEach(kpi => {
                const opt = document.createElement('option');
                opt.value = kpi;
                opt.textContent = kpi;
                selectKpi.appendChild(opt);
            });
            
            selectKpi.disabled = false;
            mostrarAlerta(`✅ ${kpisUnicos.size} KPIs disponibles`, 'success');
            
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error:', error);
            selectKpi.innerHTML = '<option value="">Error</option>';
            mostrarAlerta('❌ Error al cargar KPIs: ' + error.message, 'danger');
        })
        .obtenerDatosParaFrontend(lob, fecha, null);
});

document.getElementById('top3SelectKpi').addEventListener('change', function() {
    const lob = document.getElementById('top3SelectLob').value;
    const fecha = document.getElementById('top3SelectFecha').value;
    const kpi = this.value;
    
    if (!lob || !fecha || !kpi) return;
    
    const resultsDiv = document.getElementById('top3-all-months-results');
    resultsDiv.innerHTML = '<div class="loading">⏳ Cargando Top 3...</div>';
    
    console.log('🏆 Cargando Top 3 para:', { lob, fecha, kpi });
    mostrarAlerta('⏳ Cargando Top 3...', 'info');
    
    // ⭐ OBTENER NÚMERO DE MES DESDE LA FECHA SELECCIONADA
    const numeroMes = obtenerNumeroMesDesdeNombre(fecha);
    console.log(`📅 Mes detectado: ${fecha} → Número: ${numeroMes}`);
    
    // ⭐ CARGAR DATOS DE ESE MES ESPECÍFICO
    google.script.run
        .withSuccessHandler(function(datosRecibidos) {
            console.log(`📦 Datos recibidos para ${fecha}:`, datosRecibidos);
            
            if (!datosRecibidos.metricasPorLob || !datosRecibidos.metricasPorLob[lob]) {
                resultsDiv.innerHTML = '<div class="info-message">⚠️ No hay datos para esta LOB y fecha</div>';
                mostrarAlerta('⚠️ No hay datos disponibles', 'warning');
                return;
            }
            
            const metricas = datosRecibidos.metricasPorLob[lob];
            let top3 = [];
            
            if (!metricas.error) {
                // ⭐ BUSCAR EN TODAS LAS CATEGORÍAS
                ['keys', 'bonusStructure', 'additionalIncentives', 'decreasedKPIs'].forEach(cat => {
                    if (metricas[cat]) {
                        metricas[cat].forEach(kpiObj => {
                            if (kpiObj.kpi === kpi && kpiObj.validacion) {
                                const empleadosCumplen = kpiObj.validacion.empleadosCumplen || [];
                                
                                console.log(`   ✅ ${cat}: ${empleadosCumplen.length} empleados cumplen`);
                                
                                // ⭐ ORDENAR Y TOMAR TOP 3
                                top3 = empleadosCumplen
                                    .sort((a, b) => (b.bono || 0) - (a.bono || 0))
                                    .slice(0, 3);
                                
                                console.log(`   🏆 Top 3:`, top3);
                            }
                        });
                    }
                });
            }
            
            // ⭐ MOSTRAR SOLO ESTE MES
            mostrarTop3ResultadoUnicoMes(fecha, top3, lob, kpi);
            
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error:', error);
            resultsDiv.innerHTML = '<div class="info-message">❌ Error: ' + error.message + '</div>';
            mostrarAlerta('❌ Error: ' + error.message, 'danger');
        })
        .obtenerDatosParaFrontend(lob, fecha, numeroMes);
});

// ========================================
function generarGraficoHistoricoMeses(datosPorMes) {
    // Ordenar por número de mes
    datosPorMes.sort((a, b) => a.numeroMes - b.numeroMes);
    
    const labels = datosPorMes.map(d => d.mes);
    const data = datosPorMes.map(d => d.total);
    
    const ctx = document.getElementById('chartHistoricalMeses');
    if (chartHistoricalMeses) chartHistoricalMeses.destroy();
    
    chartHistoricalMeses = new Chart(ctx, {
        type: 'bar',
        data: {
            labels: labels,
            datasets: [{
                label: 'Total Bonos ($)',
                data: data,
                backgroundColor: 'rgba(155, 89, 182, 0.8)',
                borderColor: 'rgba(155, 89, 182, 1)',
                borderWidth: 2
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: true,
            plugins: {
                tooltip: {
                    callbacks: {

function obtenerNombreMesDeFecha(fecha) {
  if (!fecha) return 'General';
  
  console.log('🔍 Obteniendo nombre del mes para fecha:', fecha);
  
  // Buscar nombres de meses en español en la fecha
  const mesesNombres = {
    'enero': 'Enero', 'febrero': 'Febrero', 'marzo': 'Marzo', 'abril': 'Abril',
    'mayo': 'Mayo', 'junio': 'Junio', 'julio': 'Julio', 'agosto': 'Agosto',
    'septiembre': 'Septiembre', 'octubre': 'Octubre', 'noviembre': 'Noviembre', 'diciembre': 'Diciembre'
  };
  
  const fechaLower = fecha.toString().toLowerCase().trim();
  console.log('📅 Fecha en minúsculas:', fechaLower);
  
  // Buscar si la fecha contiene el nombre de algún mes
  for (const mes in mesesNombres) {
    if (fechaLower.includes(mes)) {
      console.log('✅ Mes encontrado:', mes, '->', mesesNombres[mes]);
      return mesesNombres[mes];
    }
  }
  
  // Si no encuentra nombre de mes, intentar con fecha JavaScript
  try {
    const fechaObj = new Date(fecha);
    if (!isNaN(fechaObj.getTime())) {
      const meses = ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 
                    'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'];
      const nombreMes = meses[fechaObj.getMonth()];
      console.log('📅 Mes desde Date object:', fechaObj.getMonth() + 1, '->', nombreMes);
      return nombreMes;
    }
  } catch (e) {
    console.log('⚠️ No se pudo parsear como fecha:', e);
  }
  
  console.log('❌ No se pudo determinar el mes, usando "General"');
  return 'General';
}

function obtenerNumeroMesDeFecha(fecha) {
  if (!fecha) return 0;
  
  console.log('🔍 Obteniendo número del mes para fecha:', fecha);
  
  // Mapeo CORREGIDO: Enero=1, Febrero=2, etc.
  const mesesMap = {
    'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
    'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
    'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
  };
  
  const fechaLower = fecha.toString().toLowerCase().trim();
  
  for (const mes in mesesMap) {
    if (fechaLower.includes(mes)) {
      console.log('✅ Número de mes encontrado:', mes, '->', mesesMap[mes]);
      return mesesMap[mes];
    }
  }
  
  // Si es una fecha tipo "2024-05-15"
  try {
    const fechaObj = new Date(fecha);
    if (!isNaN(fechaObj.getTime())) {
      const numeroMes = fechaObj.getMonth() + 1; // getMonth() retorna 0-11
      console.log('📅 Número de mes desde Date:', numeroMes);
      return numeroMes;
    }
  } catch (e) {
    console.log('⚠️ No se pudo parsear como fecha para número:', e);
  }
  
  console.log('❌ No se pudo determinar número de mes, usando 0');
  return 0;
}

  function aplicarTraslados() {
      const checkboxes = document.querySelectorAll('.recalc-check:checked');
      
      if (checkboxes.length === 0) {
          mostrarAlerta('⚠️ Debes seleccionar al menos un agente para trasladar', 'warning');
          return;
      }
      
      const traslados = [];
      const lobActual = selectLob.value;
      const fechaActual = selectFecha.value;
      const mesActual = mesComparativoActual;
      
      let error = false;
      
      checkboxes.forEach(check => {
          const bmsId = check.getAttribute('data-bms');
          const select = document.querySelector('.recalc-lob-select[data-bms="' + bmsId + '"]');
          const lobDestino = select.value;
          
          if (!lobDestino) {
              mostrarAlerta('⚠️ Debes seleccionar una LOB destino para BMS ' + bmsId, 'warning');
              error = true;
              return;
          }
          
          if (lobDestino === lobActual) {
              mostrarAlerta('⚠️ No puedes trasladar un agente a la misma LOB', 'warning');
              error = true;
              return;
          }
          
          const fila = document.querySelector(`tr[data-bms="${bmsId}"]`);
          const nombre = fila ? fila.querySelector('td:nth-child(4)').textContent.trim().split('\n')[0] : 'Agente ' + bmsId;
          
          traslados.push({
              bmsId: bmsId,
              nombre: nombre,
              lobOriginal: lobActual,
              lobDestino: lobDestino,
              fecha: fechaActual,
              mes: mesActual
          });
      });
      
      if (error || traslados.length === 0) return;
      
      // ⭐ CONFIRMACIÓN DETALLADA
      let mensajeConfirmacion = `⚠️ TRASLADO PERMANENTE EN H_C\n\n`;
      mensajeConfirmacion += `Vas a trasladar ${traslados.length} agente(s):\n\n`;
      
      traslados.forEach((t, i) => {
          mensajeConfirmacion += `${i+1}. ${t.nombre} (BMS ${t.bmsId})\n`;
          mensajeConfirmacion += `   De: ${t.lobOriginal} → A: ${t.lobDestino}\n\n`;
      });
      
      mensajeConfirmacion += `❗ IMPORTANTE:\n`;
      mensajeConfirmacion += `• El agente cambiará de LOB en la hoja H_C\n`;
      mensajeConfirmacion += `• Aparecerá inmediatamente en ${traslados[0].lobDestino}\n`;
      mensajeConfirmacion += `• Se registrará en "Ajustes_Traslados" para auditoría\n\n`;
      mensajeConfirmacion += `¿Continuar?`;
      
      if (!confirm(mensajeConfirmacion)) return;
      
      console.log('🔄 Aplicando traslados:', traslados);
      mostrarLoading('Trasladando ' + traslados.length + ' agente(s)...');
      
      google.script.run
          .withSuccessHandler(function(response) {
              ocultarLoading();
              
              if (response.success) {
                  mostrarAlerta('✅ ' + response.mensaje + '\n\n💡 Los agentes ya están en la nueva LOB. Recarga para ver los cambios.', 'success');
                  
                  // Recargar después de 2 segundos
                  setTimeout(() => {
                      cargarDatosCompletos(lobActual, fechaActual, mesActual);
                  }, 2000);
                  
              } else {
                  mostrarAlerta('❌ Error: ' + (response.error || 'Error desconocido'), 'danger');
              }
          })
          .withFailureHandler(function(error) {
              ocultarLoading();
              mostrarAlerta('❌ Error de conexión: ' + error.message, 'danger');
              console.error('Error completo:', error);
          })
          .trasladarAgentesTemporalmente(traslados);
  }

  /**
   * Aplicar prorrateo a los agentes seleccionados
   */
  // ========================================
  // FUNCIÓN APLICAR PRORRATEO
  // ========================================
  function aplicarProrrateo() {
      const checkboxes = document.querySelectorAll('.prorrateo-check:checked');
      
      if (checkboxes.length === 0) {
          mostrarAlerta('⚠️ Debes seleccionar al menos un agente para prorratear', 'warning');
          return;
      }
      
      const agentesProrrateo = [];
      const lobActual = selectLob.value;
      const fechaActual = selectFecha.value;
      const mesActual = mesComparativoActual;
      
      checkboxes.forEach(check => {
          const bmsId = check.getAttribute('data-bms');
          const horas = parseFloat(check.getAttribute('data-horas')) || 0;
          
          // Obtener el total del agente
          const fila = document.querySelector(`tr[data-bms="${bmsId}"]`);
          if (!fila) return;
          
          const celdas = fila.getElementsByTagName('td');
          const celdaTotal = celdas[celdas.length - 3]; // Columna Total $
          const bonoActual = parseFloat(celdaTotal.textContent.replace(/[^0-9.-]/g, ''));
          
          const nombre = fila.querySelector('td:nth-child(4)').textContent.trim().split('\n')[0];
          
          agentesProrrateo.push({
              bmsId: bmsId,
              nombre: nombre,
              horas: horas,
              bonoActual: bonoActual
          });
      });
      
      // Mostrar confirmación con detalles
      let mensajeConfirmacion = `📊 PRORRATEO DE BONOS POR HORAS\n\n`;
      mensajeConfirmacion += `Vas a prorratear ${agentesProrrateo.length} agente(s):\n\n`;
      
      const HORAS_COMPLETAS = 160;
      
      agentesProrrateo.forEach((a, i) => {
          const factorProrrateo = a.horas / HORAS_COMPLETAS;
          const bonoProrateado = a.bonoActual * factorProrrateo;
          const diferencia = a.bonoActual - bonoProrateado;
          
          mensajeConfirmacion += `${i+1}. ${a.nombre} (BMS ${a.bmsId})\n`;
          mensajeConfirmacion += `   Horas: ${a.horas}h de ${HORAS_COMPLETAS}h (${(factorProrrateo * 100).toFixed(1)}%)\n`;
          mensajeConfirmacion += `   Bono Original: $${a.bonoActual.toLocaleString('es-CO', {minimumFractionDigits: 2})}\n`;
          mensajeConfirmacion += `   Bono Prorrateado: $${bonoProrateado.toLocaleString('es-CO', {minimumFractionDigits: 2})}\n`;
          mensajeConfirmacion += `   Diferencia: -$${diferencia.toLocaleString('es-CO', {minimumFractionDigits: 2})}\n\n`;
      });
      
      mensajeConfirmacion += `¿Deseas aplicar el prorrateo?`;
      
      if (!confirm(mensajeConfirmacion)) return;
      
      console.log('📊 Aplicando prorrateo:', agentesProrrateo);
      mostrarLoading('Calculando prorrateo para ' + agentesProrrateo.length + ' agente(s)...');
      
      google.script.run
          .withSuccessHandler(function(response) {
              ocultarLoading();
              
              if (response.success) {
                  mostrarAlerta('✅ ' + response.mensaje, 'success');
                  
                  // Actualizar la tabla con los nuevos valores
                  response.resultados.forEach(resultado => {
                      if (resultado.success) {
                          const fila = document.querySelector(`tr[data-bms="${resultado.bmsId}"]`);
                          if (fila) {
                              const celdas = fila.getElementsByTagName('td');
                              const celdaTotal = celdas[celdas.length - 3];
                              
                              // Actualizar el total con animación
                              celdaTotal.style.transition = 'all 0.5s ease';
                              celdaTotal.style.background = '#27ae60';
                              celdaTotal.style.color = 'white';
                              celdaTotal.innerHTML = '<strong>$' + resultado.bonoProrateado.toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</strong><br>' +
                                                    '<small style="font-size:0.75rem;">Prorrateado (' + resultado.horasTrabajadas + 'h)</small>';
                              
                              setTimeout(() => {
                                  celdaTotal.style.background = '#d5f4e6';
                                  celdaTotal.style.color = '#000';
                              }, 2000);
                              
                              // Desmarcar checkbox
                              const checkbox = fila.querySelector('.prorrateo-check');
                              if (checkbox) checkbox.checked = false;
                          }
                      }
                  });
                  
                  // Recargar métricas
                  generarMetricas(lobActual);
                  generarGraficos();
                  
              } else {
                  mostrarAlerta('❌ Error: ' + (response.error || 'Error desconocido'), 'danger');
              }
          })
          .withFailureHandler(function(error) {
              ocultarLoading();
              mostrarAlerta('❌ Error de conexión: ' + error.message, 'danger');
              console.error('Error completo:', error);
          })
          .aplicarProrrateoAgentes(agentesProrrateo, lobActual, fechaActual, mesActual);
  }

  // ========================================
  // ⭐ CALCULAR PRORRATEO BASADO EN PROPORCIÓN DE DÍAS TRABAJADOS
  // ========================================
function calcularProrrateoInstantaneo(checkbox) {
    const bmsId = checkbox.getAttribute('data-bms');
    const horasReales = parseFloat(checkbox.getAttribute('data-horas')) || 0;
    const horasSemanales = parseFloat(checkbox.getAttribute('data-horas-semanales')) || 42;
    const bonoOriginal = parseFloat(checkbox.getAttribute('data-bono-original')) || 0;
    
    console.log(`📊 Prorrateo para BMS ${bmsId}:`, {
        checked: checkbox.checked,
        horasReales,
        horasSemanales,
        bonoOriginal
    });
    
    const fila = document.querySelector(`tr[data-bms="${bmsId}"]`);
    if (!fila) {
        console.error('❌ Fila no encontrada');
        return;
    }
    
    const celdaTotal = fila.querySelector('.celda-total');
    if (!celdaTotal) {
        console.error('❌ Celda total no encontrada');
        return;
    }
    
    if (checkbox.checked) {
        // ========================================
        // ✅ APLICAR PRORRATEO
        // ========================================
        const proporcionTrabajada = horasReales / horasSemanales;
        const diasEquivalentes = proporcionTrabajada * 6;
        const diasTotalesMes = 24;
        const bonoProrateado = (bonoOriginal / diasTotalesMes) * diasEquivalentes;
        
        console.log(`💰 Calculando:`, {
            proporcionTrabajada: proporcionTrabajada.toFixed(2),
            diasEquivalentes: diasEquivalentes.toFixed(1),
            bonoProrateado: bonoProrateado.toFixed(2)
        });
        
        // Actualizar celda con animación
        celdaTotal.style.transition = 'all 0.3s ease';
        celdaTotal.style.background = 'linear-gradient(135deg, #16a085 0%, #1abc9c 100%)';
        celdaTotal.style.color = 'white';
        celdaTotal.style.fontWeight = 'bold';
        celdaTotal.innerHTML = `
            <div style="padding:8px;">
                <div style="font-size:1.15rem;margin-bottom:4px;">
                    $${bonoProrateado.toLocaleString('es-CO', {minimumFractionDigits: 2})}
                </div>
                <div style="font-size:0.7rem;opacity:0.95;margin-bottom:3px;">
                    📊 ${diasEquivalentes.toFixed(1)} días / ${diasTotalesMes} días
                </div>
                <div style="font-size:0.65rem;opacity:0.85;border-top:1px solid rgba(255,255,255,0.3);padding-top:3px;margin-top:3px;">
                    Original: $${bonoOriginal.toLocaleString('es-CO', {minimumFractionDigits: 2})}
                </div>
            </div>
        `;
        
        // Guardar valor prorrateado
        checkbox.setAttribute('data-bono-prorrateado', bonoProrateado);
        
        console.log(`✅ Prorrateo aplicado: $${bonoProrateado.toFixed(2)}`);
        
    } else {
        // ========================================
        // ❌ QUITAR PRORRATEO (Restaurar original)
        // ========================================
        celdaTotal.style.transition = 'all 0.3s ease';
        celdaTotal.style.background = '#d5f4e6';
        celdaTotal.style.color = '#000';
        celdaTotal.style.fontWeight = 'bold';
        celdaTotal.innerHTML = '$' + bonoOriginal.toLocaleString('es-CO', {
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        });
        
        // Limpiar atributo
        checkbox.removeAttribute('data-bono-prorrateado');
        
        console.log(`✅ Valor original restaurado: $${bonoOriginal.toFixed(2)}`);
    }
    
    // Actualizar métricas totales
    actualizarMetricasConProrrateo();

// ========================================
function configurarProrrateosAutomaticos() {
    const checkboxes = document.querySelectorAll('.prorrateo-check');
    
    let habilitados = 0;
    let deshabilitados = 0;
    
    checkboxes.forEach(checkbox => {
        const horasReales = parseFloat(checkbox.getAttribute('data-horas')) || 0;
        const horasSemanales = parseFloat(checkbox.getAttribute('data-horas-semanales')) || 42;
        
        // ⭐ UMBRALES FIJOS SEGÚN JORNADA
        let umbralMinimo;
        if (horasSemanales >= 42) {
            umbralMinimo = 150; // 42h/sem → < 150h = vacaciones (7-15 días)
        } else if (horasSemanales >= 40) {
            umbralMinimo = 143; // 40h/sem → < 143h = vacaciones
        } else if (horasSemanales >= 38) {
            umbralMinimo = 136; // 38h/sem → < 136h = vacaciones
        } else {
            umbralMinimo = 130; // Otras jornadas
        }
        
        const proporcionTrabajada = horasReales / horasSemanales;
        const diasEquivalentes = proporcionTrabajada * 6;
        const diasFaltantes = 24 - diasEquivalentes;
        
        // ⭐ SOLO HABILITAR si tiene MENOS del umbral mínimo
        if (horasReales > 0 && horasReales < umbralMinimo) {
            checkbox.disabled = false;
            checkbox.checked = false; // ⭐ IMPORTANTE: No marcar automáticamente
            checkbox.title = `✅ Habilitado: ${horasReales.toFixed(0)}h (< ${umbralMinimo}h) - Aprox. ${diasFaltantes.toFixed(0)} días menos`;
            checkbox.parentElement.style.opacity = '1';
            checkbox.parentElement.style.cursor = 'pointer';
            habilitados++;
        } else {
            checkbox.disabled = true;
            checkbox.checked = false;
            if (horasReales > 0) {
                checkbox.title = `❌ No requiere: ${horasReales.toFixed(0)}h (≥ ${umbralMinimo}h - Mes completo)`;
            } else {
                checkbox.title = '❌ Sin horas registradas';
            }
            checkbox.parentElement.style.opacity = '0.3';
            checkbox.parentElement.style.cursor = 'not-allowed';
            deshabilitados++;
        }
        
        // ⭐ GARANTIZAR: Restaurar valor original si está deshabilitado
        if (checkbox.disabled && checkbox.checked) {
            checkbox.checked = false;
            const bmsId = checkbox.getAttribute('data-bms');
            const bonoOriginal = parseFloat(checkbox.getAttribute('data-bono-original')) || 0;
            const fila = document.querySelector(`tr[data-bms="${bmsId}"]`);
            if (fila) {
                const celdas = fila.getElementsByTagName('td');
                const celdaTotal = celdas[celdas.length - 3];
                if (celdaTotal) {
                    celdaTotal.textContent = '$' + bonoOriginal.toLocaleString('es-CO', {minimumFractionDigits: 2});
                    celdaTotal.style.background = '#d5f4e6';
                    celdaTotal.style.color = '#000';
                    celdaTotal.style.fontWeight = 'bold';
                }
            }
            checkbox.removeAttribute('data-bono-prorrateado');
        }
    });
    
    console.log(`✅ Checkboxes: ${habilitados} habilitados (< 150h), ${deshabilitados} deshabilitados (≥ 150h)`);
    
    // ⭐ LIMPIAR ARRAY AL RECONFIGURAR
    prorrateosMarcados = [];
    console.log('🧹 Array de prorrateos limpiado por reconfiguración automática');
    
    return habilitados; // ⭐ RETORNAR cantidad para uso externo
}

function actualizarMetricasConProrrateo() {
    const lobActual = selectLob.value;
    if (!lobActual) return;
    
    const todasFilas = document.querySelectorAll('tbody tr[data-bms]');
    let totalGeneral = 0;
    let cantidadProrrateos = 0;
    
    console.log('🔄 Recalculando métricas con prorrateos...');
    
    todasFilas.forEach(fila => {
        const celdaTotal = fila.querySelector('.celda-total');
        if (!celdaTotal) return;
        
        const bonoOriginal = parseFloat(celdaTotal.getAttribute('data-bono-original')) || 0;
        const checkbox = fila.querySelector('.prorrateo-check');
        
        let bonoFinal = bonoOriginal;
        
        if (checkbox && checkbox.checked && !checkbox.disabled) {
            const bonoProrateado = parseFloat(checkbox.getAttribute('data-bono-prorrateado')) || bonoOriginal;
            bonoFinal = bonoProrateado;
            cantidadProrrateos++;
        }
        
        // ⭐ SOLO SUMAR SI ES POSITIVO
        if (bonoFinal > 0) {
            totalGeneral += bonoFinal;
        }
    });
    
    console.log(`📊 Total recalculado: $${totalGeneral.toFixed(2)} (${cantidadProrrateos} prorrateos)`);
    
    // Actualizar métrica visual
    const metricsGrid = document.getElementById('metrics-grid');
    if (metricsGrid && metricsGrid.style.display !== 'none') {
        const metricCards = metricsGrid.querySelectorAll('.metric-card');
        if (metricCards.length > 0) {
            const totalBonosCard = metricCards[0];
            const metricValue = totalBonosCard.querySelector('.metric-value');
            const metricSubtitle = totalBonosCard.querySelector('.metric-subtitle');
            
            metricValue.textContent = '$' + totalGeneral.toLocaleString('es-CO', {minimumFractionDigits: 2});
            
            if (cantidadProrrateos > 0) {
                metricSubtitle.innerHTML = `${cantidadProrrateos} prorrateo(s) aplicado(s)`;
                totalBonosCard.style.borderLeft = '4px solid #16a085';
            } else {
                metricSubtitle.textContent = 'Suma total pagada';
                totalBonosCard.style.borderLeft = '4px solid #27ae60';
            }
        }
    }
}

function obtenerNombreMesDesdeNumero(numeroMes) {
    if (!numeroMes || isNaN(numeroMes)) return 'General';
    
    const meses = [
        'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 
        'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'
    ];
    
    // Asegurar que el número esté entre 1-12
    const numero = parseInt(numeroMes);
    if (numero >= 1 && numero <= 12) {
        return meses[numero - 1];
    }
    
    return 'General';
}

function enviaraprobados() {
    const checkboxes = document.querySelectorAll('.aprobado-check:checked');
    
    if (checkboxes.length === 0) {
        mostrarAlerta('⚠️ Debes seleccionar al menos un agente para aprobar', 'warning');
        return;
    }
    
    const aprobaciones = [];
    const lobActual = selectLob.value;

    // ⭐ CORREGIDO: Obtener el nombre del mes desde el número
    const nombreMesActual = obtenerNombreMesDesdeNumero(mesComparativoActual) || 'General';
    
    let totalGeneral = 0;
    
    // ⭐ OBTENER DATOS DE EMPLEADOS PARA CALCULAR DESGLOSE
    const empleados = empleadosPorLob[lobActual];
    
    if (!empleados) {
        mostrarAlerta('❌ No hay datos de empleados cargados', 'danger');
        return;
    }
    
    // Recopilar datos de todos los agentes seleccionados
    checkboxes.forEach(check => {
        const bmsId = check.getAttribute('data-bms');
        const nombre = check.getAttribute('data-nombre');
        const total = parseFloat(check.getAttribute('data-total')) || 0;
        
        totalGeneral += total;
        
        // ⭐ CALCULAR DESGLOSE POR CATEGORÍA PARA ESTE AGENTE
        const desglose = {
            keys: 0,
            bonusStructure: 0,
            additionalIncentives: 0,
            decreasedKPIs: 0
        };
        
        // Buscar el agente en empleadosPorLob
        const keyEmpleado = Object.keys(empleados).find(k => empleados[k].bmsId === bmsId);
        
        if (keyEmpleado) {
            const empleado = empleados[keyEmpleado];
            
            // Iterar sobre sus resultados
            Object.keys(empleado.resultados).forEach(resKey => {
                const resultado = empleado.resultados[resKey];
                
                // Solo contar si cumple y no está sin datos
                if (resultado.cumple && !resultado.sinDatos) {
                    const bono = resultado.bono || 0;
                    
                    // Clasificar por categoría
                    if (resKey.startsWith('keys_')) {
                        desglose.keys += bono;
                    } else if (resKey.startsWith('bonusStructure_')) {
                        desglose.bonusStructure += bono;
                    } else if (resKey.startsWith('additionalIncentives_')) {

// ⭐ FUNCIÓN: APLICAR DISPUTAS DESDE CACHE (MEJORADA CON DEBUG)
// ========================================
function aplicarDisputasDesdeCache() {
    const lobActual = selectLob.value;
    const fechaActual = selectFecha.value;
    
    console.log('🔍 Aplicando disputas desde cache...');
    console.log('📍 LOB actual:', lobActual);
    console.log('📍 Fecha actual:', fechaActual);
    
    if (!lobActual || !fechaActual) {
        console.warn('⚠️ No hay LOB o Fecha seleccionada');
        return;
    }
    
    const cacheKey = `disputas_${lobActual}_${fechaActual}`;
    console.log('🔑 Cache key:', cacheKey);
    
    try {
        // Método 1: Cache específico
        let disputasCache = localStorage.getItem(cacheKey);
        let disputas = [];
        
        console.log('📦 Cache específico raw:', disputasCache);
        
        if (disputasCache) {
            disputas = JSON.parse(disputasCache);
            console.log('✅ Disputas del cache específico:', disputas);
        }
        
        // Método 2: Cache global (backup)
        if (disputas.length === 0) {
            const todasDisputasRaw = localStorage.getItem('disputasEnviadas');
            console.log('📦 Cache global raw:', todasDisputasRaw);
            
            if (todasDisputasRaw) {
                const todas = JSON.parse(todasDisputasRaw);
                console.log('📋 Total disputas en cache global:', todas.length);
                
                disputas = todas.filter(d => {
                    const match = d.lob === lobActual && d.fecha === fechaActual;
                    console.log(`   Comparando: ${d.lob} === ${lobActual} && ${d.fecha} === ${fechaActual} = ${match}`);
                    return match;
                });
                
                console.log('✅ Disputas filtradas del global:', disputas);
            }
        }
        
        if (disputas.length === 0) {
            console.log('📭 No hay disputas en cache para esta LOB y fecha');
            return;
        }
        
        console.log(`🎯 Aplicando ${disputas.length} disputa(s)...`);
        
        // Esperar a que la tabla esté completamente cargada
        setTimeout(() => {
            let marcadas = 0;
            
            disputas.forEach(disputa => {
                console.log(`   Buscando fila para BMS ${disputa.bmsId}...`);
                const fila = document.querySelector(`tr[data-bms="${disputa.bmsId}"]`);
                
                if (fila) {
                    console.log(`   ✅ Fila encontrada para BMS ${disputa.bmsId}`);
                    marcarComoDisputadoCompleto(disputa);
                    marcadas++;
                } else {
                    console.warn(`   ⚠️ Fila NO encontrada para BMS ${disputa.bmsId}`);
                }
            });
            
            console.log(`✅ Total marcadas: ${marcadas} de ${disputas.length}`);
            
            if (marcadas > 0) {
                mostrarAlerta(`🚨 ${marcadas} agente(s) con disputas registradas`, 'warning');
            }
        }, 500);
        
    } catch (e) {
        console.error('❌ Error al aplicar disputas desde cache:', e);
        console.error('Stack:', e.stack);
    }
}

function marcarComoDisputadoCompleto(disputa) {

        if (!filaAlt) {
            console.error(`❌ Tampoco se encontró con ID row-${disputa.bmsId}`);
            return;
        }
        console.log(`✅ Fila encontrada con ID alternativo`);
        fila = filaAlt;
    }

    fila.style.background = 'linear-gradient(135deg, #f8d7da 0%, #f5c6cb 100%)';
    fila.style.borderLeft = '5px solid #e74c3c';
    fila.style.boxShadow = '0 4px 12px rgba(231, 76, 60, 0.3)';

    if (celdaNombre) {

        const badgeExistente = celdaNombre.querySelector('.badge-disputado');

            badge.className = 'badge-disputado';
            badge.style.cssText = 'margin-top:8px;display:inline-block;';

            `;
            celdaNombre.appendChild(badge);
            console.log(`   ✓ Badge agregado`);
        } else {
            console.log(`   ℹ️ Badge ya existe`);
        }
    } else {
        console.warn(`   ⚠️ Celda de nombre no encontrada`);
    }

    if (checkbox) {

        checkbox.checked = false;

    }

    const celdaComentario = fila.querySelector('.celda-comentario-disputa');

        if (textoSpan) {

        }

    const modal = document.getElementById('modal-disputa');
    const nombreDiv = document.getElementById('modal-disputa-nombre');
    const bmsDiv = document.getElementById('modal-disputa-bms');
    const comentario = document.getElementById('modal-disputa-comentario');
    const contador = document.getElementById('modal-disputa-contador');
    
    // Rellenar datos
    nombreDiv.textContent = disputaActual.nombre;
    bmsDiv.textContent = 'BMS: ' + disputaActual.bmsId + ' | Bono: $' + disputaActual.total.toLocaleString('es-CO', {minimumFractionDigits: 2});
    
    // Si no tiene comentario previo, limpiar
    if (!disputasPendientes[disputaActual.bmsId]) {
        comentario.value = '';
        contador.textContent = '0';
    } else {
        contador.textContent = comentario.value.length;
    }
    
    // Contador de caracteres
    comentario.oninput = function() {
        contador.textContent = this.value.length;
        
        if (this.value.length >= 10) {
            this.style.borderColor = '#27ae60';
        } else if (this.value.length > 0) {
            this.style.borderColor = '#f39c12';
        } else {
            this.style.borderColor = '#dfe6e9';
        }
    };
    
    // Mostrar modal
    modal.style.display = 'flex';
    
    // Focus en textarea
    setTimeout(() => comentario.focus(), 100);
}

function cerrarModalDisputa() {
    const modal = document.getElementById('modal-disputa');
    modal.style.display = 'none';
    
    // ⚠️ IMPORTANTE: Si se cierra sin guardar, desmarcar checkbox
    if (disputaActual && !disputasPendientes[disputaActual.bmsId]) {
        disputaActual.checkbox.checked = false;
        console.log('⚠️ Modal cerrado sin guardar - checkbox desmarcado');
    }
    
    disputaActual = null;
    console.log('✅ Modal cerrado');
}

function guardarDisputaLocal() {
    if (!disputaActual) {
        console.error('❌ No hay disputa actual');
        return;
    }
    
    const comentario = document.getElementById('modal-disputa-comentario').value.trim();
    
    console.log('💾 Intentando guardar:', { bmsId: disputaActual.bmsId, comentario });
    
    // Validación
    if (!comentario) {
        const textarea = document.getElementById('modal-disputa-comentario');
        textarea.style.borderColor = '#e74c3c';
        textarea.placeholder = '⚠️ El comentario es obligatorio';
        textarea.focus();
        mostrarAlerta('⚠️ Debes escribir un motivo para la disputa', 'warning');
        return;
    }
    
    if (comentario.length < 10) {
        mostrarAlerta('⚠️ El comentario debe tener al menos 10 caracteres', 'warning');
        return;
    }
    
    // Guardar en objeto pendientes
    disputasPendientes[disputaActual.bmsId] = {
        bmsId: disputaActual.bmsId,
        nombre: disputaActual.nombre,
        total: disputaActual.total,
        comentario: comentario,
        lob: selectLob.value,
        fecha: selectFecha.value
    };
    
    console.log('✅ Comentario guardado localmente:', disputasPendientes[disputaActual.bmsId]);
    
    // Actualizar celda de comentario
    actualizarCeldaComentario(disputaActual.bmsId, comentario);
    
    // Mostrar feedback
    mostrarAlerta('✅ Comentario guardado. Presiona "Enviar Disputas" cuando estés listo', 'success');
    
    // Cerrar modal
    cerrarModalDisputa();
}

// ⭐ 5. ACTUALIZAR CELDA DE COMENTARIO EN LA TABLA
function actualizarCeldaComentario(bmsId, comentario) {
    const celda = document.querySelector(`.celda-comentario-disputa[data-bms="${bmsId}"]`);
    if (!celda) return;
    
    const textoSpan = celda.querySelector('.texto-comentario');
    
    if (comentario && comentario.length > 0) {
        textoSpan.textContent = comentario.length > 50 ? comentario.substring(0, 50) + '...' : comentario;
        textoSpan.style.color = '#9b59b6';
        textoSpan.style.fontStyle = 'normal';
        textoSpan.style.fontWeight = '600';
        celda.style.background = '#f3e5f5';
        celda.title = comentario; // Tooltip completo
    } else {
        textoSpan.textContent = 'Sin comentario';
        textoSpan.style.color = '#7f8c8d';
        textoSpan.style.fontStyle = 'italic';
        textoSpan.style.fontWeight = 'normal';
        celda.style.background = '';
        celda.title = '';
    }
}

// ⭐ 6. ENVIAR TODAS LAS DISPUTAS PENDIENTES
// ========================================
function enviarDisputas() {
    const disputasArray = Object.values(disputasPendientes);
    
    if (disputasArray.length === 0) {
        mostrarAlerta('⚠️ No hay disputas pendientes para enviar', 'warning');
        return;
    }
    
    // Confirmación
    let mensaje = `📨 ENVIAR DISPUTAS\n\n`;
    mensaje += `Vas a enviar ${disputasArray.length} disputa(s):\n\n`;
    
    disputasArray.forEach((d, i) => {
        mensaje += `${i+1}. ${d.nombre} (BMS ${d.bmsId})\n`;
        mensaje += `   "${d.comentario.substring(0, 50)}${d.comentario.length > 50 ? '...' : ''}"\n\n`;
    });
    
    mensaje += `¿Confirmas el envío?`;
    
    if (!confirm(mensaje)) return;
    
    console.log('📤 Enviando disputas:', disputasArray);
    mostrarLoading('Enviando ' + disputasArray.length + ' disputa(s)...');
    
    google.script.run
        .withSuccessHandler(function(response) {
            ocultarLoading();
            
            if (response.success) {
                console.log('✅ Disputas enviadas exitosamente');
                
                // ⭐ GUARDAR EN CACHE CON IDENTIFICADOR ÚNICO
                const cacheKey = `disputas_${selectLob.value}_${selectFecha.value}`;
                const disputasParaCache = disputasArray.map(d => ({
                    bmsId: d.bmsId,
                    nombre: d.nombre,
                    total: d.total,
                    comentario: d.comentario,
                    lob: d.lob,
                    fecha: d.fecha,
                    fechaDisputa: new Date().toISOString(),
                    usuarioDisputa: document.getElementById('nombreUsuario').textContent
                }));
                
                // Obtener disputas existentes en cache
                let todasLasDisputas = [];
                try {
                    const cacheExistente = localStorage.getItem('disputasEnviadas');
                    if (cacheExistente) {
                        todasLasDisputas = JSON.parse(cacheExistente);
                    }
                } catch (e) {
                    console.warn('⚠️ Error al leer cache existente:', e);
                    todasLasDisputas = [];
                }
                
                // Agregar nuevas disputas
                todasLasDisputas.push(...disputasParaCache);
                
                // Guardar en localStorage
                localStorage.setItem('disputasEnviadas', JSON.stringify(todasLasDisputas));
                localStorage.setItem(cacheKey, JSON.stringify(disputasParaCache));
                
                console.log('💾 Disputas guardadas en cache:', disputasParaCache.length);
                console.log('📦 Cache key:', cacheKey);
                
                mostrarAlerta('✅ ' + disputasArray.length + ' disputa(s) enviada(s) exitosamente. Recargando tabla...', 'success');
                
                // ⭐ LIMPIAR PENDIENTES
                disputasPendientes = {};
                
                // ⭐ RECARGAR SOLO LA TABLA (no toda la página)
                setTimeout(() => {
                    actualizarSoloTabla();
                }, 1500);
                
            } else {
                mostrarAlerta('❌ Error: ' + response.error, 'danger');
            }
        })
        .withFailureHandler(function(error) {
            ocultarLoading();
            mostrarAlerta('❌ Error: ' + error.message, 'danger');
            console.error('Error completo:', error);
        })
        .guardarDisputas(disputasArray);
}

// ⭐ 7. MARCAR VISUALMENTE COMO DISPUTADO
function marcarComoDisputado(bmsId) {
    const fila = document.querySelector(`tr[data-bms="${bmsId}"]`);
    if (!fila) return;
    
    fila.style.background = 'linear-gradient(135deg, #f8d7da 0%, #f5c6cb 100%)';
    fila.style.borderLeft = '5px solid #e74c3c';
    fila.style.boxShadow = '0 4px 12px rgba(231, 76, 60, 0.3)';
    
    const celdaNombre = fila.querySelector('td:nth-child(4)');
    if (celdaNombre) {
        const badge = document.createElement('div');
        badge.style.cssText = 'margin-top:8px;display:inline-block;';
        badge.innerHTML = `
            <span style="background:#e74c3c;color:white;padding:4px 12px;border-radius:12px;font-size:0.75rem;font-weight:bold;">
                🚨 DISPUTADO
            </span>
        `;
        celdaNombre.appendChild(badge);
    }
    
    // Deshabilitar checkbox
    const checkbox = fila.querySelector('.dispute-check');
    if (checkbox) {
        checkbox.disabled = true;
        checkbox.checked = false;
    }
    
    // Actualizar celda de comentario
    const celda = fila.querySelector('.celda-comentario-disputa');
    if (celda) {
        celda.style.background = '#f8d7da';
        celda.style.color = '#721c24';
        celda.style.fontWeight = '600';
    }
    
    console.log(`✅ BMS ${bmsId} marcado como disputado`);
}

// ⭐ 8. CARGAR DISPUTAS DESDE CACHE AL INICIAR
function cargarDisputasDesdeCache() {
    const cacheData = localStorage.getItem('disputasEnviadas');
    
    if (!cacheData) {
        console.log('📭 No hay disputas en cache');
        return;
    }
    
    try {
        disputasEnviadas = JSON.parse(cacheData);
        console.log('📦 Disputas cargadas desde cache:', disputasEnviadas.length);
        
        const lobActual = selectLob.value;
        const fechaActual = selectFecha.value;
        
        // Marcar las que corresponden a esta LOB y fecha
        disputasEnviadas.forEach(disputa => {
            if (disputa.lob === lobActual && disputa.fecha === fechaActual) {
                setTimeout(() => marcarComoDisputado(disputa.bmsId), 500);
            }
        });
        
    } catch (e) {
        console.error('❌ Error al parsear cache:', e);
        localStorage.removeItem('disputasEnviadas');
    }
}
