function generarTablaConLobs(lob, empleados, metricas, categorias, todasLasLobs) {
    if (!empleados || !metricas || !categorias || !todasLasLobs) {
        console.error('❌ Parámetros inválidos en generarTablaConLobs');
        mostrarError('Datos incompletos para generar la tabla');
        return;
    }
    
    console.log('📊 Generando tabla para LOB:', lob);
    ocultarLoading();
    
    const traslados = datos.trasladosPorLob && datos.trasladosPorLob[lob] ? datos.trasladosPorLob[lob] : { salientes: [], entrantes: [] };
    const agentesTrasladados = traslados.salientes.map(t => t.bmsId);
    
    // ========================================
    // ⭐ OBTENER HORAS SEMANALES DESDE agentesPermitidos
    // ========================================
    const horasSemanalesPorAgente = {};
    
    console.log('📊 Extrayendo horas semanales de agentesPermitidos...');
    
    // ⭐ Las horas ya vienen del servidor en datos.agentesPermitidos
    const agentesPermitidos = datos.agentesPermitidos || {};
    
    Object.keys(agentesPermitidos).forEach(bmsId => {
        const agente = agentesPermitidos[bmsId];
        
        // Solo considerar agentes de esta LOB
        const normalizar = s => {
            if (!s) return "";
            return s.toString()
                .normalize("NFD")
                .replace(/[\u0300-\u036f]/g, "")
                .toLowerCase()
                .trim()
                .replace(/\s+/g, ' ');
        };
        
        if (normalizar(agente.lob) === normalizar(lob)) {
            horasSemanalesPorAgente[bmsId] = agente.horasSemanales || 42;
        }
    });
    
    console.log(`✅ Horas semanales cargadas: ${Object.keys(horasSemanalesPorAgente).length} agentes de ${lob}`);
    
    let totalGeneralTabla = 0;
    
    // ========================================
    // ⭐⭐⭐ THEAD - FILA 1 (CON COLUMNA ROLE)
    // ========================================
    let theadHTML = '<tr><th rowspan="2" onclick="ordenarTabla(0)" style="cursor:pointer;" title="Ordenar">#</th>';
    theadHTML += '<th rowspan="2" style="background:#3498db;color:white;min-width:200px;">🔄 Trasladar</th>';
    theadHTML += '<th rowspan="2" onclick="ordenarTabla(1)" style="cursor:pointer;" title="Ordenar">CCMS ID</th>';
    theadHTML += '<th rowspan="2" onclick="ordenarTabla(2)" style="cursor:pointer;" title="Ordenar">Nombre</th>';
    theadHTML += '<th rowspan="2" onclick="ordenarTabla(3)" style="cursor:pointer;background:#667eea;color:white;" title="Ordenar">Role</th>';
    theadHTML += '<th rowspan="2" style="background:#16a085;color:white;">⏱️ Horas Semanales</th>';
        
    categorias.forEach(cat => {
        const kpis = metricas[cat] || [];
        theadHTML += kpis.length > 0 ? '<th colspan="' + (kpis.length * 2) + '">' + cat + '</th>' : '<th colspan="2">' + cat + '</th>';
    });
    
    theadHTML += '<th rowspan="2" onclick="ordenarTabla(-1)" style="cursor:pointer;background:#27ae60;" title="Ordenar">Total $</th>';
    theadHTML += '<th rowspan="2" style="background:#16a085;color:white;">📊 Prorrateo</th>';
    theadHTML += '<th rowspan="2" style="background:#28a745;color:white;min-width:100px;">✅ Aprobar<br><label style="font-size:0.75rem;font-weight:normal;cursor:pointer;margin-top:5px;display:inline-block;"><input type="checkbox" id="select-all-aprobar" onchange="seleccionarTodosAprobar(this)" style="width:16px;height:16px;cursor:pointer;margin-right:5px;">Todos</label></th>';
    theadHTML += '<th rowspan="2" style="background:#9b59b6;color:white;">Disputar</th>';
    theadHTML += '<th rowspan="2" style="background:#9b59b6;color:white;">Comentario</th></tr><tr>';
    
    // ========================================
    // ⭐ THEAD - FILA 2 (Headers de KPIs)
    // ========================================
    categorias.forEach(cat => {
        const kpis = metricas[cat] || [];
        if (kpis.length === 0) {
            theadHTML += '<th colspan="2">-</th>';
        } else {
            kpis.forEach(kpiObj => {
                const esMultiTier = kpiObj.esMultiTier === true;
                
                let headerKPI = kpiObj.kpi;
                let headerScore = '';
                
                if (esMultiTier) {
                    headerScore = 'Multi-Escala<br><small style="font-size:0.6rem;color:#7f8c8d;font-weight:normal;">';
                    
                    if (kpiObj.scoresPorTier && kpiObj.scoresPorTier.length > 0) {
                        kpiObj.scoresPorTier.forEach((tier, idx) => {
                            const part = tier.participacion ? (tier.participacion * 100).toFixed(0) + '%' : '100%';
                            const sep = idx < kpiObj.scoresPorTier.length - 1 ? ' | ' : '';
                            
                            headerScore += `${tier.score} <span style="color:#f39c12;">(${part})</span>${sep}`;
                        });
                    }
                    
                    headerScore += '</small>';
                } else {
                    headerScore = kpiObj.score || '-';
                }
                
                theadHTML += `<th style="font-size:0.8rem;line-height:1.4;">${headerKPI}<br>${headerScore}</th>`;
                theadHTML += '<th style="font-size:0.8rem;background:#34495e;">Bono $</th>';
            });
        }
    });
    
    theadHTML += '</tr>';
    
    // ========================================
    // ⭐⭐⭐ TBODY
    // ========================================
    let tbodyHTML = '';
    const keys = Object.keys(empleados);
    
    if (keys.length === 0) {
        tbodyHTML = '<tr><td colspan="100" class="info-message">No hay empleados</td></tr>';
    } else {
        let contadorVisible = 0;
        
        // ⭐⭐⭐ LOOP DE EMPLEADOS ⭐⭐⭐
        keys.forEach((empKey, i) => {
            const emp = empleados[empKey];
            const fueTrasladado = agentesTrasladados.includes(emp.bmsId);
            
            let tieneSalarioPersonalizado = false;
            Object.values(emp.resultados).forEach(res => {
                if (res.salarioPersonalizado === true) {
                    tieneSalarioPersonalizado = true;
                }
            });

            let estiloFila = '';
            let claseFila = '';

            if (fueTrasladado) {
                estiloFila = 'style="background:linear-gradient(135deg, #fff3cd 0%, #ffeaa7 100%);border-left:4px solid #f39c12;"';
                claseFila = 'class="agente-trasladado"';
            } else if (tieneSalarioPersonalizado) {
                estiloFila = 'style="background:linear-gradient(135deg, #fff9c4 0%, #ffeb3b 100%);border-left:4px solid #fbc02d;"';
                claseFila = 'class="agente-salario-personalizado"';
            }
            
            tbodyHTML += `<tr data-bms="${emp.bmsId}" data-nombre="${emp.nombre.toLowerCase()}" id="row-${emp.bmsId}" ${estiloFila} ${claseFila}>`;
            tbodyHTML += '<td>' + (++contadorVisible) + '</td>';
            
            // CELDA DE TRASLADO
            tbodyHTML += '<td style="padding:8px;"><div style="display:flex;align-items:center;gap:8px;justify-content:center;">';
            
            if (fueTrasladado) {
                const trasladoInfo = traslados.salientes.find(t => t.bmsId === emp.bmsId);
                tbodyHTML += `<span style="background:#f39c12;color:white;padding:4px 8px;border-radius:10px;font-size:0.7rem;font-weight:bold;white-space:nowrap;">📤 Trasladado a ${trasladoInfo ? trasladoInfo.lobDestino : 'otra LOB'}</span>`;
            } else {
                tbodyHTML += '<input type="checkbox" class="recalc-check" data-bms="' + emp.bmsId + '" onchange="toggleRecalcLob(this)" style="width:18px;height:18px;cursor:pointer;">';
                tbodyHTML += '<select class="recalc-lob-select" data-bms="' + emp.bmsId + '" disabled style="padding:5px;border:1px solid #ddd;border-radius:4px;font-size:0.85rem;width:110px;"><option value="">LOB destino</option>';
                
                todasLasLobs.forEach(lobOption => {
                    if (lobOption !== lob) {
                        tbodyHTML += '<option value="' + lobOption + '">' + lobOption + '</option>';
                    }
                });
                
                tbodyHTML += '</select>';
            }
            
            tbodyHTML += '</div></td>';
            
            // ⭐⭐⭐ CCMS ID
            tbodyHTML += '<td>' + emp.bmsId + '</td>';
            
            // ⭐⭐⭐ NOMBRE
            tbodyHTML += '<td style="text-align:left;">' + emp.nombre;
            
            if (fueTrasladado) {
                tbodyHTML += '<br><small style="color:#e67e22;font-weight:bold;">🚨 Trasladado temporalmente</small>';
            }
            
            if (tieneSalarioPersonalizado) {
                tbodyHTML += '<br><small style="background:#fbc02d;color:#000;padding:2px 8px;border-radius:10px;font-size:0.7rem;font-weight:bold;">💰 Salario Personalizado</small>';
            }
            
            tbodyHTML += '</td>';

            // ⭐⭐⭐ NUEVA COLUMNA: ROLE
            tbodyHTML += '<td style="text-align:center;">';
            const roleAgente = agentesPermitidos[emp.bmsId]?.rol || emp.rol || '-';
            tbodyHTML += '<span style="background:#667eea15;color:#5b5fc7;padding:6px 12px;border-radius:8px;font-size:0.85rem;font-weight:700;display:inline-block;box-shadow:0 2px 4px rgba(102,126,234,0.2);">' + roleAgente + '</span>';
            tbodyHTML += '</td>';

            // ========================================
            // ⭐ CELDA DE HORAS SEMANALES (DESDE agentesPermitidos)
            // ========================================
            tbodyHTML += '<td style="text-align:center;">';
            const horasSemanales = horasSemanalesPorAgente[emp.bmsId];
            
            if (horasSemanales && horasSemanales > 0) {
                tbodyHTML += `<span style="background:#16a085;color:white;padding:4px 10px;border-radius:12px;font-weight:600;">${horasSemanales}h/sem</span>`;
            } else {
                tbodyHTML += '<span style="color:#95a5a6;font-size:0.85rem;">Sin datos</span>';
            }
            tbodyHTML += '</td>';

            // ========================================
            // ⭐⭐⭐ CALCULAR TOTAL CON LÓGICA DE KEYS
            // ========================================
            let totalKeys = 0;
            let todosKeysOK = true;
            let tieneAlgunKey = false;
            
            const kpisKeys = metricas['keys'] || [];
            kpisKeys.forEach((kpiObj, idx) => {
                const res = emp.resultados['keys_' + idx] || {};
                const bono = res.bono || 0;
                
                if (!res.sinDatos) {
                    tieneAlgunKey = true;
                    if (res.cumple) {
                        totalKeys += bono;
                    } else {
                        todosKeysOK = false;
                    }
                }
            });
            
            if (tieneAlgunKey && !todosKeysOK) {
                totalKeys = 0;
            }
            
            let totalOtrasCategorias = 0;
            let totalDecreases = 0;

            // ========================================
            // ⭐⭐⭐ GENERAR CELDAS DE KPIs ⭐⭐⭐
            // ========================================
            categorias.forEach(cat => {
                const kpis = metricas[cat] || [];
                
                if (kpis.length === 0) {
                    tbodyHTML += '<td colspan="2">-</td>';
                } else {
                    kpis.forEach((kpiObj, idx) => {
                        const res = emp.resultados[cat + '_' + idx] || {};
                        const bono = res.bono || 0;
                        const esKey = cat === 'keys';
                        const esDecrease = cat === 'decreasedKPIs';
                        
                        // ACUMULAR EN CATEGORÍAS
                        if (!esKey && res.cumple && !res.sinDatos) {
                            if (esDecrease) {
                                totalDecreases += bono;
                            } else {
                                totalOtrasCategorias += bono;
                            }
                        }
                        
                        // ========================================
                        // ⭐⭐⭐ DETERMINAR QUÉ MOSTRAR ⭐⭐⭐
                        // ========================================
                        let scoreAMostrar = kpiObj.score || '-';
                        let participacionAMostrar = null;
                        
                        // Validación: ¿Es Multi-Tier?
                        const esMultiTier = kpiObj.esMultiTier === true;
                        
                        // ⭐⭐⭐ CORRECCIÓN PRINCIPAL: OBTENER PARTICIPACIÓN CORRECTA ⭐⭐⭐
                        if (esMultiTier) {
                            const tieneScoreCumplido = res.scoreCumplido && 
                                                      res.scoreCumplido !== null && 
                                                      res.scoreCumplido !== undefined &&
                                                      res.scoreCumplido !== '';
                            
                            if (tieneScoreCumplido && res.cumple) {
                                scoreAMostrar = res.scoreCumplido;
                                
                                // ⭐ MÉTODO 1: Usar participacionUsada si está presente
                                if (res.participacionUsada !== undefined && res.participacionUsada !== null) {
                                    participacionAMostrar = (res.participacionUsada * 100).toFixed(0) + '%';
                                } else {
                                    // ⭐ MÉTODO 2: Buscar en scoresPorTier
                                    const tierMatch = kpiObj.scoresPorTier?.find(t => {
                                        const tierScore = String(t.score).trim().toLowerCase();
                                        const cumplido = String(res.scoreCumplido).trim().toLowerCase();
                                        return tierScore === cumplido;
                                    });
                                    
                                    if (tierMatch && tierMatch.participacion) {
                                        participacionAMostrar = (tierMatch.participacion * 100).toFixed(0) + '%';
                                    }
                                }
                            }
                        }
                        
                        // FORMATEAR valorF
                        let valorFormateado = '-';
                        if (res.valorF !== undefined && res.valorF !== null) {
                            const valor = parseFloat(res.valorF);
                            if (!isNaN(valor)) {
                                if (Number.isInteger(valor)) {
                                    valorFormateado = valor.toLocaleString('es-CO', {
                                        minimumFractionDigits: 0,
                                        maximumFractionDigits: 0
                                    });
                                } else {
                                    valorFormateado = valor.toLocaleString('es-CO', {
                                        minimumFractionDigits: 2,
                                        maximumFractionDigits: 2
                                    });
                                }
                            } else {
                                valorFormateado = res.valorF;
                            }
                        }
                        
                        // ========================================
                        // ⭐ RENDERIZADO SEGÚN TIPO DE KPI
                        // ========================================
                        if (esKey) {
                            // KEYS
                            if (res.sinDatos) {
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">Sin datos</td>';
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">$0.00</td>';
                            } else if (res.cumple) {
                                tbodyHTML += '<td style="background:#d4edda;color:#155724;font-weight:bold;">' + valorFormateado + '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small></td>';
                                tbodyHTML += '<td style="background:#d4edda;color:#155724;font-weight:bold;">$' + bono.toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</td>';
                            } else {
                                tbodyHTML += '<td style="background:#f8d7da;color:#721c24;font-weight:bold;">' + valorFormateado + '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small></td>';
                                tbodyHTML += '<td style="background:#f8d7da;color:#e74c3c;font-weight:bold;">$0.00</td>';
                            }
                        } else if (esDecrease) {
                            // DECREASED KPIs
                            if (res.sinDatos) {
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">Sin datos</td>';
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">$0.00</td>';
                            } else if (res.cumple) {
                                const bonoFormateado = bono < 0 
                                    ? '-$' + Math.abs(bono).toLocaleString('es-CO', {minimumFractionDigits: 2})
                                    : '$' + bono.toLocaleString('es-CO', {minimumFractionDigits: 2});
                                
                                let infoDecrease = '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small>';
                                
                                if (esMultiTier && participacionAMostrar) {
                                    infoDecrease += `<br><small style="font-size:0.65rem;color:#f39c12;">📊 ${participacionAMostrar}</small>`;
                                }
                                
                                infoDecrease += '<br><small style="font-size:0.65rem;color:#e74c3c;">❌ Penalización</small>';
                                
                                tbodyHTML += '<td style="background:#f8d7da;color:#721c24;font-weight:bold;">' + valorFormateado + infoDecrease + '</td>';
                                tbodyHTML += '<td style="background:#f8d7da;color:#e74c3c;font-weight:bold;">' + bonoFormateado + '</td>';
                            } else {
                                let infoDecrease = '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small>';
                                
                                if (esMultiTier && participacionAMostrar) {
                                    infoDecrease += `<br><small style="font-size:0.65rem;color:#f39c12;">📊 ${participacionAMostrar}</small>`;
                                }
                                
                                infoDecrease += '<br><small style="font-size:0.65rem;color:#27ae60;">✅ Sin penalización</small>';
                                
                                tbodyHTML += '<td style="background:#d4edda;color:#155724;font-weight:bold;">' + valorFormateado + infoDecrease + '</td>';
                                tbodyHTML += '<td style="background:#d4edda;color:#155724;font-weight:bold;">$0.00</td>';
                            }
                        } else {
                            // ⭐⭐⭐ BONUS STRUCTURE Y ADDITIONAL INCENTIVES ⭐⭐⭐
                            if (res.sinDatos) {
                                tbodyHTML += '<td style="background:#f0f0f0;color:#999;">Sin datos</td>';
                                tbodyHTML += '<td style="background:#f0f0f0;">$0.00</td>';
                            } else {
                                const claseEstilo = res.cumple ? 'cumple' : 'no-cumple';
                                
                                // ⭐ CONSTRUIR INFO ADICIONAL
                                let infoAdicional = '';
                                
                                if (esMultiTier && res.cumple && participacionAMostrar) {
                                    // ✅ MOSTRAR: Tier cumplido + % Participación DE ESE TIER
                                    infoAdicional = `
                                        <br><small style="font-size:0.65rem;color:#3498db;font-weight:600;">
                                            ⭐ Tier: ${scoreAMostrar}
                                        </small>
                                        <br><small style="font-size:0.65rem;color:#f39c12;font-weight:600;background:rgba(243,156,18,0.15);padding:2px 6px;border-radius:8px;display:inline-block;margin-top:2px;">
                                            📊 ${participacionAMostrar}
                                        </small>
                                    `;
                                } else if (!esMultiTier) {
                                    // KPI normal: solo mostrar score
                                    infoAdicional = '<br><small style="font-size:0.7rem;">(' + scoreAMostrar + ')</small>';
                                }
                                
                                tbodyHTML += '<td class="' + claseEstilo + '">' 
                                    + valorFormateado 
                                    + infoAdicional
                                    + '</td>';
                                tbodyHTML += '<td>$' + Math.max(0, bono).toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</td>';
                            }
                        }
                    });
                }
            });
            
            // ========================================
            // ⭐⭐⭐ CALCULAR TOTAL FINAL CON AJUSTE DE HORAS SEMANALES
            // ========================================
            let totalFinal = 0;
            let mensajeTotal = '';
            let estiloTotal = 'background:#d5f4e6;font-weight:bold;';
            
            if (tieneAlgunKey) {
                if (todosKeysOK) {
                    totalFinal = Math.max(0, totalKeys) + Math.max(0, totalOtrasCategorias) + totalDecreases;
                } else {
                    totalFinal = totalDecreases;
                    if (totalOtrasCategorias > 0) {
                        estiloTotal = 'background:#f8d7da;font-weight:bold;border-left:4px solid #e74c3c;';
                        mensajeTotal = '<br><small style="font-size:0.65rem;color:#721c24;">❌ No cumple todos los keys</small>';
                    }
                }
            } else {
                totalFinal = Math.max(0, totalOtrasCategorias) + totalDecreases;
            }
            
            totalFinal = Math.max(0, totalFinal);
            
            // ⭐⭐⭐ APLICAR AJUSTE DE HORAS SEMANALES AUTOMÁTICAMENTE ⭐⭐⭐
            const horasSem = horasSemanalesPorAgente[emp.bmsId] || 42;
            const factorHoras = horasSem / 42;
            let totalAjustado = totalFinal;
            let mensajeAjuste = '';
            
            if (horasSem !== 42 && totalFinal > 0) {
                totalAjustado = totalFinal * factorHoras;
                mensajeAjuste = `<br><small style="font-size:0.65rem;color:#16a085;font-weight:600;background:rgba(22,160,133,0.1);padding:2px 6px;border-radius:6px;display:inline-block;margin-top:3px;">⚙️ Ajustado por ${horasSem}h/sem (${(factorHoras * 100).toFixed(0)}%)</small>`;
                console.log(`💰 ${emp.nombre}: $${totalFinal.toFixed(2)} × ${factorHoras.toFixed(2)} = $${totalAjustado.toFixed(2)}`);
            }
            
            totalGeneralTabla += totalAjustado;
            
            // CELDA TOTAL
            tbodyHTML += `<td class="celda-total" data-bms="${emp.bmsId}" data-bono-original="${totalAjustado}" style="${estiloTotal}">$${totalAjustado.toLocaleString('es-CO', {minimumFractionDigits: 2})}${mensajeTotal}${mensajeAjuste}</td>`;
            
            // CELDA PRORRATEO (PARA AUSENCIAS ADICIONALES)
            tbodyHTML += '<td style="text-align:center;padding:8px;">';
            if (totalAjustado > 0) {
                const horasSem = horasSemanalesPorAgente[emp.bmsId] || 42;
                tbodyHTML += `<input type="checkbox" class="prorrateo-check" data-bms="${emp.bmsId}" data-horas-semanales="${horasSem}" data-bono-original="${totalAjustado}" onchange="calcularProrrateoInstantaneo(this)" style="width:20px;height:20px;cursor:pointer;" title="Ajustar por ausencias (Horas_N)">`;
            } else {
                tbodyHTML += '<span style="color:#95a5a6;">-</span>';
            }
            tbodyHTML += '</td>';
            
            // CELDA APROBAR
            tbodyHTML += '<td style="text-align:center;padding:8px;background:#f0fff4;">';
            tbodyHTML += `<input type="checkbox" class="aprobado-check" data-bms="${emp.bmsId}" data-nombre="${emp.nombre}" data-total="${totalAjustado}" onchange="actualizarTotalAprobar()" style="width:22px;height:22px;cursor:pointer;accent-color:#28a745;">`;
            tbodyHTML += '</td>';
            
            // DISPUTAR Y COMENTARIO
            tbodyHTML += '<td style="text-align:center;">';
            tbodyHTML += '<input type="checkbox" class="dispute-check" data-bms="' + emp.bmsId + '" data-nombre="' + emp.nombre + '" data-total="' + totalAjustado + '" onchange="toggleDisputeComment(this)" style="width:20px;height:20px;cursor:pointer;">';
            tbodyHTML += '</td>';
            tbodyHTML += '<td class="celda-comentario-disputa" data-bms="' + emp.bmsId + '" style="padding:8px;font-size:0.85rem;color:#7f8c8d;font-style:italic;">';
            tbodyHTML += '<span class="texto-comentario">Sin comentario</span>';
            tbodyHTML += '</td>';
            tbodyHTML += '</tr>';
        });
    }
    
    // FILA DE TOTALES
    if (keys.length > 0) {
        let colspanAntes = 6; // ⭐ ACTUALIZADO: Ahora son 6 columnas antes de KPIs (#, Trasladar, CCMS, Nombre, Role, Horas)
        categorias.forEach(cat => {
            const kpis = metricas[cat] || [];
            colspanAntes += kpis.length > 0 ? (kpis.length * 2) : 2;
        });
        
        tbodyHTML += '<tr style="background:#27ae60;color:white;font-weight:bold;font-size:1.1rem;">';
        tbodyHTML += '<td colspan="' + colspanAntes + '" style="text-align:right;padding:15px;">TOTAL GENERAL (CON AJUSTE AUTOMÁTICO):</td>';
        tbodyHTML += '<td style="text-align:center;padding:15px;font-size:1.3rem;">$' + totalGeneralTabla.toLocaleString('es-CO', {minimumFractionDigits: 2}) + '</td>';
        tbodyHTML += '<td colspan="4"></td>';
        tbodyHTML += '</tr>';
    }
    
    // CONSTRUCCIÓN HTML FINAL
    const container = document.getElementById('table-container');
    let html = '<div class="table-header"><h3 class="table-title">📋 Agentes - ' + lob + '</h3>';
    html += '<div style="display:flex;align-items:center;gap:15px;flex-wrap:wrap;">';
    html += '<input type="text" id="search-input" placeholder="🔍 Buscar..." onkeyup="filtrarTabla()" style="padding:10px 15px;border:2px solid #dfe6e9;border-radius:8px;font-size:1rem;width:300px;">';
