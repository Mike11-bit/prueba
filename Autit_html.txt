function procesarRespuestaExitosa(response) {
    console.log('✅ Respuesta recibida del servidor');
    console.log('Response completo:', response);
    
    try {
        if (response === undefined || response === null) {
            throw new Error('La función del servidor no retornó ningún valor');
        }
        
        if (typeof response !== 'object') {
            throw new Error('Respuesta inválida del servidor (tipo: ' + typeof response + ')');
        }
        
        if (response.error) {
            throw new Error('Error del servidor: ' + response.error);
        }
        
        console.log('✅ Validación de response exitosa');
        
        cargarInfoUsuario(response.usuario);
        window.lobsPermitidasGlobal = response.lobsPermitidas || [];
        console.log('📋 LOBs permitidas guardadas:', window.lobsPermitidasGlobal);
        
        window.permisoDisputasGlobal = response.permisoDisputas;
        console.log('🔐 Permiso disputas guardado:', window.permisoDisputasGlobal);
        
        verificarYControlarAccesoDisputas(response.permisoDisputas);
        
        datosAudit = Array.isArray(response.auditData) ? response.auditData : [];
        datosFiltrados = [...datosAudit];
        
        datosDisputas = datosAudit.filter(d => d.tipoAccion === 'Disputa');
        disputasFiltradas = [...datosDisputas];
        
        console.log(`📊 ${datosAudit.length} registros totales (${datosDisputas.length} disputas)`);
        
        actualizarUltimaActualizacion();
        inicializarDashboard();
        
        if (datosAudit.length === 0) {
            mostrarToast('📋 No hay registros en Audit todavía', 'warning');
            mostrarMensajeVacio();
        } else {
            mostrarToast(`✅ ${datosAudit.length} registros cargados exitosamente`, 'success');
        }
        
    } catch (error) {
        console.error('❌ Error procesando respuesta:', error);
        mostrarError('Error al procesar los datos: ' + error.message);
    } finally {
        mostrarCargando(false);
    }
}

    }
}

function debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
        const later = () => {
            clearTimeout(timeout);
            func(...args);
        };
        clearTimeout(timeout);
        timeout = setTimeout(later, wait);
    };
}

function obtenerValoresFiltros() {
    const fechaDesde = document.getElementById('fechaDesde')?.value;
    const fechaHasta = document.getElementById('fechaHasta')?.value;
    
    return {
        fechaDesde: fechaDesde ? new Date(fechaDesde) : null,
        fechaHasta: fechaHasta ? new Date(fechaHasta + 'T23:59:59') : null,
        usuario: document.getElementById('filtroUsuario')?.value || '',
        tipoAccion: document.getElementById('filtroTipoAccion')?.value || '',
        lob: document.getElementById('filtroLob')?.value || '',
        mesCDM: document.getElementById('filtroMesCDM')?.value || '',
        busqueda: document.getElementById('busquedaRapida')?.value?.toLowerCase() || ''
    };
}

function cumpleFiltros(dato, filtros) {
    const cumpleMesCDM = !filtros.mesCDM || 
                         (dato.mesCDM && String(dato.mesCDM).trim() === String(filtros.mesCDM).trim());
    
    let cumpleFecha = true;
    if (!filtros.mesCDM && (filtros.fechaDesde || filtros.fechaHasta)) { 
        try {
            const fReg = new Date(dato.timestamp);
            cumpleFecha = (!filtros.fechaDesde || fReg >= filtros.fechaDesde) && 
                          (!filtros.fechaHasta || fReg <= filtros.fechaHasta);
        } catch(e) { cumpleFecha = true; }
    }
    
    const cumpleUsuario = !filtros.usuario || dato.usuario === filtros.usuario;
    const cumpleTipo = !filtros.tipoAccion || dato.tipoAccion === filtros.tipoAccion;
    const cumpleLob = !filtros.lob || dato.lob === filtros.lob;
    const cumpleBusqueda = !filtros.busqueda ||
                          (dato.bmsId && dato.bmsId.toString().toLowerCase().includes(filtros.busqueda)) ||
                          (dato.nombreAgente && dato.nombreAgente.toLowerCase().includes(filtros.busqueda));
    
    return cumpleFecha && cumpleUsuario && cumpleTipo && cumpleLob && cumpleMesCDM && cumpleBusqueda;
}

                <td colspan="9" style="text-align:center;padding:40px;color:#7f8c8d;">
                    No hay registros que coincidan con los filtros aplicados
                </td>
            </tr>
        `;
        actualizarInfoPaginacion(0);
        return;
    }
    
    tbody.innerHTML = registrosPagina.map(d => crearFilaTabla(d)).join('');
    actualizarInfoPaginacion(datosFiltrados.length);
}

function crearFilaTabla(dato) {
    const monto = dato.monto 
        ? `$${Number(dato.monto).toLocaleString('es-CO', {minimumFractionDigits: 2})}` 
        : '-';
    
    const datosEscapados = JSON.stringify(dato)
        .replace(/'/g, "&#39;")
        .replace(/"/g, "&quot;");
    
    return `
        <tr>
            <td>${formatearFecha(dato.timestamp)}</td>
            <td><span class="badge ${getBadgeClass(dato.tipoAccion)}">${dato.tipoAccion}</span></td>
            <td>${dato.usuario || '-'}</td>
            <td>${dato.lob || '-'}</td>
            <td><strong>${dato.bmsId || '-'}</strong></td>
            <td>${dato.nombreAgente || '-'}</td>
            <td style="text-align:right;font-weight:600;color:var(--success);">${monto}</td>
            <td>${dato.mesCDM || '-'}</td>
            <td>
                <button class="btn-primary" 
                        style="padding:6px 12px;font-size:0.85rem;" 
                        onclick='verDetalles(${datosEscapados})'>
                    👁️ Ver
                </button>
            </td>
        </tr>
    `;
}

function formatearFecha(fecha) {
    try {
        return new Date(fecha).toLocaleString('es-CO');
    } catch (e) {
        return 'Fecha inválida';
    }
}

function getBadgeClass(tipoAccion) {
    const classes = {
        'Aprobación': 'badge-success',
        'Disputa': 'badge-warning',
        'Traslado': 'badge-info',
        'Prorrateo': 'badge-secondary',
        'Cambio Rol': 'badge-danger'
    };
    return classes[tipoAccion] || 'badge-secondary';
}

function actualizarInfoPaginacion(totalRegistros) {
    const totalPaginas = Math.ceil(totalRegistros / registrosPorPagina);
    
    const elementos = {
        paginaActual: paginaActual,
        totalPaginas: totalPaginas || 1,
        totalRegistros: totalRegistros
    };
    
    Object.entries(elementos).forEach(([id, valor]) => {
        const elemento = document.getElementById(id);
        if (elemento) {
            elemento.textContent = valor;
        }
    });
}

function cambiarPagina(direccion) {
    const totalPaginas = Math.ceil(datosFiltrados.length / registrosPorPagina);
    const nuevaPagina = paginaActual + direccion;
    
    if (nuevaPagina >= 1 && nuevaPagina <= totalPaginas) {
        paginaActual = nuevaPagina;
        renderizarTabla();
        
        const tablaContainer = document.querySelector('.table-container');
        if (tablaContainer) {
            tablaContainer.scrollIntoView({ behavior: 'smooth', block: 'start' });
        }
    }
}

function generarGraficos() {
    if (datosFiltrados.length === 0) {

function destruirGraficos() {
    if (chartTipoAccion) {
        chartTipoAccion.destroy();
        chartTipoAccion = null;
    }
    if (chartActividad) {
        chartActividad.destroy();
        chartActividad = null;
    }
}

function generarGraficoTipoAccion() {
    const ctx = document.getElementById('chartTipoAccion');
    if (!ctx) return;
    
    const contadores = {};
    datosFiltrados.forEach(d => {
        contadores[d.tipoAccion] = (contadores[d.tipoAccion] || 0) + 1;
    });
    
    const labels = Object.keys(contadores);
    const data = Object.values(contadores);
    
    const colores = {
        'Aprobación': '#27ae60',
        'Disputa': '#f39c12',
        'Traslado': '#3498db',
        'Prorrateo': '#9b59b6',
        'Cambio Rol': '#e74c3c'
    };
    
    const backgroundColors = labels.map(l => colores[l] || '#95a5a6');
    
    if (chartTipoAccion) chartTipoAccion.destroy();
    
    chartTipoAccion = new Chart(ctx, {
        type: 'doughnut',
        data: {
            labels: labels,
            datasets: [{
                data: data,
                backgroundColor: backgroundColors,
                borderWidth: 2,
                borderColor: '#fff'
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: true,
            plugins: {
                legend: {
                    position: 'bottom'
                },
                tooltip: {
                    callbacks: {
                        label: function(context) {
                            const total = context.dataset.data.reduce((a, b) => a + b, 0);
                            const porcentaje = ((context.parsed / total) * 100).toFixed(1);
                            return `${context.label}: ${context.parsed} (${porcentaje}%)`;
                        }
                    }
                }
            }
        }
    });
}

function generarGraficoActividad() {
    const ctx = document.getElementById('chartActividad');
    if (!ctx) return;
    
    if (chartActividad) { 
        chartActividad.destroy(); 
        chartActividad = null; 
    }
    
    if (!datosFiltrados || datosFiltrados.length === 0) return;
    
    const actividadPorDia = {};
    datosFiltrados.forEach(d => {
        try {
            const fecha = new Date(d.timestamp);
            if (!isNaN(fecha.getTime())) {
                const fechaStr = fecha.toLocaleDateString('es-CO');
                actividadPorDia[fechaStr] = (actividadPorDia[fechaStr] || 0) + 1;
            }
        } catch(e) {}
    });
    
    const fechasOrdenadas = Object.keys(actividadPorDia).sort((a, b) => {
        const partsA = a.split('/'); 
        const partsB = b.split('/');
        return new Date(partsA[2], partsA[1]-1, partsA[0]) - new Date(partsB[2], partsB[1]-1, partsB[0]);
    });
    
    chartActividad = new Chart(ctx, {
        type: 'line',
        data: {
            labels: fechasOrdenadas,
            datasets: [{
                label: 'Operaciones',
                data: fechasOrdenadas.map(f => actividadPorDia[f]),
                borderColor: '#3498db',
                backgroundColor: 'rgba(52, 152, 219, 0.1)',
                fill: true,
                tension: 0.3
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: true, // ⭐ CAMBIAR A true
            plugins: { 
                legend: { display: false } 
            },
            scales: {
                y: {
                    beginAtZero: true,
                    ticks: { stepSize: 1 }
                }
            }
        }
    });
}

function generarDescripcion(d) {
    const descripciones = {
        'Aprobación': `fue aprobado en <strong>${d.lob}</strong> para el mes de <strong>${d.mesCDM}</strong>`,
        'Disputa': `presentó una disputa en <strong>${d.lob}</strong>`,
        'Traslado': `fue trasladado a <strong>${d.lobDestino || 'otra LOB'}</strong>`,
        'Prorrateo': `recibió prorrateo en <strong>${d.lob}</strong>`,
        'Cambio Rol': `cambió de rol a <strong>${d.estadoNuevo}</strong>`
    };
    
    return descripciones[d.tipoAccion] || `tuvo una operación en <strong>${d.lob}</strong>`;
}

function formatearTiempoRelativo(timestamp) {
    const ahora = new Date();
    const fecha = new Date(timestamp);
    const diferencia = ahora - fecha;
    
    const minutos = Math.floor(diferencia / 60000);
    const horas = Math.floor(diferencia / 3600000);
    const dias = Math.floor(diferencia / 86400000);
    
    if (minutos < 1) return 'Ahora mismo';
    if (minutos < 60) return `Hace ${minutos} min`;
    if (horas < 24) return `Hace ${horas} hora${horas > 1 ? 's' : ''}`;
    if (dias < 30) return `Hace ${dias} día${dias > 1 ? 's' : ''}`;
    return fecha.toLocaleDateString('es-CO');
}

function obtenerColorEstado(estado) {
    const colores = {
        'Pendiente': '#f39c12',
        '⏳ En Proceso': '#3498db',
        '✅ Aprobada': '#27ae60',
        '❌ Rechazada': '#e74c3c',
        '📋 Más Información': '#9b59b6'
    };
    
    for (const key in colores) {
        if (estado.includes(key) || estado.includes(key.replace(/[⏳✅❌📋]/g, '').trim())) {
            return colores[key];
        }
    }
    
    return '#95a5a6';
}

function filtrarDisputas() {
    const estadoFiltro = document.getElementById('filtroEstadoDisputa')?.value || '';
    const lobFiltro = document.getElementById('filtroLobDisputa')?.value || '';
    const fechaDesde = document.getElementById('fechaDesdeDisputa')?.value;
    const fechaHasta = document.getElementById('fechaHastaDisputa')?.value;
    const busqueda = document.getElementById('busquedaDisputa')?.value?.toLowerCase() || '';
    
    const fechaDesdeObj = fechaDesde ? new Date(fechaDesde) : null;
    const fechaHastaObj = fechaHasta ? new Date(fechaHasta + 'T23:59:59') : null;
    
    disputasFiltradas = datosDisputas.filter(d => {
        const fechaRegistro = new Date(d.timestamp);
        
        const cumpleEstado = !estadoFiltro || d.estado.includes(estadoFiltro);
        const cumpleLob = !lobFiltro || d.lob === lobFiltro;
        const cumpleFecha = (!fechaDesdeObj || fechaRegistro >= fechaDesdeObj) &&
                           (!fechaHastaObj || fechaRegistro <= fechaHastaObj);
        const cumpleBusqueda = !busqueda ||
                              (d.bmsId && d.bmsId.toString().toLowerCase().includes(busqueda)) ||
                              (d.nombreAgente && d.nombreAgente.toLowerCase().includes(busqueda));
        
        return cumpleEstado && cumpleLob && cumpleFecha && cumpleBusqueda;
    });
    
    paginaActualDisputas = 1;
    renderizarTablaDisputas(disputasFiltradas);
}

function cargarFiltrosDisputas() {
    let lobs;
    
    if (window.lobsPermitidasGlobal && window.lobsPermitidasGlobal.length > 0) {
        lobs = window.lobsPermitidasGlobal.sort();
        console.log(`📋 Cargando ${lobs.length} LOBs permitidas en filtro de disputas:`, lobs);
    } else {
        lobs = [...new Set(datosDisputas.map(d => d.lob))].sort();
        console.log(`⚠️ Fallback: Usando LOBs de datosDisputas:`, lobs);
    }
    
    const select = document.getElementById('filtroLobDisputa');
    
    if (select) {
        select.innerHTML = '<option value="">Todas las LOBs</option>';
        lobs.forEach(lob => {
            const opt = document.createElement('option');
            opt.value = lob;
            opt.textContent = lob;
            select.appendChild(opt);
        });
    }
}

function actualizarInfoPaginacionDisputas(totalRegistros) {

function cambiarPaginaDisputas(direccion) {
    const totalPaginas = Math.ceil(disputasFiltradas.length / registrosPorPagina);
    const nuevaPagina = paginaActualDisputas + direccion;
    
    if (nuevaPagina >= 1 && nuevaPagina <= totalPaginas) {
        paginaActualDisputas = nuevaPagina;
        renderizarTablaDisputas(disputasFiltradas);
        
        const tablaContainer = document.querySelector('.table-container');
        if (tablaContainer) {
            tablaContainer.scrollIntoView({ behavior: 'smooth', block: 'start' });
        }
    }
}

function exportarDisputas() {
    if (disputasFiltradas.length === 0) {
        mostrarToast('⚠️ No hay disputas para exportar', 'warning');
        return;
    }
    
    try {
        const datosExcel = disputasFiltradas.map(d => ({
            'Timestamp': formatearFecha(d.timestamp),
            'BMS ID': d.bmsId || '',
            'Nombre': d.nombreAgente || '',
            'LOB': d.lob || '',
            'Mes CDM': d.mesCDM || '',
            'Monto': d.monto || 0,
            'Fecha KPI': d.fechaKPI || '',
            'Comentario': d.detalles || '',
            'Estado': d.estado || '',
            'Usuario': d.usuario || '',
            'Correo': d.correo || ''
        }));
        
        const wb = XLSX.utils.book_new();
        const ws = XLSX.utils.json_to_sheet(datosExcel);
        
        ws['!cols'] = [
            {wch: 20}, {wch: 10}, {wch: 30}, {wch: 15},
            {wch: 12}, {wch: 15}, {wch: 12}, {wch: 50},
            {wch: 20}, {wch: 25}, {wch: 30}
        ];
        
        XLSX.utils.book_append_sheet(wb, ws, 'Disputas');
        
        const nombreArchivo = `Disputas_${new Date().toISOString().slice(0, 10)}.xlsx`;
        XLSX.writeFile(wb, nombreArchivo);
        
        mostrarToast(`✅ Archivo exportado: ${nombreArchivo}`, 'success');
        
    } catch (error) {
        console.error('Error al exportar:', error);
        mostrarError('Error al exportar disputas: ' + error.message);
    }
}

function actualizarCamposDecision(decision) {
    const divMonto = document.getElementById('divMontoAjustado');
    const inputMonto = document.querySelector('[name="montoAjustado"]');
    
    if (decision === 'APROBADA') {
        divMonto.style.display = 'block';
        inputMonto.required = true;
    } else {
        divMonto.style.display = 'none';
        inputMonto.required = false;

}

function enviarRespuestaDisputa(event, disputa) {
    event.preventDefault();
    
    const form = event.target;
    const formData = new FormData(form);
    
    const decision = formData.get('decision');
    const respuesta = formData.get('respuesta');
    const montoAjustado = formData.get('montoAjustado');
    const comentarioInterno = formData.get('comentarioInterno');
    
    if (!decision || !respuesta) {
        mostrarToast('⚠️ Completa los campos obligatorios', 'warning');
        return;
    }
    
    if (decision === 'APROBADA' && !montoAjustado) {
        mostrarToast('⚠️ Debes ingresar el monto ajustado', 'warning');
        return;
    }
    
    const mensajesConfirmacion = {
        'APROBADA': '¿Confirmas APROBAR esta disputa? Se ajustará el bono en Audit.',
        'RECHAZADA': '¿Confirmas RECHAZAR esta disputa?',
        'MAS_INFO': '¿Confirmas solicitar más información al agente?',
        'EN_PROCESO': '¿Confirmas marcar esta disputa como EN PROCESO?'
    };
    
    if (!confirm(mensajesConfirmacion[decision] || '¿Confirmas esta acción?')) {
        return;
    }
    
    const datosRespuesta = {
        fila: disputa.fila,
        bmsId: disputa.bmsId,
        nombre: disputa.nombreAgente,
        lob: disputa.lob,
        fechaDisputa: disputa.timestamp,
        estadoOriginal: disputa.estado,
        decision: decision,
        respuesta: respuesta,
        montoAjustado: montoAjustado ? parseFloat(montoAjustado) : null,
        comentarioInterno: comentarioInterno || ''
    };
    
    const btnSubmit = form.querySelector('button[type="submit"]');
    btnSubmit.disabled = true;
    btnSubmit.innerHTML = '⏳ Procesando...';
    
    mostrarToast('⏳ Procesando respuesta...', 'info');
    
    google.script.run
        .withSuccessHandler(function(response) {
            console.log('✅ Respuesta procesada:', response);
            
            if (response.success) {
                mostrarToast(`✅ ${response.mensaje}`, 'success');
                cerrarModalRespuesta();
                cargarDatosIniciales();
            } else {
                mostrarToast(`❌ Error: ${response.error}`, 'error');
                btnSubmit.disabled = false;
                btnSubmit.innerHTML = '✅ Enviar Respuesta';
            }
        })
        .withFailureHandler(function(error) {
            console.error('❌ Error:', error);
            mostrarToast('❌ Error al procesar la respuesta', 'error');
            btnSubmit.disabled = false;
            btnSubmit.innerHTML = '✅ Enviar Respuesta';
        })
        .responderDisputa(datosRespuesta);
}

function cargarTipoAccesoActual() {
    google.script.run
        .withSuccessHandler(function(tipoAcceso) {
            const select = document.getElementById('selectTipoAcceso');
            if (select && tipoAcceso) {
                select.value = tipoAcceso;
                console.log('✅ Tipo de acceso actual:', tipoAcceso);
            }
        })
        .withFailureHandler(function(error) {
            console.log('❌ Error al obtener tipo de acceso:', error);
        })
        .obtenerTipoAccesoActual();
}

function cambiarTipoAccesoYRedirigir() {
    const select = document.getElementById('selectTipoAcceso');
    const nuevoTipoAcceso = select.value;
    
    if (!nuevoTipoAcceso) {
        return;
    }
    
    if (!confirm(`¿Cambiar vista a ${nuevoTipoAcceso}?`)) {
        cargarTipoAccesoActual();
        return;
    }
    
    mostrarToast('⏳ Cambiando vista...', 'info');
    
    const emailUsuario = document.getElementById('correoUsuario').textContent;
    
    google.script.run
        .withSuccessHandler(function(response) {
            if (response.success) {
                mostrarToast(`✅ ${response.mensaje}`, 'success');
                
                setTimeout(() => {
                    window.location.href = `${window.location.origin}${window.location.pathname}?page=${nuevoTipoAcceso}`;
                }, 1000);
            } else {
                mostrarToast(`❌ ${response.error}`, 'error');
                cargarTipoAccesoActual();
            }
        })
        .withFailureHandler(function(error) {
            mostrarToast('❌ Error al cambiar vista', 'error');
            console.log('Error:', error);
            cargarTipoAccesoActual();
        })
        .cambiarTipoAcceso(emailUsuario, nuevoTipoAcceso);
}
