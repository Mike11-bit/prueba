        const ClientCache = {
            storage: window.sessionStorage,

            get: function (key) {
                const item = this.storage.getItem(key);
                if (!item) return null;
                try {
                    const record = JSON.parse(item);
                    if (new Date().getTime() > record.expiration) {
                        this.storage.removeItem(key);
                        return null;
                    }
                    console.log(`⚡ Cache Hit: ${key}`);
                    return record.value;
                } catch (e) { return null; }
            },

            set: function (key, value, minutes) {
                const record = {
                    value: value,
                    expiration: new Date().getTime() + (minutes * 60 * 1000)
                };
                try {
                    this.storage.setItem(key, JSON.stringify(record));
                } catch (e) { console.error('Cache Full'); }
            },

            clear: function (prefix) {
                Object.keys(this.storage).forEach(k => {
                    if (k.startsWith(prefix)) this.storage.removeItem(k);
                });
            }
        };

function validarPorcentaje(input) {
    input.addEventListener('input', function () {
        const val = this.value;
        // Permitir número con punto decimal en progreso (ej: "50.")
        // Solo limpiar caracteres no numéricos excepto el punto
        let limpio = val.replace(/[^0-9.]/g, '');

        // Evitar múltiples puntos
        const partes = limpio.split('.');
        if (partes.length > 2) {
            limpio = partes[0] + '.' + partes.slice(1).join('');
        }

        // Máximo 2 decimales — solo cortar si hay más de 2 después del punto
        if (partes[1] !== undefined && partes[1].length > 2) {
            limpio = partes[0] + '.' + partes[1].substring(0, 2);
        }

        // ⭐ Solo reasignar si cambió algo para no interrumpir el cursor
        if (this.value !== limpio) {
            const cursor = this.selectionStart;
            this.value = limpio;
            // Restaurar posición del cursor
            try { this.setSelectionRange(cursor, cursor); } catch(e) {}
        }
    });

    input.addEventListener('blur', function () {
        const val = parseFloat(this.value);
        if (!isNaN(val)) {
            // Formatear al salir: quitar cero innecesario al final
            this.value = Number.isInteger(val) ? val.toString() : parseFloat(val.toFixed(2)).toString();
        }
    });
}

// ⭐ NUEVA: Poblar el formulario de registro con una estructura existente
function cargarEstructuraEnFormulario(estructura) {
    if (!estructura || !estructura.encontrado) return;
    
    console.log('📥 Poblando formulario con estructura existente para:', estructura.lob, estructura.mes);
    
    // 1. Configuración Salarial
    const _fmt = (val) => String(val || '').replace('.', ',');
    document.getElementById('salario').value = _fmt(estructura.salario);
    document.getElementById('porcentajeBono').value = _fmt(estructura.porcentaje);
    document.getElementById('pvePresupuesto').value = _fmt(estructura.pve);
    
    // 2. Tipo de Hoja
    if (estructura.tipoHoja) {
        document.getElementById('tipoHoja').value = estructura.tipoHoja;
        actualizarLimites();
    }
    
    // 3. Limpiar tablas
    ['tbodyKeys', 'tbodyBonus', 'tbodyAdditional', 'tbodyDecreased'].forEach(id => {
        const tb = document.getElementById(id);
        if (tb) tb.innerHTML = '';
    });
    
    // 4. Agregar filas por categoría
    if (estructura.kpis) {
        if (estructura.kpis.keys) {
            estructura.kpis.keys.forEach(k => {
                const row = agregarFilaKPI_Datos('keys', k);
            });
        }
        if (estructura.kpis.bonusStructure) {
            estructura.kpis.bonusStructure.forEach(k => {
                const row = agregarFilaKPI_Datos('bonus', k);
            });
        }
        if (estructura.kpis.additionalIncentives) {
            estructura.kpis.additionalIncentives.forEach(k => {
                const row = agregarFilaKPI_Datos('additional', k);
            });
        }
        if (estructura.kpis.decreasedKPIs) {
            estructura.kpis.decreasedKPIs.forEach(k => {
                const row = agregarFilaKPI_Datos('decreased', k);
            });
        }
    }
    
    calcularBonoMaximo();
    actualizarIndicadorBono();
}

// ⭐ Función auxiliar para agregar fila con datos pre-cargados
function agregarFilaKPI_Datos(tipo, kpiData) {
    const tbody = document.getElementById('tbody' + capitalize(tipo));
    if (!tbody) return;
    
    const tipoHoja = document.getElementById('tipoHoja').value;
    const esPisos = (tipoHoja === 'Registro_pisos');
    const esBonus = (tipo === 'bonus');
    
    const row = tbody.insertRow();
    
    // Preparar opciones de KPI
    let kpis = kpisDisponibles ? kpisDisponibles[tipo === 'bonus' ? 'bonusStructure' :
        tipo === 'additional' ? 'additionalIncentives' :
            tipo === 'decreased' ? 'decreasedKPIs' : tipo] : [];

    let optsHTML = '<option value="">-- Seleccionar --</option>';
    if (kpis && Array.isArray(kpis)) {
        kpis.forEach(k => {
            const selected = (normalizarKey(k) === normalizarKey(kpiData.nombre)) ? 'selected' : '';
            optsHTML += `<option value="${k}" ${selected}>${k}</option>`;
        });
    }

    // Procesar Score y Gate
    let scoreLimpio = kpiData.score || '';
    let comparacion = '>=';
    let isKeyChecked = false;

    // Detectar comparación
    const comps = ['<=', '>=', '<', '>', '='];
    for (let c of comps) {
        if (scoreLimpio.startsWith(c)) {
            comparacion = c;
            scoreLimpio = scoreLimpio.substring(c.length);
            break;
        }
    }

    // Detectar Gate flag (",key")
    if (scoreLimpio.toLowerCase().includes(',key')) {
        isKeyChecked = true;
        scoreLimpio = scoreLimpio.replace(/,key/gi, '');
    }

    // REEMPLAZO DE COMAS Y PORCENTAJES (Para inputs de tipo number)
    scoreLimpio = scoreLimpio.toString().replace('%', '').replace(',', '.');
    const valorPct = (kpiData.porcentaje || '').toString().replace('%', '').replace(',', '.');
    const valorPart = (kpiData.participacion || '').toString().replace('%', '').replace(',', '.');

    const gateCell = esBonus
        ? `<td style="text-align:center;">
             <input type="checkbox" class="kpi-is-key" ${isKeyChecked ? 'checked' : ''}
                    style="width:20px; height:20px; cursor:pointer;"
                    onchange="validarMaximoGates(this)">
           </td>`
        : `<td style="display:none;"><input type="checkbox" class="kpi-is-key"></td>`;

    if (tipo === 'keys') {
        row.innerHTML = `
        <td><select class="kpi-nombre">${optsHTML}</select></td>
        <td>
            <select class="kpi-comparacion">
                <option value="<" ${comparacion === '<' ? 'selected' : ''}>&lt;</option>
                <option value="<=" ${comparacion === '<=' ? 'selected' : ''}>&lt;=</option>
                <option value=">" ${comparacion === '>' ? 'selected' : ''}>&gt;</option>
                <option value=">=" ${comparacion === '>=' ? 'selected' : ''}>&gt;=</option>
                <option value="=" ${comparacion === '=' ? 'selected' : ''}>=</option>
            </select>
        </td>
        <td><input type="number" step="0.01" class="kpi-score form-control" placeholder="0.00" value="${scoreLimpio.replace(',', '.')}"></td>
        <td><button type="button" class="btn-delete" onclick="eliminarFila(this)">🗑️</button></td>
        `;
    } else {
        row.innerHTML = `
        <td><select class="kpi-nombre">${optsHTML}</select></td>
        <td>
            <select class="kpi-comparacion">
                <option value="<" ${comparacion === '<' ? 'selected' : ''}>&lt;</option>
                <option value="<=" ${comparacion === '<=' ? 'selected' : ''}>&lt;=</option>
                <option value=">" ${comparacion === '>' ? 'selected' : ''}>&gt;</option>
                <option value=">=" ${comparacion === '>=' ? 'selected' : ''}>&gt;=</option>
                <option value="=" ${comparacion === '=' ? 'selected' : ''}>=</option>
            </select>
        </td>
        <td><input type="number" step="0.01" class="kpi-score form-control" placeholder="0.00" value="${scoreLimpio.replace(',', '.')}"></td>
        <td><input type="number" step="0.01" class="kpi-porcentaje form-control" placeholder="0.00" value="${valorPct.replace(',', '.')}" oninput="actualizarIndicadorBono()"></td>
        ${gateCell}
        <td>
            <input type="number" step="0.01" class="kpi-participacion form-control" value="${valorPart.replace(',', '.')}"
                   placeholder="100" oninput="calcularValoresPesosEnTiempoReal()">
        </td>
        <td class="celda-valor-pesos" style="text-align:right; font-weight:bold; color:#27ae60;">—</td>
        <td><button type="button" class="btn-delete" onclick="eliminarFila(this)">🗑️</button></td>
        `;
    }
    
    // ⭐ Refrescar visibilidad de la nueva fila
    actualizarLimites();
    return row;
}

// ============================================================
// 📥 EXPORTAR ESTRUCTURA A PDF
// ============================================================
function exportarEstructuraPDF(estructura, lob, mes) {
    if (!estructura || !window.jspdf) {
        mostrarToast('❌ Error: Biblioteca PDF no cargada', 'danger');
        return;
    }

    const { jsPDF } = window.jspdf;
    const doc = new jsPDF();
    const margin = 15;
    let y = 20;

    const bonoMaximo = (estructura.salario * estructura.porcentaje) / 100;

    // Título Principal
    doc.setFontSize(22);
    doc.setTextColor(52, 152, 219); // var(--primary)
    doc.text('Estructura de Métricas', margin, y);
    y += 12;

    // Información de Cabecera - Fila 1
    doc.setFontSize(10);
    doc.setTextColor(44, 62, 80);
    doc.setFont("helvetica", "bold");
    doc.text(`LOB:`, margin, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${lob}`, margin + 10, y);
    
    doc.setFont("helvetica", "bold");
    doc.text(`Mes:`, margin + 70, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${mes}`, margin + 80, y);

    doc.setFont("helvetica", "bold");
    doc.text(`Tipo:`, margin + 130, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${estructura.tipoHoja || '-'}`, margin + 140, y);
    y += 7;

    // Información de Cabecera - Fila 2
    doc.setFont("helvetica", "bold");
    doc.text(`ACCM:`, margin, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${estructura.accm || '-'}`, margin + 13, y);

    doc.setFont("helvetica", "bold");
    doc.text(`OM:`, margin + 70, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${estructura.om || '-'}`, margin + 78, y);
    
    doc.setFont("helvetica", "bold");
    doc.text(`Fecha Reg:`, margin + 130, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${estructura.fecha || estructura.mes || '-'}`, margin + 150, y);
    y += 10;

    // Cuadro de Resumen Salarial
    doc.setFillColor(248, 249, 250);
    doc.rect(margin, y, 180, 22, 'F');
    doc.setDrawColor(222, 226, 230);
    doc.rect(margin, y, 180, 22, 'S');
    
    y += 13;
    doc.setFontSize(11);
    doc.setFont("helvetica", "bold");
    doc.text(`HC:`, margin + 10, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${estructura.hc || '-'}`, margin + 18, y);

    doc.setFont("helvetica", "bold");
    doc.text(`Salario:`, margin + 40, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${formatearMoneda(estructura.salario || 0)}`, margin + 55, y);

    doc.setFont("helvetica", "bold");
    doc.text(`% Bono:`, margin + 95, y);
    doc.setFont("helvetica", "normal");
    doc.text(`${estructura.porcentaje || 0}%`, margin + 112, y);

    doc.setTextColor(39, 174, 96); // success green
    doc.setFont("helvetica", "bold");
    doc.text(`Bono Máx:`, margin + 135, y);
    doc.text(`${formatearMoneda(bonoMaximo)}`, margin + 158, y);
    y += 18;

    // Sección de Comentarios
    if (estructura.comentarios) {
        doc.setTextColor(133, 100, 4); // warning brown
        doc.setFillColor(255, 243, 205);
        doc.setFontSize(10);
        
        const splitComments = doc.splitTextToSize(`Comentarios: ${estructura.comentarios}`, 170);
        const commentHeight = (splitComments.length * 5) + 6;
        
        doc.rect(margin, y - 5, 180, commentHeight, 'F');
        doc.text(splitComments, margin + 5, y);
        y += commentHeight + 5;
    }

    // Función para añadir secciones con tablas
    const añadirSeccion = (titulo, kpis, color) => {
        if (!kpis || kpis.length === 0) return;
        
        // Verificar si necesitamos nueva página
        if (y > 230) { doc.addPage(); y = 20; }

        doc.setFontSize(14);
        doc.setTextColor(color[0], color[1], color[2]);
        doc.setFont("helvetica", "bold");
        doc.text(titulo, margin, y);
        y += 6;

        const body = kpis.map(k => {
            const pctAsignado = parseFloat(k.porcentaje) || 0;
            const pctPart = parseFloat(k.participacion) || 100;
            const bonoKPI = (bonoMaximo * pctAsignado * pctPart) / 10000;

            return [
                k.nombre,
                k.score || '-',
                pctAsignado > 0 ? pctAsignado.toFixed(1) + '%' : '-',
                pctPart.toFixed(0) + '%',
                bonoKPI > 0 ? formatearMoneda(bonoKPI) : '-'
            ];
        });

        doc.autoTable({
            startY: y,
            head: [['KPI', 'Meta/Score', '% Asignado', '% Part.', 'Monto ($)']],
            body: body,
            headStyles: { fillColor: color },
            margin: { left: margin },
            theme: 'striped',
            styles: { fontSize: 8, cellPadding: 2 },
            columnStyles: {
                4: { halign: 'right', fontStyle: 'bold' }
            }
        });

        y = doc.lastAutoTable.finalY + 12;
    };

    añadirSeccion('Keys (Mandatorios)', estructura.kpis.keys, [52, 152, 219]); // #3498db
    añadirSeccion('Bonus Structure', estructura.kpis.bonusStructure, [46, 204, 113]); // #2ecc71
    añadirSeccion('Additional Incentives', estructura.kpis.additionalIncentives, [243, 156, 18]); // #f39c12
    añadirSeccion('Decreased', estructura.kpis.decreasedKPIs, [231, 76, 60]); // #e74c3c

    // Pie de página
    const totalPages = doc.internal.getNumberOfPages();
    for(let i = 1; i <= totalPages; i++) {
        doc.setPage(i);
        doc.setFontSize(8);
        doc.setTextColor(150);
        doc.text(`Documento generado el ${new Date().toLocaleString()} - Sistema de Gestión de Métricas`, margin, 285);
        doc.text(`Página ${i} de ${totalPages}`, 180, 285);
    }

function manejarSubmitRegistro(e) {

    }

        mes:         mes,

    // ========================================
    // 3️⃣ DESHABILITAR BOTÓN Y MOSTRAR LOADING
    // ========================================
    const btn = document.getElementById('btnGuardar');
    if (btn) {
        btn.disabled = true;
        btn.textContent = '⏳ Guardando...';
    }
    mostrarLoading('Guardando registro...');

    // ========================================
    // 4️⃣ LLAMAR AL BACKEND
    // ========================================
    google.script.run
        .withSuccessHandler(function(response) {
            ocultarLoading();
            if (btn) {
                btn.disabled = false;
                btn.textContent = '💾 GUARDAR REGISTRO';

            ocultarLoading();
            if (btn) {
                btn.disabled = false;
                btn.textContent = '💾 GUARDAR REGISTRO';
            }
            console.error('❌ Error al guardar:', err);
            mostrarToast('❌ Error de conexión: ' + (err.message || err), 'danger');
        })
        .REGISTRO_guardarRegistroMetricas(datos);
}

function extraerKPIs(tipo) {
    const kpis = [];
    try {
        const capitalize = (s) => s.charAt(0).toUpperCase() + s.slice(1);
        const tbodyId = `tbody${capitalize(tipo)}`;
        const rows = document.querySelectorAll(`#${tbodyId} tr`);
        
        rows.forEach(row => {
            const nombre = row.querySelector('.kpi-nombre')?.value;
            if (!nombre) return;

            const _getVal = (selector) => {
                const inner = row.querySelector(selector)?.value || '';
                // Normalizar: cambiar coma por punto para el parseFloat del backend
                return inner.replace(',', '.');
            };

            const obj = {
                nombre: nombre,
                comparacion: row.querySelector('.kpi-comparacion')?.value || '>=',
                score: row.querySelector('.kpi-score')?.value || '',
                porcentaje: _getVal('.kpi-porcentaje') || '0'
            };

            // Participación (si existe)
            const partInput = row.querySelector('.kpi-participacion');
            if (partInput) {
                obj.participacion = _getVal('.kpi-participacion') || '100';
            }
            
            // Gate (solo para bonus)
            if (tipo === 'bonus') {
                obj.gate = row.querySelector('.kpi-gate')?.checked || false;
            }
            
            kpis.push(obj);
        });
    } catch (e) {
        console.error(`❌ Error en extraerKPIs(${tipo}):`, e);
    }
    return kpis;
}

function capitalize(s) {
    if (typeof s !== 'string') return '';
    return s.charAt(0).toUpperCase() + s.slice(1);
}

function calcularValoresPesosEnTiempoReal() {
    const salario = parseFloat(document.getElementById('salario').value.replace(',', '.')) || 0;
    const porcentajeBono = parseFloat(document.getElementById('porcentajeBono').value) || 0;
    const bonoMaximo = (salario * porcentajeBono) / 100;
    const tipoHoja = document.getElementById('tipoHoja').value;
    const esPisos = (tipoHoja === 'Registro_pisos');

    if (bonoMaximo <= 0) {
        document.querySelectorAll('.celda-valor-pesos').forEach(c => {
            c.textContent = '—';
            c.style.color = '#27ae60';
        });
        return;
    }

    ['Bonus', 'Additional', 'Decreased'].forEach(cat => {
        const kpisProcesados = {};

        document.querySelectorAll(`#tbody${cat} tr`).forEach(row => {
            const nombreInput = row.querySelector('.kpi-nombre');
            const pctInput    = row.querySelector('.kpi-porcentaje');
            const cellMonto   = row.querySelector('.celda-valor-pesos');

            if (!cellMonto) return;

            const nombre = nombreInput?.value?.trim().toUpperCase();
            const pctRaw = pctInput?.value || '';

            if (!nombre || !pctRaw) {
                cellMonto.textContent = '—';
                return;
            }

            let pct = parseFloat(pctRaw.replace(',', '.')) || 0;
            if (pct > 0 && pct < 1) pct = pct * 100;

            // Registro normal = participación 100%; Registro_pisos = leer input
            let part = 100;
            if (esPisos) {
                const partInput = row.querySelector('.kpi-participacion');
                const partRaw   = partInput?.value || '100';
                part = parseFloat(partRaw.replace(',', '.')) || 100;
                if (part > 0 && part < 1) part = part * 100;
            }

            const montoBase  = (bonoMaximo * pct) / 100;
            const montoFinal = (montoBase * part) / 100;
            const esRepetido = !!kpisProcesados[nombre];

            if (montoFinal > 0) {
                const prefix = cat === 'Decreased' ? '-' : '';
                cellMonto.textContent = prefix + formatearMoneda(montoFinal);
                cellMonto.style.color = esRepetido
                    ? '#95a5a6'
                    : (cat === 'Decreased' ? '#e74c3c' : '#27ae60');
            } else {
                cellMonto.textContent = '—';
                cellMonto.style.color = '#27ae60';
            }

            kpisProcesados[nombre] = true;
        });
    });
}

function calcularBonoMaximo() {
    const salario = parseFloat(document.getElementById('salario').value) || 0;
    const porcentaje = parseFloat(document.getElementById('porcentajeBono').value) || 0;
    const bonoMaximo = (salario * porcentaje) / 100;
    
    const elTexto = document.getElementById('bonoMaximoTexto');
    const elIndicator = document.getElementById('bonoIndicator');
    
    if (elTexto) elTexto.textContent = formatearMoneda(bonoMaximo);

function actualizarIndicadorBono() {
    let totalPct = 0;
    const _p = (val) => parseFloat(String(val).replace(',', '.')) || 0;
    
    document.querySelectorAll('#tbodyBonus .kpi-porcentaje').forEach(input => {
        totalPct += _p(input.value);
    });

    const progressBar = document.getElementById('progressBar');
    const pctAsignadoTexto = document.getElementById('porcentajeAsignadoTexto');
    const pctDisponibleTexto = document.getElementById('porcentajeDisponibleTexto');
    const montoAsignadoTexto = document.getElementById('montoAsignadoTexto');
    
    const salario = _p(document.getElementById('salario').value);
    const porcentajeBono = _p(document.getElementById('porcentajeBono').value);
    const bonoMaximo = (salario * porcentajeBono) / 100;
    const montoAsignado = (bonoMaximo * totalPct) / 100;

    if (progressBar) {
        progressBar.style.width = Math.min(totalPct, 100) + '%';
        progressBar.textContent = totalPct.toFixed(1) + '%';
        if (totalPct > 100) {
            progressBar.className = 'progress-bar danger';
        } else if (totalPct > 80) {
            progressBar.className = 'progress-bar warning';
        } else {
            progressBar.className = 'progress-bar';
        }
    }

    if (pctAsignadoTexto) pctAsignadoTexto.textContent = totalPct.toFixed(1) + '%';
    if (pctDisponibleTexto) pctDisponibleTexto.textContent = (100 - totalPct).toFixed(1) + '%';
    if (montoAsignadoTexto) montoAsignadoTexto.textContent = formatearMoneda(montoAsignado);
}

function actualizarLimites() {
    const tipoHoja = document.getElementById('tipoHoja').value;
    const esPisos = (tipoHoja === 'Registro_pisos');
    
    // Identificadores de encabezados
    const yellowHeaders = ['headerPartYellowBonus', 'headerPartYellowAdditional', 'headerPartYellowDecreased'];
    const greenHeaders = ['headerPartGreenBonus', 'headerPartGreenAdditional', 'headerPartGreenDecreased'];

    if (esPisos) {
        // En Registro_pisos mostramos AMBAS columnas: % Part y Valor Part
        yellowHeaders.forEach(id => { const el = document.getElementById(id); if (el) el.style.display = 'table-cell'; });
        greenHeaders.forEach(id => { const el = document.getElementById(id); if (el) el.style.display = 'table-cell'; });
        
        document.querySelectorAll('.kpi-participacion').forEach(el => el.closest('td').style.display = 'table-cell');
        document.querySelectorAll('.celda-valor-pesos').forEach(el => el.style.display = 'table-cell');
    } else {
        // En Registro OCULTAMOS % Part (Yellow) y MOSTRAMOS Valor Part (Green)
        yellowHeaders.forEach(id => { const el = document.getElementById(id); if (el) el.style.display = 'none'; });
        greenHeaders.forEach(id => { const el = document.getElementById(id); if (el) el.style.display = 'table-cell'; });
        
        document.querySelectorAll('.kpi-participacion').forEach(el => {
            el.closest('td').style.display = 'none';
            // Si el valor está vacío, ponemos 100 por defecto para que el cálculo de Valor Part funcione
            if (!el.value) el.value = '100'; 
        });
        document.querySelectorAll('.celda-valor-pesos').forEach(el => el.style.display = 'table-cell');
    }

    calcularValoresPesosEnTiempoReal();
}

// 📥 EXPORTAR REPORTE UNIFICADO
// ============================================================

function exportarReporteUnificadoExcel() {
    if (!window.reporteUnificadoActual) {
        mostrarToast('⚠️ No hay datos para exportar', 'warning');
        return;
    }
    
    mostrarLoading('Generando Excel unificado...');
    
    const data = window.reporteUnificadoActual;
    
    const datosExportar = {
        encabezados: [
            'LOB', 'HC', 'Cobran', 'No Cobran', 
            'Inv. Teórica', 'Inv. Real', 'Ahorro', 
            'Efic. %', 'Tasa Cumpli. %', '% PVE'
        ],
        filas: data.resultados.map(function(r) {
            const agentesNoCobran = r.hc - (r.estadisticas?.agentesCobran || 0);
            const pctPVE = data.presupuestoPVE > 0 ? ((r.totalLOB / data.presupuestoPVE) * 100) : 0;
            
            return [
                r.lob,
                r.hc,
                r.estadisticas?.agentesCobran || 0,
                agentesNoCobran,
                r.inversionTeorica,
                r.totalLOB,
                r.ahorro,
                parseFloat(r.eficiencia),
                parseFloat(r.tasaCumplimiento),
                pctPVE.toFixed(2)
            ];
        }),
        resumen: [
            ['TOTAL LOBs', data.resultados.length],
            ['TOTAL AGENTES', data.resumen.totalAgentes],
            ['INVERSIÓN TEÓRICA', data.resumen.inversionTeorica],
            ['INVERSIÓN REAL (CDM)', data.resumen.inversionTotal],
            ['AHORRO TOTAL', data.resumen.ahorro],
            ['EFICIENCIA GLOBAL', data.resumen.eficienciaGlobal.toFixed(1) + '%'],
            ['PRESUPUESTO PVE', data.presupuestoPVE || 0],
            ['% USO PVE', data.presupuestoPVE > 0 ? ((data.resumen.inversionTotal / data.presupuestoPVE) * 100).toFixed(1) + '%' : 'N/A']
        ]
    };
    
    google.script.run
        .withSuccessHandler(function(url) {
            ocultarLoading();
            if (url) {
                window.open(url, '_blank');
                mostrarToast('✅ Excel exportado exitosamente', 'success');
            } else {
                mostrarToast('✅ Exportación completada', 'success');
            }
        })
        .withFailureHandler(function(err) {
            ocultarLoading();
            mostrarToast('❌ Error: ' + err, 'danger');
        })
        .REGISTRO_exportarSimulacionAExcel(datosExportar);
}

        function filtrarLOBsPorRegion() {
            const region = document.getElementById('simSelectRegion').value;

            if (!region) {
                cargarLobsRegistradasEnSimulacion();
                return;
            }

            mostrarLoading('Cargando LOBs de ' + region + '...');

            google.script.run
                .withSuccessHandler(function (lobs) {
                    ocultarLoading();
                    const select = document.getElementById('simSelectLob');
                    select.innerHTML = '<option value="">-- Seleccione LOB --</option>';

                    if (lobs && lobs.length > 0) {
                        lobs.forEach(function (lob) {
                            select.appendChild(new Option(lob, lob));
                        });
                        mostrarToast('✅ ' + lobs.length + ' LOBs en ' + region, 'success');

                        mostrarToast('⚠️ No hay LOBs con registros en ' + region, 'warning');
                    }
                })
                .withFailureHandler(function (err) {
                    ocultarLoading();
                    mostrarToast('❌ Error: ' + err, 'danger');
                })
                .REGISTRO_obtenerLobsRegistradasPorRegion(region);
        }

        let pct = parseFloat(pisos[0].porcentaje) || 0;

        pisos.forEach(piso => {
            let part = parseFloat(piso.participacion) || 100;
            if (part > 0 && part < 1) part = part * 100;

function generarTablaKPIsConPisos(kpisPorNombre, bonoMaximo, hcCount, tipo, esPisos) {
    if (Object.keys(kpisPorNombre).length === 0) {
        return '<p style="color: #999; font-style: italic;">Sin KPIs configurados</p>';
    }
    
    const colorBorde = tipo === 'bonus' ? '#27ae60' : tipo === 'additional' ? '#f39c12' : '#e74c3c';
    
    let html = `
    <table class="kpi-table" style="border-left: 4px solid ${colorBorde};">
        <thead>
            <tr>
                <th>KPI</th>
                ${esPisos ? '<th>Piso</th>' : ''}
                <th>Meta</th>
                <th>% Base</th>`;
    
    // ⭐ AGREGAR COLUMNAS SOLO SI ES PISOS
    if (esPisos) {
        html += `<th style="background: #fff3cd; color: #856404; font-weight: bold;">% Participación</th>`;
        html += `<th style="background: #e8f5e9; color: #27ae60; font-weight: bold;">💰 Valor/Agente</th>`;
    }
    
    html += `<th>Total x HC</th>
            </tr>
        </thead>
        <tbody>
    `;
    
    let totalCategoria = 0;
    
    Object.keys(kpisPorNombre).forEach(nombreKPI => {
        const pisos = kpisPorNombre[nombreKPI];
        const numPisos = pisos.length;
        
        // ⭐ USAR SOLO EL PORCENTAJE DEL PRIMER PISO
        let pct = parseFloat(pisos[0].porcentaje) || 0;
        if (pct > 0 && pct < 1) pct = pct * 100;
        
        const valorBase = (bonoMaximo * pct) / 100;
        
        // ⭐ ENCONTRAR LA PARTICIPACIÓN MÁS ALTA
        let participacionMaxima = 100;
        if (esPisos) {
            participacionMaxima = 0;
            pisos.forEach(piso => {
                let part = parseFloat(piso.participacion) || 100;
                if (part > 0 && part < 1) part = part * 100;
                if (part > participacionMaxima) participacionMaxima = part;
            });
        }
        
        // ⭐ CALCULAR CON LA PARTICIPACIÓN MÁS ALTA
        const valorConPartMax = esPisos ? (valorBase * participacionMaxima) / 100 : valorBase;
        const totalKPI = valorConPartMax * hcCount;
        totalCategoria += totalKPI;
        
        console.log(`📊 ${nombreKPI}: ${pct}% base × ${participacionMaxima}% part = $${valorConPartMax.toFixed(2)}/agente`);
        
        // ⭐ MOSTRAR CADA PISO
        if (esPisos && numPisos > 1) {
            // Mostrar pisos separados
            pisos.forEach((piso, idx) => {
                let participacion = parseFloat(piso.participacion) || 100;
                if (participacion > 0 && participacion < 1) participacion = participacion * 100;
                
                const esMaximo = (participacion === participacionMaxima);
                
                html += `
                <tr ${esMaximo ? 'style="background: #e8f5e9; font-weight: 600;"' : ''}>
                    ${idx === 0 ? `<td rowspan="${numPisos}" style="font-weight: 600; vertical-align: middle;">${nombreKPI}</td>` : ''}
                    <td style="text-align: center; background: #f8f9fa;">${idx + 1} ${esMaximo ? '🏆' : ''}</td>
                    <td style="text-align: center;">${piso.comparacion || '>='} ${piso.score || '-'}</td>
                    ${idx === 0 ? `<td rowspan="${numPisos}" style="text-align: center; font-weight: 600; color: ${colorBorde}; vertical-align: middle;">${pct.toFixed(1)}%</td>` : ''}
                    <td style="text-align: center; background: #fff3cd; font-weight: ${esMaximo ? 'bold' : 'normal'}; color: ${esMaximo ? '#27ae60' : 'inherit'};">
                        ${participacion.toFixed(0)}% ${esMaximo ? '👑' : ''}
                    </td>
                    ${idx === 0 ? `<td rowspan="${numPisos}" style="text-align: right; vertical-align: middle; font-weight: bold; color: #27ae60; background: #e8f5e9;">
                        ${formatearMoneda(valorConPartMax)}
                        <div style="font-size: 0.75em; color: #666; margin-top: 3px;">(Usa: ${participacionMaxima}%)</div>
                    </td>` : ''}
                    ${idx === 0 ? `<td rowspan="${numPisos}" style="text-align: right; vertical-align: middle; font-weight: bold; color: #27ae60;">
                        ${formatearMoneda(totalKPI)}
                    </td>` : ''}
                </tr>
                `;
            });
        } else {
            // ⭐ UNA SOLA FILA (SIN PISOS O REGISTRO ESTÁNDAR)
            html += `
            <tr>
                <td style="font-weight: 600;">${nombreKPI}</td>
                <td style="text-align: center;">${pisos[0].comparacion || '>='} ${pisos[0].score || '-'}</td>
                <td style="text-align: center; font-weight: 600; color: ${colorBorde};">${pct.toFixed(1)}%</td>`;
            
            if (esPisos) {
                html += `
                <td style="text-align: center; background: #fff3cd; font-weight: 600;">${participacionMaxima.toFixed(0)}%</td>
                <td style="text-align: right; background: #e8f5e9; font-weight: bold; color: #27ae60;">${formatearMoneda(valorConPartMax)}</td>`;

            `;
        }
    });
    
    html += `
            <tr style="background: ${tipo === 'bonus' ? '#d5f4e6' : tipo === 'additional' ? '#fef5e7' : '#ffebee'}; font-weight: bold;">
                <td colspan="${esPisos ? '6' : '4'}" style="text-align: right;">SUBTOTAL ${tipo.toUpperCase()}:</td>
                <td style="text-align: right;">${formatearMoneda(totalCategoria)}</td>
            </tr>
        </tbody>
    </table>
    `;
    
    return html;
}

        function agregarDestinatario() {
            const input = document.getElementById('inputNuevoCorreo');
            const correo = input.value.trim();

            // Validar formato de correo
            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (!emailRegex.test(correo)) {
                mostrarToast('⚠️ Ingrese un correo válido', 'warning');
                return;
            }

            // Verificar que no esté duplicado
            if (window.destinatariosCorreo.includes(correo)) {
                mostrarToast('⚠️ Este correo ya está en la lista', 'warning');
                return;
            }

            window.destinatariosCorreo.push(correo);
            input.value = '';
            actualizarListaDestinatarios();
            mostrarToast('✅ Destinatario agregado', 'success');
        }

        function eliminarDestinatario(correo) {
            window.destinatariosCorreo = window.destinatariosCorreo.filter(c => c !== correo);
            actualizarListaDestinatarios();
        }

        function actualizarListaDestinatarios() {
            const container = document.getElementById('listaDestinatarios');
            if (!container) return;

            if (window.destinatariosCorreo.length === 0) {
                container.innerHTML = '<span style="color: #999; font-style: italic;">No hay destinatarios agregados</span>';
                return;
            }

            let html = '';
            window.destinatariosCorreo.forEach(correo => {
                html += `
                <span style="display: inline-flex; align-items: center; background: #667eea; color: white; 
                            padding: 5px 10px; border-radius: 20px; margin: 3px; font-size: 0.9em;">

                            style="background: none; border: none; color: white; margin-left: 8px; cursor: pointer; font-size: 1.1em;">
                        ×
                    </button>
                </span>
                `;
            });
            container.innerHTML = html;
        }

                }

        mostrarToast('⚠️ Seleccione LOB y Mes (o deje LOB vacío para ver todas)', 'warning');
        return;
    }

    mostrarLoading('Consultando estructura de ' + lob + '...');

    google.script.run
        .withSuccessHandler(function (response) {
            ocultarLoading();
            // ⭐ FIX: proteger ante null
            if (!response) {
                mostrarToast('❌ No se recibió respuesta del servidor', 'danger');
                return;

            if (response.encontrado) {

        .withFailureHandler(function (err) {
            ocultarLoading();
            mostrarToast('❌ Error: ' + err, 'danger');
        })
        .REGISTRO_obtenerEstructuraRegistrada(lob, mes, datosUsuario);

function abrirModalSimulacion() {

        .withSuccessHandler(function (configOficial) {

                .withSuccessHandler(function (estructura) {
                    ocultarLoading();

                    if (!estructura || !estructura.encontrado) {
                        mostrarToast('❌ No hay estructura registrada para esta LOB/Mes', 'danger');

                    const inputPorcentaje = parseFloat(document.getElementById('simInputPorcentaje').value);

                        porcentaje = configOficial.porcentaje;

    function llenarKPIsEnFormulario(kpis) {
    console.log('📝 Llenando KPIs en formulario:', kpis);
    
    // Limpiar tablas existentes
    ['Keys', 'Bonus', 'Additional', 'Decreased'].forEach(tipo => {
        const tbody = document.getElementById('tbody' + tipo);
        if (tbody) tbody.innerHTML = '';
    });

    // Llenar Keys
    if (kpis?.keys) {
        kpis.keys.forEach(kpi => agregarFilaKPIConDatos('keys', kpi));
    }

    // Llenar Bonus
    if (kpis?.bonusStructure) {
        kpis.bonusStructure.forEach(kpi => {
            console.log('  Bonus KPI:', kpi);
            agregarFilaKPIConDatos('bonus', kpi);
        });
    }

    // Llenar Additional
    if (kpis?.additionalIncentives) {
        kpis.additionalIncentives.forEach(kpi => {
            console.log('  Additional KPI:', kpi);
            agregarFilaKPIConDatos('additional', kpi);
        });
    }

    // Llenar Decreased
    if (kpis?.decreasedKPIs) {
        kpis.decreasedKPIs.forEach(kpi => agregarFilaKPIConDatos('decreased', kpi));
    }
    
    // ⭐ ACTUALIZAR INDICADOR DESPUÉS DE CARGAR
    setTimeout(() => {
        actualizarIndicadorBono();
    }, 100);
}

function agregarFilaKPIConDatos(tipo, kpi) {
    const tbody = document.getElementById('tbody' + capitalize(tipo));
    if (!tbody) return;
    
    const row = tbody.insertRow();
    
    let comparacion = kpi.comparacion || '>=';
    let score = kpi.score || '';
    
    // ⭐ DETECTAR SUFIJO ",key" EN EL SCORE
    let esKeyManual = false;
    if (typeof score === 'string' && score.includes(',key')) {
        esKeyManual = true;
        score = score.replace(',key', '');
    }

    if (score && typeof score === 'string') {
        const match = score.match(/^([<>=]+)(.*)$/);
        if (match) {
            comparacion = match[1];
            score = match[2].trim();
        }
    }
    
    if (tipo === 'keys') {
        row.innerHTML = `
        <td><select class="kpi-nombre"><option value="${kpi.nombre}" selected>${kpi.nombre}</option></select></td>
        <td>
            <select class="kpi-comparacion">
                <option value="<" ${comparacion === '<' ? 'selected' : ''}>&lt;</option>
                <option value="<=" ${comparacion === '<=' ? 'selected' : ''}>&lt;=</option>
                <option value=">" ${comparacion === '>' ? 'selected' : ''}>&gt;</option>
                <option value=">=" ${comparacion === '>=' ? 'selected' : ''}>&gt;=</option>
                <option value="=" ${comparacion === '=' ? 'selected' : ''}>=</option>
            </select>
        </td>
        <td><input type="number" class="kpi-score" value="${score}" step="0.01"></td>
        <td><button type="button" class="btn-delete" onclick="eliminarFila(this)">🗑️</button></td>
        `;
    } else {
        const tipoHoja = document.getElementById('tipoHoja').value;
        const esPisos = (tipoHoja === 'Registro_pisos');
        const styleParticipacion = esPisos ? '' : 'display:none;';
        
        row.innerHTML = `
        <td><select class="kpi-nombre"><option value="${kpi.nombre}" selected>${kpi.nombre}</option></select></td>
        <td>
            <select class="kpi-comparacion">
                <option value="<" ${comparacion === '<' ? 'selected' : ''}>&lt;</option>
                <option value="<=" ${comparacion === '<=' ? 'selected' : ''}>&lt;=</option>
                <option value=">" ${comparacion === '>' ? 'selected' : ''}>&gt;</option>
                <option value=">=" ${comparacion === '>=' ? 'selected' : ''}>&gt;=</option>
            </select>
        </td>
        <td><input type="number" class="kpi-score" value="${score}" step="0.01"></td>
        <td><input type="number" class="kpi-porcentaje" value="${kpi.porcentaje || ''}" step="0.01" oninput="actualizarIndicadorBono()"></td>


${(tipo === 'bonus')
    ? `<td style="text-align:center;">
         <input type="checkbox" class="kpi-is-key" ${esKeyManual ? 'checked' : ''}
                style="width:20px; height:20px; cursor:pointer;"
                onchange="validarMaximoGates(this)">
       </td>`
    : `<td style="display:none;"><input type="checkbox" class="kpi-is-key"></td>`
}
        <td style="${styleParticipacion}">
            <input type="number" class="kpi-participacion" value="${kpi.participacion || ''}" step="0.01" placeholder="Opcional" oninput="calcularValoresPesosEnTiempoReal()">
        </td>
        <td class="celda-valor-pesos" style="${styleParticipacion} text-align:right; font-weight:bold; color:#27ae60;">—</td>
        <td><button type="button" class="btn-delete" onclick="eliminarFila(this)">🗑️</button></td>
        `;
    }
}



function calcularValoresPesosEnTiempoReal() {
    const salario = parseFloat(document.getElementById('salario').value.replace(',', '.')) || 0;
    const porcentajeBono = parseFloat(document.getElementById('porcentajeBono').value) || 0;
    const bonoMaximo = (salario * porcentajeBono) / 100;
    const tipoHoja = document.getElementById('tipoHoja').value;
    const esPisos = (tipoHoja === 'Registro_pisos');

    if (bonoMaximo <= 0) {
        document.querySelectorAll('.celda-valor-pesos').forEach(c => {
            c.textContent = '—';
            c.style.color = '#27ae60';
        });
        return;
    }

    ['Bonus', 'Additional', 'Decreased'].forEach(cat => {
        const kpisProcesados = {};
        document.querySelectorAll(`#tbody${cat} tr`).forEach(row => {
            const nombreInput = row.querySelector('.kpi-nombre');
            const pctInput    = row.querySelector('.kpi-porcentaje');
            const cellMonto   = row.querySelector('.celda-valor-pesos');

            if (!cellMonto) return;

            const nombre = nombreInput?.value?.trim().toUpperCase();
            const pctRaw = pctInput?.value || '';

            if (!nombre || !pctRaw) {
                cellMonto.textContent = '—';
                return;
            }

            let pct = parseFloat(pctRaw.replace(',', '.')) || 0;
            if (pct > 0 && pct < 1) pct = pct * 100;

            let part = 100;
            if (esPisos) {
                const partInput = row.querySelector('.kpi-participacion');
                const partRaw   = partInput?.value || '100';
                part = parseFloat(partRaw.replace(',', '.')) || 100;
                if (part > 0 && part < 1) part = part * 100;
            }

            const montoBase  = (bonoMaximo * pct) / 100;
            const montoFinal = (montoBase * part) / 100;
            const esRepetido = !!kpisProcesados[nombre];

            if (montoFinal > 0) {
                const prefix = cat === 'Decreased' ? '-' : '';
                cellMonto.textContent = prefix + formatearMoneda(montoFinal);
                cellMonto.style.color = esRepetido
                    ? '#95a5a6'
                    : (cat === 'Decreased' ? '#e74c3c' : '#27ae60');
            } else {
                cellMonto.textContent = '—';
                cellMonto.style.color = '#27ae60';
            }

            kpisProcesados[nombre] = true;
        });
    });
}

function actualizarIndicadorBono() {
    const salario = parseFloat(document.getElementById('salario').value) || 0;
    const porcentajeBono = parseFloat(document.getElementById('porcentajeBono').value) || 0;
    const bonoMaximo = (salario * porcentajeBono) / 100;
    
    console.log(`\n📊 Actualizando indicador...`);
    console.log(`   Bono Máximo: ${formatearMoneda(bonoMaximo)}`);
    
    // ⭐ SOLO BONUS STRUCTURE (no Additional ni Decreased)
    const kpisPorNombre = {};
    
    document.querySelectorAll('#tbodyBonus tr').forEach((row, idx) => {
        let nombreKPI = row.querySelector('.kpi-nombre')?.value;
        if (nombreKPI) nombreKPI = nombreKPI.trim().toUpperCase();
        const porcentajeInput = row.querySelector('.kpi-porcentaje')?.value;
        
        if (nombreKPI && porcentajeInput) {
            let porcentaje = parseFloat(porcentajeInput) || 0;
            if (porcentaje > 0 && porcentaje < 1) porcentaje = porcentaje * 100;
            
            // Solo contar la primera vez cada KPI (pisos no suman al 100%)
            if (!kpisPorNombre[nombreKPI]) {
                kpisPorNombre[nombreKPI] = porcentaje;
                console.log(`   [${idx + 1}] ${nombreKPI}: ${porcentaje}% (NUEVO)`);
            } else {
                console.log(`   [${idx + 1}] ${nombreKPI}: Piso detectado, no suma al total.`);
            }
        }
    });
    
    // ⭐ SUMAR SOLO UNA VEZ CADA KPI
    let totalAsignado = 0;
    Object.keys(kpisPorNombre).forEach(nombreKPI => {
        totalAsignado += kpisPorNombre[nombreKPI];
    });
    
    const disponible = 100 - totalAsignado;
    
    console.log(`   Total Bonus: ${totalAsignado.toFixed(1)}%`);
    console.log(`   Disponible: ${disponible.toFixed(1)}%`);
    
    // ========================================
    // 🎨 ACTUALIZAR UI
    // ========================================
    const progressBar = document.getElementById('progressBar');
    if (progressBar) {
        progressBar.style.width = Math.min(totalAsignado, 100) + '%';
        progressBar.textContent = totalAsignado.toFixed(1) + '%';
        
        progressBar.className = 'progress-bar';
        if (totalAsignado > 100) {
            progressBar.classList.add('danger');
        } else if (totalAsignado > 90) {

    if (porcentajeAsignado) {
        porcentajeAsignado.textContent = totalAsignado.toFixed(1) + '%';
    }
    
    if (porcentajeDisponible) {
        porcentajeDisponible.textContent = Math.max(0, disponible).toFixed(1) + '%';

function validarAntesDeGuardar() {
    const tipoHoja = document.getElementById('tipoHoja').value;

    // 🔍 DEBUG TEMPORAL - BORRAR DESPUÉS
    // ========================================

        const pct = row.querySelector('.kpi-porcentaje')?.value;
        console.log(`  [${idx}] Nombre: "${nombre}" | %: "${pct}"`);
    });
    
    // ========================================
    // ✅ VALIDACIÓN 1: AL MENOS 1 KEY OBLIGATORIO
    // ========================================
    const tbodyKeys = document.getElementById('tbodyKeys');
    const totalKeys = tbodyKeys ? tbodyKeys.rows.length : 0;
    
    if (totalKeys === 0) {
        alert('❌ ERROR: Debe agregar al menos 1 KPI en la sección Keys\n\n' +
              'Los Keys son obligatorios para cualquier estructura.');
        return false;
    }
    
    let keysValidos = 0;
    for (let row of tbodyKeys.rows) {
        const nombre = row.querySelector('.kpi-nombre')?.value;
        const score = row.querySelector('.kpi-score')?.value;
        if (nombre && score) keysValidos++;
    }
    
    if (keysValidos === 0) {
        alert('❌ ERROR: Los Keys deben tener Nombre y Meta/Score completos');
        return false;
    }
    
    // ========================================
    // ✅ VALIDACIÓN 2: BONUS STRUCTURE (100% MAX) - DEDUPLICADO
    // ========================================
    const kpisPorNombre = {};
    let totalBonusStructure = 0;
    
    document.querySelectorAll('#tbodyBonus tr').forEach((row, idx) => {
        let nombreKPI = row.querySelector('.kpi-nombre')?.value;
        if (nombreKPI) nombreKPI = nombreKPI.trim().toUpperCase();
        const porcentajeInput = row.querySelector('.kpi-porcentaje')?.value;
        
        if (!nombreKPI || !porcentajeInput) return;
        
        let porcentaje = parseFloat(porcentajeInput) || 0;
        if (porcentaje > 0 && porcentaje < 1) porcentaje = porcentaje * 100;
        
        if (!kpisPorNombre[nombreKPI]) {
            kpisPorNombre[nombreKPI] = porcentaje;
            totalBonusStructure += porcentaje;
        }
        // Pisos duplicados no se suman
    });
    
    // ========================================
    // ✅ VALIDACIÓN 3: ADDITIONAL Y DECREASED (SOLO INFORMATIVO)
    // ========================================
    let totalAdditional = 0;
    let totalDecreased = 0;
    
    document.querySelectorAll('#tbodyAdditional tr').forEach(row => {
        const nombre = row.querySelector('.kpi-nombre')?.value;
        const pctInput = row.querySelector('.kpi-porcentaje')?.value;
        if (nombre && pctInput) {
            let pct = parseFloat(pctInput) || 0;
            if (pct > 0 && pct < 1) pct = pct * 100;
            totalAdditional += pct;
        }
    });
    
    document.querySelectorAll('#tbodyDecreased tr').forEach(row => {
        const nombre = row.querySelector('.kpi-nombre')?.value;
        const pctInput = row.querySelector('.kpi-porcentaje')?.value;
        if (nombre && pctInput) {
            let pct = parseFloat(pctInput) || 0;
            if (pct > 0 && pct < 1) pct = pct * 100;
            totalDecreased += pct;
        }
    });
    
    console.log(`   Bonus Structure: ${totalBonusStructure.toFixed(1)}%`);
    console.log(`   Additional: ${totalAdditional.toFixed(1)}% (independiente)`);
    console.log(`   Decreased: ${totalDecreased.toFixed(1)}% (independiente)`);
    
    // ========================================
    // 🚨 VALIDAR SOLO BONUS STRUCTURE
    // ========================================
    if (totalBonusStructure > 100) {
        alert(`❌ ERROR: BONUS STRUCTURE SUPERA 100%\n\n` +
              `📊 Distribución actual:\n` +
              `• Bonus Structure: ${totalBonusStructure.toFixed(1)}% ❌ EXCEDE\n` +
              `• Additional: ${totalAdditional.toFixed(1)}% ✅ OK (no cuenta)\n` +
              `• Decreased: ${totalDecreased.toFixed(1)}% ✅ OK (no cuenta)\n\n` +
              `⚠️ SOLO BONUS STRUCTURE debe ser ≤ 100%\n` +
              `Additional y Decreased son independientes.\n\n` +
              `✅ Solución: Reduce los porcentajes de Bonus Structure.`);
        return false;
    }
    
    // ========================================
    // ✅ VALIDACIÓN EXITOSA
    // ========================================
    console.log(`\n✅ ===== VALIDACIÓN EXITOSA =====\n`);
    
    // Mensaje de confirmación solo si hay Additional o Decreased
    if (totalAdditional > 0 || totalDecreased > 0) {
        const msg = `✅ ESTRUCTURA VÁLIDA\n\n` +
                   `📊 Resumen:\n` +
                   `• Bonus Structure: ${totalBonusStructure.toFixed(1)}% ✅\n` +
                   `• Additional: ${totalAdditional.toFixed(1)}%\n` +
                   `• Decreased: ${totalDecreased.toFixed(1)}%\n\n` +
                   `💡 Nota: Additional y Decreased son independientes\n` +
                   `y NO cuentan para el límite de 100% de Bonus.\n\n` +
                   `¿Continuar con el guardado?`;
        
        return confirm(msg);
    }
    
    return true;
}

        function cambiarTipoAccesoUsuario() {
            const select = document.getElementById('tipoAccesoSelect');
            const statusSpan = document.getElementById('tipoAccesoStatus');

            if (!select) {
                console.error('❌ No se encontró el select de tipo de acceso');
                mostrarToast('❌ Error: No se encontró el selector', 'danger');
                return;
            }

            const tipoAcceso = select.value;
            console.log('🔄 Cambiando tipo de acceso a:', tipoAcceso);

            // Validar selección
            if (!tipoAcceso || tipoAcceso === '') {
                mostrarToast('⚠️ Por favor selecciona un tipo de acceso', 'warning');
                return;
            }

            // Deshabilitar botón temporalmente
            const boton = event.target;
            if (boton && boton.tagName === 'BUTTON') {
                boton.disabled = true;
                boton.textContent = '⏳ Procesando...';
            }

            // Usar correo del usuario actual
            const email = datosUsuario.correo || (document.getElementById('correoUsuario') ? document.getElementById('correoUsuario').textContent : '');

            mostrarLoading(`Cambiando acceso a ${tipoAcceso}...`);

            google.script.run
                .withSuccessHandler(function (response) {
                    ocultarLoading();

                    if (response.success) {
                        mostrarToast('✅ Tipo de acceso cambiado a: ' + tipoAcceso + '. Recargando...', 'success');

                        setTimeout(function () {
                            // Recargar para aplicar cambios
                            google.script.run.withSuccessHandler(function (url) {
                                if (url) {
                                    const sep = url.includes('?') ? '&' : '?';
                                    window.top.location.href = url + sep + 'page=' + tipoAcceso;
                                } else {
                                    window.top.location.reload();
                                }
                            }).REGISTRO_getUrl();
                        }, 1200);

                    } else {
                        mostrarToast('❌ Error: ' + response.error, 'danger');
                        if (boton && boton.tagName === 'BUTTON') {
                            boton.disabled = false;
                            boton.textContent = '🔄 Cambiar';
                        }
                    }
                })
                .withFailureHandler(function (error) {
                    ocultarLoading();
                    if (boton && boton.tagName === 'BUTTON') {
                        boton.disabled = false;
                        boton.textContent = '🔄 Cambiar';
                    }
                    mostrarToast('❌ Error de conexión: ' + error.message, 'danger');
                })
                .REGISTRO_cambiarTipoAccesoUsuario(email, tipoAcceso);
        }

// ⭐ FUNCIÓN: Renderizar dinámicamente el ranking basado en el filtro
function renderizarRankingAgentes(filtro) {
    const data = window.simulacionRealActual;
    if (!data || !data.top10Ganadores) return;
    
    const container = document.getElementById('contenedorTablaRanking');
    if (!container) return;
    
    let agentesAMostrar = data.top10Ganadores;
    if (filtro === 'top10') {
        agentesAMostrar = data.top10Ganadores.slice(0, 10);
    }
    
    container.innerHTML = generarHtmlTablaRanking(agentesAMostrar, data.configuracion);
}

// ⭐ FUNCIÓN: Generar solo el HTML de la tabla para el re-renderizado
function generarHtmlTablaRanking(agentes, configuracion) {
    let html = `
    <table class="kpi-table">
        <thead>
            <tr>
                <th>#</th>
                <th>BMS ID</th>
                <th>Nombre</th>
                <th>Horas Reales</th>
                <th>Factor</th>
                <th style="color: #27ae60;">Keys</th>
                <th style="color: #27ae60;">Bonus</th>
                <th style="color: #27ae60;">Additional</th>
                <th style="color: #e74c3c;">Penalties</th>
                <th style="background: #f8f9fa;">TOTAL</th>
                <th>Detalle</th>
            </tr>
        </thead>
        <tbody>`;
        
    agentes.forEach((agente, idx) => {
        const medalla = idx === 0 ? '🥇' : idx === 1 ? '🥈' : idx === 2 ? '🥉' : (idx + 1);
        const totalKeys = parseFloat(agente.totalKeys) || 0;
        const totalBonus = parseFloat(agente.totalBonus) || 0;
        const totalAdditional = parseFloat(agente.totalAdditional) || 0;
        const totalDecreased = parseFloat(agente.totalDecreased) || 0;
        const bonoFinal = parseFloat(agente.bonoFinal) || 0;
        
        const bonoMaximoCheck = configuracion?.bonoMaximo || 0;
        const maxEsperado = bonoMaximoCheck * 1.5;
        
        let estiloFila = '';
        let colorTotal = '#27ae60';
        if (bonoFinal > maxEsperado && bonoMaximoCheck > 0) {
            estiloFila = 'background: #ffebee; border-left: 3px solid #e74c3c;';
            colorTotal = '#e74c3c';
        }
        
        const horasSemanales = parseFloat(agente.horasSemanales) || 0;
        const factorProrrateo = parseFloat(agente.factorProrrateo) || 1;
        const horasDisplay = horasSemanales > 0 ? horasSemanales + 'h/sem' : 'N/A';
        const factorDisplay = factorProrrateo !== 1 ? (factorProrrateo * 100).toFixed(0) + '%' : '100%';
        const colorFactor = factorProrrateo < 0.9 ? '#e74c3c' : '#27ae60';
        
        html += `
            <tr style="${estiloFila}">
                <td style="text-align: center; font-size: 1.2rem;">${medalla}</td>
                <td style="text-align: center;">${agente.bmsId}</td>
                <td style="font-weight: 600;">${agente.nombre}</td>
                <td style="text-align: center; color: #2c3e50; font-weight: 600;">${horasDisplay}</td>
                <td style="text-align: center; font-size: 0.85rem; color: ${colorFactor}; font-weight: 600;">${factorDisplay}</td>
                <td style="text-align: right;">${formatearMoneda(totalKeys)}</td>
                <td style="text-align: right;">${formatearMoneda(totalBonus)}</td>
                <td style="text-align: right;">${formatearMoneda(totalAdditional)}</td>
                <td style="text-align: right; color: #e74c3c;">-${formatearMoneda(totalDecreased)}</td>
                <td style="text-align: right; font-weight: bold; color: ${colorTotal}; background: #f8f9fa;">${formatearMoneda(Math.max(0, bonoFinal))}</td>
                <td style="text-align: center;">
                    <button onclick="verDetalleKPIsAgente('${agente.bmsId}')" class="btn btn-primary" style="padding: 5px 10px; font-size: 0.8rem;">
                        👁️ Ver
                    </button>
                </td>
            </tr>`;
    });
    
    html += `</tbody></table>`;
    return html;
}

// 🔧 FUNCIÓN AUXILIAR: Generar tablas CON PISOS para CDM
// ============================================================
function generarTablaKPIsRealesConPisos(kpis, bonoMaximo, hc, tipo, esPisos) {
    const colorBorde = tipo === 'bonus' ? '#27ae60' : tipo === 'additional' ? '#f39c12' : '#e74c3c';
    const titulo = tipo === 'bonus' ? '💰 Bonus Structure' : tipo === 'additional' ? '⭐ Additional Incentive' : '🔴 Decreased';
    
    let html = `
    <div style="margin-bottom: 30px;">
        <h4 style="color: ${colorBorde};">${titulo}</h4>
        <table class="kpi-table" style="border-left: 4px solid ${colorBorde};">
            <thead>
                <tr>
                    <th>KPI</th>`;
    
    if (esPisos) {
        html += `<th>Piso</th>`;
    }
    
    html += `
                    <th>Meta</th>
                    <th>% Base</th>`;
    
    if (esPisos) {
        html += `
                    <th style="background: #fff3cd; font-weight: bold;">% Part.</th>
                    <th style="background: #e8f5e9; color: #27ae60; font-weight: bold;">💰 Valor/Agente</th>`;
    }
    
    html += `
                    <th>Cumplen</th>
                    <th>Total x HC</th>
                </tr>
            </thead>
            <tbody>
    `;
    
    let totalCategoria = 0;
    
    // ⭐ AGRUPAR POR NOMBRE DE KPI
    const kpisPorNombre = {};
    kpis.forEach(kpi => {
        const nombre = kpi.kpi;
        if (!kpisPorNombre[nombre]) {
            kpisPorNombre[nombre] = [];
        }
        kpisPorNombre[nombre].push(kpi);
    });
    
    Object.keys(kpisPorNombre).forEach(nombreKPI => {
        const pisos = kpisPorNombre[nombreKPI];
        const numPisos = pisos.length;
        
        let pct = parseFloat(pisos[0].porcentaje) || 0;
        if (pct > 0 && pct < 1) pct = pct * 100;
        
        // Encontrar máxima participación
        let participacionMaxima = 100;
        if (esPisos) {
            participacionMaxima = 0;
            pisos.forEach(piso => {
                let part = parseFloat(piso.participacion) || 100;
                if (part > 0 && part < 1) part = part * 100;
                if (part > participacionMaxima) participacionMaxima = part;
            });
        }
        
        const valorBase = (bonoMaximo * pct) / 100;
        const valorConPartMax = esPisos ? (valorBase * participacionMaxima) / 100 : valorBase;
        
        // Total cumplen (sumando de todos los pisos)
        const totalCumplen = pisos.reduce((sum, p) => sum + (p.validacion?.totalCumplenCondicion || 0), 0);
        
        const montoTotal = pisos.reduce((sum, p) => {
            return sum + (p.validacion?.empleadosCumplen || []).reduce((s, e) => s + (tipo === 'decreased' ? Math.abs(e.bono || 0) : (e.bono || 0)), 0);
        }, 0);
        
        totalCategoria += montoTotal;
        
        if (esPisos && numPisos > 1) {
            // Mostrar pisos separados
            pisos.forEach((piso, idx) => {
                let participacion = parseFloat(piso.participacion) || 100;
                if (participacion > 0 && participacion < 1) participacion = participacion * 100;
                
                const esMaximo = (participacion === participacionMaxima);
                const cumplenPiso = piso.validacion?.totalCumplenCondicion || 0;
                
                html += `
                <tr ${esMaximo ? 'style="background: #e8f5e9; font-weight: 600;"' : ''}>
                    ${idx === 0 ? `<td rowspan="${numPisos}" style="font-weight: 600; vertical-align: middle;">${nombreKPI}</td>` : ''}
                    <td style="text-align: center; background: #f8f9fa;">${idx + 1} ${esMaximo ? '🏆' : ''}</td>
                    <td style="text-align: center;">${piso.score || '-'}</td>
                    ${idx === 0 ? `<td rowspan="${numPisos}" style="text-align: center; font-weight: 600; color: ${colorBorde}; vertical-align: middle;">${pct.toFixed(1)}%</td>` : ''}
                    <td style="text-align: center; background: #fff3cd; font-weight: ${esMaximo ? 'bold' : 'normal'}; color: ${esMaximo ? '#27ae60' : 'inherit'};">
                        ${participacion.toFixed(0)}% ${esMaximo ? '👑' : ''}
                    </td>
                    ${idx === 0 ? `<td rowspan="${numPisos}" style="text-align: right; vertical-align: middle; font-weight: bold; color: #27ae60; background: #e8f5e9;">
                        ${formatearMoneda(valorConPartMax)}
                    </td>` : ''}
                    <td style="text-align: center;">${cumplenPiso} / ${hc}</td>
                    ${idx === 0 ? `<td rowspan="${numPisos}" style="text-align: right; vertical-align: middle; font-weight: bold; color: ${tipo === 'decreased' ? '#e74c3c' : '#27ae60'};">
                        ${tipo === 'decreased' ? '-' : ''}${formatearMoneda(montoTotal)}
                    </td>` : ''}
                </tr>
                `;
            });
        } else {
            // Una sola fila
            html += `
            <tr>
                <td style="font-weight: 600;">${nombreKPI}</td>
                <td style="text-align: center;">${pisos[0].score || '-'}</td>
                <td style="text-align: center; font-weight: 600; color: ${colorBorde};">${pct.toFixed(1)}%</td>`;
            
            if (esPisos) {
                html += `
                <td style="text-align: center; background: #fff3cd; font-weight: 600;">${participacionMaxima.toFixed(0)}%</td>
                <td style="text-align: right; background: #e8f5e9; font-weight: bold; color: #27ae60;">${formatearMoneda(valorConPartMax)}</td>`;
            }
            
            html += `
                <td style="text-align: center;">${totalCumplen} / ${hc}</td>
                <td style="text-align: right; font-weight: bold; color: ${tipo === 'decreased' ? '#e74c3c' : '#27ae60'};">
                    ${tipo === 'decreased' ? '-' : ''}${formatearMoneda(montoTotal)}
                </td>
            </tr>
            `;
        }
    });
    
    html += `
            <tr style="background: ${tipo === 'bonus' ? '#d5f4e6' : tipo === 'additional' ? '#fef5e7' : '#ffebee'}; font-weight: bold;">
                <td colspan="${esPisos ? '7' : '5'}" style="text-align: right;">SUBTOTAL ${tipo.toUpperCase()}:</td>
                <td style="text-align: right; color: ${tipo === 'decreased' ? '#e74c3c' : '#27ae60'};">
                    ${tipo === 'decreased' ? '-' : ''}${formatearMoneda(totalCategoria)}
                </td>
            </tr>
        </tbody>
    </table>
</div>
    `;
    
    return html;
}

// 📊 FUNCIÓN AUXILIAR: Obtener número de mes desde nombre
// ============================================================
function obtenerNumeroMesDesdeNombre(nombreMes) {
    const mesesMap = {
        'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
        'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
        'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
    };
    
    const mesLower = nombreMes.toLowerCase().trim();
    return mesesMap[mesLower] || null;
}

// 👁️ FUNCIÓN: Ver detalle de KPIs por Agente
// ============================================================
// ============================================================
// 👁️ FIX: verDetalleKPIsAgente()
// Correcciones:
// 1. Decreased ya no aparece duplicado
// 2. "Tier Logrado" muestra el tier real alcanzado (no el objetivo)
// ============================================================
function verDetalleKPIsAgente(bmsId) {
    const data = window.simulacionRealActual;
    if (!data) return;

    const agenteRanking = data.top10Ganadores.find(a => a.bmsId === bmsId);
    const nombreAgente  = agenteRanking ? agenteRanking.nombre : bmsId;

    let html = `
    <div style="padding: 10px;">
        <h2 style="color: #2c3e50; border-bottom: 3px solid #3498db; padding-bottom: 10px; margin-bottom: 20px;">
            🔍 Detalle de KPIs: ${nombreAgente} <span style="font-size:0.75em; color:#666;">(${bmsId})</span>
        </h2>

        <table class="kpi-table" style="width: 100%; border-collapse: collapse;">
            <thead>
                <tr style="background: #f8f9fa;">
                    <th>Categoría</th>
                    <th>KPI</th>
                    <th style="text-align:center;">Meta Objetivo</th>
                    <th style="text-align:center; background:#e3f2fd;">📊 Score Real CDM</th>
                    <th style="text-align:center; background:#fff3cd;">% Participación</th>
                    <th style="text-align:center;">Resultado</th>
                    <th style="text-align:right;">Monto</th>
                </tr>
            </thead>
            <tbody>
    `;

    const categorias = [
        { id: 'keys',                 label: '🔑 Key KPIs',            color: '#3498db' },
        { id: 'bonusStructure',       label: '💰 Bonus Structure',      color: '#27ae60' },
        { id: 'additionalIncentives', label: '⭐ Additional Incentive', color: '#f39c12' },
        { id: 'decreasedKPIs',        label: '🔴 Decreased',           color: '#e74c3c' }
    ];

    let totalCalculado = 0;
    let hayFilas = false;

    categorias.forEach(cat => {
        const listaKPIs = data.resultadosDetallados[cat.id] || [];

        // Deduplicar KPIs por nombre (consolidar pisos)
        const kpisPorNombre = {};
        listaKPIs.forEach(kpi => {
            const nombre = kpi.kpi;
            if (!kpisPorNombre[nombre]) kpisPorNombre[nombre] = [];
            kpisPorNombre[nombre].push(kpi);
        });

        Object.keys(kpisPorNombre).forEach(nombreKPI => {
            const pisos = kpisPorNombre[nombreKPI];

            // Buscar al agente en todos los pisos
            let cumple   = null;
            let noCumple = null;
            let sinDatos = null;

            for (const piso of pisos) {
                const val = piso.validacion || {};
                const c = (val.empleadosCumplen   || []).find(e => e.bmsId === bmsId);
                const n = (val.empleadosNoCumplen || []).find(e => e.bmsId === bmsId);
                const s = (val.empleadosSinDatos  || []).find(e => e.bmsId === bmsId);

                if (c && (!cumple || Math.abs(c.bono || 0) > Math.abs(cumple.bono || 0))) cumple = c;
                if (!cumple && n) noCumple = n;
                if (!cumple && !noCumple && s) sinDatos = s;
            }

            if (!cumple && !noCumple && !sinDatos) return;
            hayFilas = true;

            const esMultiTier   = pisos.length > 1 || pisos[0]?.esMultiTier;
            const entradaAgente = cumple || noCumple;

            // ── Meta objetivo ─────────────────────────────────
            let metaDisplay;
            if (esMultiTier) {
                const scores = pisos.map(p => p.score).filter(s => s && s !== 'Multi-Escala').join(' | ');
                metaDisplay = `<span style="color:#8e44ad;font-style:italic;" title="${scores}">Multi-Escala 🎚️</span>`;
            } else {
                metaDisplay = pisos[0].score || '—';
            }

            // ✅ FIX BUG 2: mostrar el score REAL del CDM del agente
            let scoreRealDisplay = '—';
            if (entradaAgente) {
                if (entradaAgente.valorReal !== undefined && entradaAgente.valorReal !== null) {
                    // ⭐ FIX: Formatear a máximo 2 decimales, sin % automático
                    const val = parseFloat(entradaAgente.valorReal);
                    const valDisplay = isNaN(val) ? entradaAgente.valorReal : val.toFixed(2);
                    scoreRealDisplay = `<strong style="color:#1565c0;">${valDisplay}</strong>`;
                } else if (entradaAgente.scoreCumplido) {
                    // Fallback: usar el score del tier cumplido
                    scoreRealDisplay = `<strong style="color:#1565c0;">${entradaAgente.scoreCumplido}</strong>`;
                }
            } else if (sinDatos) {
                scoreRealDisplay = '<span style="color:#f39c12;">Sin datos CDM</span>';
            }

            // ── Participación ─────────────────────────────────
            let participacionUI = '—';
            if (entradaAgente) {
                if (entradaAgente.participacionUsada !== undefined && entradaAgente.participacionUsada !== null) {
                    let pUI = parseFloat(entradaAgente.participacionUsada);
                    if (!isNaN(pUI)) {
                        if (pUI > 0 && pUI <= 1) pUI = pUI * 100;
                        participacionUI = pUI.toFixed(0) + '%';
                    }
                } else if (!esMultiTier) {
                    participacionUI = '100%';
                } else {
                    for (const piso of pisos) {
                        const val    = piso.validacion || {};
                        const enPiso = (val.empleadosCumplen || []).find(e => e.bmsId === bmsId)
                                    || (val.empleadosNoCumplen || []).find(e => e.bmsId === bmsId);
                        if (enPiso) {
                            let pUI = parseFloat(piso.participacion) || 100;
                            if (pUI > 0 && pUI <= 1) pUI = pUI * 100;
                            participacionUI = pUI.toFixed(0) + '%';
                            break;
                        }
                    }
                }
            }

            const partDisplay = participacionUI !== '—'
                ? `<span style="background:#fff3cd;padding:3px 8px;border-radius:4px;font-weight:bold;">${participacionUI}</span>`
                : '—';

            // ── Resultado y monto ─────────────────────────────
            let resultadoText = '', resultadoColor = '', monto = 0;
            if (cumple) {
                resultadoText  = '✅ CUMPLE';
                resultadoColor = '#27ae60';
                monto = cumple.bono || 0;
            } else if (noCumple) {
                resultadoText  = '❌ NO CUMPLE';
                resultadoColor = '#e74c3c';
            } else if (sinDatos) {
                resultadoText  = cat.id === 'decreasedKPIs' ? '✅ SIN DATOS (No Penaliza)' : '⚠️ SIN DATOS CDM';
                resultadoColor = cat.id === 'decreasedKPIs' ? '#27ae60' : '#f39c12';
            }

            totalCalculado += monto;

            // ✅ FIX BUG 1 display: para decreased mostrar monto como negativo
            const montoDisplay = monto !== 0
                ? `<span style="color:${monto < 0 ? '#e74c3c' : '#27ae60'};font-weight:bold;">${formatearMoneda(monto)}</span>`
                : '—';

            html += `
            <tr>
                <td style="font-weight:bold; color:${cat.color}; white-space:nowrap;">${cat.label}</td>
                <td style="font-weight:600;">${nombreKPI}</td>
                <td style="text-align:center;">${metaDisplay}</td>
                <td style="text-align:center; background:#f0f8ff;">${scoreRealDisplay}</td>
                <td style="text-align:center;">${partDisplay}</td>
                <td style="text-align:center; font-weight:bold; color:${resultadoColor};">${resultadoText}</td>
                <td style="text-align:right;">${montoDisplay}</td>
            </tr>
            `;
        });
    });

    if (!hayFilas) {
        html += `<tr><td colspan="7" style="text-align:center;padding:20px;color:#999;font-style:italic;">
            ⚠️ No se encontraron KPIs para este agente
        </td></tr>`;
    }

    html += `
            </tbody>
            <tfoot>
                <tr style="background:#2c3e50; color:white; font-size:1.2rem;">
                    <td colspan="6" style="text-align:right; padding:15px; font-weight:bold;">TOTAL BONO:</td>
                    <td style="text-align:right; padding:15px; font-weight:bold;">${formatearMoneda(Math.max(0, totalCalculado))}</td>
                </tr>
            </tfoot>
        </table>

        <div style="margin-top:25px; text-align:center;">
            <button onclick="cerrarModalDetalleAgente()" class="btn btn-primary" style="padding:12px 30px;">CERRAR</button>
        </div>
    </div>
    `;

    mostrarModalDetalleAgente(html);
}
function mostrarModalDetalleAgente(html) {
    // Si no existe un div para el segundo modal, lo creamos
    let modal = document.getElementById('modalDetalleAgente');
    if (!modal) {
        modal = document.createElement('div');
        modal.id = 'modalDetalleAgente';
        modal.className = 'modal';
        modal.innerHTML = `
            <div class="modal-content" style="max-width: 800px;">
                <span class="close-btn" onclick="cerrarModalDetalleAgente()">&times;</span>
                <div id="modalDetalleAgenteContent"></div>
            </div>
        `;
        document.body.appendChild(modal);

function exportarSimulacionRealExcel() {
    if (!window.simulacionRealActual) {
        mostrarToast('⚠️ No hay datos para exportar', 'warning');
        return;
    }
    
    mostrarLoading('Generando Excel con métricas reales del CDM...');
    
    const data = window.simulacionRealActual;
    
    // ⭐ Preparar encabezados
    const encabezados = [
        'Categoría', 'KPI', 'Piso', 'Meta', '% Base', '% Participación',
        'Valor/Agente', 'Cumplen', 'No Cumplen', 'Sin Datos', 'Total x HC'
    ];
    
    const filas = [];
    
    // ========================================
    // 🔑 KEYS
    // ========================================
    if (data.resultadosDetallados?.keys) {
        data.resultadosDetallados.keys.forEach(kpi => {
            const val = kpi.validacion || {};
            filas.push([
                'KEYS',
                kpi.kpi,
                '-',
                kpi.score || '-',
                '-',
                '-',
                '-',
                val.totalCumplenCondicion || 0,
                val.empleadosNoCumplen?.length || 0,
                val.empleadosSinDatos?.length || 0,
                '-'
            ]);
        });
    }
    
    // ========================================
    // 💰 BONUS STRUCTURE
    // ========================================
    if (data.resultadosDetallados?.bonusStructure) {
        const kpisPorNombre = {};
        
        // Agrupar por nombre
        data.resultadosDetallados.bonusStructure.forEach(kpi => {
            const nombre = kpi.kpi;
            if (!kpisPorNombre[nombre]) kpisPorNombre[nombre] = [];
            kpisPorNombre[nombre].push(kpi);
        });
        
        Object.keys(kpisPorNombre).forEach(nombreKPI => {
            const pisos = kpisPorNombre[nombreKPI];
            let pct = parseFloat(pisos[0].porcentaje) || 0;
            if (pct > 0 && pct < 1) pct = pct * 100;
            
            const bonoMaximo = data.configuracion?.bonoMaximo || 0;
            const valorBase = (bonoMaximo * pct) / 100;
            
            const montoTotal = pisos.reduce((sum, p) => {
                return sum + (p.validacion?.empleadosCumplen || []).reduce((s, e) => s + (e.bono || 0), 0);
            }, 0);
            
            const totalCumplen = pisos.reduce((sum, p) => sum + (p.validacion?.totalCumplenCondicion || 0), 0);
            
            if (pisos.length > 1) {
                // Múltiples pisos
                pisos.forEach((piso, idx) => {
                    let participacion = parseFloat(piso.participacion) || 100;
                    if (participacion > 0 && participacion < 1) participacion = participacion * 100;
                    
                    const valorConPart = (valorBase * participacion) / 100;
                    const cumplenPiso = piso.validacion?.totalCumplenCondicion || 0;
                    
                    filas.push([
                        idx === 0 ? 'BONUS' : '',
                        idx === 0 ? nombreKPI : '',

                        idx === 0 ? pct.toFixed(1) + '%' : '',
                        participacion.toFixed(0) + '%',
                        idx === 0 ? valorConPart : '',

                        '-',
                        '-',
                        idx === 0 ? montoTotal : ''
                    ]);
                });
            } else {
                // Un solo piso
                const participacion = parseFloat(pisos[0].participacion) || 100;
                const participacionDisplay = participacion > 0 && participacion < 1 ? participacion * 100 : participacion;
                const valorConPart = (valorBase * participacionDisplay) / 100;
                
                filas.push([
                    'BONUS',
                    nombreKPI,
                    '-',
                    pisos[0].score || '-',
                    pct.toFixed(1) + '%',
                    participacionDisplay.toFixed(0) + '%',
                    valorConPart,
                    totalCumplen,
                    '-',
                    '-',
                    montoTotal
                ]);
            }
        });
    }
    
    // ========================================
    // ⭐ ADDITIONAL INCENTIVE
    // ========================================
    if (data.resultadosDetallados?.additionalIncentives) {
        const kpisPorNombre = {};
        
        data.resultadosDetallados.additionalIncentives.forEach(kpi => {
            const nombre = kpi.kpi;
            if (!kpisPorNombre[nombre]) kpisPorNombre[nombre] = [];

                return sum + (p.validacion?.empleadosCumplen || []).reduce((s, e) => s + (e.bono || 0), 0);
            }, 0);

                        idx === 0 ? 'ADDITIONAL' : '',
                        idx === 0 ? nombreKPI : '',
                        idx + 1,
                        piso.score || '-',
                        idx === 0 ? pct.toFixed(1) + '%' : '',
                        participacion.toFixed(0) + '%',
                        idx === 0 ? valorConPart : '',
                        cumplenPiso,

                        idx === 0 ? montoTotal : ''
                    ]);

            } else {
                const participacion = parseFloat(pisos[0].participacion) || 100;
                const participacionDisplay = participacion > 0 && participacion < 1 ? participacion * 100 : participacion;
                const valorConPart = (valorBase * participacionDisplay) / 100;

                    nombreKPI,

                    pct.toFixed(1) + '%',
                    participacionDisplay.toFixed(0) + '%',
                    valorConPart,
                    totalCumplen,
                    '-',
                    '-',
                    montoTotal
                ]);
            }
        });
    }
    
    // ========================================
    // 🔴 DECREASED
    // ========================================
    if (data.resultadosDetallados?.decreasedKPIs) {
        const kpisPorNombre = {};
        
        data.resultadosDetallados.decreasedKPIs.forEach(kpi => {
            const nombre = kpi.kpi;

function togglePanelCorreoSimulacionReal() {
    const panel = document.getElementById('panelCorreoSimulacionReal');
    if (panel.style.display === 'none') {
        panel.style.display = 'block';
        actualizarListaDestinatariosReal();
    } else {
        panel.style.display = 'none';
    }
}

function agregarDestinatarioReal() {
    const input = document.getElementById('inputNuevoCorreoReal');
    const correo = input.value.trim();
    
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(correo)) {
        mostrarToast('⚠️ Ingrese un correo válido', 'warning');
        return;
    }
    
    if (window.destinatariosCorreoReal.includes(correo)) {
        mostrarToast('⚠️ Este correo ya está en la lista', 'warning');
        return;
    }
    
    window.destinatariosCorreoReal.push(correo);
    input.value = '';
    actualizarListaDestinatariosReal();
    mostrarToast('✅ Destinatario agregado', 'success');
}

function eliminarDestinatarioReal(correo) {
    window.destinatariosCorreoReal = window.destinatariosCorreoReal.filter(c => c !== correo);
    actualizarListaDestinatariosReal();
}

function actualizarListaDestinatariosReal() {
    const container = document.getElementById('listaDestinatariosReal');
    if (!container) return;
    
    if (window.destinatariosCorreoReal.length === 0) {
        container.innerHTML = '<span style="color: #999; font-style: italic;">No hay destinatarios adicionales</span>';
        return;
    }
    
    let html = '';
    window.destinatariosCorreoReal.forEach(correo => {
        html += `
        <span style="display: inline-flex; align-items: center; background: #667eea; color: white; 
                    padding: 8px 15px; border-radius: 20px; margin: 5px; font-size: 0.95em;">
            ${correo}
            <button onclick="eliminarDestinatarioReal('${correo}')" 
                    style="background: none; border: none; color: white; margin-left: 10px; cursor: pointer; font-size: 1.2em;">
                ×
            </button>
        </span>
        `;
    });
    container.innerHTML = html;
}

function enviarReporteRealPorCorreo() {
    const data = window.simulacionRealActual;
    if (!data) {
        mostrarToast('❌ No hay datos de simulación', 'danger');
        return;
    }
    
    // Agregar correo del usuario actual
    const correoUsuario = datosUsuario.correo;
    const destinatariosFinales = [...window.destinatariosCorreoReal];
    
    if (correoUsuario && !destinatariosFinales.includes(correoUsuario)) {
        destinatariosFinales.push(correoUsuario);
    }
    
    if (destinatariosFinales.length === 0) {
        mostrarToast('⚠️ No hay destinatarios válidos', 'warning');
        return;
    }
    
    const asunto = document.getElementById('inputAsuntoCorreoReal').value.trim();
    const mensaje = document.getElementById('inputMensajeCorreoReal').value.trim();
    
    if (!asunto) {
        mostrarToast('⚠️ El asunto no puede estar vacío', 'warning');
        return;
    }
    
    mostrarLoading('Enviando correo con reporte CDM...');
    
    const datosCorreo = {
        destinatarios: destinatariosFinales,
        asunto: asunto,
        mensaje: mensaje || 'Adjunto reporte de simulación con métricas reales del CDM.',
        simulacion: data
    };
    
    google.script.run
        .withSuccessHandler(function(response) {
            ocultarLoading();
            if (response.success) {
                mostrarToast('✅ Correo enviado exitosamente (incluyendo copia a ti)', 'success');
                window.destinatariosCorreoReal = [];
                actualizarListaDestinatariosReal();
                togglePanelCorreoSimulacionReal();
            } else {
                mostrarToast('❌ Error: ' + response.error, 'danger');
            }
        })
        .withFailureHandler(function(err) {
            ocultarLoading();
            mostrarToast('❌ Error al enviar: ' + err, 'danger');
        })
        .REGISTRO_enviarReporteSimulacionPorCorreo(datosCorreo);
}


function _ejecutarSimulacionReal(lob, mesInput) {
    const mesesMap = {
        'enero': 1, 'febrero': 2, 'marzo': 3, 'abril': 4,
        'mayo': 5, 'junio': 6, 'julio': 7, 'agosto': 8,
        'septiembre': 9, 'octubre': 10, 'noviembre': 11, 'diciembre': 12
    };
    const mesesNombres = [
        'Enero','Febrero','Marzo','Abril','Mayo','Junio',
        'Julio','Agosto','Septiembre','Octubre','Noviembre','Diciembre'
    ];

    // Normalizar mesInput: si viene como yyyy-MM-dd convertir a nombre
    let mes = mesInput;
    if (mes && mes.includes('-')) {
        const partes = mes.split('-');
        if (partes.length >= 2) {
            const numMes = parseInt(partes[1]) - 1;
            mes = mesesNombres[numMes] || mes;
        }
    }

    const mesLower = (mes || '').toLowerCase().trim();
    const numeroMes = mesesMap[mesLower] || new Date().getMonth() + 1;

    console.log(`\n🎯 _ejecutarSimulacionReal`);
    console.log(`   LOB: "${lob}"`);
    console.log(`   Mes normalizado: "${mes}"`);
    console.log(`   NumMes: ${numeroMes}`);

    mostrarLoading('🔄 Calculando con métricas reales del CDM...\nEsto puede tomar 30-60 segundos...');

    google.script.run
        .withSuccessHandler(function(resultado) {
            ocultarLoading();
            if (resultado && resultado.success) {
                mostrarResultadoMetricasReales(resultado);
            } else {
                console.error('❌ Error:', resultado ? resultado.error : 'Sin respuesta');
                mostrarToast('❌ Error: ' + (resultado ? resultado.error : 'No se recibió respuesta'), 'danger');
            }
        })
        .withFailureHandler(function(error) {
            ocultarLoading();
            mostrarToast('❌ Error: ' + (error.message || error), 'danger');
        })
        // ⭐ SIN esProrrateado — mismo call en todos los casos → mismo total
        .REGISTRO_simularConMetricasReales(lob, mes, numeroMes);
}
